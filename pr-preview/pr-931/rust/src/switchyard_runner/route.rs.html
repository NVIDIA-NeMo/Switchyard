20/// Capabilities declared for one route.
21///
22/// `GET /v1/models` includes `context_window`, `tool_calling`, and `vision` in each
23/// standard `data` entry, using `null` for unset values. It omits `reasoning`.
24/// The server rejects tool inputs, reasoning controls, or images when the
25/// corresponding declaration is explicitly `false` for the selected route.
26#[derive(Clone, Copy, Default)]
27pub struct ModelCapabilities {
28    pub context_window: Option<u32>,
29    pub tool_calling: Option<bool>,
30    /// Whether the routed model accepts reasoning controls, as declared in config.
31    pub reasoning: Option<bool>,
32    /// Whether the routed model accepts image input, as declared in config.
33    pub vision: Option<bool>,
34}

36/// Caller credential family required by a forwarded-auth route.
37#[derive(Clone, Copy, Debug, Eq, PartialEq)]
38pub enum CallerAuthKind {
39    Anthropic,
40    OpenAi,
41}
42
43impl CallerAuthKind {
44    /// Stable provider name used by the server compatibility API.
45    pub const fn as_str(self) -> &'static str {
46        match self {
47            Self::Anthropic => "anthropic",
48            Self::OpenAi => "openai",
49        }
50    }
51
52    const fn accepts(self, wire_format: WireFormat) -> bool {
53        matches!(
54            (self, wire_format),
55            (Self::Anthropic, WireFormat::AnthropicMessages)
56                | (
57                    Self::OpenAi,
58                    WireFormat::OpenAiChat | WireFormat::OpenAiResponses
59                )
60        )
61    }
62}

71/// Error returned while loading or executing configured routes.
72#[derive(Debug, Error)]
73pub enum RunnerError {
74    #[error("{message}")]
75    Configuration {
76        message: String,
77        #[source]
78        source: Option<Box<dyn Error + Send + Sync>>,
79    },
80    #[error("unknown route model {0:?}")]
81    UnknownRouteModel(String),
82    #[error("caller format is incompatible with {} credentials", .0.as_str())]
83    IncompatibleCallerFormat(CallerAuthKind),
84    #[error("route has no compatible target for the auxiliary operation")]
85    AuxiliaryUnsupported,
86    #[error(transparent)]
87    Algorithm(#[from] LibsyError),
88    #[error(transparent)]
89    Client(#[from] LlmClientError),
90}
91
92impl RunnerError {
93    pub(crate) fn configuration(message: impl Into<String>) -> Self {
94        Self::Configuration {
95            message: message.into(),
96            source: None,
97        }
98    }
99
100    pub(crate) fn configuration_source(
101        message: impl Into<String>,
102        source: impl Error + Send + Sync + 'static,
103    ) -> Self {
104        Self::Configuration {
105            message: message.into(),
106            source: Some(Box::new(source)),
107        }
108    }
109}

111// Configured algorithm and targets for one execution lane.
112pub(crate) struct ExecutionLane {
113    algorithm: Arc<dyn Algorithm>,
114    // Resolves each offloaded call to the client configured for the target the algorithm
115    // selected. A route is a synthetic model with no upstream of its own, so this is a
116    // per-target lookup, never one client serving the whole route.
117    clients: ClientRouter,
118    anthropic_auxiliary_target: Option<AuxiliaryTarget>,
119    responses_auxiliary_target: Option<AuxiliaryTarget>,
120    models: Arc<RuntimeModels>,
121}

215    fn select_lane(&self, request: &Request) -> Result<LaneSelection<'_>, RunnerError> {
216        if let Some(local) = &self.local {
217            let local_only = local.policy.admit(request)?;
218            let privacy = Some(PrivacyDecision::from_local_requirement(
219                request, local_only,
220            )?);
221            let lane = if local_only {
222                &local.lane
223            } else {
224                &self.standard
225            };
226            tracing::debug!(
227                switchyard.route = request.llm_request.model.as_deref().unwrap_or_default(),
228                switchyard.algorithm = self.algorithm_name(),
229                local_only,
230                privacy.lane = privacy.as_ref().map(PrivacyDecision::lane),
231                privacy.source = privacy.as_ref().map(PrivacyDecision::source),
232                privacy.reason_code = privacy.as_ref().map(PrivacyDecision::reason_code),
233                "execution lane selected"
234            );
235            return Ok(LaneSelection { lane, privacy });
236        }
237        if has_local_requirement(request) {
238            return Err(LlmClientError::InvalidRequest {
239                message: "route does not accept a local-only signal".to_string(),
240            }
241            .into());
242        }
243        Ok(LaneSelection {
244            lane: &self.standard,
245            privacy: None,
246        })
247    }

287    /// Executes the configured route without consuming or proxying streamed responses.
288    pub async fn execute(
289        &self,
290        request: Request,
291        observer: Option<RunObserver>,
292    ) -> Result<RunOutput, RunnerError> {
293        self.execute_with_privacy_observer(request, observer, |_| {})
294            .await
295    }
296
297    /// Executes the route and reports standard or local-only admission before dispatch.
298    pub async fn execute_with_privacy_observer(
299        &self,
300        request: Request,
301        observer: Option<RunObserver>,
302        on_privacy_decision: impl FnOnce(&PrivacyDecision) + Send,
303    ) -> Result<RunOutput, RunnerError> {
304        let selection = self.select_lane(&request)?;
305        if let Some(decision) = &selection.privacy {
306            on_privacy_decision(decision);
307        }
308        let lane = selection.lane;
309        let (selected_model, response) = switchyard_llm_client::run(
310            Arc::clone(&lane.algorithm),
311            lane.clients.clone(),
312            request,
313            Arc::clone(&lane.models),
314            observer,
315        )
316        .await?;
317        Ok(RunOutput {
318            selected_model,
319            response,
320        })
321    }

323    /// Completes routing-time calls without serving a post-routing completion.
324    pub async fn decide(&self, request: Request) -> Result<RoutingOutcome, RunnerError> {
325        let selection = self.select_lane(&request)?;
326        let lane = selection.lane;
327        switchyard_llm_client::decide(
328            Arc::clone(&lane.algorithm),
329            lane.clients.clone(),
330            request,
331            Arc::clone(&lane.models),
332        )
333        .await
334        .map_err(RunnerError::from)
335    }

342    /// Apply the optional queue policy to a completed task decision.
343    ///
344    /// The host must hold its in-flight accounting lock through this call and the
345    /// increment for the selected model. Ordinary `decide` leaves task routing unchanged.
346    pub fn apply_infra(
347        &self,
348        outcome: &mut RoutingOutcome,
349        signals: &switchyard_protocol::InfraSignals,
350    ) -> Result<bool, RunnerError> {
351        match &self.infra {
352            Some(policy) => {
353                let selection = self.select_lane(&outcome.request)?;
354                policy
355                    .apply(outcome, &selection.lane.models, signals)
356                    .map_err(Into::into)
357            }
358            None => Ok(false),
359        }
360    }

362    /// Routes with a host-owned client for routing-time LLM calls, such as judges.
363    ///
364    /// The host owns transport, credentials, and target request settings. Configured
365    /// HTTP clients, target prompts, and decision clients are not used on this path.
366    /// The same algorithm instance retains its state across calls. No post-routing
367    /// completion is dispatched. Target eligibility still applies to the host-owned path.
368    pub async fn decide_with_client(
369        &self,
370        request: Request,
371        client: Arc<dyn RoutedLlmClient>,
372    ) -> Result<RoutingOutcome, RunnerError> {
373        let selection = self.select_lane(&request)?;
374        let lane = selection.lane;
375        switchyard_llm_client::decide(
376            Arc::clone(&lane.algorithm),
377            ClientRouter::single(client),
378            request,
379            Arc::clone(&lane.models),
380        )
381        .await
382        .map_err(RunnerError::from)
383    }
384
385    /// Executes a model-bearing provider operation through a compatible target.
386    pub async fn call_auxiliary(
387        &self,
388        request: Request,
389        operation: AuxiliaryOperation,
390    ) -> Result<Value, RunnerError> {
391        let selection = self.select_lane(&request)?;
392        let lane = selection.lane;
393        let target = match operation {
394            AuxiliaryOperation::AnthropicCountTokens => &lane.anthropic_auxiliary_target,
395            AuxiliaryOperation::ResponsesInputTokens | AuxiliaryOperation::ResponsesCompact => {
396                &lane.responses_auxiliary_target
397            }
398        }
399        .as_ref()
400        .ok_or(RunnerError::AuxiliaryUnsupported)?;
401        target
402            .client
403            .call_auxiliary(&target.model, request, operation)
404            .await
405            .map_err(Into::into)
406    }
