4//! Trusted local-only admission and bounded task retention.

6use std::collections::HashSet;
7
8use parking_lot::Mutex;
9use serde_json::Value;
10use strum_macros::IntoStaticStr;
11use switchyard_protocol::{LlmClientError, Request, WireFormat};
12
13const LOCAL_ONLY_KEY: &str = "switchyard.internal.local_only";
14const MAX_TASK_ID_BYTES: usize = 512;
15const MAX_TASKS: usize = 4_096;
16const RESPONSES_STATE_FIELDS: [&str; 2] = ["previous_response_id", "conversation"];
17
18/// Admits a local-only flag and optionally retains it for the task.
19pub(crate) struct LocalPolicy {
20    require_explicit_signal: bool,
21    tasks: Option<Mutex<HashSet<String>>>,
22}
23
24impl LocalPolicy {
25    pub(crate) fn new(require_explicit_signal: bool, task_scope: bool) -> Self {
26        Self {
27            require_explicit_signal,
28            tasks: task_scope.then(Mutex::default),
29        }
30    }
31
32    // Retain a restriction even if later context validation rejects this turn.
33    pub(crate) fn admit(&self, request: &Request) -> Result<bool, LlmClientError> {
34        let signal = local_requirement(request)?;
35        if self.require_explicit_signal && signal.is_none() {
36            return Err(invalid("route requires an explicit local-only signal"));
37        }
38        let mut local_only = signal.unwrap_or(false);
39        if let Some(tasks) = &self.tasks {
40            let task_id = request
41                .metadata
42                .as_ref()
43                .and_then(|metadata| metadata.task_id.as_deref())
44                .filter(|id| !id.is_empty() && id.len() <= MAX_TASK_ID_BYTES && id.trim() == *id)
45                .ok_or_else(|| invalid("local-only routing needs a valid task id"))?;
46            let mut tasks = tasks.lock();
47            let retained = tasks.contains(task_id);
48            if !retained && tasks.len() >= MAX_TASKS {
49                return Err(invalid("local-only task capacity exhausted"));
50            }
51            if local_only && !retained {
52                tasks.insert(task_id.to_owned());
53            }
54            local_only |= retained;
55        }
56        validate_request_context(request)?;
57        Ok(local_only)
58    }
59}

61/// Attaches a trusted local-only assessment to an in-process request.
62/// False records an explicit signal but cannot remove an earlier true value.
63/// The host owns classification, task authorization and target approvals.
64pub fn require_local_targets(
65    request: &mut Request,
66    local_only: bool,
67) -> Result<(), LlmClientError> {
68    let local_only = local_requirement(request)?.unwrap_or(false) || local_only;
69    request
70        .metadata
71        .get_or_insert_default()
72        .extra_metadata
73        .get_or_insert_default()
74        .insert(LOCAL_ONLY_KEY.to_string(), local_only.to_string());
75    Ok(())
76}
77
78pub(crate) fn has_local_requirement(request: &Request) -> bool {
79    request
80        .metadata
81        .as_ref()
82        .and_then(|metadata| metadata.extra_metadata.as_ref())
83        .is_some_and(|metadata| metadata.contains_key(LOCAL_ONLY_KEY))
84}
85
86fn local_requirement(request: &Request) -> Result<Option<bool>, LlmClientError> {
87    match request
88        .metadata
89        .as_ref()
90        .and_then(|metadata| metadata.extra_metadata.as_ref())
91        .and_then(|metadata| metadata.get(LOCAL_ONLY_KEY))
92        .map(String::as_str)
93    {
94        None => Ok(None),
95        Some("true") => Ok(Some(true)),
96        Some("false") => Ok(Some(false)),
97        Some(_) => Err(invalid("invalid local-only signal")),
98    }
99}

101#[derive(Clone, Copy, IntoStaticStr)]
102#[strum(serialize_all = "snake_case")]
103enum PrivacyLane {
104    Standard,
105    Restricted,
106}
107
108#[derive(Clone, Copy, IntoStaticStr)]
109#[strum(serialize_all = "snake_case")]
110enum PrivacySource {
111    Policy,
112    ExternalSignal,
113    TaskRetention,
114}
115
116/// The admitted standard or local-only lane and its trusted source.
117pub struct PrivacyDecision {
118    lane: PrivacyLane,
119    source: PrivacySource,
120}
121
122impl PrivacyDecision {
123    /// Returns `restricted` for local-only handling, otherwise `standard`.
124    pub fn lane(&self) -> &'static str {
125        self.lane.into()
126    }
127
128    /// Returns the policy component that selected the lane.
129    pub fn source(&self) -> &'static str {
130        self.source.into()
131    }
132
133    /// Returns a bounded reason for the decision.
134    pub const fn reason_code(&self) -> &'static str {
135        match self.source {
136            PrivacySource::Policy => "all_clear",
137            PrivacySource::ExternalSignal => "restricted",
138            PrivacySource::TaskRetention => "retained_restriction",
139        }
140    }
141
142    pub(crate) fn from_local_requirement(
143        request: &Request,
144        local_only: bool,
145    ) -> Result<Self, LlmClientError> {
146        Ok(Self {
147            lane: if local_only {
148                PrivacyLane::Restricted
149            } else {
150                PrivacyLane::Standard
151            },
152            source: if !local_only {
153                PrivacySource::Policy
154            } else if local_requirement(request)? == Some(true) {
155                PrivacySource::ExternalSignal
156            } else {
157                PrivacySource::TaskRetention
158            },
159        })
160    }
161}
162
163fn invalid(message: &'static str) -> LlmClientError {
164    LlmClientError::InvalidRequest {
165        message: message.to_string(),
166    }
167}

169// Provider-held history may have been admitted under a different restriction.
170fn validate_request_context(request: &Request) -> Result<(), LlmClientError> {
171    let has_state = |body: &Value| {
172        RESPONSES_STATE_FIELDS
173            .iter()
174            .any(|field| body.get(*field).is_some_and(|value| !value.is_null()))
175    };
176    if RESPONSES_STATE_FIELDS.iter().any(|field| {
177        request
178            .llm_request
179            .extensions
180            .fields
181            .get(*field)
182            .is_some_and(|value| !value.is_null())
183    }) || request
184        .llm_request
185        .preservation
186        .requests
187        .iter()
188        .any(|(format, body)| {
189            format.as_str() == WireFormat::OpenAiResponses.as_str() && has_state(body)
190        })
191    {
192        return Err(invalid(
193            "local-only routes do not support OpenAI Responses continuation state",
194        ));
195    }
196    Ok(())
197}

199#[cfg(test)]
200mod tests {
201    use super::*;
202
203    fn task(id: &str, local_only: bool) -> Request {
204        let mut request = Request::default();
205        request.metadata.get_or_insert_default().task_id = Some(id.into());
206        require_local_targets(&mut request, local_only).expect("trusted signal");
207        request
208    }
209
210    #[test]
211    fn explicit_signals_cannot_relax_a_request_or_task() {
212        let policy = LocalPolicy::new(true, true);
213        assert!(policy.admit(&Request::default()).is_err());
214        assert!(!policy.admit(&task("one", false)).expect("explicit false"));
215        let mut request = task("one", true);
216        require_local_targets(&mut request, false).expect("cannot clear true");
217        assert!(policy.admit(&request).expect("local-only"));
218        assert!(policy.admit(&task("one", false)).expect("retained"));
219        assert!(!policy.admit(&task("two", false)).expect("other task"));
220        let retained =
221            PrivacyDecision::from_local_requirement(&task("one", false), true).expect("decision");
222        assert_eq!(
223            (retained.lane(), retained.source()),
224            ("restricted", "task_retention")
225        );
226    }
227
228    #[test]
229    fn malformed_signal_and_full_task_store_fail_closed() {
230        let policy = LocalPolicy::new(false, true);
231        let mut request = task("one", true);
232        request
233            .metadata
234            .as_mut()
235            .unwrap()
236            .extra_metadata
237            .as_mut()
238            .unwrap()
239            .insert(LOCAL_ONLY_KEY.into(), "unknown".into());
240        assert!(require_local_targets(&mut request, false).is_err());
241        assert!(policy.admit(&request).is_err());
242        for id in 0..MAX_TASKS {
243            policy
244                .admit(&task(&id.to_string(), true))
245                .expect("capacity available");
246        }
247        assert!(policy.admit(&task("new", false)).is_err());
248        assert!(policy.admit(&task("0", false)).expect("existing task"));
249    }
250
251    #[test]
252    fn provider_held_history_is_rejected_without_erasing_task_restriction() {
253        let policy = LocalPolicy::new(false, true);
254        for field in RESPONSES_STATE_FIELDS {
255            let mut request = task(field, true);
256            request
257                .llm_request
258                .extensions
259                .fields
260                .insert(field.into(), Value::String("hidden".into()));
261            assert!(policy.admit(&request).is_err());
262            assert!(
263                policy
264                    .admit(&task(field, false))
265                    .expect("retained after rejection")
266            );
267            let mut preserved = task("preserved", false);
268            preserved.llm_request.preservation.requests.insert(
269                "openai_responses".into(),
270                serde_json::json!({field: "hidden"}),
271            );
272            assert!(policy.admit(&preserved).is_err());
273        }
274    }
