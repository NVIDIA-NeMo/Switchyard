21/// Default instruction added while the capable model is planning.
22pub const DEFAULT_PLANNING_PROMPT: &str =
23    include_str!("../prompts/plan-execute/planning-system-prompt.md");

25const MAX_EXECUTING_SESSIONS: usize = 4_096;

27/// Configuration for [`PlanExecute`].
28#[derive(Clone, Debug)]
29pub struct PlanExecuteConfig {
30    /// LLM or decision judge used by the judge trigger.
31    pub judge: Option<TaskClassifierConfig>,
32    /// Additional tool names whose mutations trigger handoff.
33    pub tool_semantics: ToolSemantics,
34    /// System instruction added until handoff.
35    pub planning_prompt: String,
36    /// Optional instruction appended to the handoff request.
37    pub handoff_prompt: Option<String>,
38    /// Replays visible planner reasoning summaries as assistant text at handoff.
39    pub planner_reasoning_as_text: bool,
40}
41
42impl Default for PlanExecuteConfig {
43    fn default() -> Self {
44        Self {
45            judge: None,
46            tool_semantics: ToolSemantics::default(),
47            planning_prompt: DEFAULT_PLANNING_PROMPT.trim().to_string(),
48            handoff_prompt: None,
49            planner_reasoning_as_text: false,
50        }
51    }
52}

54#[derive(Clone, Copy, Debug, Eq, PartialEq)]
55enum Phase {
56    Plan,
57    Handoff,
58    Execute,
59}

61/// Routes planning turns to the runtime capable model, then latches execution
62/// to the runtime efficient model after an edit or judge approval.
63pub struct PlanExecute {
64    config: PlanExecuteConfig,
65    judge: Option<LlmTaskClassifier>,
66    executing_sessions: Mutex<HashSet<RoutingIdentity>>,
67}
68
69impl PlanExecute {
70    /// Creates a plan/execute router.
71    ///
72    /// Returns an error when a prompt is blank or tool semantics are invalid.
73    pub fn new(mut config: PlanExecuteConfig) -> Result<Self> {
74        if config.planning_prompt.trim().is_empty() {
75            return Err(LibsyError::AlgorithmError {
76                message: "planning_prompt must not be empty".to_string(),
77            });
78        }
79        if config
80            .handoff_prompt
81            .as_deref()
82            .is_some_and(|prompt| prompt.trim().is_empty())
83        {
84            return Err(LibsyError::AlgorithmError {
85                message: "handoff_prompt must not be empty".to_string(),
86            });
87        }
88        config.tool_semantics.validate()?;
89        let judge = if let Some(mut judge) = config.judge.take() {
90            judge.recent_turn_window.get_or_insert(10);
91            let prompt = include_str!("../prompts/plan-execute/handoff-decision.md");
92            match &mut judge.judge {
93                CapabilityJudgeConfig::Llm(judge) if judge.contract.prompt().is_none() => {
94                    judge.contract = judge.contract.clone().with_prompt(format!(
95                        "{}\n{prompt}",
96                        include_str!("../prompts/capability-classifier/prompt.md")
97                    ));
98                }
99                CapabilityJudgeConfig::Decision(judge) if judge.instructions.is_none() => {
100                    judge.instructions = Some(serde_json::json!(prompt));
101                }
102                _ => {}
103            }
104            Some(LlmTaskClassifier::new(LlmClassifierConfig::Capability {
105                config: judge,
106            })?)
107        } else {
108            None
109        };
110        Ok(Self {
111            config,
112            judge,
113            executing_sessions: Mutex::new(HashSet::new()),
114        })
115    }
116
117    fn phase(&self, request: &Request, handoff: bool) -> Result<Phase> {
118        let Some(identity) = RoutingIdentity::from_request(request) else {
119            return Ok(if handoff { Phase::Handoff } else { Phase::Plan });
120        };
121        let is_session_final = request
122            .metadata
123            .as_ref()
124            .and_then(|metadata| metadata.session_final)
125            == Some(true);
126
127        let mut sessions = self.executing_sessions.lock();
128        if self.judge.is_some() && starts_user_turn(request) {
129            sessions.remove(&identity);
130        }
131        let phase = if sessions.contains(&identity) {
132            Phase::Execute
133        } else if handoff {
134            if !is_session_final {
135                if sessions.len() >= MAX_EXECUTING_SESSIONS {
136                    return Err(LibsyError::AlgorithmError {
137                        message: format!(
138                            "plan_execute reached its limit of {MAX_EXECUTING_SESSIONS} executing sessions. \
139                             Finish an existing session with session_final before retrying the handoff"
140                        ),
141                    });
142                }
143                sessions.insert(identity.clone());
144            }
145            Phase::Handoff
146        } else {
147            Phase::Plan
148        };
149        if is_session_final {
150            sessions.remove(&identity);
151        }
152        Ok(phase)
153    }
154
155    fn replay_planner_reasoning_as_text(request: &mut Request) -> usize {
156        let mut converted = 0;
157        for message in &mut request.llm_request.messages {
158            message.content = std::mem::take(&mut message.content)
159                .into_iter()
160                .filter_map(|block| match block {
161                    ContentBlock::Reasoning { text, .. } => {
162                        converted += 1;
163                        (!text.is_empty()).then_some(ContentBlock::Text { text })
164                    }
165                    other => Some(other),
166                })
167                .collect();
168        }
169        request
170            .llm_request
171            .messages
172            .retain(|message| !message.content.is_empty());
173        if converted > 0 {
174            drop_exact_replay(request);
175        }
176        converted
177    }
178
179    fn route_to(driver: &Driver, category: Category, request: Request) -> Result<RoutingOutcome> {
180        let models = driver.models_for(&category);
181        let Some((selected, fallbacks)) = models.split_first() else {
182            return Err(LibsyError::AlgorithmError {
183                message: format!("no models available for category {}", category.as_str()),
184            });
185        };
186        Ok(RoutingOutcome::route_to(
187            selected.clone(),
188            fallbacks.to_vec(),
189            request,
190        ))
191    }
192}

194#[async_trait::async_trait]
195impl Algorithm for PlanExecute {
196    fn name(&self) -> &str {
197        "plan_execute"
198    }
199
200    async fn route(
201        self: Arc<Self>,
202        driver: Driver,
203        mut request: Request,
204    ) -> Result<RoutingOutcome> {
205        let mut phase = self.phase(&request, false)?;
206        if phase == Phase::Plan {
207            let handoff = if let Some(judge) = &self.judge {
208                if starts_user_turn(&request) {
209                    false
210                } else {
211                    judge
212                        .score(&mut State::default(), &mut request, &driver)
213                        .await?
214                        .0
215                        .argmax(false)?
216                        .is_some_and(|score| score.category == Some(Category::Efficient))
217                }
218            } else {
219                let signals = ToolSignals::from_request_with_semantics(
220                    &request,
221                    None,
222                    &self.config.tool_semantics,
223                );
224                signals.edit_count > 0 || signals.write_count > 0
225            };
226            if handoff {
227                phase = self.phase(&request, true)?;
228            }
229        }
230        match phase {
231            Phase::Plan => {
232                prepend_system_prompt(&mut request, &self.config.planning_prompt);
233                tracing::debug!(phase = "plan", "plan-execute selected capable tier");
234                Self::route_to(&driver, Category::Capable, request)
235            }
236            Phase::Handoff => {
237                let reasoning_converted = if self.config.planner_reasoning_as_text {
238                    Self::replay_planner_reasoning_as_text(&mut request)
239                } else {
240                    0
241                };
242                let prompt_applied = if let Some(prompt) = &self.config.handoff_prompt {
243                    append_note(&mut request, prompt);
244                    true
245                } else {
246                    false
247                };
248                tracing::debug!(
249                    phase = "handoff",
250                    handoff_prompt_applied = prompt_applied,
251                    planner_reasoning_converted = reasoning_converted,
252                    "plan-execute selected efficient tier"
253                );
254                Self::route_to(&driver, Category::Efficient, request)
255            }
256            Phase::Execute => {
257                tracing::debug!(phase = "execute", "plan-execute selected efficient tier");
258                Self::route_to(&driver, Category::Efficient, request)
259            }
260        }
261    }
262}

468    #[tokio::test]
469    async fn capacity_preserves_execution_after_compaction() {
470        let algorithm = algorithm(PlanExecuteConfig::default());
471        let sessions: Vec<_> = (0..MAX_EXECUTING_SESSIONS)
472            .map(|index| format!("task-{index}"))
473            .collect();
474        for session in &sessions {
475            let first_edit = request(
476                vec![tool_call("Write", json!({"file_path": "task.py"}))],
477                Some(session),
478            );
479            let (selected, _) = route_and_capture(Arc::clone(&algorithm), first_edit).await;
480            assert_eq!(selected, EFFICIENT);
481        }
482
483        let overflow = request(
484            vec![tool_call("Write", json!({"file_path": "task.py"}))],
485            Some("overflow"),
486        );
487        let (driver, _) = Driver::new("plan_execute", Arc::new(models()));
488        let result = Arc::clone(&algorithm).route(driver, overflow.clone()).await;
489        assert!(matches!(
490            result,
491            Err(LibsyError::AlgorithmError { message })
492                if message.contains("limit of 4096 executing sessions")
493        ));
494
495        for session in &sessions {
496            let mut compacted = request(
497                vec![Message::text(Role::User, "Continue after compaction")],
498                Some(session),
499            );
500            if Some(session) == sessions.last() {
501                compacted.metadata.as_mut().unwrap().session_final = Some(true);
502            }
503            let (selected, routed) = route_and_capture(Arc::clone(&algorithm), compacted).await;
504            assert_eq!(selected, EFFICIENT, "session {session} lost its latch");
505            assert!(routed.llm_request.instructions.is_empty());
506        }
507
508        let (selected, _) = route_and_capture(Arc::clone(&algorithm), overflow).await;
509        assert_eq!(selected, EFFICIENT);
510        let compacted = request(
511            vec![Message::text(Role::User, "Continue after compaction")],
512            Some("overflow"),
513        );
514        let (selected, routed) = route_and_capture(algorithm, compacted).await;
515        assert_eq!(selected, EFFICIENT);
516        assert!(routed.llm_request.instructions.is_empty());
517    }

519    #[tokio::test]
520    async fn final_handoff_does_not_evict_an_executing_session_at_capacity() {
521        let algorithm = Arc::new(PlanExecute::new(PlanExecuteConfig::default()).unwrap());
522        algorithm.executing_sessions.lock().extend(
523            (0..MAX_EXECUTING_SESSIONS)
524                .map(|index| RoutingIdentity::Session(format!("task-{index}"))),
525        );
526        let mut final_request = request(
527            vec![tool_call("Write", json!({"file_path": "task.py"}))],
528            Some("final-handoff"),
529        );
530        final_request.metadata.as_mut().unwrap().session_final = Some(true);
531
532        let (selected, routed) = route_and_capture(algorithm.clone(), final_request).await;
533        assert_eq!(selected, EFFICIENT);
534        assert!(routed.llm_request.instructions.is_empty());
535        let sessions = algorithm.executing_sessions.lock();
536        assert_eq!(sessions.len(), MAX_EXECUTING_SESSIONS);
537        assert!(!sessions.contains(&RoutingIdentity::Session("final-handoff".to_string())));
538    }
