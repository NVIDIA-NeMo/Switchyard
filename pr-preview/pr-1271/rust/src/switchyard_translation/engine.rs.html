31/// A stream of wire-format event objects in one format — the unframed body of an
32/// SSE response. The serving layer frames each `Value` (e.g. as an SSE
33/// `data:`/`event:` block).
34pub type RawEventStream =
35    Pin<Box<dyn Stream<Item = std::result::Result<Value, LlmStreamError>> + Send>>;

58/// Registry mapping wire formats to buffered codecs.
59#[derive(Default)]
60pub struct FormatRegistry {
61    codecs: BTreeMap<FormatId, Arc<dyn FormatCodec>>,
62}
63
64impl FormatRegistry {
65    /// Creates an empty format registry.
66    pub fn new() -> Self {
67        Self::default()
68    }
69
70    /// Creates a registry populated with the built-in provider codecs.
71    pub fn with_builtins() -> Self {
72        let mut registry = Self::new();
73        registry.register(OpenAiChatCodec);
74        registry.register(AnthropicMessagesCodec);
75        registry.register(OpenAiResponsesCodec);
76        registry
77    }
78
79    /// Registers or replaces a codec for its declared format.
80    pub fn register(&mut self, codec: impl FormatCodec + 'static) {
81        self.codecs.insert(codec.format(), Arc::new(codec));
82    }
83
84    /// Looks up a codec by format identifier.
85    pub fn codec(&self, format: impl Into<FormatId>) -> Result<Arc<dyn FormatCodec>> {
86        let format = format.into();
87        self.codecs
88            .get(&format)
89            .cloned()
90            .ok_or_else(|| TranslationError::Other(format!("no codec registered for {format}")))
91    }
92}

126    /// Decodes a request body into the neutral request IR.
127    pub fn decode_request(
128        &self,
129        source: impl Into<FormatId>,
130        body: &Value,
131        policy: &TranslationPolicy,
132    ) -> Result<RequestIrOutput> {
133        let source = source.into();
134        let mut decoded = self.registry.codec(source)?.decode_request(body, policy)?;
135        crate::normalize_compaction(&mut decoded.request, None);
136        Ok(RequestIrOutput {
137            request: decoded.request,
138            diagnostics: decoded.diagnostics,
139        })
140    }

142    /// Encodes a neutral request IR into a target wire format.
143    ///
144    /// Preserved target bodies take precedence over normalized fields, including
145    /// `output.is_schema_enforced`. Clear the target's preservation entry or use
146    /// [`crate::policy::PreservationPolicy::Disabled`] to encode IR edits.
147    pub fn encode_request(
148        &self,
149        target: impl Into<FormatId>,
150        request: &LlmRequest,
151        policy: &TranslationPolicy,
152    ) -> Result<TranslationOutput> {
153        let target = target.into();
154        let encoded = self
155            .registry
156            .codec(target)?
157            .encode_request(request, policy)?;
158        Ok(TranslationOutput {
159            body: encoded.body,
160            diagnostics: encoded.diagnostics,
161        })
162    }

164    /// Translates a request body from source format to target format.
165    pub fn translate_request(
166        &self,
167        source: impl Into<FormatId>,
168        target: impl Into<FormatId>,
169        body: &Value,
170        policy: &TranslationPolicy,
171    ) -> Result<TranslationOutput> {
172        let source = source.into();
173        let target = target.into();
174        let decoded = self
175            .registry
176            .codec(source.clone())?
177            .decode_request(body, policy)?;
178        let encoded = self
179            .registry
180            .codec(target.clone())?
181            .encode_request(&decoded.request, policy)?;
182        Ok(TranslationOutput {
183            body: encoded.body,
184            diagnostics: with_formats(decoded.diagnostics, encoded.diagnostics, source, target),
185        })
186    }

221    /// Encodes a neutral response IR while applying metadata retained from its request.
222    ///
223    /// Request extensions are used to restore caller-specific response details, such as
224    /// Codex tool namespaces, without exposing those details to the upstream provider.
225    pub fn encode_response_with_extensions(
226        &self,
227        target: impl Into<FormatId>,
228        response: &AggLlmResponse,
229        request_extensions: &ProviderExtensions,
230        policy: &TranslationPolicy,
231    ) -> Result<TranslationOutput> {
232        let mut output = self.encode_response(target, response, policy)?;
233        if !request_extensions.fields.is_empty() {
234            crate::codex_namespaces::restore_qualified_tool_names(
235                &mut output.body,
236                &crate::codex_namespaces::qualified_tool_origins(request_extensions),
237            );
238            crate::codex_custom_tools::restore_custom_tool_calls(
239                &mut output.body,
240                &crate::codex_custom_tools::custom_tool_names(request_extensions),
241            );
242        }
243        Ok(output)
244    }

263    /// Translates a response while restoring details from the original request's extensions.
264    ///
265    /// Request extensions apply after encoding, including when the target body is preserved.
266    pub fn translate_response_with_extensions(
267        &self,
268        source: impl Into<FormatId>,
269        target: impl Into<FormatId>,
270        body: &Value,
271        request_extensions: &ProviderExtensions,
272        policy: &TranslationPolicy,
273    ) -> Result<TranslationOutput> {

276        let decoded = self

278            .codec(source.clone())?
279            .decode_response(body, policy)?;

282            &decoded.response,
283            request_extensions,

286        Ok(TranslationOutput {
287            body: encoded.body,
288            diagnostics: with_formats(decoded.diagnostics, encoded.diagnostics, source, target),

292    /// Translates one streaming source event into zero or more target events.
293    pub fn translate_event(
294        &self,
295        state: &mut StreamTranslationState,
296        source: impl Into<FormatId>,
297        target: impl Into<FormatId>,
298        event: &Value,
299    ) -> Result<Vec<Value>> {
300        let source = source.into();
301        let target = target.into();
302        let source_codec = self.stream_registry.codec(source.clone())?;
303        let target_codec = self.stream_registry.codec(target.clone())?;
304        if let Err(error) =
305            crate::codecs::responses::validate_stream_output(&source, &target, event)
306        {
307            return Ok(target_codec.encode_event(
308                state,
309                crate::LlmResponseChunk::DecodeError {
310                    message: error.to_string(),
311                },
312            ));
313        }
314        let canonical = source_codec.decode_event(state, event);
315        state.source = Some(source);
316        state.target = Some(target);
317        Ok(canonical
318            .into_iter()
319            .flat_map(|event| target_codec.encode_event(state, event))
320            .collect())
321    }

323    /// Decodes one provider event while retaining its parsed source JSON value.
324    ///
325    /// Takes ownership of `event` so preservation does not deep-copy provider JSON
326    /// on the per-event streaming path.
327    pub fn decode_stream_event(
328        &self,
329        state: &mut StreamTranslationState,
330        source: impl Into<FormatId>,
331        event: Value,
332    ) -> Result<LlmResponseStreamEvent> {
333        let source = source.into();
334        let source_codec = self.stream_registry.codec(source.clone())?;
335        state.source = Some(source.clone());
336        let normalized = source_codec.decode_event(state, &event);
337        Ok(LlmResponseStreamEvent::preserved(source, event, normalized))
338    }

373    /// Encodes a stream of IR chunks into a stream of target-format wire events.
374    ///
375    /// `served_model` is exposed as the response model (via the stream state's
376    /// `target_model`); `None` falls back to the id observed on the source stream.
377    /// The target stream codec is resolved once and reused per chunk; terminal events
378    /// (`message_stop` / `response.completed`) come from `finish`.
379    pub fn encode_stream(
380        &self,
381        target: impl Into<FormatId>,
382        chunks: LlmResponseStream,
383        served_model: Option<String>,
384    ) -> std::result::Result<RawEventStream, LlmClientError> {
385        self.encode_stream_with_extensions(
386            target,
387            chunks,
388            served_model,
389            &ProviderExtensions::default(),
390        )
391    }

393    /// Encodes a response stream, honouring request extensions.
394    ///
395    /// Identical to [`Self::encode_stream`] except that Codex tool namespaces recorded on
396    /// the request are restored on each encoded event.
397    pub fn encode_stream_with_extensions(
398        &self,
399        target: impl Into<FormatId>,
400        chunks: LlmResponseStream,
401        served_model: Option<String>,
402        request_extensions: &ProviderExtensions,
403    ) -> std::result::Result<RawEventStream, LlmClientError> {
404        let origins = crate::codex_namespaces::qualified_tool_origins(request_extensions);
405        let custom_tools = crate::codex_custom_tools::custom_tool_names(request_extensions);
406        let mut custom_state = crate::codex_custom_tools::CustomToolCallStreamState::default();
407        let target_format: FormatId = target.into();
408        let codec = self
409            .stream_registry
410            .codec(target_format.clone())
411            // Currently the only error is that the codec is missing, which is Configuration
412            .map_err(|err| LlmClientError::Configuration {
413                message: err.to_string(),
414            })?;
415
416        let served_model_for_events = served_model.clone();
417        let mut state = StreamTranslationState {
418            target: Some(target_format.clone()),
419            target_model: served_model,
420            ..Default::default()
421        };
422        let mut chunks = chunks;
423
424        let events = try_stream! {
425            while let Some(item) = chunks.next().await {
426                let event = item.map_err(LlmStreamError::Client)?;
427                let mut encoded =
428                    encode_response_stream_event(&mut state, codec.as_ref(), &target_format, event);
429                for value in &mut encoded {
430                    if let Some(model) = served_model_for_events.as_deref() {
431                        codec.stamp_streamed_response_model(value, model);
432                    }
433                    crate::codex_namespaces::restore_qualified_tool_names(value, &origins);
434                }
435                // Argument deltas for a freeform tool cannot be expressed on the wire; the
436                // rewritten completed item carries the input instead.
437                let mut encoded: Vec<Value> = encoded
438                    .into_iter()
439                    .filter_map(|mut value| {
440                        crate::codex_custom_tools::restore_custom_tool_calls_in_event(
441                            &mut value,
442                            &custom_tools,
443                            &mut custom_state,
444                        )
445                        .then_some(value)
446                    })
447                    .collect();
448                let terminal = encoded.pop();
449                for value in encoded {
450                    yield value;
451                }
452                match (state.errored, terminal) {
453                    (false, Some(value)) => yield value,
454                    (true, Some(value)) => Err(LlmStreamError::Upstream(value))?,
455                    (true, None) => Err(LlmStreamError::Client(LlmClientError::ResponseTranslation(
456                        "stream ended after an error without a terminal event".to_string(),
457                    )))?,
458                    _ => ()
459                }
460            }
461            for mut value in codec.finish(&mut state) {
462                if let Some(model) = served_model_for_events.as_deref() {
463                    codec.stamp_streamed_response_model(&mut value, model);
464                }
465                crate::codex_namespaces::restore_qualified_tool_names(&mut value, &origins);
466                if crate::codex_custom_tools::restore_custom_tool_calls_in_event(
467                    &mut value,
468                    &custom_tools,
469                    &mut custom_state,
470                ) {
471                    yield value;
472                }
473            }
474        };
475
476        Ok(Box::pin(events))
477    }

479    /// Decodes provider SSE bytes into normalized stream events.
480    ///
481    /// Operates on raw bytes, not any HTTP client type: the caller adapts its
482    /// transport's body stream into `Stream<Item = Result<Vec<u8>, _>>`. Frames are
483    /// buffered across chunks (a partial frame waits for its boundary); the source
484    /// stream codec is resolved once and reused for every frame.
485    ///
486    /// The decoder yields the terminal JSON event before it stops reading. Chat Completions
487    /// keeps reading after `finish_reason` to include usage before `[DONE]` or EOF.
488    /// If EOF arrives without a required terminal event, the stream yields a deferred
489    /// [`LlmClientError::ResponseTranslation`] error after any valid decoded events.
490    pub fn decode_stream<S>(
491        &self,
492        source: impl Into<FormatId>,
493        bytes: S,
494    ) -> std::result::Result<LlmResponseStream, LlmClientError>
495    where
496        S: Stream<Item = std::result::Result<Vec<u8>, LlmClientError>> + Send + 'static,
497    {
498        let marker = sse::done_marker();
499        let source_format: FormatId = source.into();
500        let codec = self
501            .stream_registry
502            .codec(source_format.clone())
503            .map_err(|error| LlmClientError::ResponseTranslation(error.to_string()))?;
504        // Adapt the byte-chunk stream into an async line reader. The BufReader
505        // reassembles data split across network chunks (including multi-byte UTF-8),
506        // and `lines()` yields one SSE field line at a time. The stream is boxed to
507        // an `io::Error` item so `into_async_read`'s error bound resolves cleanly. The
508        // source error is boxed intact rather than stringified, so
509        // `llm_client_error_from_io` can recover its original variant on the way out.
510        let io_bytes: Pin<Box<dyn Stream<Item = std::io::Result<Vec<u8>>> + Send>> =
511            Box::pin(bytes.map(|item| item.map_err(std::io::Error::other)));
512        let lines = futures::io::BufReader::new(io_bytes.into_async_read()).lines();
513
514        let mut state = StreamTranslationState {
515            source: Some(source_format.clone()),
516            ..StreamTranslationState::default()
517        };
518        let mut frame = String::new();
519        let mut saw_terminal = false;
520        let mut saw_error = false;
521        let stream = Box::pin(try_stream! {
522            futures::pin_mut!(lines);
523            while let Some(line) = lines.next().await {
524                let line = line.map_err(llm_client_error_from_io)?;
525                // A blank line (allowing a bare CR for CRLF streams) ends the frame.
526                if line.trim_end().is_empty() {
527                    let parsed = sse::parse_json_sse_frame(&frame, marker)
528                        .map_err(|error| LlmClientError::ResponseTranslation(error.to_string()))?;
529                    frame.clear();
530                    match parsed {
531                        sse::SseFrame::Empty => {}
532                        sse::SseFrame::Done => {
533                            saw_terminal |= codec.sse_done_is_terminal();
534                            break;
535                        }
536                        sse::SseFrame::Data(value) => {
537                            saw_terminal |= codec.is_terminal_event(&value);
538                            let normalized = codec.decode_event(&mut state, &value);
539                            saw_error |= normalized.iter().any(|chunk| matches!(
540                                chunk,
541                                LlmResponseChunk::DecodeError { .. }
542                                    | LlmResponseChunk::StreamError { .. }
543                            ));
544                            yield LlmResponseStreamEvent::preserved(
545                                source_format.clone(),
546                                value,
547                                normalized,
548                            );
549                            // Chat usage can follow finish_reason, so the decoder keeps reading.
550                            if saw_terminal && source_format != WireFormat::OpenAiChat.into() {
551                                break;
552                            }
553                        }
554                    }
555                } else {
556                    frame.push_str(&line);
557                    frame.push('\n');
558                }
559            }
560
561            // A non-standard upstream might omit the final blank line; parse a trailing
562            // complete frame instead of losing its last chunk.
563            #[allow(clippy::collapsible_if)]
564            if !frame.trim_end().is_empty() {
565                let parsed = sse::parse_json_sse_frame(&frame, marker)
566                    .map_err(|error| LlmClientError::ResponseTranslation(error.to_string()))?;
567                match parsed {
568                    sse::SseFrame::Done => {
569                        saw_terminal |= codec.sse_done_is_terminal();
570                    }
571                    sse::SseFrame::Data(value) => {
572                        saw_terminal |= codec.is_terminal_event(&value);
573                        let normalized = codec.decode_event(&mut state, &value);
574                        saw_error |= normalized.iter().any(|chunk| matches!(
575                            chunk,
576                            LlmResponseChunk::DecodeError { .. }
577                                | LlmResponseChunk::StreamError { .. }
578                        ));
579                        yield LlmResponseStreamEvent::preserved(source_format.clone(), value, normalized);
580                    }
581                    sse::SseFrame::Empty => {}
582                }
583            }
584
585            if !saw_terminal && !saw_error {
586                Err(LlmClientError::ResponseTranslation(format!(
587                    "{source_format} stream ended before a terminal event"
588                )))?;
589            }
590        });
591        Ok(stream)
592    }
593}

595// Recover transport errors wrapped for `AsyncRead`; other reader failures are
596// invalid upstream responses.
597fn llm_client_error_from_io(error: std::io::Error) -> LlmClientError {
598    let kind = error.kind();
599    let message = error.to_string();
600    match error.into_inner() {
601        Some(source) => match source.downcast::<LlmClientError>() {
602            Ok(error) => *error,
603            Err(source) => LlmClientError::InvalidResponse { source },
604        },
605        None => LlmClientError::InvalidResponse {
606            source: Box::new(std::io::Error::new(kind, message)),
607        },
608    }
609}

611// Attaches source and target formats to every diagnostic emitted across both passes.
612fn with_formats(
613    decoded: Vec<TranslationDiagnostic>,
614    encoded: Vec<TranslationDiagnostic>,
615    source: FormatId,
616    target: FormatId,
617) -> Vec<TranslationDiagnostic> {
618    decoded
619        .into_iter()
620        .chain(encoded)
621        .map(|diagnostic| diagnostic.with_formats(source.clone(), target.clone()))
622        .collect()
623}
