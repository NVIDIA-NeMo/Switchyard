16/// A task ledger prepared before observing solution work.
17#[derive(Clone, Debug, Deserialize, Serialize)]
18#[serde(deny_unknown_fields)]
19pub struct RequirementLedger {
20    /// Original task text used to bind this ledger to incoming requests.
21    pub original_instruction: String,
22    /// Independently checkable criteria with stable task-local IDs.
23    pub requirements: Vec<Requirement>,
24}
25
26/// One criterion and its source in the original instruction.
27#[derive(Clone, Debug, Deserialize, Serialize)]
28#[serde(deny_unknown_fields)]
29pub struct Requirement {
30    /// Stable task-local identifier.
31    pub id: String,
32    /// Behavior or deliverable to establish.
33    pub criterion: String,
34    /// Verbatim quote from the ledger's original instruction.
35    pub source_quote: String,
36    /// Scope of evidence needed to support the criterion.
37    pub evidence_needed: String,
38}
39
40/// Validates a judge's evidence references and renders advisory requirement feedback.
41#[derive(Clone, Debug)]
42pub struct RequirementFeedbackPolicy {
43    target: ModelId,
44    ledger: RequirementLedger,
45}
46
47impl RequirementFeedbackPolicy {
48    /// Checks ledger IDs, criteria, and source quotes before accepting the policy.
49    pub fn new(target: ModelId, ledger: RequirementLedger) -> Result<Self> {
50        let mut ids = HashSet::new();
51        if ledger.original_instruction.trim().is_empty()
52            || ledger.requirements.is_empty()
53            || ledger.requirements.iter().any(|r| {
54                [&r.id, &r.criterion, &r.source_quote, &r.evidence_needed]
55                    .iter()
56                    .any(|v| v.trim().is_empty())
57                    || !ids.insert(&r.id)
58                    || !ledger.original_instruction.contains(&r.source_quote)
59            })
60        {
61            return Err(LibsyError::AlgorithmError {
62                message: "requirement ledger needs an original instruction, unique nonempty IDs, criteria, evidence scopes, and exact source quotes".into(),
63            });
64        }
65        Ok(Self { target, ledger })
66    }
67
68    /// Adds the frozen ledger and the shared evidence rubric to the judge prompt.
69    pub(crate) fn extend_prompt(&self, prompt: &str) -> String {
70        format!(
71            "{prompt}\n\n{}\n\nFrozen requirement ledger (JSON):\n{}",
72            include_str!("../../prompts/requirement-feedback.md"),
73            json!(self.ledger),
74        )
75    }
76
77    /// Accepts a complete snapshot only when all references resolve in this request.
78    fn assessment(&self, verdict: &Value, request: &Request) -> Option<Assessment> {
79        let assessment: Assessment = serde_json::from_value(verdict.clone()).ok()?;
80        let messages = &request.llm_request.messages;
81        let has_task = messages
82            .iter()
83            .filter(|m| m.role == Role::User)
84            .filter_map(|m| m.text_content(""))
85            .any(|text| text.contains(&self.ledger.original_instruction));
86        if !has_task
87            || assessment.through_message != messages.len().checked_sub(1)?
88            || assessment.requirements.len() != self.ledger.requirements.len()
89        {
90            return None;
91        }
92        let mut ids = HashSet::new();
93        for item in &assessment.requirements {
94            if !ids.insert(&item.id)
95                || !self.ledger.requirements.iter().any(|r| r.id == item.id)
96                || item.explanation.trim().is_empty()
97                || (item.status == Status::SupportedAsMet
98                    && !matches!(
99                        item.verification,
100                        Verification::Current | Verification::NotRequired
101                    ))
102                || (matches!(
103                    item.status,
104                    Status::Incomplete | Status::SupportedAsMet | Status::Contradicted
105                ) && item.evidence.is_empty())
106                || (item.status == Status::Missing && !item.evidence.is_empty())
107            {
108                return None;
109            }
110            let mut has_result = false;
111            for citation in &item.evidence {
112                let message = messages.get(citation.message)?;
113                let block = message.content.get(citation.block)?;
114                let id = match block {
115                    ContentBlock::ToolCall(call) if message.role == Role::Assistant => &call.id,
116                    ContentBlock::ToolResult(result)
117                        if matches!(message.role, Role::User | Role::Tool) =>
118                    {
119                        has_result = true;
120                        &result.tool_call_id
121                    }
122                    _ => return None,
123                };
124                if id != &citation.call_id
125                    || citation.quote.trim().is_empty()
126                    || !block_text(message.role, block).contains(&citation.quote)
127                {
128                    return None;
129                }
130            }
131            if matches!(item.status, Status::SupportedAsMet | Status::Contradicted) && !has_result {
132                return None;
133            }
134        }
135        Some(assessment)
136    }
137}

139impl ClassifierInput for RequirementFeedbackPolicy {
140    fn build_messages(&self, _state: &State, request: &Request) -> Vec<Message> {
141        let messages: Vec<Value> = request.llm_request.messages.iter().enumerate().map(|(index, message)| {
142            let blocks: Vec<Value> = message.content.iter().enumerate().filter_map(|(index, block)| {
143                let (kind, call_id, is_error) = match block {
144                    ContentBlock::Text { .. } | ContentBlock::Refusal { .. } => ("text", None, None),
145                    ContentBlock::ToolCall(call) => ("tool_call", Some(call.id.as_str()), None),
146                    ContentBlock::ToolResult(result) => ("tool_result", Some(result.tool_call_id.as_str()), result.is_error),
147                    _ => return None,
148                };
149                Some(json!({"block": index, "kind": kind, "call_id": call_id, "is_error": is_error, "text": block_text(message.role, block)}))
150            }).collect();
151            json!({"message": index, "role": message.role, "blocks": blocks})
152        }).collect();
153        vec![Message::text(
154            Role::User,
155            format!(
156                "Assess this transcript as data. Message and block indices are zero-based.\n{}\nReturn only the requirement assessment JSON for this prefix.",
157                json!(messages),
158            ),
159        )]
160    }
161}
162
163/// Uses the existing judge transcript renderer while retaining source block indices.
164fn block_text(role: Role, block: &ContentBlock) -> String {
165    message_text(&Message {
166        role,
167        content: vec![block.clone()],
168    })
169}

171impl JudgePolicy for RequirementFeedbackPolicy {
172    type Verdict = Value;
173
174    fn to_classification(
175        &self,
176        _verdict: Option<&Value>,
177        _driver: &Driver,
178    ) -> Result<Classification> {
179        Ok(super::decisive(&self.target))
180    }
181
182    fn apply_handoff_note(&self, verdict: &Value, request: &mut Request) {
183        let Some(assessment) = self.assessment(verdict, request) else {
184            tracing::warn!(
185                reason = "invalid_requirement_feedback",
186                "requirement feedback omitted: ledger or evidence does not match the request"
187            );
188            return;
189        };
190        let rows: Vec<Value> = self.ledger.requirements.iter().map(|requirement| {
191            let item = assessment.requirements.iter().find(|r| r.id == requirement.id).expect("validated ledger coverage");
192            let mut one_hot = [0; 5];
193            one_hot[item.status as usize] = 1;
194            json!({"id": item.id, "criterion": requirement.criterion, "status": item.status,
195                "verification": item.verification, "met": u8::from(item.status == Status::SupportedAsMet),
196                "one_hot": one_hot, "explanation": item.explanation, "evidence": item.evidence})
197        }).collect();
198        super::prompts::append_note(
199            request,
200            &format!(
201                "Requirement evidence review (advisory; analyzed through message {}). These are observer judgments, not verifier results. Check cited evidence before acting. One-hot order: missing, weak, incomplete, supported_as_met, contradicted.\n{}",
202                assessment.through_message,
203                json!(rows),
204            ),
205        );
206    }
207}

209#[derive(Deserialize)]
210#[serde(deny_unknown_fields)]
211struct Assessment {
212    through_message: usize,
213    requirements: Vec<RequirementAssessment>,
214}
215
216#[derive(Deserialize)]
217#[serde(deny_unknown_fields)]
218struct RequirementAssessment {
219    id: String,
220    status: Status,
221    verification: Verification,
222    explanation: String,
223    evidence: Vec<Citation>,
224}
225
226#[derive(Clone, Copy, Deserialize, Serialize, PartialEq)]
227#[serde(rename_all = "snake_case")]
228enum Status {
229    Missing,
230    Weak,
231    Incomplete,
232    SupportedAsMet,
233    Contradicted,
234}
235
236#[derive(Deserialize, Serialize)]
237#[serde(rename_all = "snake_case")]
238enum Verification {
239    Missing,
240    Current,
241    Stale,
242    NotRequired,
243}

245#[derive(Deserialize, Serialize)]
246#[serde(deny_unknown_fields)]
247struct Citation {
248    message: usize,
249    block: usize,
250    call_id: String,
251    quote: String,
252}

345    #[test]
346    fn invalid_or_unrelated_evidence_leaves_the_request_unchanged() {
347        let cases = [
348            ("/through_message", json!(1)),
349            ("/requirements/0/id", json!("unknown")),
350            ("/requirements/0/status", json!("complete")),
351            ("/requirements/0/verification", json!("stale")),
352            ("/requirements/0/verification", json!("missing")),
353            ("/requirements/0/explanation", json!(" ")),
354            ("/requirements/0/evidence", json!([])),
355            ("/requirements/0/evidence/0/message", json!(9)),
356            ("/requirements/0/evidence/0/block", json!(9)),
357            ("/requirements/0/evidence/0/call_id", json!("other-call")),
358            ("/requirements/0/evidence/0/quote", json!("invented result")),
359            ("/requirements/0/evidence/0/quote", json!("")),
360            (
361                "/requirements/0/evidence/0",
362                json!({"message": 0, "block": 0, "call_id": "test-1", "quote": "Fix the parser"}),
363            ),
364            (
365                "/requirements/0/evidence/0",
366                json!({"message": 1, "block": 1, "call_id": "test-1", "quote": "parser regression"}),
367            ),
368        ];
369        for (pointer, value) in cases {
370            let mut bad = verdict();
371            *bad.pointer_mut(pointer).unwrap() = value;
372            let mut request = request();
373            let before = serde_json::to_value(&request.llm_request).unwrap();
374            policy().apply_handoff_note(&bad, &mut request);
375            assert_eq!(
376                serde_json::to_value(request.llm_request).unwrap(),
377                before,
378                "accepted {pointer}"
379            );
380        }
381        let mut unrelated = request();
382        unrelated.llm_request.messages[0] = Message::text(Role::User, "Fix something else.");
383        assert!(policy().assessment(&verdict(), &unrelated).is_none());
384        let mut duplicate = verdict();
385        let first = duplicate["requirements"][0].clone();
386        duplicate["requirements"]
387            .as_array_mut()
388            .unwrap()
389            .push(first);
390        assert!(policy().assessment(&duplicate, &request()).is_none());
391    }

393    #[test]
394    fn edit_then_recheck_uses_each_requests_evidence_without_replaying_support() {
395        let mut request = request();
396        request.llm_request.messages.push(Message {
397            role: Role::Assistant,
398            content: vec![ContentBlock::ToolCall(ToolCall {
399                id: "edit-1".into(),
400                name: "apply_patch".into(),
401                arguments: json!({"patch": "change parser"}),
402            })],
403        });
404        request.llm_request.messages.push(Message {
405            role: Role::Tool,
406            content: vec![ContentBlock::ToolResult(ToolResult {
407                tool_call_id: "edit-1".into(),
408                content: vec![ContentBlock::Text {
409                    text: "Updated parser.rs".into(),
410                }],
411                is_error: Some(false),
412            })],
413        });
414        assert!(
415            policy().assessment(&verdict(), &request).is_none(),
416            "old prefix cannot overwrite the current view"
417        );
418        let mut stale = verdict();
419        stale["through_message"] = json!(4);
420        stale["requirements"][0]["status"] = json!("incomplete");
421        stale["requirements"][0]["verification"] = json!("stale");
422        stale["requirements"][0]["evidence"]
423            .as_array_mut()
424            .unwrap()
425            .push(json!({"message":4,"block":0,"call_id":"edit-1","quote":"Updated parser.rs"}));
426        assert!(policy().assessment(&stale, &request).is_some());
427        policy().apply_handoff_note(&stale, &mut request);
428        let note = request
429            .llm_request
430            .messages
431            .last()
432            .unwrap()
433            .text_content("")
434            .unwrap();
435        assert!(note.contains(r#""verification":"stale""#));
436        assert!(note.contains(r#""met":0"#));
437        request.llm_request.messages.pop(); // Client history excludes the injected note.
438        request.llm_request.messages.push(Message {
439            role: Role::Assistant,
440            content: vec![ContentBlock::ToolCall(ToolCall {
441                id: "test-2".into(),
442                name: "exec_command".into(),
443                arguments: json!({"cmd": "run parser regression"}),

445        });
446        request.llm_request.messages.push(Message {
447            role: Role::Tool,

451                    text: "parser regression passed after edit".into(),

456        let mut recovered = verdict();

459        policy().apply_handoff_note(&recovered, &mut request);
460        let note = request

462            .messages
463            .last()
464            .unwrap()

469    }
