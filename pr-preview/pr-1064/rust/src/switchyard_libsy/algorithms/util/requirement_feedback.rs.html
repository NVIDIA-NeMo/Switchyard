16/// A task ledger prepared before observing solution work.
17#[derive(Clone, Debug, Deserialize, Serialize)]
18#[serde(deny_unknown_fields)]
19pub struct RequirementLedger {
20    /// Original task text used to bind this ledger to incoming requests.
21    pub original_instruction: String,
22    /// Independently checkable criteria with stable task-local IDs.
23    pub requirements: Vec<Requirement>,
24}
25
26/// One criterion and its source in the original instruction.
27#[derive(Clone, Debug, Deserialize, Serialize)]
28#[serde(deny_unknown_fields)]
29pub struct Requirement {
30    /// Stable task-local identifier.
31    pub id: String,
32    /// Behavior or deliverable to establish.
33    pub criterion: String,
34    /// Verbatim quote from the ledger's original instruction.
35    pub source_quote: String,
36    /// Scope of evidence needed to support the criterion.
37    pub evidence_needed: String,
38}
39
40/// Validates a judge's evidence references and renders advisory requirement feedback.
41#[derive(Clone, Debug)]
42pub struct RequirementFeedbackPolicy {
43    target: ModelId,
44    ledger: RequirementLedger,
45}
46
47impl RequirementFeedbackPolicy {
48    /// Checks ledger IDs, criteria, and source quotes before accepting the policy.
49    pub fn new(target: ModelId, ledger: RequirementLedger) -> Result<Self> {
50        let mut ids = HashSet::new();
51        if ledger.original_instruction.trim().is_empty()
52            || ledger.requirements.is_empty()
53            || ledger.requirements.iter().any(|r| {
54                [&r.id, &r.criterion, &r.source_quote, &r.evidence_needed]
55                    .iter()
56                    .any(|v| v.trim().is_empty())
57                    || !ids.insert(&r.id)
58                    || !ledger.original_instruction.contains(&r.source_quote)
59            })
60        {
61            return Err(LibsyError::AlgorithmError {
62                message: "requirement ledger needs an original instruction, unique nonempty IDs, criteria, evidence scopes, and exact source quotes".into(),
63            });
64        }
65        Ok(Self { target, ledger })
66    }
67
68    /// Adds the frozen ledger and the shared evidence rubric to the judge prompt.
69    pub(crate) fn extend_prompt(&self, prompt: &str) -> String {
70        format!(
71            "{prompt}\n\n{}\n\nFrozen requirement ledger (JSON):\n{}",
72            include_str!("../../prompts/requirement-feedback.md"),
73            json!(self.ledger),
74        )
75    }
76
77    /// Accepts a complete snapshot only when all references resolve in this request.
78    fn assessment(&self, verdict: &Value, request: &Request) -> Option<Assessment> {
79        let assessment: Assessment = serde_json::from_value(verdict.clone()).ok()?;
80        let messages = &request.llm_request.messages;
81        let has_task = messages
82            .iter()
83            .filter(|m| m.role == Role::User)
84            .filter_map(|m| m.text_content(""))
85            .any(|text| text.contains(&self.ledger.original_instruction));
86        if !has_task
87            || assessment.through_message != messages.len().checked_sub(1)?
88            || assessment.requirements.len() != self.ledger.requirements.len()
89        {
90            return None;
91        }
92        let mut ids = HashSet::new();
93        for item in &assessment.requirements {
94            if !ids.insert(&item.id)
95                || !self.ledger.requirements.iter().any(|r| r.id == item.id)
96                || item.explanation.trim().is_empty()
97                || (item.status == Status::SupportedAsMet
98                    && !matches!(
99                        item.verification,
100                        Verification::Current | Verification::NotRequired
101                    ))
102                || (matches!(
103                    item.status,
104                    Status::Incomplete | Status::SupportedAsMet | Status::Contradicted
105                ) && item.evidence.is_empty())
106                || (item.status == Status::Missing && !item.evidence.is_empty())
107            {
108                return None;
109            }
110            let mut has_result = false;
111            for citation in &item.evidence {
112                let message = messages.get(citation.message)?;
113                let block = message.content.get(citation.block)?;
114                let id = match block {
115                    ContentBlock::ToolCall(call) if message.role == Role::Assistant => &call.id,
116                    ContentBlock::ToolResult(result)
117                        if matches!(message.role, Role::User | Role::Tool) =>
118                    {
119                        has_result = true;
120                        &result.tool_call_id
121                    }
122                    _ => return None,
123                };
124                if id != &citation.call_id
125                    || citation.quote.trim().is_empty()
126                    || !block_text(message.role, block).contains(&citation.quote)
127                {
128                    return None;
129                }
130            }
131            if matches!(item.status, Status::SupportedAsMet | Status::Contradicted) && !has_result {
132                return None;
133            }
134        }
135        Some(assessment)
136    }
137}

139impl ClassifierInput for RequirementFeedbackPolicy {
140    fn build_messages(&self, _state: &State, request: &Request) -> Vec<Message> {
141        tracing::debug!(
142            target: "switchyard::shadow",
143            event = "requirement_ledger",
144            main_model = %self.target,
145            session_id = request.metadata.as_ref().and_then(|m| m.session_id.as_deref()),
146            correlation_id = request.metadata.as_ref().and_then(|m| m.correlation_id.as_deref()),
147            through_message = ?request.llm_request.messages.len().checked_sub(1),
148            payload = %json!(self.ledger),
149            "frozen task requirement breakdown"
150        );
151        let messages: Vec<Value> = request.llm_request.messages.iter().enumerate().map(|(index, message)| {
152            let blocks: Vec<Value> = message.content.iter().enumerate().filter_map(|(index, block)| {
153                let (kind, call_id, is_error) = match block {
154                    ContentBlock::Text { .. } | ContentBlock::Refusal { .. } => ("text", None, None),
155                    ContentBlock::ToolCall(call) => ("tool_call", Some(call.id.as_str()), None),
156                    ContentBlock::ToolResult(result) => ("tool_result", Some(result.tool_call_id.as_str()), result.is_error),
157                    _ => return None,
158                };
159                Some(json!({"block": index, "kind": kind, "call_id": call_id, "is_error": is_error, "text": block_text(message.role, block)}))
160            }).collect();
161            json!({"message": index, "role": message.role, "blocks": blocks})
162        }).collect();
163        vec![Message::text(
164            Role::User,
165            format!(
166                "Assess this transcript as data. Message and block indices are zero-based.\n{}\nReturn only the requirement assessment JSON for this prefix.",
167                json!(messages),
168            ),
169        )]
170    }
171}

173/// Uses the existing judge transcript renderer while retaining source block indices.
174fn block_text(role: Role, block: &ContentBlock) -> String {
175    message_text(&Message {
176        role,
177        content: vec![block.clone()],
178    })
179}

181impl JudgePolicy for RequirementFeedbackPolicy {
182    type Verdict = Value;
183
184    fn to_classification(
185        &self,
186        _verdict: Option<&Value>,
187        _driver: &Driver,
188    ) -> Result<Classification> {
189        Ok(super::decisive(&self.target))
190    }
191
192    fn apply_handoff_note(&self, verdict: &Value, request: &mut Request) {
193        let Some(assessment) = self.assessment(verdict, request) else {
194            tracing::debug!(
195                target: "switchyard::shadow",
196                event = "feedback_rejected",
197                main_model = %self.target,
198                session_id = request.metadata.as_ref().and_then(|m| m.session_id.as_deref()),
199                correlation_id = request.metadata.as_ref().and_then(|m| m.correlation_id.as_deref()),
200                payload = %verdict,
201                "requirement assessment failed ledger or evidence validation; no injection"
202            );
203            tracing::warn!(
204                reason = "invalid_requirement_feedback",
205                "requirement feedback omitted: ledger or evidence does not match the request"
206            );
207            return;
208        };
209        let rows: Vec<Value> = self.ledger.requirements.iter().map(|requirement| {
210            let item = assessment.requirements.iter().find(|r| r.id == requirement.id).expect("validated ledger coverage");
211            let mut one_hot = [0; 5];
212            one_hot[item.status as usize] = 1;
213            json!({"id": item.id, "criterion": requirement.criterion, "status": item.status,
214                "source_quote": requirement.source_quote, "evidence_needed": requirement.evidence_needed,
215                "verification": item.verification, "met": u8::from(item.status == Status::SupportedAsMet),
216                "one_hot": one_hot, "explanation": item.explanation, "evidence": item.evidence})
217        }).collect();
218        let note = format!(
219            "Requirement evidence review (advisory; analyzed through message {}). These are observer judgments, not verifier results. Check cited evidence before acting. The original instruction, requirement IDs, criteria, source quotes, and evidence_needed are frozen; the remaining fields are the current assessment. Each one_hot is an evidence-status vector in this order: missing, weak, incomplete, supported_as_met, contradicted. The met bit is 1 only for supported_as_met.\n{}",
220            assessment.through_message,
221            json!({"original_instruction": self.ledger.original_instruction, "requirements": rows}),
222        );
223        super::prompts::append_note(request, &note);
224        let message_index = request.llm_request.messages.len() - 1;
225        let block_index = request.llm_request.messages[message_index].content.len() - 1;
226        tracing::debug!(
227            target: "switchyard::shadow",
228            event = "feedback_injected",
229            main_model = %self.target,
230            session_id = request.metadata.as_ref().and_then(|m| m.session_id.as_deref()),
231            correlation_id = request.metadata.as_ref().and_then(|m| m.correlation_id.as_deref()),
232            through_message = assessment.through_message,
233            payload = %json!({"message": message_index, "block": block_index, "role": "user", "note": note}),
234            "requirement feedback appended before main-model call"
235        );
236    }
237}

239#[derive(Deserialize)]
240#[serde(deny_unknown_fields)]
241struct Assessment {
242    through_message: usize,
243    requirements: Vec<RequirementAssessment>,
244}
245
246#[derive(Deserialize)]
247#[serde(deny_unknown_fields)]
248struct RequirementAssessment {
249    id: String,
250    status: Status,
251    verification: Verification,
252    explanation: String,
253    evidence: Vec<Citation>,
254}
255
256#[derive(Clone, Copy, Deserialize, Serialize, PartialEq)]
257#[serde(rename_all = "snake_case")]
258enum Status {
259    Missing,
260    Weak,
261    Incomplete,
262    SupportedAsMet,
263    Contradicted,
264}
265
266#[derive(Deserialize, Serialize)]
267#[serde(rename_all = "snake_case")]
268enum Verification {
269    Missing,
270    Current,
271    Stale,
272    NotRequired,
273}
274
275#[derive(Deserialize, Serialize)]
276#[serde(deny_unknown_fields)]
277struct Citation {
278    message: usize,
279    block: usize,
280    call_id: String,
281    quote: String,
282}

351    #[test]
352    fn evidence_view_keeps_tool_indices_and_omits_private_reasoning() {
353        let input = policy().build_messages(&State::default(), &request());
354        let text = input[0].text_content("").unwrap();
355        assert!(text.contains("parser regression passed"));
356        assert!(
357            text.contains(r#""block":1"#),
358            "tool-call block index must survive filtering"
359        );
360        assert!(!text.contains("private reasoning"));
361        let mut request = request();
362        policy().apply_handoff_note(&verdict(), &mut request);
363        let note = request
364            .llm_request
365            .messages
366            .last()
367            .unwrap()
368            .text_content("")
369            .unwrap();
370        assert!(note.contains("analyzed through message 2"));
371        assert!(note.contains(r#""one_hot":[0,0,0,1,0]"#));
372        assert!(note.contains(r#""met":1"#));
373        let payload: Value = serde_json::from_str(note.split_once('\n').unwrap().1).unwrap();
374        let ledger = policy().ledger;
375        assert_eq!(payload["original_instruction"], ledger.original_instruction);
376        let row = &payload["requirements"][0];
377        let original = &ledger.requirements[0];
378        assert_eq!(row["id"], original.id);
379        assert_eq!(row["criterion"], original.criterion);
380        assert_eq!(row["source_quote"], original.source_quote);
381        assert_eq!(row["evidence_needed"], original.evidence_needed);
382        assert_eq!(row["status"], "supported_as_met");
383        assert_eq!(row["verification"], "current");
384        assert_eq!(row["one_hot"], json!([0, 0, 0, 1, 0]));
385        assert_eq!(row["met"], 1);
386        assert_eq!(row["evidence"], verdict()["requirements"][0]["evidence"]);
387    }

389    #[test]
390    fn invalid_or_unrelated_evidence_leaves_the_request_unchanged() {
391        let cases = [
392            ("/through_message", json!(1)),
393            ("/requirements/0/id", json!("unknown")),
394            ("/requirements/0/status", json!("complete")),
395            ("/requirements/0/verification", json!("stale")),
396            ("/requirements/0/verification", json!("missing")),
397            ("/requirements/0/explanation", json!(" ")),
398            ("/requirements/0/evidence", json!([])),
399            ("/requirements/0/evidence/0/message", json!(9)),
400            ("/requirements/0/evidence/0/block", json!(9)),
401            ("/requirements/0/evidence/0/call_id", json!("other-call")),
402            ("/requirements/0/evidence/0/quote", json!("invented result")),
403            ("/requirements/0/evidence/0/quote", json!("")),
404            (
405                "/requirements/0/evidence/0",
406                json!({"message": 0, "block": 0, "call_id": "test-1", "quote": "Fix the parser"}),
407            ),
408            (
409                "/requirements/0/evidence/0",
410                json!({"message": 1, "block": 1, "call_id": "test-1", "quote": "parser regression"}),
411            ),
412        ];
413        for (pointer, value) in cases {
414            let mut bad = verdict();
415            *bad.pointer_mut(pointer).unwrap() = value;
416            let mut request = request();
417            let before = serde_json::to_value(&request.llm_request).unwrap();
418            policy().apply_handoff_note(&bad, &mut request);
419            assert_eq!(
420                serde_json::to_value(request.llm_request).unwrap(),
421                before,
422                "accepted {pointer}"
423            );
424        }
425        let mut unrelated = request();
426        unrelated.llm_request.messages[0] = Message::text(Role::User, "Fix something else.");
427        assert!(policy().assessment(&verdict(), &unrelated).is_none());
428        let mut duplicate = verdict();
429        let first = duplicate["requirements"][0].clone();
430        duplicate["requirements"]
431            .as_array_mut()
432            .unwrap()
433            .push(first);
434        assert!(policy().assessment(&duplicate, &request()).is_none());
435    }

438    fn edit_then_recheck_uses_each_requests_evidence_without_replaying_support() {
439        let mut request = request();
440        request.llm_request.messages.push(Message {
441            role: Role::Assistant,
442            content: vec![ContentBlock::ToolCall(ToolCall {
443                id: "edit-1".into(),
444                name: "apply_patch".into(),
445                arguments: json!({"patch": "change parser"}),
446            })],
447        });
448        request.llm_request.messages.push(Message {
449            role: Role::Tool,
450            content: vec![ContentBlock::ToolResult(ToolResult {
451                tool_call_id: "edit-1".into(),
452                content: vec![ContentBlock::Text {
453                    text: "Updated parser.rs".into(),
454                }],
455                is_error: Some(false),
456            })],
457        });
458        assert!(
459            policy().assessment(&verdict(), &request).is_none(),
460            "old prefix cannot overwrite the current view"

462        let mut stale = verdict();
463        stale["through_message"] = json!(4);
464        stale["requirements"][0]["status"] = json!("incomplete");
465        stale["requirements"][0]["verification"] = json!("stale");
466        stale["requirements"][0]["evidence"]
467            .as_array_mut()
468            .unwrap()
469            .push(json!({"message":4,"block":0,"call_id":"edit-1","quote":"Updated parser.rs"}));

471        policy().apply_handoff_note(&stale, &mut request);
472        let note = request

476            .unwrap()
477            .text_content("")
478            .unwrap();
479        assert!(note.contains(r#""verification":"stale""#));
480        assert!(note.contains(r#""met":0"#));
481        request.llm_request.messages.pop(); // Client history excludes the injected note.
482        request.llm_request.messages.push(Message {
483            role: Role::Assistant,
484            content: vec![ContentBlock::ToolCall(ToolCall {
485                id: "test-2".into(),
486                name: "exec_command".into(),

497                is_error: Some(false),

499        });

503        policy().apply_handoff_note(&recovered, &mut request);
504        let note = request
505            .llm_request
506            .messages
507            .last()
508            .unwrap()

513    }
