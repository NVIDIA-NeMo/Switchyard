16/// Deployment-calibrated work scores, not predicted response latency.
17#[derive(Clone, Debug, Deserialize)]
18#[serde(deny_unknown_fields)]
19pub struct InfraRouterConfig {
20    /// Deadline for the host lookup. Expiry preserves the task decision.
21    pub probe_timeout_ms: u64,
22    /// Maximum observation age at the final routing decision.
23    pub max_signal_age_ms: u64,
24    /// Require at least this many in-flight requests on the efficient candidate.
25    pub min_active_requests: u64,
26    /// Protect the capable candidate when it has more in-flight requests than this.
27    pub max_spillover_active_requests: u64,
28    /// Relative prefill costs, ordered efficient then capable.
29    pub prefill_costs: [f64; 2],
30    /// Extra work per in-flight request, expressed in equivalent prefill tokens.
31    pub request_penalty_tokens: f64,
32    /// Required fractional score reduction, in [0, 1).
33    pub min_improvement: f64,
34}
35
36impl InfraRouterConfig {
37    fn validate(&self) -> Result<()> {
38        if self.probe_timeout_ms == 0
39            || self.max_signal_age_ms == 0
40            || self.min_active_requests == 0
41            || self
42                .prefill_costs
43                .iter()
44                .any(|v| !v.is_finite() || *v <= 0.0)
45            || !self.request_penalty_tokens.is_finite()
46            || self.request_penalty_tokens <= 0.0
47            || !(0.0..1.0).contains(&self.min_improvement)
48        {
49            return Err(LibsyError::AlgorithmError {
50                message: "infra requires positive age, source load, costs and request penalty, and min_improvement in [0, 1)".into(),
51            });
52        }
53        Ok(())
54    }
55
56    fn score(&self, signal: &ModelInfraSignals, cost: f64, now: Instant) -> Option<f64> {
57        if now.checked_duration_since(signal.sampled_at)?
58            > Duration::from_millis(self.max_signal_age_ms)
59        {
60            return None;
61        }
62        let work = signal.effective_prefill_tokens? as f64
63            + signal.active_prefill_tokens? as f64
64            + signal.active_requests? as f64 * self.request_penalty_tokens;
65        let score = work * cost;
66        score.is_finite().then_some(score)
67    }
68}

70/// Runs task routing first, then optionally moves an efficient choice to the capable tier.
71///
72/// The host supplies trusted observations through a read-only call after task routing. This policy does not
73/// select or reserve a worker. Task affinity stays in the inner algorithm, so a
74/// temporary spillover does not latch the session onto the capable model.
75pub struct InfraRouter {
76    inner: Arc<dyn Algorithm>,
77    config: InfraRouterConfig,
78}
79
80impl InfraRouter {
81    /// Wraps a decision-only tier router. Both tiers must contain exactly one model.
82    pub fn new(inner: Arc<dyn Algorithm>, config: InfraRouterConfig) -> Result<Self> {
83        config.validate()?;
84        Ok(Self { inner, config })
85    }
86
87    fn evaluate(
88        &self,
89        signals: &InfraSignals,
90        efficient: &ModelId,
91        capable: &ModelId,
92    ) -> (&'static str, Option<(f64, f64)>) {
93        let (Some(source), Some(destination)) = (signals.get(efficient), signals.get(capable))
94        else {
95            return ("signals_unavailable", None);
96        };
97        let now = Instant::now();
98        let (Some(source_score), Some(destination_score)) = (
99            self.config.score(source, self.config.prefill_costs[0], now),
100            self.config
101                .score(destination, self.config.prefill_costs[1], now),
102        ) else {
103            return ("stale_or_incomplete_signals", None);
104        };
105        let scores = Some((source_score, destination_score));
106        if source.active_requests.unwrap() < self.config.min_active_requests {
107            return ("source_not_busy", scores);
108        }
109        if destination.active_requests.unwrap() > self.config.max_spillover_active_requests {
110            return ("destination_busy", scores);
111        }
112        if destination_score < source_score * (1.0 - self.config.min_improvement) {
113            ("lower_work_score", scores)
114        } else {
115            ("insufficient_improvement", scores)
116        }
117    }
118}

120#[async_trait::async_trait]
121impl Algorithm for InfraRouter {
122    fn name(&self) -> &str {
123        self.inner.name()
124    }
125
126    async fn route(self: Arc<Self>, driver: Driver, request: Request) -> Result<RoutingOutcome> {
127        let delegated = request
128            .metadata
129            .as_ref()
130            .is_some_and(|m| m.is_subagent_work());
131        let mut outcome = self.inner.clone().route(driver.clone(), request).await?;
132        // Sub-agent routing has separate target groups and capability requirements.
133        if outcome.response.is_some() || delegated {
134            return Ok(outcome);
135        }
136        let ([efficient], [capable]) = (
137            driver.models_for(&Category::Efficient),
138            driver.models_for(&Category::Capable),
139        ) else {
140            return Err(LibsyError::AlgorithmError {
141                message: "infra requires exactly one efficient and one capable model".into(),
142            });
143        };
144        if outcome.selected_model_id()? != efficient || efficient == capable {
145            return Ok(outcome);
146        }
147        let lookup = tokio::time::timeout(
148            Duration::from_millis(self.config.probe_timeout_ms),
149            driver.call_infra(
150                outcome.request.clone(),
151                vec![efficient.clone(), capable.clone()],
152            ),
153        )
154        .await;
155        let (reason, scores) = match lookup {
156            Ok(Ok(signals)) => self.evaluate(&signals, efficient, capable),
157            Ok(Err(_)) => ("signals_unavailable", None),
158            Err(_) => ("probe_timeout", None),
159        };
160        let spillover = reason == "lower_work_score";
161        driver.annotate_evidence(
162            "infra",
163            serde_json::json!({
164                "task_model": efficient,
165                "alternate_model": capable,
166                "spillover": spillover,
167                "reason": reason,
168                "source_score": scores.map(|s| s.0),
169                "destination_score": scores.map(|s| s.1),
170            }),
171        );
172        tracing::info!(
173            task_model = %efficient, alternate_model = %capable, spillover, reason,
174            source_score = scores.map(|s| s.0), destination_score = scores.map(|s| s.1),
175            "infrastructure routing decision"
176        );
177        if spillover {
178            outcome.selected_model_ids.retain(|model| model != capable);
179            outcome.selected_model_ids.insert(0, capable.clone());
180            outcome.request.llm_request.model = Some(capable.to_string());
181        }
182        Ok(outcome)
183    }
184}

187mod tests {
188    use super::*;
189    use crate::{Call, RuntimeModels};
190    use switchyard_protocol::text_request;

192    struct TaskChoice(&'static str);
193
194    #[async_trait::async_trait]
195    impl Algorithm for TaskChoice {
196        fn name(&self) -> &str {
197            "task"
198        }
199
200        async fn route(
201            self: Arc<Self>,
202            driver: Driver,
203            request: Request,
204        ) -> Result<RoutingOutcome> {
205            driver.set_evidence(serde_json::json!({"task_reason": "classified"}));
206            Ok(RoutingOutcome::route_to(self.0.into(), vec![], request))
207        }
208    }

210    fn config() -> InfraRouterConfig {
211        InfraRouterConfig {
212            probe_timeout_ms: 10,
213            max_signal_age_ms: 1000,
214            min_active_requests: 8,
215            max_spillover_active_requests: 1,
216            prefill_costs: [1.0, 4.0],
217            request_penalty_tokens: 4096.0,
218            min_improvement: 0.2,
219        }

222    #[tokio::test]
223    async fn spillover_requires_a_fresh_and_better_capable_candidate() {
224        for case in [
225            "busy",
226            "quiet",
227            "destination_busy",
228            "cold",
229            "missing",
230            "incomplete",
231            "stale",
232            "future",
233            "timeout",
234            "capable_task",
235        ] {
236            let mut signals: InfraSignals = [("small", 8), ("large", 0)]
237                .into_iter()
238                .map(|(id, active)| {
239                    (
240                        id.into(),
241                        ModelInfraSignals {
242                            sampled_at: Instant::now(),
243                            effective_prefill_tokens: Some(1000),
244                            active_prefill_tokens: Some(0),
245                            active_requests: Some(active),
246                        },
247                    )
248                })
249                .collect();
250            let source = signals.get_mut(&ModelId::from("small")).unwrap();
251            match case {
252                "quiet" => source.active_requests = Some(0),
253                "incomplete" => source.active_requests = None,
254                "stale" => source.sampled_at -= Duration::from_secs(2),
255                "future" => source.sampled_at += Duration::from_secs(2),
256                "missing" => signals.clear(),
257                "destination_busy" => {
258                    signals
259                        .get_mut(&ModelId::from("large"))
260                        .unwrap()
261                        .active_requests = Some(2)
262                }
263                "cold" => {
264                    signals
265                        .get_mut(&ModelId::from("large"))
266                        .unwrap()
267                        .effective_prefill_tokens = Some(20_000)
268                }
269                _ => {}
270            }
271            let task_model = if case == "capable_task" {
272                "large"
273            } else {
274                "small"
275            };
276            let algorithm =
277                Arc::new(InfraRouter::new(Arc::new(TaskChoice(task_model)), config()).unwrap());
278            let models = RuntimeModels::new(
279                [
280                    (Category::Efficient, vec!["small".into()]),
281                    (Category::Capable, vec!["large".into()]),
282                ]
283                .into(),
284            );
285            let request = Request {
286                llm_request: text_request(None, "hello"),
287                ..Default::default()
288            };
289            let expected = if matches!(case, "busy" | "capable_task") {
290                "large"
291            } else {
292                "small"
293            };
294            let outcome = crate::drive(algorithm, request.clone(), Arc::new(models), move |call| {
295                let signals = signals.clone();
296                async move {
297                    let Call::Infra(call) = call else {
298                        panic!("unexpected model call")
299                    };
300                    assert_ne!(case, "capable_task");
301                    assert_eq!(call.request.llm_request.model.as_deref(), Some("small"));
302                    assert_eq!(
303                        call.candidates,
304                        vec![ModelId::from("small"), ModelId::from("large")]
305                    );
306                    call.respond(async move {
307                        if case == "timeout" {
308                            std::future::pending::<()>().await;
309                        }
310                        signals
311                    })
312                    .await
313                }
314            })
315            .await
316            .unwrap();
317            assert_eq!(
318                outcome.selected_model_id().unwrap().as_str(),
319                expected,
320                "{case}"
321            );
322            let mut actual_request = outcome.request.llm_request;
323            assert_eq!(actual_request.model.take().as_deref(), Some(expected));
324            assert_eq!(actual_request, request.llm_request);
325            let evidence = outcome.metadata.unwrap().evidence.unwrap();
326            assert_eq!(evidence["task_reason"], "classified");
327            if case != "capable_task" {
328                assert_eq!(evidence["infra"]["spillover"], case == "busy");
329            }
330        }
331    }
