4//! Optional queue-based moves from the efficient model to the capable model.
5
6use crate::{LibsyError, Result, RoutingOutcome, RuntimeModels};
7use serde::Deserialize;
8use std::{
9    collections::BTreeSet,
10    time::{Duration, Instant},
11};
12use switchyard_protocol::{Category, InfraSignals, ModelInfraSignals};
13
14/// Queue and capacity limits for moving requests to the capable model.
15#[derive(Clone, Debug, Deserialize)]
16#[serde(deny_unknown_fields)]
17pub struct InfraPolicyConfig {
18    /// Maximum age of every required worker observation.
19    pub max_signal_age_ms: u64,
20    /// Every efficient worker must have at least this many waiting requests.
21    pub min_waiting_requests: u64,
22    /// Stop moving requests when observed running or locally tracked capable work reaches this limit.
23    pub max_capable_inflight: u64,
24    /// Complete set of efficient worker IDs in the host's metrics inventory.
25    pub efficient_workers: BTreeSet<String>,
26    /// Complete set of capable worker IDs in the host's metrics inventory.
27    pub capable_workers: BTreeSet<String>,
28}
29
30/// Applies a queue policy after task routing; the host owns collection and request lifetimes.
31pub struct InfraPolicy {
32    config: InfraPolicyConfig,
33}
34
35impl InfraPolicy {
36    /// Validate the policy once at configuration load.
37    pub fn new(config: InfraPolicyConfig) -> Result<Self> {
38        if config.max_signal_age_ms == 0
39            || config.min_waiting_requests == 0
40            || config.max_capable_inflight == 0
41            || config.efficient_workers.is_empty()
42            || config.capable_workers.is_empty()
43            || !config
44                .efficient_workers
45                .is_disjoint(&config.capable_workers)
46            || config
47                .efficient_workers
48                .iter()
49                .chain(&config.capable_workers)
50                .any(String::is_empty)
51        {
52            return Err(LibsyError::AlgorithmError {
53                message: "infra requires positive limits and distinct nonempty worker IDs".into(),
54            });
55        }
56        Ok(Self { config })
57    }
58
59    fn fresh(&self, model: &ModelInfraSignals, expected: &BTreeSet<String>, now: Instant) -> bool {
60        expected.len() == model.workers.len()
61            && expected.iter().all(|id| {
62                model.workers.get(id).is_some_and(|worker| {
63                    now.checked_duration_since(worker.sampled_at)
64                        .is_some_and(|age| {
65                            age <= Duration::from_millis(self.config.max_signal_age_ms)
66                        })
67                })
68            })
69    }
70
71    fn evaluate(
72        &self,
73        source: &ModelInfraSignals,
74        destination: &ModelInfraSignals,
75    ) -> (bool, &'static str) {
76        let now = Instant::now();
77        if !self.fresh(source, &self.config.efficient_workers, now)
78            || !self.fresh(destination, &self.config.capable_workers, now)
79        {
80            return (false, "unknown_or_stale_signals");
81        }
82        if source
83            .workers
84            .values()
85            .any(|w| w.waiting < self.config.min_waiting_requests)
86        {
87            return (false, "small_workers_not_all_queued");
88        }
89        let running = destination
90            .workers
91            .values()
92            .try_fold(0u64, |sum, w| sum.checked_add(w.running));
93        if destination.workers.values().any(|w| w.waiting > 0)
94            || running.is_none_or(|n| {
95                n.max(destination.local_inflight) >= self.config.max_capable_inflight
96            })
97        {
98            return (false, "protect_capable_capacity");
99        }
100        (true, "small_workers_queued")
101    }
102
103    /// Adjust a decision-only task result. Returns true only when the model changes.
104    ///
105    /// Hosts must serialize this check with their local in-flight increment, then decrement
106    /// when the response ends or is cancelled. Worker selection remains in the serving layer.
107    pub fn apply(
108        &self,
109        outcome: &mut RoutingOutcome,
110        models: &RuntimeModels,
111        signals: &InfraSignals,
112    ) -> Result<bool> {
113        if outcome.response.is_some()
114            || outcome
115                .request
116                .metadata
117                .as_ref()
118                .is_some_and(|m| m.is_subagent_work())
119        {
120            return Ok(false);
121        }
122        let ([efficient], [capable]) = (
123            models.models_for(&Category::Efficient),
124            models.models_for(&Category::Capable),
125        ) else {
126            return Err(LibsyError::AlgorithmError {
127                message: "infra requires one efficient and one capable model".into(),
128            });
129        };
130        if outcome.selected_model_id()? != efficient || efficient == capable {
131            return Ok(false);
132        }
133        let (spillover, reason) = match (signals.get(efficient), signals.get(capable)) {
134            (Some(source), Some(destination)) => self.evaluate(source, destination),
135            _ => (false, "unknown_or_stale_signals"),
136        };
137        tracing::info!(task_model = %efficient, alternate_model = %capable, spillover, reason,
138            "queue routing decision");
139        if spillover {
140            outcome.selected_model_ids.retain(|id| id != capable);
141            outcome.selected_model_ids.insert(0, capable.clone());
142            outcome.request.llm_request.model = Some(capable.to_string());
143        }
144        Ok(spillover)
145    }
146}

148#[cfg(test)]
149mod tests {
150    use super::*;
151    use switchyard_protocol::{ModelId, Request, WorkerLoad, text_request};
152
153    fn config() -> InfraPolicyConfig {
154        InfraPolicyConfig {
155            max_signal_age_ms: 1000,
156            min_waiting_requests: 4,
157            max_capable_inflight: 2,
158            efficient_workers: ["a".into(), "b".into()].into(),
159            capable_workers: ["c".into()].into(),
160        }
161    }
162
163    #[test]
164    fn queue_policy_requires_all_workers_and_preserves_task_choices() {
165        for case in [
166            "busy",
167            "one_quiet",
168            "capable_queued",
169            "capable_running",
170            "local_limit",
171            "missing",
172            "stale",
173            "future",
174            "below_threshold",
175            "at_threshold",
176            "capable_task",
177        ] {
178            let model = |ids: &[&str], waiting| ModelInfraSignals {
179                workers: ids
180                    .iter()
181                    .map(|id| {
182                        (
183                            id.to_string(),
184                            WorkerLoad {
185                                sampled_at: Instant::now(),
186                                running: 0,
187                                waiting,
188                            },
189                        )
190                    })
191                    .collect(),
192                ..Default::default()
193            };
194            let mut source = model(&["a", "b"], 8);
195            let mut destination = model(&["c"], 0);
196            match case {
197                "one_quiet" => source.workers.get_mut("b").unwrap().waiting = 0,
198                "capable_queued" => destination.workers.get_mut("c").unwrap().waiting = 1,
199                "capable_running" => destination.workers.get_mut("c").unwrap().running = 2,
200                "local_limit" => destination.local_inflight = 2,
201                "missing" => {
202                    source.workers.remove("a");
203                }
204                "stale" => {
205                    source.workers.get_mut("a").unwrap().sampled_at -= Duration::from_secs(2)
206                }
207                "future" => {
208                    destination.workers.get_mut("c").unwrap().sampled_at += Duration::from_secs(2)
209                }
210                "below_threshold" => source.workers.get_mut("b").unwrap().waiting = 3,
211                "at_threshold" => source.workers.values_mut().for_each(|w| w.waiting = 4),
212                _ => {}
213            }
214            let signals = [
215                (ModelId::from("small"), source),
216                (ModelId::from("large"), destination),
217            ]
218            .into();
219            let models = RuntimeModels::new(
220                [
221                    (Category::Efficient, vec!["small".into()]),
222                    (Category::Capable, vec!["large".into()]),
223                ]
224                .into(),
225            );
226            let request = Request {
227                llm_request: text_request(None, "hello"),
228                ..Default::default()
229            };
230            let mut outcome = RoutingOutcome::route_to(
231                if case == "capable_task" {
232                    "large"
233                } else {
234                    "small"
235                }
236                .into(),
237                vec![],
238                request.clone(),
239            );
240            let moved = InfraPolicy::new(config())
241                .unwrap()
242                .apply(&mut outcome, &models, &signals)
243                .unwrap();
244            assert_eq!(moved, matches!(case, "busy" | "at_threshold"), "{case}");
245            let expected = if matches!(case, "busy" | "at_threshold" | "capable_task") {
246                "large"
247            } else {
248                "small"
249            };
250            assert_eq!(
251                outcome.selected_model_id().unwrap().as_str(),
252                expected,
253                "{case}"
254            );
255            outcome.request.llm_request.model = None;
256            assert_eq!(outcome.request.llm_request, request.llm_request);
257        }
258        let mut invalid = config();
259        invalid.min_waiting_requests = 0;
260        assert!(InfraPolicy::new(invalid).is_err());
261    }
