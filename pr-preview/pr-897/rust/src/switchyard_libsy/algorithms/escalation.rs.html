27const STREAK_KEY: &str = "escalation_streak";
28/// Session-state key holding the category currently being confirmed.
29const CATEGORY_KEY: &str = "escalation_category";
30const STRONG_CALLS_KEY: &str = "escalation_strong_calls";
31const RELEASE_STREAK_KEY: &str = "escalation_release_streak";
32const WEAK_COOLDOWN_KEY: &str = "escalation_weak_cooldown";
33
34fn count(state: &State, key: &str) -> u32 {
35    match state.extra.get(key) {
36        Some(StateValue::Count(n)) => *n,
37        _ => 0,
38    }
39}
40
41fn set_count(state: &mut State, key: &str, value: u32) {
42    state
43        .extra
44        .insert(key.to_string(), StateValue::Count(value));
45}
46
47/// Reads the failure category whose confirmation streak is in progress, if any.
48fn category(state: &State) -> Option<&str> {
49    match state.extra.get(CATEGORY_KEY) {
50        Some(StateValue::String(category)) => Some(category),
51        _ => None,
52    }
53}
54
55/// Clears the confirmation streak together with the category it was confirming.
56fn clear_streak(state: &mut State) {
57    set_count(state, STREAK_KEY, 0);
58    state.extra.remove(CATEGORY_KEY);
59}

61fn assistant_message(response: &AggLlmResponse) -> Message {
62    Message {
63        role: Role::Assistant,
64        content: response
65            .first_output()
66            .map(|output| output.content.clone())
67            .unwrap_or_default(),
68    }
69}

71/// Calls the efficient model, judges its response, and latches to capable once the streak
72/// confirms. Returns the efficient response directly when not escalating so the caller does
73/// not pay for a second model call.
74struct EscalationClassifier {
75    escalation_judge: JudgeClassifier<EscalationJudge, EscalationPolicy>,
76    /// Consecutive escalate verdicts required to latch.
77    confirmations: u32,
78    deescalation: Option<DeescalationPolicy>,
79    missing_session_warning_emitted: AtomicBool,
80}
81
82struct DeescalationPolicy {
83    config: DeescalationConfig,
84    judge: JudgeClassifier<EscalationJudge, EscalationPolicy>,
85}
86
87/// Builds the escalation classifier used by the shared LLM classifier route shell.
88pub(super) fn build_classifier(
89    contract_config: ClassifierContractConfig,
90    config: EscalationJudgeConfig,
91    max_output_tokens: u64,
92) -> Result<Arc<dyn Classifier<State>>> {
93    let confirmations = config.confirmations;
94    let deescalation = match config.deescalation {
95        Some(deescalation) => Some(DeescalationPolicy {
96            config: deescalation,
97            judge: escalation::build_judge(
98                &contract_config,
99                config.clone(),
100                Some(EvaluationPhase::Strong),
101                max_output_tokens,
102            )?,
103        }),
104        None => None,
105    };
106    let is_phase_aware = deescalation.is_some();
107    let classifier: Arc<dyn Classifier<State>> = Arc::new(EscalationClassifier {
108        escalation_judge: escalation::build_judge(
109            &contract_config,
110            config,
111            is_phase_aware.then_some(EvaluationPhase::Efficient),
112            max_output_tokens,
113        )?,
114        confirmations,
115        deescalation,
116        missing_session_warning_emitted: AtomicBool::new(false),
117    });
118    Ok(classifier)
119}

121/// The two tiers an escalation route moves between.
122struct Tiers<'a> {
123    capable: &'a ModelId,
124    efficient: &'a ModelId,
125}
126
127impl EscalationClassifier {
128    async fn review_capable(
129        &self,
130        deescalation: &DeescalationPolicy,
131        state: &mut State,
132        request: &Request,
133        driver: &Driver,
134        tiers: Tiers<'_>,
135        strong_calls: u32,
136    ) -> Result<(Classification, Option<Response>)> {
137        let Tiers { capable, efficient } = tiers;
138        let next_strong_call = strong_calls.saturating_add(1);
139
140        // The call that confirms escalation is the first capable call.
141        if next_strong_call < deescalation.config.strong_min_calls {
142            set_count(state, STRONG_CALLS_KEY, next_strong_call);
143            driver.set_evidence(serde_json::json!({"source": "escalation", "verdict": "latched"}));
144            return Ok((decisive(capable), None));
145        }
146
147        let capable_response = driver
148            .call_model(request.clone(), vec![capable.clone(), efficient.clone()])
149            .await?;
150        if capable_response.served_model() == Some(efficient) {
151            set_count(state, RELEASE_STREAK_KEY, 0);
152            driver.set_evidence(serde_json::json!({"source": "fallback"}));
153            return Ok((decisive(efficient), Some(capable_response)));
154        }
155        let capable_response = match buffer_response(capable.as_str(), capable_response).await {
156            Ok(response) => response,
157            Err(LibsyError::ClientCall {
158                source: LlmClientError::Transport { .. },
159                ..
160            }) => {
161                set_count(state, RELEASE_STREAK_KEY, 0);
162                driver.set_evidence(
163                    serde_json::json!({"source": "fallback", "reason_code": "transport"}),
164                );
165                return Ok((decisive(efficient), None));
166            }
167            Err(error) => return Err(error),
168        };
169        let mut judge_request = request.clone();
170        judge_request
171            .llm_request
172            .messages
173            .push(assistant_message(&capable_response.agg));
174
175        let (classification, _) = deescalation
176            .judge
177            .score(state, &mut judge_request, driver)
178            .await?;
179        let best = classification.argmax(false)?;
180        let release_streak = match &best {
181            Some(score) if score.target == *efficient => {
182                count(state, RELEASE_STREAK_KEY).saturating_add(1)
183            }
184            _ => 0,
185        };
186        set_count(state, STRONG_CALLS_KEY, next_strong_call);
187        set_count(state, RELEASE_STREAK_KEY, release_streak);
188
189        if release_streak >= deescalation.config.confirmations {
190            clear_streak(state);
191            set_count(state, STRONG_CALLS_KEY, 0);
192            set_count(state, RELEASE_STREAK_KEY, 0);
193            tracing::debug!(
194                target = %efficient,
195                "de-escalation policy released session to efficient tier"
196            );
197        } else if release_streak > 0 {
198            driver.set_evidence(serde_json::json!({"source": "escalation", "verdict": "pending"}));
199        }
200
201        // A confirmed release applies on the next request; this turn is already complete.
202        Ok((decisive(capable), Some(capable_response.into_response())))
203    }
204}

206#[async_trait]
207impl Classifier<State> for EscalationClassifier {
208    async fn score(
209        &self,
210        state: &mut State,
211        request: &mut Request,
212        driver: &Driver,
213    ) -> Result<(Classification, Option<Response>)> {
214        let capable = driver.first_model_for(&Category::Capable)?.clone();
215        let efficient = driver.first_model_for(&Category::Efficient)?.clone();
216
217        let has_session_id = request
218            .metadata
219            .as_ref()
220            .and_then(|metadata| metadata.session_id.as_deref())
221            .is_some_and(|session_id| !session_id.is_empty());
222        if (self.deescalation.is_some() || self.confirmations > 1)
223            && !has_session_id
224            && !self
225                .missing_session_warning_emitted
226                .swap(true, Ordering::Relaxed)
227        {
228            tracing::warn!(
229                target: "libsy",
230                confirmations = self.confirmations,
231                deescalation = self.deescalation.is_some(),
232                required_header = "x-switchyard-session-id",
233                "stateful escalation has no session ID; routing state will not persist"
234            );
235        }
236
237        let mut strong_calls = count(state, STRONG_CALLS_KEY);
238        if let Some(deescalation) = &self.deescalation
239            && deescalation
240                .config
241                .strong_max_calls
242                .is_some_and(|strong_max_calls| strong_calls >= strong_max_calls)
243        {
244            clear_streak(state);
245            set_count(state, STRONG_CALLS_KEY, 0);
246            set_count(state, RELEASE_STREAK_KEY, 0);
247            set_count(
248                state,
249                WEAK_COOLDOWN_KEY,
250                deescalation.config.weak_cooldown_calls,
251            );
252            strong_calls = 0;
253            tracing::debug!(
254                target = %efficient,
255                "de-escalation policy reached its hard limit and returned to efficient tier"
256            );
257        }
258        if let Some(deescalation) = &self.deescalation
259            && strong_calls > 0
260        {
261            return self
262                .review_capable(
263                    deescalation,
264                    state,
265                    request,
266                    driver,
267                    Tiers {
268                        capable: &capable,
269                        efficient: &efficient,
270                    },
271                    strong_calls,
272                )
273                .await;
274        }
275
276        // A confirmed permanent escalation stays capable without a judge call.
277        if self.deescalation.is_none() && count(state, STREAK_KEY) >= self.confirmations {
278            driver.set_evidence(serde_json::json!({
279                "source": "escalation",
280                "verdict": "latched",
281            }));
282            return Ok((decisive(&capable), None));
283        }
284
285        // Call efficient model and buffer the response so the judge can read it.
286        //
287        // If the efficient model exceeds its context window, fall through to capable. This call
288        // deliberately has one candidate so the classifier sees the efficient model's error.
289        tracing::info!(
290            target = %efficient,
291            "escalation classifier selected efficient tier"
292        );
293        let efficient_response = match driver
294            .call_model(request.clone(), vec![efficient.clone()])
295            .await
296        {
297            Ok(r) => r,
298            Err(LibsyError::ClientCall {
299                source: LlmClientError::ContextWindowExceeded { .. },
300                ..
301            }) => {
302                driver.set_evidence(serde_json::json!({
303                    "source": "fallback",
304                    "reason_code": "context_window",
305                }));
306                return Ok((decisive(&capable), None));
307            }
308            Err(e) => return Err(e),
309        };
310        // The call resolves when its stream handle arrives; context and transport failures can
311        // still occur while buffering.
312        let efficient_response = match buffer_response(efficient.as_str(), efficient_response).await
313        {
314            Ok(response) => response,
315            Err(LibsyError::ClientCall {
316                source: LlmClientError::ContextWindowExceeded { .. },
317                ..
318            }) => {
319                driver.set_evidence(serde_json::json!({
320                    "source": "fallback",
321                    "reason_code": "context_window",
322                }));
323                return Ok((decisive(&capable), None));
324            }
325            Err(LibsyError::ClientCall {
326                source: LlmClientError::Transport { .. },
327                ..
328            }) => {
329                driver.set_evidence(serde_json::json!({
330                    "source": "fallback",
331                    "reason_code": "transport",
332                }));
333                return Ok((decisive(&capable), None));
334            }
335            Err(error) => return Err(error),
336        };
337        // Append the efficient reply so the judge reads this turn's completed trajectory.
338        let mut judge_request = request.clone();
339        judge_request
340            .llm_request
341            .messages
342            .push(assistant_message(&efficient_response.agg));
343
344        let weak_cooldown = count(state, WEAK_COOLDOWN_KEY);
345        if weak_cooldown > 0 {
346            set_count(state, WEAK_COOLDOWN_KEY, weak_cooldown - 1);
347            driver
348                .set_evidence(serde_json::json!({"source": "deescalation", "verdict": "cooldown"}));
349            return Ok((
350                decisive(&efficient),
351                Some(efficient_response.into_response()),
352            ));
353        }
354
355        let judge_models = driver.models_for(&Category::Judge);
356        if judge_models.is_empty() {
357            return Err(LibsyError::AlgorithmError {
358                message: "no models available for category Judge".to_string(),
359            });
360        }
361        let verdict = self
362            .escalation_judge
363            .verdict(state, &judge_request, driver, judge_models)
364            .await;
365
366        let held = count(state, STREAK_KEY);
367        let held_category = category(state).map(str::to_string);
368        let (escalate, pending, pending_category) = match verdict.as_ref() {
369            Some(verdict) => {
370                let category = verdict.category.label();
371                tracing::info!(
372                    escalate = verdict.escalate,
373                    category,
374                    new_evidence = verdict.new_evidence,
375                    "escalation judge verdict"
376                );
377                if verdict.escalate
378                    && verdict.new_evidence
379                    && verdict.category != EscalationCategory::None
380                {
381                    let next = if held_category.as_deref() == Some(category) {
382                        held.saturating_add(1)
383                    } else {
384                        1
385                    };
386                    (true, next, Some(category.to_string()))
387                } else {
388                    (false, 0, None)
389                }
390            }
391            None => (false, held, held_category),
392        };
393        set_count(state, STREAK_KEY, pending);
394        if let Some(category) = pending_category {
395            state
396                .extra
397                .insert(CATEGORY_KEY.to_string(), StateValue::String(category));
398        } else {
399            state.extra.remove(CATEGORY_KEY);
400        }
401
402        if escalate && pending >= self.confirmations {
403            driver.set_evidence(serde_json::json!({
404                "source": "escalation",
405                "verdict": "escalate",
406            }));
407            if self.deescalation.is_some() {
408                set_count(state, STRONG_CALLS_KEY, 1);
409                set_count(state, RELEASE_STREAK_KEY, 0);
410            }
411            return Ok((decisive(&capable), None));
412        }
413
414        if escalate {
415            driver.set_evidence(serde_json::json!({
416                "source": "escalation",
417                "verdict": "pending",
418            }));
419        } else if verdict.is_some() {
420            driver.set_evidence(serde_json::json!({
421                "source": "escalation",
422                "verdict": "continue",
423            }));
424        }
425
426        Ok((
427            decisive(&efficient),
428            Some(efficient_response.into_response()),
429        ))
430    }
431}

449    /// A queue of replies, drained in order.
450    struct Queue(Mutex<VecDeque<String>>);
451
452    impl Queue {
453        fn new(replies: impl IntoIterator<Item = &'static str>) -> Arc<Self> {
454            Arc::new(Self(Mutex::new(
455                replies.into_iter().map(String::from).collect(),
456            )))
457        }
458
459        fn take(&self) -> String {
460            self.0
461                .lock()
462                .pop_front()
463                .unwrap_or_else(|| "unexpected call".to_string())
464        }
465    }

467    /// Serves the judge and model targets from separate reply queues.
468    fn queued(model: Arc<Queue>, judge: Arc<Queue>) -> impl Serve {
469        move |target: ModelId, request: Request| {

471                Arc::clone(&judge)
472            } else {
473                Arc::clone(&model)

475            async move {
476                Ok(Response {
477                    llm_response: LlmResponse::Agg(text_response(None, queue.take())),
478                    metadata: request.metadata,
479                    upstream_headers: http::HeaderMap::new(),
480                })
481            }
482        }
483    }
