4//! Minimal SSE frame parser for decoding streamed provider responses.
5//!
6//! One copy backs all neutral-IR stream decoding ([`decode_stream`](crate::decode_stream)).
7//! Errors are boxed `std::error::Error`s — the item error type of a streamed
8//! response — so this module stays free of any HTTP client or server types.
9
10use serde_json::Value;
11
12/// Boxed, thread-safe error carried by a streamed item.
13pub(crate) type BoxError = Box<dyn std::error::Error + Send + Sync>;
14
15/// Parsed contents of one SSE frame.
16pub(crate) enum SseFrame {
17    Empty,
18    Done,
19    Data(Value),
20}
21
22/// The optional terminal SSE marker accepted for all supported wire formats.
23/// Native Anthropic streams can end at `message_stop`, while compatible
24/// providers may additionally send `[DONE]`.
25pub(crate) const fn done_marker() -> &'static str {
26    "[DONE]"
27}
28
29/// Value of a `data` field line; the space after the colon is optional framing.
30fn data_field_value(line: &str) -> Option<String> {
31    let value = match line.split_once(':') {
32        Some(("data", value)) => value,
33        None if line == "data" => "",
34        _ => return None,
35    };
36    Some(value.strip_prefix(' ').unwrap_or(value).to_string())
37}
38
39pub(crate) fn parse_json_sse_frame(frame: &str, done_marker: &str) -> Result<SseFrame, BoxError> {
40    let data = frame
41        .lines()
42        .filter(|line| !line.is_empty() && !line.starts_with(':'))
43        .filter_map(data_field_value)
44        .fold(String::new(), |mut a, b| {
45            a.reserve(b.len() + 1);
46            a.push_str(&b);
47            a.push('\n');
48            a
49        });
50    let data = data.trim_end();
51
52    if data.is_empty() {
53        return Ok(SseFrame::Empty);
54    }
55    if data == done_marker {
56        return Ok(SseFrame::Done);
57    }
58    let value = serde_json::from_str::<Value>(data)?;
59    Ok(SseFrame::Data(value))
60}

62#[cfg(test)]
63mod tests {
64    use serde_json::json;
65
66    use super::*;
67
68    const DONE: &str = done_marker();
69
70    #[test]
71    fn parses_a_data_line_as_json() -> Result<(), BoxError> {
72        let SseFrame::Data(value) = parse_json_sse_frame("data: {\"text\":\"hi\"}\n", DONE)? else {
73            return Err("expected a payload".into());
74        };
75        assert_eq!(value, json!({"text": "hi"}));
76        Ok(())
77    }
78
79    #[test]
80    fn ignores_comment_and_non_data_fields() -> Result<(), BoxError> {
81        // Only `data:` fields contribute; comments (`:`) and other fields (`event:`) are dropped.
82        let frame = ": keep-alive\nevent: message\ndata: {\"n\":1}\n";
83        let SseFrame::Data(value) = parse_json_sse_frame(frame, DONE)? else {
84            return Err("expected a payload".into());
85        };
86        assert_eq!(value, json!({"n": 1}));
87        Ok(())
88    }
89
90    #[test]
91    fn parses_a_data_line_without_a_space_after_the_colon() -> Result<(), BoxError> {
92        // The space after `data:` is optional framing, not part of the value.
93        let SseFrame::Data(value) = parse_json_sse_frame("data:{\"text\":\"hi\"}\n", DONE)? else {
94            return Err("expected a payload".into());
95        };
96        assert_eq!(value, json!({"text": "hi"}));
97        Ok(())
98    }
99
100    #[test]
101    fn done_marker_is_recognised_without_a_space() -> Result<(), BoxError> {
102        assert!(matches!(
103            parse_json_sse_frame("data:[DONE]\n", DONE)?,
104            SseFrame::Done
105        ));
106        Ok(())
107    }
108
109    #[test]
110    fn field_names_are_matched_exactly() -> Result<(), BoxError> {
111        // `database:` must not be read as a `data` field.
112        assert!(matches!(
113            parse_json_sse_frame("database: {\"n\":1}\n", DONE)?,
114            SseFrame::Empty
115        ));
116        Ok(())
117    }
118
119    #[test]
120    fn done_marker_yields_no_payload() -> Result<(), BoxError> {
121        assert!(matches!(
122            parse_json_sse_frame("data: [DONE]\n", DONE)?,
123            SseFrame::Done
124        ));
125        Ok(())
126    }
127
128    #[test]
129    fn frame_without_data_yields_no_payload() -> Result<(), BoxError> {
130        // A comment-only frame carries no data payload.
131        assert!(matches!(
132            parse_json_sse_frame(": keep-alive\n", DONE)?,
133            SseFrame::Empty
134        ));
135        // An empty frame likewise decodes to nothing.
136        assert!(matches!(parse_json_sse_frame("", DONE)?, SseFrame::Empty));
137        Ok(())
138    }
139
140    #[test]
141    fn invalid_json_is_an_error() {
142        assert!(parse_json_sse_frame("data: {not json}\n", DONE).is_err());
143    }
144}
