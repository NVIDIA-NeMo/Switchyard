4//! Initialize and reuse one Dynamo download pool per target.
5
6use std::sync::Arc;
7
8use dynamo_multimodal::fetch::{FetchPolicy, MediaFetcher};
9use tokio::sync::Semaphore;
10
11use super::{MediaConfig, MediaError, MediaProcessor, Result};
12
13impl MediaProcessor {
14    pub(crate) fn new(config: MediaConfig, slots: Arc<Semaphore>) -> Result<Self> {
15        let mut policy = FetchPolicy::default();
16        policy.allow_direct_ip = true;
17        policy.allow_direct_port = true;
18        policy.max_bytes = config.max_input_bytes as u64;
19        // Dynamo caps the encoded payload; our setting counts decoded bytes.
20        policy.max_data_url_bytes = config.max_input_bytes.div_ceil(3) * 4;
21        // Preparation bounds the whole call, including queueing and decoding.
22        policy.timeout = None;
23        let fetcher = MediaFetcher::new(policy)
24            .map_err(|_| MediaError::Invalid("cannot create media fetcher"))?;
25        Ok(Self {
26            config,
27            fetcher,
28            slots,
29        })
30    }
31
32    pub(super) async fn load(&self, source: String) -> Result<Vec<u8>> {
33        // Dynamo errors may contain URLs or hostnames. Keep request data out of errors.
34        let bytes =
35            self.fetcher.fetch(source).await.map_err(|_| {
36                MediaError::Invalid("media fetch failed or exceeds max_input_bytes")
37            })?;
38        if bytes.len() > self.config.max_input_bytes {
39            return Err(MediaError::Invalid("media exceeds max_input_bytes"));
40        }
41        Ok(bytes)
42    }
43}

45#[cfg(test)]
46mod tests {
47    use super::*;
48
49    #[tokio::test]
50    async fn enforces_target_limits_and_redacts_errors() {
51        let slots = Arc::new(Semaphore::new(2));
52        let target = |max_input_bytes| {
53            MediaProcessor::new(
54                MediaConfig {
55                    max_input_bytes,
56                    ..Default::default()
57                },
58                slots.clone(),
59            )
60            .unwrap()
61        };
62        let fetcher = target(3);
63        let source = "data:image/png;base64,AQID";
64        assert_eq!(fetcher.load(source.into()).await.unwrap(), [1, 2, 3]);
65        for limit in [1, 2] {
66            assert!(target(limit).load(source.into()).await.is_err());
67        }
68        assert_eq!(fetcher.load(source.into()).await.unwrap(), [1, 2, 3]);
69        let fetcher = target(100);
70        for source in [
71            "file:///tmp/secret",
72            "https://127.0.0.1/secret",
73            "https://[::1]/secret",
74            "https://169.254.169.254/secret",
75            "https://user:secret@example.com/a",
76            "https://metadata.google.internal./secret",
77            "data:image/png,secret",
78        ] {
79            let error = fetcher.load(source.into()).await.unwrap_err();
80            assert_eq!(
81                error.to_string(),
82                "media fetch failed or exceeds max_input_bytes"
83            );
84        }
85    }
