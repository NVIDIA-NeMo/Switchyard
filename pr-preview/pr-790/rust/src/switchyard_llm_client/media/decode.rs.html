25const MAX_DIMENSION: u32 = 16_384;
26const MAX_ALLOC: u64 = 128 * 1024 * 1024;

28type EncodedFrames = Vec<(f64, Vec<u8>)>;

30pub(crate) async fn resize(
31    bytes: Vec<u8>,
32    edge: u32,
33    slots: Arc<Semaphore>,
34) -> Result<(&'static str, Vec<u8>)> {
35    let permit = slots
36        .acquire_owned()
37        .await
38        .map_err(|_| MediaError::Invalid("media worker closed"))?;
39    tokio::task::spawn_blocking(move || {
40        // Keep the permit in the worker even if the caller cancels its wait.
41        let _permit = permit;
42        // ImageReader limits adapted from Dynamo (Apache-2.0):
43        // https://github.com/ai-dynamo/dynamo/blob/c3deae7507717409d4d1ff0d6f7e575180bd03d7/lib/llm/src/preprocessor/media/decoders/image/backends/image_reader.rs
44        let mut reader = ImageReader::new(Cursor::new(&bytes))
45            .with_guessed_format()
46            .map_err(image::ImageError::IoError)?;
47        let mut limits = image::Limits::default();
48        limits.max_image_width = Some(MAX_DIMENSION);
49        limits.max_image_height = Some(MAX_DIMENSION);
50        limits.max_alloc = Some(MAX_ALLOC);
51        reader.limits(limits);
52        let format = reader
53            .format()
54            .ok_or(MediaError::Invalid("unsupported image format"))?;
55        let mut decoder = reader.into_decoder()?;
56        if decoder.total_bytes() > MAX_ALLOC {
57            return Err(MediaError::Invalid("image exceeds decode allocation limit"));
58        }
59        let orientation = decoder.orientation()?;
60        let decoded = if format == ImageFormat::Jpeg {
61            drop(decoder);
62            let (rgb, height, width) = decode_rgb_exif_transposed(
63                &bytes,
64                &DecodeLimits {
65                    max_width: Some(MAX_DIMENSION as usize),
66                    max_height: Some(MAX_DIMENSION as usize),
67                    max_pixels: Some(MAX_ALLOC as usize / 3),
68                },
69            )
70            .map_err(|_| MediaError::Invalid("image decoding failed"))?;
71            DynamicImage::ImageRgb8(
72                RgbImage::from_raw(width as u32, height as u32, rgb)
73                    .ok_or(MediaError::Invalid("invalid decoded image dimensions"))?,
74            )
75        } else {
76            // Retain transparency and higher bit depths supported by the image crate.
77            let mut decoded = DynamicImage::from_decoder(decoder)?;
78            decoded.apply_orientation(orientation);
79            decoded
80        };
81        if decoded.width() <= edge
82            && decoded.height() <= edge
83            && orientation == Orientation::NoTransforms
84        {
85            return Ok((format.to_mime_type(), bytes));
86        }
87        let resized = if decoded.width() > edge || decoded.height() > edge {
88            if decoded.color().has_alpha() {
89                // Dynamo's RGB resampler cannot preserve transparency.
90                decoded.resize(edge, edge, image::imageops::FilterType::Triangle)
91            } else {
92                let rgb = decoded.into_rgb8();
93                let (width, height) = rgb.dimensions();
94                let scale = f64::from(edge) / f64::from(width.max(height));
95                let out_width = (f64::from(width) * scale).round().max(1.0) as u32;
96                let out_height = (f64::from(height) * scale).round().max(1.0) as u32;
97                let pixels = resize_lanczos_rgb(
98                    rgb.as_raw(),
99                    height as usize,
100                    width as usize,
101                    out_height as usize,
102                    out_width as usize,
103                );
104                DynamicImage::ImageRgb8(
105                    RgbImage::from_raw(out_width, out_height, pixels)
106                        .ok_or(MediaError::Invalid("invalid resized image dimensions"))?,
107                )
108            }
109        } else {
110            decoded
111        };
112        let mut output = Cursor::new(Vec::new());
113        let mime = if resized.color().has_alpha() {
114            resized.write_to(&mut output, ImageFormat::Png)?;
115            "image/png"
116        } else {
117            image::codecs::jpeg::JpegEncoder::new_with_quality(&mut output, 85)
118                .encode_image(&resized.into_rgb8())?;
119            "image/jpeg"
120        };
121        Ok((mime, output.into_inner()))
122    })
123    .await
124    .map_err(|_| MediaError::Invalid("image worker failed"))?
125}

127pub(crate) async fn frames(
128    bytes: Vec<u8>,
129    count: usize,
130    edge: u32,
131    slots: Arc<Semaphore>,
132    remaining_bytes: usize,
133) -> Result<EncodedFrames> {
134    let permit = slots
135        .acquire_owned()
136        .await
137        .map_err(|_| MediaError::Invalid("media worker closed"))?;
138    tokio::task::spawn_blocking(move || {
139        // Retain the worker permit even if the request deadline cancels the wait.
140        let _permit = permit;
141        decode_video_frames(&bytes, count, edge, remaining_bytes)
142    })
143    .await
144    .map_err(|_| MediaError::Invalid("video worker failed"))?
145}
146
147fn decode_video_frames(
148    bytes: &[u8],
149    count: usize,
150    edge: u32,
151    mut remaining_bytes: usize,
152) -> Result<EncodedFrames> {
153    let mut limits = VideoDecodeLimits::default();
154    limits.max_width = MAX_DIMENSION;
155    limits.max_height = MAX_DIMENSION;
156    limits.max_output_bytes = MAX_ALLOC as usize;
157    let decoded = match probe(bytes).and_then(|info| {
158        // Dynamo's output limit applies after resizing; bound its full-size RGB scratch too.
159        if u64::from(info.width) * u64::from(info.height) > MAX_ALLOC / 3 {
160            return Err(MmError::limit_exceeded(
161                "video frame exceeds decode allocation limit",
162            ));
163        }
164        let indices = sample_indices(info.frame_count, count);
165        let scale = (f64::from(edge) / f64::from(info.width.max(info.height))).min(1.0);
166        let fit = FrameResize {
167            width: (f64::from(info.width) * scale).round().max(1.0) as u32,
168            height: (f64::from(info.height) * scale).round().max(1.0) as u32,
169            filter: Resample::Pil(Filter::Lanczos),
170        };
171        decode_frames(bytes, &indices, Some(&fit), &limits)
172    }) {
173        Ok(frames) => frames,
174        Err(MmError::Unsupported { .. }) => {
175            return Err(MediaError::Invalid(
176                "video codec, format, or decoder is unsupported",
177            ));
178        }
179        Err(MmError::LimitExceeded { .. }) => {
180            return Err(MediaError::Invalid("video exceeds decode allocation limit"));
181        }
182        Err(_) => return Err(MediaError::Invalid("video decoding failed")),
183    };
184    let frames = encode_jpeg_frames(&decoded, 85)
185        .map_err(|_| MediaError::Invalid("video frame encoding failed"))?;
186    frames
187        .into_iter()
188        .map(|frame| {
189            remaining_bytes =
190                remaining_bytes
191                    .checked_sub(frame.bytes.len())
192                    .ok_or(MediaError::Invalid(
193                        "prepared media exceeds max_output_bytes",
194                    ))?;
195            Ok((frame.timestamp_secs, frame.bytes))
196        })
197        .collect::<Result<Vec<_>>>()
198}
199
200fn sample_indices(total: u64, count: usize) -> Vec<u64> {
201    let count = (count as u64).min(total);
202    (0..count)
203        .map(|i| {
204            ((total - 1) as f64 * i as f64 / count.saturating_sub(1).max(1) as f64)
205                .round_ties_even() as u64
206        })
207        .collect()
208}

214    #[test]
215    fn uniform_samples_preserve_endpoints_and_clamp_to_available_frames() {
216        assert_eq!(sample_indices(10, 1), [0]);
217        assert_eq!(sample_indices(10, 3), [0, 4, 9]);
218        assert_eq!(sample_indices(2, 4), [0, 1]);
219    }

221    #[tokio::test]
222    async fn rgb_resize_keeps_thin_images_nonempty_and_does_not_upscale() {
223        let slots = Arc::new(Semaphore::new(1));
224        for (format, width, height, expected) in [ImageFormat::Png, ImageFormat::Jpeg]
225            .into_iter()
226            .flat_map(|format| {
227                [(80, 1, (20, 1)), (1, 80, (1, 20)), (8, 4, (8, 4))]
228                    .into_iter()
229                    .map(move |(w, h, expected)| (format, w, h, expected))
230            })
231        {
232            let image = RgbImage::from_pixel(width, height, image::Rgb([210, 40, 20]));
233            let mut input = Cursor::new(Vec::new());
234            image.write_to(&mut input, format).unwrap();
235            let original = input.into_inner();
236            let (_, output) = resize(original.clone(), 20, slots.clone()).await.unwrap();
237            let resized = image::load_from_memory(&output).unwrap();
238            assert_eq!((resized.width(), resized.height()), expected);
239            let pixel = resized.to_rgb8().get_pixel(0, 0).0;
240            assert!(pixel[0] > 190 && pixel[1] < 60 && pixel[2] < 40);
241            if width <= 20 && height <= 20 {
242                assert_eq!(output, original);
243            }
244        }
245    }

247    #[tokio::test]
248    async fn image_resize_applies_exif() {
249        let slots = Arc::new(Semaphore::new(1));
250        let image = image::RgbImage::from_pixel(80, 40, image::Rgb([255, 0, 0]));
251        let mut jpeg = Cursor::new(Vec::new());
252        image.write_to(&mut jpeg, ImageFormat::Jpeg).unwrap();
253        let mut jpeg = jpeg.into_inner();
254        // EXIF orientation 6: rotate 90 degrees clockwise.
255        jpeg.splice(
256            2..2,
257            [
258                0xff, 0xe1, 0, 34, b'E', b'x', b'i', b'f', 0, 0, b'I', b'I', 42, 0, 8, 0, 0, 0, 1,
259                0, 0x12, 1, 3, 0, 1, 0, 0, 0, 6, 0, 0, 0, 0, 0, 0, 0,
260            ],
261        );
262        let (_, bytes) = resize(jpeg, 40, slots).await.unwrap();
263        let rotated = image::load_from_memory(&bytes).unwrap();
264        assert_eq!((rotated.width(), rotated.height()), (20, 40));
265    }
