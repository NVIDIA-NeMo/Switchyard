4//! Condenses history from the efficient tier into a recorded handoff digest.
5//!
6//! Later requests reuse the digest only when their original history prefix
7//! matches. User updates and tool content are clipped, so the digest is lossy.

24const RECORD_KEY: &str = "vgr_compaction";
25
26#[derive(Serialize, Deserialize)]
27struct Recorded {
28    head: u32,
29    cut: u32,
30    anchor: String,
31    summary: String,
32}
33
34/// Tool steps shown in detail; earlier steps receive short entries.
35const RECENT_STEPS: usize = 3;
36const RECENT_ARGUMENT_BUDGET: usize = 600;
37const RECENT_RESULT_BUDGET: usize = 800;
38const EARLY_ARGUMENT_BUDGET: usize = 100;
39const EARLY_RESULT_BUDGET: usize = 80;
40const EARLY_SECTION_BUDGET: usize = 2_500;
41const NARRATION_BUDGET: usize = 200;
42const RECENT_USER_TEXTS: usize = 2;
43
44/// Condenses everything after the task message and records how, so later
45/// requests of this user turn can be rewritten identically.
46///
47/// Returns `false`, leaving the request untouched, when there is no task
48/// message followed by work from the efficient tier.
49pub(super) fn compact(request: &mut Request, state: &mut State, note: &str) -> bool {
50    let messages = &request.llm_request.messages;
51    let start = turn_start(request);
52    let Some(task) = messages[start..].iter().position(is_task_message) else {
53        return false;
54    };
55    let head = start + task + 1;
56    let cut = messages.len();
57    if cut <= head {
58        return false;
59    }
60    let mut summary = digest(&messages[head..cut]);
61    summary.push_str(note);
62    let anchor = fingerprint(&messages[..cut]);
63    let (Ok(head_count), Ok(cut_count)) = (u32::try_from(head), u32::try_from(cut)) else {
64        return false;
65    };
66    let record = Recorded {
67        head: head_count,
68        cut: cut_count,
69        anchor,
70        summary,
71    };
72    let Ok(serialized) = serde_json::to_string(&record) else {
73        return false;
74    };
75    rewrite(request, head, cut, &record.summary);
76    state
77        .extra
78        .insert(RECORD_KEY.into(), StateValue::String(serialized));
79    true
80}

82/// Reapplies a recorded compaction to a later request of the same user turn.
83///
84/// Registered ahead of every other classifier and never votes.
85pub(super) struct Compactor;
86
87#[async_trait]
88impl Classifier for Compactor {
89    async fn score(
90        &self,
91        state: &mut State,
92        request: &mut Request,
93        _driver: &Driver,
94    ) -> Result<(Classification, Option<Response>)> {
95        if initiated_by_user(request)
96            .unwrap_or_else(|| has_new_user_turn(&request.llm_request.messages))
97        {
98            state.extra.remove(RECORD_KEY);
99        } else if let Some((head, cut, summary)) = recorded(state, request) {
100            rewrite(request, head, cut, &summary);
101        }
102        Ok((Classification::Scores(Vec::new()), None))
103    }
104}
105
106/// The recorded compaction, when the request still begins with the condensed history.
107fn recorded(state: &State, request: &Request) -> Option<(usize, usize, String)> {
108    let StateValue::String(serialized) = state.extra.get(RECORD_KEY)? else {
109        return None;
110    };
111    let record: Recorded = serde_json::from_str(serialized).ok()?;
112    let head = usize::try_from(record.head).ok()?;
113    let cut = usize::try_from(record.cut).ok()?;
114    let messages = &request.llm_request.messages;
115    (head > 0
116        && head < cut
117        && cut <= messages.len()
118        && fingerprint(&messages[..cut]) == record.anchor)
119        .then(|| (head, cut, record.summary.clone()))
120}
121
122fn rewrite(request: &mut Request, head: usize, cut: usize, summary: &str) {
123    let messages = &mut request.llm_request.messages;
124    messages[head - 1].content.push(ContentBlock::Text {
125        text: summary.to_string(),
126    });
127    messages.drain(head..cut);
128    drop_exact_replay(request);
129}
130
131fn is_task_message(message: &Message) -> bool {
132    message.role == Role::User
133        && message
134            .content
135            .iter()
136            .any(|block| matches!(block, ContentBlock::Text { .. }))
137}
138
139fn fingerprint(messages: &[Message]) -> String {
140    let mut hasher = DefaultHasher::new();
141    serde_json::to_string(messages)
142        .unwrap_or_default()
143        .hash(&mut hasher);
144    format!("{:016x}", hasher.finish())
145}

147/// One tool call and what came back, in conversation order.
148struct Step<'a> {
149    narration: Vec<&'a str>,
150    call: &'a ToolCall,
151    result: Option<&'a ToolResult>,
152}
153
154fn result_text(result: &ToolResult) -> String {
155    result
156        .content
157        .iter()
158        .map(|part| match part {
159            ContentBlock::Text { text } => text.as_str(),
160            _ => "[non-text content]",
161        })
162        .collect::<Vec<_>>()
163        .join("\n")
164}
165
166fn digest(messages: &[Message]) -> String {
167    let mut steps = Vec::<Step<'_>>::new();
168    let mut narration = Vec::new();
169    let mut user_text = Vec::new();
170    for message in messages {
171        for block in &message.content {
172            match block {
173                ContentBlock::Text { text } if message.role == Role::Assistant => {
174                    narration.push(text.as_str());
175                }
176                ContentBlock::Text { text } if message.role == Role::User => {
177                    user_text.push(clip_mid(text, RECENT_RESULT_BUDGET, 0.5));
178                }
179                ContentBlock::ToolCall(call) => steps.push(Step {
180                    narration: std::mem::take(&mut narration),
181                    call,
182                    result: None,
183                }),
184                ContentBlock::ToolResult(result) => {
185                    if let Some(step) = steps
186                        .iter_mut()
187                        .rev()
188                        .find(|step| step.call.id == result.tool_call_id && step.result.is_none())
189                    {
190                        step.result = Some(result);
191                    }
192                }
193                _ => {}
194            }
195        }
196    }
197
198    let recent_from = steps.len().saturating_sub(RECENT_STEPS);
199    let mut early = String::new();
200    for (index, step) in steps[..recent_from].iter().enumerate() {
201        let body = step.result.map(result_text).unwrap_or_default();
202        let status = match step.result {
203            Some(result) if result.is_error == Some(true) => "error",
204            Some(_) => "ok",
205            None => "no result",
206        };
207        let first_line = body
208            .lines()
209            .find(|line| !line.trim().is_empty())
210            .unwrap_or("");
211        early.push_str(&format!(
212            "[{}] {} {} -> {status}: {}\n",
213            index + 1,
214            step.call.name,
215            clip_mid(&step.call.arguments.to_string(), EARLY_ARGUMENT_BUDGET, 0.7),
216            clip_mid(first_line, EARLY_RESULT_BUDGET, 0.7),
217        ));
218    }
219    let mut out = String::from(
220        "\n\n[Context note from the serving infrastructure, not from the user] The tool-using \
221         work that followed this message was condensed into the digest below. Earlier steps \
222         are one line each; the most recent steps are shown in detail.\n<condensed_history>\n",
223    );
224    out.push_str(&clip_mid(&early, EARLY_SECTION_BUDGET, 0.3));
225    for (offset, step) in steps[recent_from..].iter().enumerate() {
226        let narration = step.narration.concat();
227        if !narration.trim().is_empty() {
228            out.push_str(&format!(
229                "assistant: {}\n",
230                clip_mid(narration.trim(), NARRATION_BUDGET, 0.5)
231            ));
232        }
233        out.push_str(&format!(
234            "[{}] {} {}\n",
235            recent_from + offset + 1,
236            step.call.name,
237            clip_mid(
238                &step.call.arguments.to_string(),
239                RECENT_ARGUMENT_BUDGET,
240                0.5
241            )
242        ));
243        match step.result {
244            Some(result) => out.push_str(&format!(
245                "result{}:\n{}\n",
246                if result.is_error == Some(true) {
247                    " (error)"
248                } else {
249                    ""
250                },
251                clip_mid(&result_text(result), RECENT_RESULT_BUDGET, 0.3)
252            )),
253            None => out.push_str("result: none\n"),
254        }
255    }
256    let narration = narration.concat();
257    if !narration.trim().is_empty() {
258        out.push_str(&format!(
259            "assistant: {}\n",
260            clip_mid(narration.trim(), NARRATION_BUDGET, 0.5)
261        ));
262    }
263    for text in &user_text[user_text.len().saturating_sub(RECENT_USER_TEXTS)..] {
264        out.push_str(&format!("user: {text}\n"));
265    }
266    out.push_str("</condensed_history>");
267    out
268}

275    fn call(id: &str, command: &str) -> Message {
276        Message {
277            role: Role::Assistant,
278            content: vec![ContentBlock::ToolCall(ToolCall {
279                id: id.into(),
280                name: "bash".into(),
281                arguments: serde_json::json!({ "command": command }),
282            })],
283        }
284    }
285
286    fn result(id: &str, text: &str) -> Message {
287        Message {
288            role: Role::Tool,
289            content: vec![ContentBlock::ToolResult(ToolResult {
290                tool_call_id: id.into(),
291                content: vec![ContentBlock::Text { text: text.into() }],
292                is_error: None,
293            })],
294        }
295    }

297    fn session(steps: usize) -> Request {
298        let mut messages = vec![Message::text(Role::User, "build the thing")];
299        for index in 0..steps {
300            let id = format!("call-{index}");
301            messages.push(call(&id, &format!("step {index}")));
302            messages.push(result(&id, &format!("output {index}")));
303        }
304        Request {
305            llm_request: LlmRequest {
306                messages,
307                ..LlmRequest::default()
308            },
309            raw_request: None,
310            metadata: None,
311        }
312    }

314    async fn reapply(state: &mut State, request: &mut Request) {
315        let driver = Driver::new(
316            "test",
317            std::sync::Arc::new(crate::core::algorithm::RuntimeModels::new(
318                std::collections::HashMap::new(),
319            )),
320        )
321        .0;
322        Compactor
323            .score(state, request, &driver)
324            .await
325            .expect("the compactor never fails");
326    }

328    #[tokio::test]
329    async fn later_requests_see_the_same_condensed_prefix() {
330        let mut state = State::default();
331        let mut handoff = session(12);
332        assert!(compact(&mut handoff, &mut state, "NOTE"));
333        assert_eq!(handoff.llm_request.messages.len(), 1);
334        let condensed = handoff.llm_request.messages[0].clone();
335        let digest = condensed.text_content("").expect("task text");
336        assert!(digest.contains("step 11") && digest.ends_with("NOTE"));
337
338        let saved = serde_json::to_string(&state).expect("serializable routing state");
339        let mut state: State = serde_json::from_str(&saved).expect("restored routing state");
340
341        let mut follow_up = session(12);
342        follow_up.llm_request.messages.push(call("capable-1", "ls"));
343        follow_up
344            .llm_request
345            .messages
346            .push(result("capable-1", "a b"));
347        reapply(&mut state, &mut follow_up).await;
348
349        let messages = &follow_up.llm_request.messages;
350        assert_eq!(
351            messages.len(),
352            3,
353            "task, then only the capable tier's own work"
354        );
355        assert_eq!(
356            messages[0], condensed,
357            "the cached prefix is byte-identical"
358        );
359    }

442    fn batched_results_are_paired_by_tool_call_id() {
443        let calls = Message {
444            role: Role::Assistant,
445            content: ["a", "b"]
446                .into_iter()
447                .map(|id| {
448                    ContentBlock::ToolCall(ToolCall {
449                        id: id.into(),
450                        name: format!("tool_{id}"),
451                        arguments: serde_json::json!({}),
452                    })
453                })
454                .collect(),
455        };
456        for order in [["a", "b"], ["b", "a"]] {
457            let results = Message {
458                role: Role::Tool,
459                content: order
460                    .into_iter()
461                    .map(|id| {
462                        ContentBlock::ToolResult(ToolResult {
463                            tool_call_id: id.into(),
464                            content: vec![ContentBlock::Text {
465                                text: format!("output_{id}"),
466                            }],
467                            is_error: Some(id == "b"),
468                        })
469                    })
470                    .collect(),
471            };
472            let view = digest(&[calls.clone(), results]);
473            assert!(view.contains("tool_a {}\nresult:\noutput_a"), "{view}");
474            assert!(
475                view.contains("tool_b {}\nresult (error):\noutput_b"),
476                "{view}"
477            );
478        }
