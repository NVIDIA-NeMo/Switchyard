22/// Capabilities declared for one route.
23///
24/// `GET /v1/models` includes `context_window`, `tool_calling`, and `vision` in each
25/// standard `data` entry, using `null` for unset values. It omits `reasoning`.
26/// The server rejects tool inputs, reasoning controls, or images when the
27/// corresponding declaration is explicitly `false` for the selected route.
28#[derive(Clone, Copy, Default)]
29pub struct ModelCapabilities {
30    pub context_window: Option<u32>,
31    pub tool_calling: Option<bool>,
32    /// Whether the routed model accepts reasoning controls, as declared in config.
33    pub reasoning: Option<bool>,
34    /// Whether the routed model accepts image input, as declared in config.
35    pub vision: Option<bool>,
36}
37
38/// Caller credential family required by a forwarded-auth route.
39#[derive(Clone, Copy, Debug, Eq, PartialEq)]
40pub enum CallerAuthKind {
41    Anthropic,
42    OpenAi,
43}
44
45impl CallerAuthKind {
46    /// Stable provider name used by the server compatibility API.
47    pub const fn as_str(self) -> &'static str {
48        match self {
49            Self::Anthropic => "anthropic",
50            Self::OpenAi => "openai",
51        }
52    }
53
54    const fn accepts(self, wire_format: WireFormat) -> bool {
55        matches!(
56            (self, wire_format),
57            (Self::Anthropic, WireFormat::AnthropicMessages)
58                | (
59                    Self::OpenAi,
60                    WireFormat::OpenAiChat | WireFormat::OpenAiResponses
61                )
62        )
63    }
64}

73/// Error returned while loading or executing configured routes.
74#[derive(Debug, Error)]
75pub enum RunnerError {
76    #[error("{message}")]
77    Configuration {
78        message: String,
79        #[source]
80        source: Option<Box<dyn Error + Send + Sync>>,
81    },
82    #[error("unknown route model {0:?}")]
83    UnknownRouteModel(String),
84    #[error("caller format is incompatible with {} credentials", .0.as_str())]
85    IncompatibleCallerFormat(CallerAuthKind),
86    #[error("route has no compatible target for the auxiliary operation")]
87    AuxiliaryUnsupported,
88    #[error(transparent)]
89    Algorithm(#[from] LibsyError),
90    #[error(transparent)]
91    Client(#[from] LlmClientError),
92}
93
94impl RunnerError {
95    pub(crate) fn configuration(message: impl Into<String>) -> Self {
96        Self::Configuration {
97            message: message.into(),
98            source: None,
99        }
100    }
101
102    pub(crate) fn configuration_source(
103        message: impl Into<String>,
104        source: impl Error + Send + Sync + 'static,
105    ) -> Self {
106        Self::Configuration {
107            message: message.into(),
108            source: Some(Box::new(source)),
109        }
110    }
111}

113// Configured algorithm and targets for one execution lane.
114pub(crate) struct ExecutionLane {
115    algorithm: Arc<dyn Algorithm>,
116    // Resolves each offloaded call to the client configured for the target the algorithm
117    // selected. A route is a synthetic model with no upstream of its own, so this is a
118    // per-target lookup, never one client serving the whole route.
119    clients: ClientRouter,
120    anthropic_auxiliary_target: Option<AuxiliaryTarget>,
121    responses_auxiliary_target: Option<AuxiliaryTarget>,
122    models: Arc<RuntimeModels>,
123}

134            algorithm,

212    pub(crate) fn with_eligibility(
213        mut self,

221    fn select_lane(&self, request: &Request) -> Result<LaneSelection, RunnerError> {
222        if let Some(eligibility) = &self.eligibility {
223            let required = eligibility.policy.admit(request)?;
224            let lane = if required.required_labels().is_empty() {
225                Arc::clone(&self.standard)
226            } else {
227                eligibility
228                    .lanes
229                    .select(required.required_labels(), &self.standard)?
230            };
231            let requirements = Some(required);
232            tracing::debug!(
233                switchyard.route = request.llm_request.model.as_deref().unwrap_or_default(),
234                switchyard.algorithm = self.algorithm_name(),
235                privacy.lane = requirements.as_ref().map(TargetRequirements::lane),
236                privacy.source = requirements.as_ref().map(TargetRequirements::source),
237                privacy.reason_code = requirements.as_ref().map(TargetRequirements::reason_code),
238                "eligible targets selected"
239            );
240            return Ok(LaneSelection { lane, requirements });
241        }
242        if has_target_requirements(request) {
243            return Err(LlmClientError::InvalidRequest {
244                message: "route does not accept target requirements".to_string(),
245            }
246            .into());
247        }
248        Ok(LaneSelection {
249            lane: Arc::clone(&self.standard),
250            requirements: None,
251        })
252    }

282    /// Rejects a caller format incompatible with forwarded credentials.
283    pub fn check_caller_format(&self, input_format: WireFormat) -> Result<(), RunnerError> {
284        if let Some(kind) = self.caller_auth
285            && !kind.accepts(input_format)
286        {
287            return Err(RunnerError::IncompatibleCallerFormat(kind));

292    /// Executes the configured route without consuming or proxying streamed responses.
293    pub async fn execute(
294        &self,
295        request: Request,
296        observer: Option<RunObserver>,
297    ) -> Result<RunOutput, RunnerError> {
298        self.execute_with_privacy_observer(request, observer, |_| {})
299            .await
300    }
301
302    /// Executes the route and reports standard or restricted admission before dispatch.
303    pub async fn execute_with_privacy_observer(
304        &self,
305        request: Request,
306        observer: Option<RunObserver>,
307        on_privacy_decision: impl FnOnce(&TargetRequirements) + Send,
308    ) -> Result<RunOutput, RunnerError> {
309        let selection = self.select_lane(&request)?;
310        if let Some(decision) = &selection.requirements {
311            on_privacy_decision(decision);
312        }
313        let lane = selection.lane;
314        let (selected_model, response) = switchyard_llm_client::run(
315            Arc::clone(&lane.algorithm),
316            lane.clients.clone(),
317            request,
318            Arc::clone(&lane.models),
319            observer,
320        )
321        .await?;
322        Ok(RunOutput {
323            selected_model,
324            response,
325        })
326    }

328    /// Completes routing-time calls without serving a post-routing completion.
329    pub async fn decide(&self, request: Request) -> Result<RoutingOutcome, RunnerError> {
330        let selection = self.select_lane(&request)?;
331        let lane = selection.lane;
332        switchyard_llm_client::decide(
333            Arc::clone(&lane.algorithm),
334            lane.clients.clone(),
335            request,
336            Arc::clone(&lane.models),
337        )
338        .await
339        .map_err(RunnerError::from)
340    }

347    /// Apply the optional queue policy to a completed task decision.
348    ///
349    /// The host must hold its in-flight accounting lock through this call and the
350    /// increment for the selected model. Ordinary `decide` leaves task routing unchanged.
351    pub fn apply_infra(
352        &self,
353        outcome: &mut RoutingOutcome,
354        signals: &switchyard_protocol::InfraSignals,
355    ) -> Result<bool, RunnerError> {
356        match &self.infra {
357            Some(policy) => {
358                let selection = self.select_lane(&outcome.request)?;
359                if outcome.selected_model_ids.iter().any(|model| {
360                    !selection
361                        .lane
362                        .models
363                        .models_for(&switchyard_protocol::Category::Any)
364                        .contains(model)
365                }) {
366                    return Err(LlmClientError::InvalidRequest {
367                        message: "target requirements changed after the routing decision".into(),
368                    }
369                    .into());
370                }
371                policy
372                    .apply(outcome, &selection.lane.models, signals)
373                    .map_err(Into::into)
374            }
375            None => Ok(false),
376        }
377    }

379    /// Routes with a host-owned client for routing-time LLM calls, such as judges.
380    ///
381    /// The host owns transport, credentials, and target request settings. Configured
382    /// HTTP clients, target prompts, and decision clients are not used on this path.
383    /// The same algorithm instance retains its state across calls. No post-routing
384    /// completion is dispatched. Target eligibility still applies to the host-owned path.
385    pub async fn decide_with_client(
386        &self,
387        request: Request,
388        client: Arc<dyn RoutedLlmClient>,
389    ) -> Result<RoutingOutcome, RunnerError> {
390        let selection = self.select_lane(&request)?;
391        let lane = selection.lane;
392        switchyard_llm_client::decide(
393            Arc::clone(&lane.algorithm),
394            ClientRouter::single(client),
395            request,
396            Arc::clone(&lane.models),
397        )
398        .await
399        .map_err(RunnerError::from)
400    }
401
402    /// Executes a model-bearing provider operation through a compatible target.
403    pub async fn call_auxiliary(
404        &self,
405        request: Request,
406        operation: AuxiliaryOperation,
407    ) -> Result<Value, RunnerError> {
408        let selection = self.select_lane(&request)?;
409        let lane = selection.lane;
410        let target = match operation {
411            AuxiliaryOperation::AnthropicCountTokens => &lane.anthropic_auxiliary_target,
412            AuxiliaryOperation::ResponsesInputTokens | AuxiliaryOperation::ResponsesCompact => {
413                &lane.responses_auxiliary_target
414            }
415        }
416        .as_ref()
417        .ok_or(RunnerError::AuxiliaryUnsupported)?;
418        target
419            .client
420            .call_auxiliary(&target.model, request, operation)
421            .await
422            .map_err(Into::into)
423    }

461    // Different requirements that resolve to the same bindings share algorithm state.
462    #[test]
463    fn resolved_bindings_reuse_the_algorithm_and_standard_setup() {
464        let source = config(POOL);
465        let runner = Runner::from_toml(&source).expect("eligible route");
466        let route = runner.route("route").expect("route");
467        let private = route
468            .select_lane(&request("one", &["private"]))
469            .expect("private");
470        let both = route
471            .select_lane(&request("two", &["private", "customer_data"]))
472            .expect("both labels");
473        assert!(Arc::ptr_eq(&private.lane, &both.lane));
474        assert!(Arc::ptr_eq(&private.lane.algorithm, &both.lane.algorithm));
475        assert!(!Arc::ptr_eq(&private.lane, &route.standard));
476
477        let runner = Runner::from_toml(&source.replace("labels = []", "labels = ['private']"))
478            .expect("standard models already approved");
479        let route = runner.route("route").expect("route");
480        let approved = route
481            .select_lane(&request("one", &["private"]))
482            .expect("approved");
483        assert!(Arc::ptr_eq(&approved.lane, &route.standard));
484    }

486    // Queue spillover cannot introduce targets outside this request's approved pool.
487    #[test]
488    fn infra_spillover_stays_within_request_requirements() {
489        let algorithm = "type = 'llm_classifier'\nclassifier_target = 'judge'\n\
490            strong_target = 'capable'\nweak_target = 'fast'\nbase_threshold = 0.5";
491        let source = format!(
492            "{}\n[routes.test.infra]\nmax_signal_age_ms = 1000\n\
493             min_waiting_requests = 4\nmax_capable_inflight = 2\n",
494            config(algorithm).replace("labels = []", "labels = ['private']")
495        );
496        let runner = Runner::from_toml(&source).expect("eligible queue policy");
497        let route = runner.route("route").expect("route");
498        let signals: InfraSignals = [("fast", 8), ("capable", 0)]
499            .into_iter()
500            .map(|(model, waiting)| {
501                (
502                    model.into(),
503                    ModelInfraSignals {
504                        load: Some(PoolLoad {
505                            sampled_at: std::time::Instant::now(),
506                            min_waiting: waiting,
507                            max_waiting: waiting,
508                            running: 0,
509                        }),
510                        ..Default::default()
511                    },
512                )
513            })
514            .collect();
515
516        let mut stale =
517            RoutingOutcome::route_to("outside".into(), vec![], request("request", &["private"]));
518        let error = route
519            .apply_infra(&mut stale, &signals)
520            .expect_err("stale decision");
521        assert!(
522            error.to_string().contains("requirements changed"),
523            "{error}"
524        );
525
526        for labels in [&[] as &[&str], &["private"]] {
527            let mut outcome =
528                RoutingOutcome::route_to("fast".into(), vec![], request("request", labels));
529            assert!(
530                route
531                    .apply_infra(&mut outcome, &signals)
532                    .expect("approved spillover")
533            );
534            assert_eq!(
535                outcome.selected_model_id().expect("selected model"),
536                "capable"
537            );
538        }
539    }
