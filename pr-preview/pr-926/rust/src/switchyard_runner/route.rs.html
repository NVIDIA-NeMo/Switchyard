23/// Capabilities declared for one route.
24///
25/// `GET /v1/models` includes `context_window`, `tool_calling`, and `vision` in each
26/// standard `data` entry, using `null` for unset values. It omits `reasoning`.
27/// The server rejects tool inputs, reasoning controls, or images when the
28/// corresponding declaration is explicitly `false` for the selected route.
29#[derive(Clone, Copy, Default)]
30pub struct ModelCapabilities {
31    pub context_window: Option<u32>,
32    pub tool_calling: Option<bool>,
33    /// Whether the routed model accepts reasoning controls, as declared in config.
34    pub reasoning: Option<bool>,
35    /// Whether the routed model accepts image input, as declared in config.
36    pub vision: Option<bool>,
37}

39/// Caller credential family required by a forwarded-auth route.
40#[derive(Clone, Copy, Debug, Eq, PartialEq)]
41pub enum CallerAuthKind {
42    Anthropic,
43    OpenAi,
44}
45
46impl CallerAuthKind {
47    /// Stable provider name used by the server compatibility API.
48    pub const fn as_str(self) -> &'static str {
49        match self {
50            Self::Anthropic => "anthropic",
51            Self::OpenAi => "openai",
52        }
53    }
54
55    const fn accepts(self, wire_format: WireFormat) -> bool {
56        matches!(
57            (self, wire_format),
58            (Self::Anthropic, WireFormat::AnthropicMessages)
59                | (
60                    Self::OpenAi,
61                    WireFormat::OpenAiChat | WireFormat::OpenAiResponses
62                )
63        )
64    }
65}

74/// Error returned while loading or executing configured routes.
75#[derive(Debug, Error)]
76pub enum RunnerError {
77    #[error("{message}")]
78    Configuration {
79        message: String,
80        #[source]
81        source: Option<Box<dyn Error + Send + Sync>>,
82    },
83    #[error("unknown route model {0:?}")]
84    UnknownRouteModel(String),
85    #[error("caller format is incompatible with {} credentials", .0.as_str())]
86    IncompatibleCallerFormat(CallerAuthKind),
87    #[error("route has no compatible target for the auxiliary operation")]
88    AuxiliaryUnsupported,
89    #[error(transparent)]
90    Algorithm(#[from] LibsyError),
91    #[error(transparent)]
92    Client(#[from] LlmClientError),
93}
94
95impl RunnerError {
96    pub(crate) fn configuration(message: impl Into<String>) -> Self {
97        Self::Configuration {
98            message: message.into(),
99            source: None,
100        }
101    }
102
103    pub(crate) fn configuration_source(
104        message: impl Into<String>,
105        source: impl Error + Send + Sync + 'static,
106    ) -> Self {
107        Self::Configuration {
108            message: message.into(),
109            source: Some(Box::new(source)),
110        }
111    }
112}

114/// A configured algorithm and the per-target clients its calls resolve through.
115pub(crate) struct ExecutionLane {
116    algorithm: Arc<dyn Algorithm>,
117    // Resolves each offloaded call to the client configured for the target the algorithm
118    // selected. A route is a synthetic model with no upstream of its own, so this is a
119    // per-target lookup, never one client serving the whole route.
120    clients: ClientRouter,
121    anthropic_auxiliary_target: Option<AuxiliaryTarget>,
122    responses_auxiliary_target: Option<AuxiliaryTarget>,
123    models: Arc<RuntimeModels>,
124}

128        algorithm: Arc<dyn Algorithm>,
129        clients: ClientRouter,
130        anthropic_auxiliary_target: Option<AuxiliaryTarget>,

132        models: RuntimeModels,
133    ) -> Self {
134        Self {
135            algorithm,
136            clients,
137            anthropic_auxiliary_target,
138            responses_auxiliary_target,
139            models: Arc::new(models),
140        }
141    }

154/// A configured route with optional target eligibility checks.
155#[derive(Clone)]
156pub struct Route {
157    standard: Arc<ExecutionLane>,
158    eligibility: Option<Arc<EligibilityExecution>>,
159    targets: Arc<HashMap<ModelId, DecisionTarget>>,
160    caller_auth: Option<CallerAuthKind>,

165/// The selected model and untouched response produced by a route execution.
166pub struct RunOutput {
167    pub selected_model: ModelId,
168    pub response: Response,
169}

171impl Route {
172    /// Creates a fully configured execution route.
173    #[allow(clippy::too_many_arguments)]
174    pub fn new(
175        algorithm: Arc<dyn Algorithm>,
176        clients: ClientRouter,
177        caller_auth: Option<CallerAuthKind>,
178        capabilities: ModelCapabilities,
179        anthropic_auxiliary_target: Option<AuxiliaryTarget>,
180        responses_auxiliary_target: Option<AuxiliaryTarget>,
181        targets: Vec<DecisionTarget>,
182        models: RuntimeModels,
183    ) -> Self {
184        let models = models.with_provider_models(
185            targets
186                .iter()
187                .map(|target| (ModelId::from(target.target.as_str()), target.model.clone()))
188                .collect(),
189        );
190        Self::from_lane(
191            ExecutionLane::new(
192                algorithm,
193                clients,
194                anthropic_auxiliary_target,
195                responses_auxiliary_target,
196                models,
197            ),
198            caller_auth,
199            capabilities,
200            targets,
201        )
202    }
203
204    pub(crate) fn from_lane(
205        lane: ExecutionLane,
206        caller_auth: Option<CallerAuthKind>,
207        capabilities: ModelCapabilities,
208        targets: Vec<DecisionTarget>,
209    ) -> Self {
210        let targets = targets
211            .into_iter()
212            .map(|target| (ModelId::from(target.target.as_str()), target))
213            .collect();
214        Self {
215            standard: Arc::new(lane),
216            eligibility: None,
217            targets: Arc::new(targets),
218            caller_auth,
219            capabilities,
220            infra: None,
221        }
222    }
223
224    pub(crate) fn with_eligibility(
225        mut self,
226        policy: EligibilityPolicy,
227        lanes: EligibleLanes,
228    ) -> Self {
229        self.eligibility = Some(Arc::new(EligibilityExecution { policy, lanes }));
230        self
231    }
232
233    fn select_lane(&self, request: &Request) -> Result<LaneSelection, RunnerError> {
234        if let Some(eligibility) = &self.eligibility {
235            let required = eligibility.policy.admit(request)?;
236            let lane = if required.required_labels().is_empty() {
237                Arc::clone(&self.standard)
238            } else {
239                eligibility
240                    .lanes
241                    .select(required.required_labels(), &self.standard)?
242            };
243            let requirements = Some(required);
244            tracing::debug!(
245                switchyard.route = request.llm_request.model.as_deref().unwrap_or_default(),
246                switchyard.algorithm = self.algorithm_name(),
247                privacy.lane = requirements.as_ref().map(TargetRequirements::lane),
248                privacy.source = requirements.as_ref().map(TargetRequirements::source),
249                privacy.reason_code = requirements.as_ref().map(TargetRequirements::reason_code),
250                "eligible targets selected"
251            );
252            return Ok(LaneSelection { lane, requirements });
253        }
254        if has_target_requirements(request) {
255            return Err(LlmClientError::InvalidRequest {
256                message: "route does not accept target requirements".to_string(),
257            }
258            .into());
259        }
260        Ok(LaneSelection {
261            lane: Arc::clone(&self.standard),
262            requirements: None,
263        })
264    }

291    /// The provider model that serves `selected`, or `selected` itself when it names no

305    /// Rejects a caller format incompatible with forwarded credentials.
306    pub fn check_caller_format(&self, input_format: WireFormat) -> Result<(), RunnerError> {
307        if let Some(kind) = self.caller_auth
308            && !kind.accepts(input_format)
309        {
310            return Err(RunnerError::IncompatibleCallerFormat(kind));
311        }
312        Ok(())
313    }

315    /// Executes the configured route without consuming or proxying streamed responses.
316    pub async fn execute(
317        &self,
318        request: Request,
319        observer: Option<RunObserver>,
320    ) -> Result<RunOutput, RunnerError> {
321        self.execute_with_privacy_observer(request, observer, |_| {})
322            .await
323    }
324
325    /// Executes the route, reporting admission before dispatch when eligibility is enabled.
326    pub async fn execute_with_privacy_observer(
327        &self,
328        request: Request,
329        observer: Option<RunObserver>,
330        on_privacy_decision: impl FnOnce(&TargetRequirements) + Send,
331    ) -> Result<RunOutput, RunnerError> {
332        let selection = self.select_lane(&request)?;
333        if let Some(decision) = &selection.requirements {
334            on_privacy_decision(decision);
335        }
336        let lane = selection.lane;
337        let (selected_model, response) = switchyard_llm_client::run(
338            Arc::clone(&lane.algorithm),
339            lane.clients.clone(),
340            request,
341            Arc::clone(&lane.models),
342            observer,
343        )
344        .await?;
345        Ok(RunOutput {
346            selected_model,
347            response,
348        })
349    }

351    /// Completes routing-time calls without serving a post-routing completion.
352    pub async fn decide(&self, request: Request) -> Result<RoutingOutcome, RunnerError> {
353        let selection = self.select_lane(&request)?;
354        let lane = selection.lane;
355        switchyard_llm_client::decide(
356            Arc::clone(&lane.algorithm),
357            lane.clients.clone(),
358            request,
359            Arc::clone(&lane.models),
360        )
361        .await
362        .map_err(RunnerError::from)
363    }

370    /// Apply the optional queue policy to a completed task decision.
371    ///
372    /// The host must hold its in-flight accounting lock through this call and the
373    /// increment for the selected model. Ordinary `decide` leaves task routing unchanged.
374    pub fn apply_infra(
375        &self,
376        outcome: &mut RoutingOutcome,
377        signals: &switchyard_protocol::InfraSignals,
378    ) -> Result<bool, RunnerError> {
379        match &self.infra {
380            Some(policy) => {
381                let selection = self.select_lane(&outcome.request)?;
382                if outcome.selected_model_ids.iter().any(|model| {
383                    !selection
384                        .lane
385                        .models
386                        .models_for(&switchyard_protocol::Category::Any)
387                        .contains(model)
388                }) {
389                    return Err(LlmClientError::InvalidRequest {
390                        message: "target requirements changed after the routing decision".into(),
391                    }
392                    .into());
393                }
394                policy
395                    .apply(outcome, &selection.lane.models, signals)
396                    .map_err(Into::into)
397            }
398            None => Ok(false),
399        }
400    }

402    /// Routes with a host-owned client for routing-time LLM calls, such as judges.
403    ///
404    /// The host owns transport, credentials, and target request settings. Configured
405    /// HTTP clients, target prompts, and decision clients are not used on this path.
406    /// Algorithm state is retained for each target setup. No post-routing completion
407    /// is dispatched. Target eligibility still applies to the host-owned path.
408    pub async fn decide_with_client(
409        &self,
410        request: Request,
411        client: Arc<dyn RoutedLlmClient>,
412    ) -> Result<RoutingOutcome, RunnerError> {
413        let selection = self.select_lane(&request)?;
414        let lane = selection.lane;
415        switchyard_llm_client::decide(
416            Arc::clone(&lane.algorithm),
417            ClientRouter::single(client),
418            request,
419            Arc::clone(&lane.models),
420        )
421        .await
422        .map_err(RunnerError::from)
423    }
424
425    /// Executes a model-bearing provider operation through a compatible target.
426    pub async fn call_auxiliary(
427        &self,
428        request: Request,
429        operation: AuxiliaryOperation,
430    ) -> Result<Value, RunnerError> {
431        let selection = self.select_lane(&request)?;
432        let lane = selection.lane;
433        let target = match operation {
434            AuxiliaryOperation::AnthropicCountTokens => &lane.anthropic_auxiliary_target,
435            AuxiliaryOperation::ResponsesInputTokens | AuxiliaryOperation::ResponsesCompact => {
436                &lane.responses_auxiliary_target
437            }
438        }
439        .as_ref()
440        .ok_or(RunnerError::AuxiliaryUnsupported)?;
441        target
442            .client
443            .call_auxiliary(&target.model, request, operation)
444            .await
445            .map_err(Into::into)
446    }

473        ));

477    fn request(task: &str, labels: &[&str]) -> Request {
478        let mut request = Request::default();

484    #[test]
485    fn resolved_bindings_reuse_the_algorithm_and_standard_setup() {
486        let source = config(POOL);
487        let runner = Runner::from_toml(&source).expect("eligible route");
488        let route = runner.route("route").expect("route");
489        let cloned = route.clone();
490        let private = route
491            .select_lane(&request("one", &["private"]))
492            .expect("private");
493        let both = route
494            .select_lane(&request("two", &["private", "customer_data"]))
495            .expect("both labels");
496        assert!(Arc::ptr_eq(&private.lane, &both.lane));
497        assert!(Arc::ptr_eq(&private.lane.algorithm, &both.lane.algorithm));
498        assert!(!Arc::ptr_eq(&private.lane, &route.standard));
499        assert_eq!(
500            private.lane.models.provider_model(&"approved_fast".into()),
501            "approved/fast"
502        );
503        let shared = cloned
504            .select_lane(&request("three", &["private"]))
505            .expect("cloned route");
506        assert!(Arc::ptr_eq(&private.lane, &shared.lane));
507
508        let runner = Runner::from_toml(&source.replace("labels = []", "labels = ['private']"))
509            .expect("standard models already approved");
510        let route = runner.route("route").expect("route");
511        let approved = route
512            .select_lane(&request("one", &["private"]))
513            .expect("approved");
514        assert!(Arc::ptr_eq(&approved.lane, &route.standard));
515    }

518    fn infra_spillover_stays_within_request_requirements() {
519        let algorithm = "type = 'llm_classifier'\nclassifier_target = 'judge'\n\

527        let route = runner.route("route").expect("route");
528        let signals: InfraSignals = [("fast", 8), ("capable", 0)]
529            .into_iter()

532                    model.into(),
533                    ModelInfraSignals {

536                            min_waiting: waiting,

541                    },
542                )
543            })
544            .collect();
545

547            RoutingOutcome::route_to("outside".into(), vec![], request("request", &["private"]));
548        let error = route
