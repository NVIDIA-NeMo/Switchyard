30/// Separator marking where [`truncate_middle`] dropped a message's interior.
31const TRIM_MARKER: &str = " ...[trimmed] ";
32
33/// Suffix marking a transcript cut off by [`MAX_REQUEST_CHARS`].
34const TRUNCATION_SUFFIX: &str = "...<truncated>";
35
36/// Per-message cap for system and developer anchors, which carry no trajectory signal but
37/// which coding-agent harnesses make very large.
38const SYSTEM_CHARS: usize = 1_000;
39
40/// Per-message cap for task-framing user messages — every user message that precedes the first
41/// assistant reply. Coding-agent harnesses often send environment boilerplate as the first user
42/// message and the task itself as the second, so anchoring only the first would pin the
43/// boilerplate and let the task scroll out of the window. Feature specifications run to several
44/// thousand characters, so this gets the widest anchor budget.
45const TASK_CHARS: usize = 4_000;
46
47/// Backstop on the assembled transcript; the per-message caps normally bind first.
48const MAX_REQUEST_CHARS: usize = 18_000;

50/// Optional policy for returning an escalated session to the efficient tier.
51#[derive(Clone, Copy, Debug, Deserialize)]
52#[serde(deny_unknown_fields)]
53pub struct DeescalationConfig {
54    /// Minimum number of capable-tier turns before the judge may release the session.
55    pub strong_min_calls: u32,
56    /// Optional hard limit on capable-tier turns before a forced return.
57    #[serde(default)]
58    pub strong_max_calls: Option<u32>,
59    /// Consecutive judge declines required to return to the efficient tier.
60    pub confirmations: u32,
61    /// Efficient calls served without judging after a hard-limit return.
62    #[serde(default)]
63    pub weak_cooldown_calls: u32,
64}
65
66impl DeescalationConfig {
67    fn validate(&self) -> Result<()> {
68        let reject = |message: &str| {
69            Err(LibsyError::AlgorithmError {
70                message: message.to_string(),
71            })
72        };
73        if self.strong_min_calls == 0 {
74            return reject("deescalation.strong_min_calls must be at least 1");
75        }
76        if self.confirmations == 0 {
77            return reject("deescalation.confirmations must be at least 1");
78        }
79        if self
80            .strong_max_calls
81            .is_some_and(|strong_max_calls| strong_max_calls < self.strong_min_calls)
82        {
83            return reject(
84                "deescalation.strong_max_calls must be at least deescalation.strong_min_calls",
85            );
86        }
87        Ok(())
88    }
89}

91/// The tuning surface for the trajectory judge.
92///
93/// The routing settings retain their benchmarked defaults. Everything else is a fixed invariant
94/// (the constants above).
95#[derive(Clone, Debug, Deserialize)]
96#[serde(default, deny_unknown_fields)]
97pub struct EscalationJudgeConfig {
98    /// Consecutive fresh-evidence verdicts in the same category required before a turn moves to
99    /// the capable tier, which is also the turn that latches the session. Any decline or stale
100    /// evidence clears the streak.
101    /// `1` escalates on the first verdict; the router's main cost dial.
102    /// `2` or higher needs a session id, since the streak is retained per session.
103    pub confirmations: u32,
104    /// Trailing messages shown on top of the anchors. A loop longer than this is invisible.
105    pub recent_turn_window: usize,
106    /// Per-message cap inside the trailing window.
107    pub window_message_chars: usize,
108    /// De-escalation policy. `None` preserves permanent latching to the capable tier.
109    pub deescalation: Option<DeescalationConfig>,
110}
111
112impl EscalationJudgeConfig {
113    /// Rejects settings that would leave the judge with nothing useful to read.
114    fn validate(&self) -> Result<()> {
115        let reject = |message: String| Err(LibsyError::AlgorithmError { message });
116        if self.confirmations == 0 {
117            return reject("confirmations must be at least 1".to_string());
118        }
119        if self.recent_turn_window == 0 {
120            return reject("recent_turn_window must be at least 1".to_string());
121        }
122        if self.window_message_chars < 50 {
123            return reject(format!(
124                "window_message_chars must be at least 50, got {}",
125                self.window_message_chars
126            ));
127        }
128        if let Some(deescalation) = self.deescalation {
129            deescalation.validate()?;
130        }
131        Ok(())
132    }
133}
134
135impl Default for EscalationJudgeConfig {
136    fn default() -> Self {
137        Self {
138            confirmations: 2,
139            recent_turn_window: 28,
140            window_message_chars: 500,
141            deescalation: None,
142        }
143    }

220/// Maps the judge's verdict to a classification. A verdict names the tier to serve — capable
221/// on escalate, efficient on decline — so the caller reads it straight off the winning score.
222/// [`Classification::Ambiguous`] carries the unavailable case, which names no tier: both a
223/// decline and an outage stay efficient, but only a decline is evidence, so only a decline
224/// clears the streak.
225pub(crate) struct EscalationPolicy {
226    phase: Option<EvaluationPhase>,
227}
228
229impl JudgePolicy for EscalationPolicy {
230    type Verdict = EscalationVerdict;
231
232    fn to_classification(
233        &self,
234        verdict: Option<&EscalationVerdict>,
235        driver: &Driver,
236    ) -> Result<Classification> {
237        if let Some(verdict) = verdict {
238            tracing::debug!(
239                escalate = verdict.escalate,
240                category = verdict.category.label(),
241                new_evidence = verdict.new_evidence,
242                "escalation judge verdict"
243            );
244        }
245        match verdict {
246            Some(verdict) if verdict.escalate => Ok(Classification::Scores(vec![Score {
247                target: driver.first_model_for(&Category::Capable)?.clone(),
248                confidence: 1.0,
249                category: Some(Category::Capable),
250            }])),
251            Some(_) => Ok(Classification::Scores(vec![Score {
252                target: driver.first_model_for(&Category::Efficient)?.clone(),
253                confidence: 1.0,
254                category: Some(Category::Efficient),
255            }])),
256            None => Ok(Classification::Ambiguous(Vec::new())),
257        }
258    }
259}

261/// Maps present verdicts to phase-aware evidence; absent verdicts add nothing.
262fn escalation_evidence(
263    policy: &EscalationPolicy,
264    verdict: Option<&EscalationVerdict>,
265) -> Option<Value> {
266    verdict.map(|verdict| {
267        let verdict = match (policy.phase, verdict.escalate) {
268            (Some(EvaluationPhase::Strong), true) => "retain",
269            (Some(EvaluationPhase::Strong), false) => "deescalate",
270            (_, true) => "escalate",
271            (_, false) => "continue",
272        };
273        serde_json::json!({
274            "source": "escalation",
275            "verdict": verdict,
276        })
277    })
278}

280/// Builds the trajectory judge, scoring the runtime capable category when it escalates.
281///
282/// Loads the packaged prompt and schema, so an unusable asset or an unusable `config` value
283/// fails here rather than on the first request.
284pub(crate) fn build_judge(
285    contract_config: &ClassifierContractConfig,
286    config: EscalationJudgeConfig,
287    phase: Option<EvaluationPhase>,
288    max_output_tokens: u64,
289    judge_deadline_ms: Option<u64>,
290) -> Result<JudgeClassifier<EscalationJudge, EscalationPolicy>> {
291    config.validate()?;
292    let contract = build_contract(contract_config, phase.is_some())?;
293    Ok(JudgeClassifier::new(
294        StructuredJudge::new(
295            EscalationInput { config, phase },
296            contract,
297            SerdeDecoder::new(),
298            JudgeRuntimeConfig::new(max_output_tokens)?.with_deadline_ms(judge_deadline_ms)?,
299        ),
300        EscalationPolicy { phase },
301    )
302    .with_evidence(escalation_evidence))
303}

305fn build_contract(
306    contract_config: &ClassifierContractConfig,
307    phase_aware: bool,
308) -> Result<ClassifierContract> {
309    let prompt = contract_config.prompt().unwrap_or(PROMPT_TEMPLATE);
310    validate_prompt(prompt)?;
311    let phase_aware_config = phase_aware.then(|| {
312        contract_config.clone().with_prompt(format!(
313            "{}\n\n{}",
314            prompt.trim_end(),
315            DEESCALATION_PROMPT.trim()
316        ))
317    });
318    let contract_config = phase_aware_config.as_ref().unwrap_or(contract_config);
319    ClassifierContract::from_config(contract_config, PROMPT_TEMPLATE, SCHEMA_TEMPLATE)
320}

322/// The 1-indexed model invocation the transcript ends on: one per assistant reply.
323///
324/// The judge reads the turn *including* the reply it is judging, so the newest assistant
325/// message is this turn — no `+ 1`. Counting the caller's request instead would report the
326/// turn after the one under judgement.
327///
328/// Messages are already normalized by `switchyard-protocol`, so this needs no
329/// per-format branching.
330pub(crate) fn conversation_turn(request: &Request) -> usize {
331    request
332        .llm_request
333        .messages
334        .iter()
335        .filter(|message| message.role == Role::Assistant)
336        .count()
337}

339/// Keeps the head and tail of `text` within `limit` characters.
340///
341/// The head gets two thirds of the surviving budget: for a trajectory judge the command or
342/// error signature that opens a message carries more signal than its trailing output.
343fn truncate_middle(text: &str, limit: usize) -> String {
344    let chars: Vec<char> = text.chars().collect();
345    if chars.len() <= limit {
346        return text.to_string();
347    }
348    let keep = limit
349        .saturating_sub(TRIM_MARKER.chars().count())
350        .max(20)
351        .min(chars.len());
352    let head = keep * 2 / 3;
353    let tail = keep - head;
354    let mut out: String = chars[..head].iter().collect();
355    out.push_str(TRIM_MARKER);
356    out.extend(chars[chars.len() - tail..].iter());
357    out
358}

360/// Renders a compact role-labelled transcript for the judge.
361///
362/// Task-framing user messages are capped individually. System/developer instructions share
363/// the remaining budget after reserving space for task framing and the newest window entry.
364/// The trailing window carries recent activity. A coverage header states how much history is
365/// not shown, so the judge can reason about pace rather than assuming it sees everything.
366///
367/// When the assembled text still exceeds `max_request_chars`, the oldest window lines go
368/// first: for a trajectory judge the newest evidence is strictly the most valuable.
369fn summarize_for_judge(
370    instructions: &[InstructionBlock],
371    messages: &[Message],
372    turn: usize,
373    phase: Option<EvaluationPhase>,
374    config: &EscalationJudgeConfig,
375) -> String {
376    let mut instruction_anchors: Vec<String> = Vec::new();
377    let mut anchors: Vec<String> = Vec::new();
378    let mut window: Vec<String> = Vec::new();
379    let mut assistant_seen = false;
380
381    for instruction in instructions {
382        let mut parts = Vec::new();
383        collect_text(&instruction.content, &mut parts, &[]);
384        instruction_anchors.push(format!(
385            "[{}] {}",
386            role_label(instruction.role),
387            truncate_middle(&parts.join(" "), SYSTEM_CHARS)
388        ));
389    }
390
391    for message in messages {
392        let text = message_text(message);
393        match message.role {
394            Role::System | Role::Developer => instruction_anchors.push(format!(
395                "[{}] {}",
396                role_label(message.role),
397                truncate_middle(&text, SYSTEM_CHARS)
398            )),
399            // Everything the user said before the agent first replied is task framing.
400            Role::User if !assistant_seen => {
401                anchors.push(format!(
402                    "[user (task)] {}",
403                    truncate_middle(&text, TASK_CHARS)
404                ));
405            }
406            role => {
407                if role == Role::Assistant {
408                    assistant_seen = true;
409                }
410                window.push(format!(
411                    "[{}] {}",
412                    role_label(role),
413                    truncate_middle(&text, config.window_message_chars)
414                ));
415            }
416        }
417    }
418
419    if window.len() > config.recent_turn_window {
420        window.drain(..window.len() - config.recent_turn_window);
421    }
422
423    let assemble = |instructions: Option<&str>, window: &[String]| {
424        let header = format!(
425            "Conversation turn {turn}; showing the last {} of {} messages after the task framing.",
426            window.len(),
427            messages.len(),
428        );
429        phase
430            .map(|phase| format!("Routing phase: {}", phase.marker()))
431            .into_iter()
432            .chain(std::iter::once(header))
433            .chain(instructions.map(str::to_owned))
434            .chain(anchors.iter().cloned())
435            .chain(window.iter().cloned())
436            .collect::<Vec<_>>()
437            .join("\n")
438    };
439
440    let reserved = assemble(None, &window[window.len().saturating_sub(1)..])
441        .chars()
442        .count();
443    let instruction_budget = MAX_REQUEST_CHARS.saturating_sub(reserved + 1);
444    // The remaining budget may be smaller than truncate_middle's minimum retained span.
445    let instruction_text = truncate_middle(&instruction_anchors.join("\n"), instruction_budget)
446        .chars()
447        .take(instruction_budget)
448        .collect::<String>();
449    let instructions = (!instruction_text.is_empty()).then_some(instruction_text.as_str());
450    let mut text = assemble(instructions, &window);
451    while text.chars().count() > MAX_REQUEST_CHARS && !window.is_empty() {
452        window.remove(0);
453        text = assemble(instructions, &window);
454    }
455    if text.chars().count() > MAX_REQUEST_CHARS {
456        let keep = MAX_REQUEST_CHARS.saturating_sub(TRUNCATION_SUFFIX.chars().count() + 1);
457        text = text.chars().take(keep).collect::<String>() + TRUNCATION_SUFFIX;
458    }
459    text
460}

462/// The transcript label for a role.
463fn role_label(role: Role) -> &'static str {
464    match role {
465        Role::System => "system",
466        Role::Developer => "developer",
467        Role::User => "user",
468        Role::Assistant => "assistant",
469        Role::Tool => "tool",
470    }
471}

800        // Codex sends environment boilerplate as the first user message and the task as the

802        let mut messages = vec![
803            Message::text(

813        for i in 0..40 {

815            messages.push(Message::text(Role::User, format!("later user note {i}")));
816        }
817        let config = EscalationJudgeConfig {
818            recent_turn_window: 3,

841    #[test]
842    fn summary_drops_oldest_window_lines_under_the_char_cap() {
843        // MAX_REQUEST_CHARS is a backstop, not a dial: at default settings the window caps
844        // bind first (28 x 500 plus anchors sits under it), so reaching it takes an unusually
845        // wide per-message cap. That is the point — it only fires on pathological input.
846        let mut messages = vec![
847            Message::text(Role::System, "framing"),
848            Message::text(Role::User, "task"),
849        ];
850        for i in 0..20 {
851            messages.push(Message::text(
852                Role::Assistant,
853                format!("{i} {}", "x".repeat(2_000)),
854            ));
855        }
856        let config = EscalationJudgeConfig {
857            window_message_chars: 2_000,
858            ..EscalationJudgeConfig::default()
859        };
860
861        let summary = summarize_for_judge(&[], &messages, 21, None, &config);
862
863        assert!(
864            summary.chars().count() <= MAX_REQUEST_CHARS,
865            "{}",
866            summary.chars().count()
867        );
868        // Anchors are never dropped, and the newest activity outlives the oldest.
869        assert!(summary.contains("[system] framing"), "{summary}");
870        assert!(summary.contains("[user (task)] task"), "{summary}");
871        assert!(summary.contains("19 xxx"), "{summary}");
872        assert!(!summary.contains("0 xxx"), "{summary}");
873    }
