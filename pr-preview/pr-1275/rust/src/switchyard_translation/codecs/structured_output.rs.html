4//! Structured-output enforcement across provider formats.
5
6use serde_json::Value;
7
8use crate::diagnostic::TranslationDiagnostic;
9use crate::error::Result;
10use crate::llm::OutputParams;
11use crate::policy::TranslationPolicy;
12use crate::util::push_lossy;
13
14pub(super) fn decode_openai_schema_enforcement(format: Option<&Value>) -> Option<bool> {
15    let format = format.filter(|format| format["type"] == "json_schema")?;
16    Some(format["json_schema"]["strict"].as_bool().unwrap_or(false))
17}
18
19pub(super) fn encode_openai_format(
20    output: &OutputParams,
21    diagnostics: &mut Vec<TranslationDiagnostic>,
22    policy: &TranslationPolicy,
23) -> Result<Option<Value>> {
24    let mut format = output.response_format.clone();
25    let Some(is_enforced) = output.is_schema_enforced else {
26        return Ok(format);
27    };
28    let schema = format
29        .as_mut()
30        .filter(|format| format["type"] == "json_schema")
31        .and_then(|format| format.get_mut("json_schema"))
32        .and_then(Value::as_object_mut);
33    let Some(schema) = schema.filter(|schema| schema.get("schema").is_some_and(Value::is_object))
34    else {
35        if is_enforced {
36            push_lossy(
37                diagnostics,
38                policy,
39                "Structured-output schema enforcement requires a json_schema object with a schema",
40            )?;
41        }
42        return Ok(format);
43    };
44    // Preserve native strict schemas, including constructs outside our checked subset.
45    if is_enforced
46        && schema.get("strict").and_then(Value::as_bool) != Some(true)
47        && !schema
48            .get("schema")
49            .is_some_and(is_openai_strict_compatible)
50    {
51        push_lossy(
52            diagnostics,
53            policy,
54            "Structured-output schema was left non-strict: compatibility with OpenAI strict mode could not be established",
55        )?;
56    } else {
57        schema.insert("strict".into(), Value::Bool(is_enforced));
58    }
59    Ok(format)
60}
61
62// A conservative subset shared by OpenAI base and fine-tuned models. Fine-tuning
63// excludes format, pattern, minLength/maxLength, minimum/maximum, multipleOf,
64// and minItems/maxItems. These and unchecked constructs such as references retain
65// their schema with a lossiness diagnostic.
66fn is_openai_strict_compatible(schema: &Value) -> bool {
67    #[derive(Default)]
68    struct Size {
69        properties: usize,
70        enum_values: usize,
71        chars: usize,
72    }
73
74    fn visit(schema: &Value, depth: usize, size: &mut Size) -> bool {
75        let Some(object) = schema.as_object() else {
76            return false;
77        };
78        if object.keys().any(|key| {
79            !matches!(
80                key.as_str(),
81                "type"
82                    | "title"
83                    | "description"
84                    | "properties"
85                    | "required"
86                    | "additionalProperties"
87                    | "items"
88                    | "enum"
89                    | "const"
90                    | "anyOf"
91            )
92        }) {
93            return false;
94        }
95        if let Some(values) = object.get("enum") {
96            let Some(values) = values.as_array().filter(|values| !values.is_empty()) else {
97                return false;
98            };
99            if values
100                .iter()
101                .any(|value| value.is_object() || value.is_array())
102            {
103                return false;
104            }
105            let chars = values
106                .iter()
107                .filter_map(Value::as_str)
108                .map(|value| value.chars().count())
109                .sum::<usize>();
110            if values.len() > 250 && chars > 15_000 {
111                return false;
112            }
113            size.enum_values += values.len();
114            size.chars += chars;
115        }
116        if let Some(value) = object.get("const") {
117            if value.is_object() || value.is_array() {
118                return false;
119            }
120            size.chars += value.as_str().map_or(0, |value| value.chars().count());
121        }
122        if let Some(choices) = object.get("anyOf") {
123            // Limit siblings to annotations so every constraint is checked in a branch.
124            return object
125                .keys()
126                .all(|key| matches!(key.as_str(), "anyOf" | "title" | "description"))
127                && choices.as_array().is_some_and(|choices| {
128                    !choices.is_empty() && choices.iter().all(|choice| visit(choice, depth, size))
129                });
130        }
131        let schema_type = match &schema["type"] {
132            Value::String(kind) => Some(kind.as_str()),
133            Value::Array(kinds)
134                if kinds.len() == 2 && kinds.iter().any(|kind| kind.as_str() == Some("null")) =>
135            {
136                kinds
137                    .iter()
138                    .filter_map(Value::as_str)
139                    .find(|kind| *kind != "null")
140            }
141            _ => None,
142        };
143        match schema_type {
144            Some("object") if depth < 10 => {
145                let (Some(fields), Some(required)) = (
146                    schema["properties"].as_object(),
147                    schema["required"].as_array(),
148                ) else {
149                    return false;
150                };
151                size.properties += fields.len();
152                size.chars += fields.keys().map(|key| key.chars().count()).sum::<usize>();
153                schema["additionalProperties"] == false
154                    && required.len() == fields.len()
155                    && fields.iter().all(|(name, value)| {
156                        required.iter().any(|value| value.as_str() == Some(name))
157                            && visit(value, depth + 1, size)
158                    })
159            }
160            Some("array") if depth < 10 => visit(&schema["items"], depth + 1, size),
161            Some("string" | "number" | "integer" | "boolean" | "null") => true,
162            _ => false,
163        }
164    }
165    let mut size = Size::default();
166    schema["type"] == "object"
167        && visit(schema, 0, &mut size)
168        && size.properties <= 5000
169        && size.enum_values <= 1000
170        && size.chars <= 120_000
171}
