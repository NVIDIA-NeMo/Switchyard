4//! No-op router. Always returns a hard coded response. Does not route to a backend.
5//! For testing.
6
7use std::sync::Arc;
8
9use switchyard_protocol::{
10    AggLlmResponse, ContentBlock, LlmResponse, Request, Response, ResponseOutput, Role, StopReason,
11};
12
13use crate::Result;
14use crate::core::algorithm::{Algorithm, Driver};
15use switchyard_protocol::{Context, Decision};
16
17/// A routing algorithm that does not route. It returns a hard-coded response.
18pub struct Noop {}
19
20/// How [`Noop`] records which model it chose. This will be the model on the Request if any,
21/// otherwise a hard coded placeholder. Neither is actually used.
22pub struct NoopDecision {
23    model: String,
24}
25
26impl Decision for NoopDecision {
27    fn selected_model(&self) -> &str {
28        &self.model
29    }
30    fn reasoning(&self) -> Option<&str> {
31        None
32    }
33    fn as_any(&self) -> &dyn std::any::Any {
34        self
35    }
36}
37
38#[async_trait::async_trait]
39impl Algorithm for Noop {
40    fn name(&self) -> &str {
41        "noop"
42    }
43
44    async fn create_run_task(
45        self: Arc<Self>,
46        ctx: Context,
47        driver: Driver,
48        request: Request,
49    ) -> Result<Response> {
50        let model = request
51            .requested_model()
52            .unwrap_or("switchyard/noop")
53            .to_string();
54        let decision: Arc<dyn Decision> = Arc::new(NoopDecision {
55            model: model.clone(),
56        });
57        driver.info(ctx, decision.clone()).await?;
58
59        let llm_response = LlmResponse::Agg(AggLlmResponse {
60            id: Some("switchyard-noop".to_string()),
61            model: Some(model),
62            outputs: vec![ResponseOutput {
63                role: Role::Assistant,
64                content: vec![ContentBlock::Text {
65                    text: "OK".to_string(),
66                }],
67                stop_reason: Some(StopReason::EndTurn),
68            }],
69            ..Default::default()
70        });
71        let response = Response {
72            llm_response,
73            metadata: request.metadata.clone(),
74        };
75        Ok(response)
76    }
77}

79#[cfg(test)]
80mod tests {
81    use switchyard_protocol::{LlmRequest, Message, Role};
82
83    use super::*;
84
85    #[tokio::test]
86    async fn test_noop_algo() -> Result<()> {
87        const TEST_MODEL: &str = "test_noop_algo";
88        let request = Request {
89            llm_request: LlmRequest {
90                model: Some(TEST_MODEL.to_string()),
91                messages: vec![Message::text(Role::User, "hi")],
92                ..LlmRequest::default()
93            },
94            raw_request: None,
95            metadata: None,
96        };
97
98        let a: Arc<dyn Algorithm> = Arc::new(Noop {});
99        let (decisions, response) = a.run(Context::default(), request).await?;
100        let Some(decision) = decisions.first() else {
101            panic!("Expected exactly one Decision");
102        };
103        assert_eq!(decision.selected_model(), TEST_MODEL);
104        assert_eq!(response.selected_model(), Some(TEST_MODEL));
105        Ok(())
106    }
107}
