4//! Round-trips OpenAI Responses freeform ("custom") tools through the neutral IR.
5//!
6//! Codex drives GPT-5 models with freeform tools: the definition is `{"type": "custom", "name",
7//! "description", "format"}` and the model answers with `custom_tool_call` items whose `input`
8//! is a raw string rather than JSON arguments. The IR only knows function-style tools, so a
9//! custom tool is represented as a function whose single argument is `input`, and the verbatim
10//! definitions are kept on the request extensions. When the response is encoded back to
11//! Responses, calls to those tools are rewritten into `custom_tool_call` items again.
12
13use std::collections::HashSet;
14
15use serde_json::{Map, Value, json};
16use switchyard_protocol::ProviderExtensions;
17
18/// Request-extension key holding the verbatim custom tool definitions, keyed by tool name.
19///
20/// Prefixed so it cannot collide with a real provider field, and so a codec that allowlists
21/// provider fields never forwards it.
22pub const CUSTOM_TOOLS_KEY: &str = "switchyard_codex_custom_tools";
23
24/// Request-extension key holding the verbatim `tools` array of a Responses-lite
25/// `additional_tools` input item, so the request can be re-emitted in the same shape.
26///
27/// Codex sends GPT-5 requests in a "lite" shape: no top-level `tools`, empty `instructions`,
28/// and the tool definitions inside `input[0]` as `{"type": "additional_tools", "role":
29/// "developer", "tools": [...]}`.
30pub const ADDITIONAL_TOOLS_KEY: &str = "switchyard_codex_additional_tools";
31
32/// Stores the verbatim tools array of an `additional_tools` input item.
33pub fn attach_additional_tools(extensions: &mut ProviderExtensions, tools: Vec<Value>) {
34    if !tools.is_empty() {
35        extensions
36            .fields
37            .insert(ADDITIONAL_TOOLS_KEY.to_string(), Value::Array(tools));
38    }
39}
40
41/// Reads the verbatim `additional_tools` array back off a request's extensions.
42pub fn additional_tools(extensions: &ProviderExtensions) -> Option<&Vec<Value>> {
43    extensions
44        .fields
45        .get(ADDITIONAL_TOOLS_KEY)
46        .and_then(Value::as_array)
47}
48
49/// Request-extension key holding the `call_id`s of `custom_tool_call_output` items whose
50/// call is not in the request, so the outputs keep their type when re-encoded.
51pub const CUSTOM_CALL_OUTPUTS_KEY: &str = "switchyard_codex_custom_call_outputs";
52
53/// Stores the call ids of stored-state custom tool outputs, when there are any.
54pub fn attach_custom_call_outputs(extensions: &mut ProviderExtensions, call_ids: Vec<String>) {
55    if !call_ids.is_empty() {
56        extensions.fields.insert(
57            CUSTOM_CALL_OUTPUTS_KEY.to_string(),
58            Value::Array(call_ids.into_iter().map(Value::String).collect()),
59        );
60    }
61}
62
63/// Reads the stored-state custom tool output call ids back off a request's extensions.
64pub fn custom_call_outputs(extensions: &ProviderExtensions) -> HashSet<String> {
65    extensions
66        .fields
67        .get(CUSTOM_CALL_OUTPUTS_KEY)
68        .and_then(Value::as_array)
69        .map(|ids| {
70            ids.iter()
71                .filter_map(Value::as_str)
72                .map(str::to_owned)
73                .collect()
74        })
75        .unwrap_or_default()
76}
77
78/// Argument name used to carry a custom tool's freeform input through the IR.
79pub const INPUT_ARGUMENT: &str = "input";
80
81/// The IR parameter schema for a custom tool: one required string, `input`.
82pub fn input_schema() -> Value {
83    json!({
84        "type": "object",
85        "properties": {INPUT_ARGUMENT: {"type": "string"}},
86        "required": [INPUT_ARGUMENT],
87        "additionalProperties": false,
88    })
89}
90
91/// Stores the collected definitions on a request's extensions, when there are any.
92pub fn attach_custom_tools(extensions: &mut ProviderExtensions, tools: Map<String, Value>) {
93    if !tools.is_empty() {
94        extensions
95            .fields
96            .insert(CUSTOM_TOOLS_KEY.to_string(), Value::Object(tools));
97    }
98}
99
100/// Reads the definitions back off a request's extensions.
101pub fn custom_tools(extensions: &ProviderExtensions) -> Option<&Map<String, Value>> {
102    extensions
103        .fields
104        .get(CUSTOM_TOOLS_KEY)
105        .and_then(Value::as_object)
106}
107
108/// Names of the custom tools recorded on a request.
109pub fn custom_tool_names(extensions: &ProviderExtensions) -> HashSet<String> {
110    custom_tools(extensions)
111        .map(|tools| tools.keys().cloned().collect())
112        .unwrap_or_default()
113}

115/// Extracts the freeform input from IR tool arguments, falling back to the serialized
116/// arguments when the model did not use the `input` convention.
117pub fn input_from_arguments(arguments: &Value) -> String {
118    match arguments {
119        Value::Object(object) => match object.get(INPUT_ARGUMENT) {
120            Some(Value::String(input)) => input.clone(),
121            Some(other) => other.to_string(),
122            None => arguments.to_string(),
123        },
124        Value::String(text) => match serde_json::from_str::<Value>(text) {
125            Ok(Value::Object(object)) => match object.get(INPUT_ARGUMENT) {
126                Some(Value::String(input)) => input.clone(),
127                _ => text.clone(),
128            },
129            _ => text.clone(),
130        },
131        other => other.to_string(),
132    }
133}

135/// Rewrites a `function_call` output item into a `custom_tool_call` when the tool is custom.
136/// Returns whether the item was rewritten.
137fn rewrite_item(item: &mut Value, custom: &HashSet<String>) -> bool {
138    let Some(object) = item.as_object_mut() else {
139        return false;
140    };
141    if object.get("type").and_then(Value::as_str) != Some("function_call") {
142        return false;
143    }
144    let Some(name) = object.get("name").and_then(Value::as_str) else {
145        return false;
146    };
147    if !custom.contains(name) {
148        return false;
149    }
150    let input = object
151        .remove("arguments")
152        .map(|arguments| input_from_arguments(&arguments))
153        .unwrap_or_default();
154    object.insert(
155        "type".to_string(),
156        Value::String("custom_tool_call".to_string()),
157    );
158    object.insert("input".to_string(), Value::String(input));
159    // OpenAI validates replayed item ids by prefix: a custom tool call must be `ctc_...`.
160    if let Some(Value::String(id)) = object.get_mut("id")
161        && let Some(rest) = id.strip_prefix("fc_")
162    {
163        *id = format!("ctc_{rest}");
164    }
165    true
166}
167
168/// Rewrites custom tool calls inside a buffered Responses body's `output` array.
169pub fn restore_custom_tool_calls(body: &mut Value, custom: &HashSet<String>) {
170    if custom.is_empty() {
171        return;
172    }
173    if let Some(items) = body.get_mut("output").and_then(Value::as_array_mut) {
174        for item in items {
175            rewrite_item(item, custom);
176        }
177    }
178}

180/// Per-stream bookkeeping for [`restore_custom_tool_calls_in_event`].
181#[derive(Default)]
182pub struct CustomToolCallStreamState {
183    /// Output indexes whose item was rewritten into a custom tool call.
184    custom_indexes: HashSet<u64>,
185}
186
187/// Rewrites a streamed Responses event so a custom tool's call reaches the client in the shape
188/// it expects. Item events are rewritten in place; argument delta events for a rewritten item
189/// are dropped (returns `false`), because a partial JSON delta cannot be turned into a
190/// freeform input delta and clients read the completed item instead.
191pub fn restore_custom_tool_calls_in_event(
192    event: &mut Value,
193    custom: &HashSet<String>,
194    state: &mut CustomToolCallStreamState,
195) -> bool {
196    if custom.is_empty() {
197        return true;
198    }
199    let kind = event
200        .get("type")
201        .and_then(Value::as_str)
202        .unwrap_or_default()
203        .to_string();
204    let index = event.get("output_index").and_then(Value::as_u64);
205    match kind.as_str() {
206        "response.output_item.added" | "response.output_item.done" => {
207            if let Some(item) = event.get_mut("item")
208                && rewrite_item(item, custom)
209                && let Some(index) = index
210            {
211                state.custom_indexes.insert(index);
212            }
213            true
214        }
215        "response.function_call_arguments.delta" | "response.function_call_arguments.done" => {
216            !index.is_some_and(|index| state.custom_indexes.contains(&index))
217        }
218        "response.completed" | "response.incomplete" | "response.failed" => {
219            if let Some(response) = event.get_mut("response") {
220                restore_custom_tool_calls(response, custom);
221            }
222            true
223        }
224        _ => true,
225    }
226}

228#[cfg(test)]
229mod tests {
230    use super::*;
231
232    #[test]
233    fn input_is_read_from_the_input_argument_or_left_verbatim() {
234        assert_eq!(input_from_arguments(&json!({"input": "ls -la"})), "ls -la");
235        assert_eq!(input_from_arguments(&json!("{\"input\":\"pwd\"}")), "pwd");
236        assert_eq!(input_from_arguments(&json!("raw text")), "raw text");
237        assert_eq!(
238            input_from_arguments(&json!({"cmd": "x"})),
239            "{\"cmd\":\"x\"}"
240        );
241    }
242
243    #[test]
244    fn function_call_items_for_custom_tools_become_custom_tool_calls() {
245        let custom: HashSet<String> = ["exec".to_string()].into_iter().collect();
246        let mut body = json!({"output": [
247            {"type": "function_call", "call_id": "c1", "name": "exec", "arguments": "{\"input\":\"ls\"}"},
248            {"type": "function_call", "call_id": "c2", "name": "update_plan", "arguments": "{}"}
249        ]});
250        restore_custom_tool_calls(&mut body, &custom);
251        assert_eq!(body["output"][0]["type"], "custom_tool_call");
252        assert_eq!(body["output"][0]["input"], "ls");
253        assert!(body["output"][0].get("arguments").is_none());
254        assert_eq!(body["output"][1]["type"], "function_call");
255    }
256
257    #[test]
258    fn rewritten_custom_tool_calls_take_the_ctc_id_prefix() {
259        let custom: HashSet<String> = ["exec".to_string()].into_iter().collect();
260        let mut body = json!({"output": [
261            {"type": "function_call", "id": "fc_abc_1", "call_id": "c1", "name": "exec", "arguments": "{\"input\":\"ls\"}"}
262        ]});
263        restore_custom_tool_calls(&mut body, &custom);
264        assert_eq!(body["output"][0]["type"], "custom_tool_call");
265        assert_eq!(body["output"][0]["id"], "ctc_abc_1");
266    }
267
268    #[test]
269    fn streamed_argument_deltas_for_custom_tools_are_dropped() {
270        let custom: HashSet<String> = ["exec".to_string()].into_iter().collect();
271        let mut state = CustomToolCallStreamState::default();
272        let mut added = json!({"type": "response.output_item.added", "output_index": 1,
273            "item": {"type": "function_call", "call_id": "c1", "name": "exec", "arguments": ""}});
274        assert!(restore_custom_tool_calls_in_event(
275            &mut added, &custom, &mut state
276        ));
277        assert_eq!(added["item"]["type"], "custom_tool_call");
278        let mut delta = json!({"type": "response.function_call_arguments.delta", "output_index": 1, "delta": "{\"in"});
279        assert!(!restore_custom_tool_calls_in_event(
280            &mut delta, &custom, &mut state
281        ));
282        let mut other = json!({"type": "response.function_call_arguments.delta", "output_index": 2, "delta": "{}"});
283        assert!(restore_custom_tool_calls_in_event(
284            &mut other, &custom, &mut state
285        ));
286        let mut done = json!({"type": "response.output_item.done", "output_index": 1,
287            "item": {"type": "function_call", "call_id": "c1", "name": "exec", "arguments": "{\"input\":\"ls -la\"}"}});
288        assert!(restore_custom_tool_calls_in_event(
289            &mut done, &custom, &mut state
290        ));
291        assert_eq!(done["item"]["input"], "ls -la");
292    }
