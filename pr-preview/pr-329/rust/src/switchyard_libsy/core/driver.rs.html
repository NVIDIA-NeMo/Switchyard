4//! # driver — a type-erased promise-over-a-stream request pump
5//!
6//! [`TypeErasedDriver`] is the generic offload primitive; its sibling
7//! [`algorithm`](super::algorithm) builds the libsy-typed `Driver` on top of it. This
8//! module has no dependency on the rest of the crate — the coupling is one-directional
9//! (`core/algorithm.rs` → `core/driver.rs`).
10//!
11//! A [`TypeErasedDriver`] lets a *producer* (e.g. a routing algorithm) fulfill
12//! arbitrary requests by publishing promises onto a stream that a single *consumer*
13//! drains. It is the type-erased generalization of the crate's `run_stream` offload:
14//! instead of one fixed request/response shape, a producer calls
15//! [`fulfill_request`](TypeErasedDriver::fulfill_request)
16//! with *any* `REQ` and awaits *any* `RES`.
17//!
18//! - [`fulfill_request`](TypeErasedDriver::fulfill_request) enqueues a [`DriverStep::Request`]
19//!   carrying a [`DriverRequest`], then awaits the consumer's response.
20//! - [`info`](TypeErasedDriver::info) pushes a fire-and-forget [`DriverStep::Info`] — no promise
21//!   to await.
22//! - [`done`](TypeErasedDriver::done) emits the terminal [`DriverStep::Done`] with a final payload.
23//! - [`stream`](TypeErasedDriver::stream) hands the single consumer the [`Stream`] of steps; for
24//!   each [`DriverStep::Request`] the consumer downcasts the request, computes a
25//!   response, and writes it back with [`DriverRequest::respond`].
26//!
27//! Payloads are erased to `Box<dyn Any + Send>`, so one `TypeErasedDriver` serves any request
28//! type; the consumer downcasts to the concrete type it expects. `TypeErasedDriver` is `Clone`
29//! (many producer tasks may call it concurrently — multi-producer), while the stream
30//! is single-consumer. Each request rides its own `oneshot`, so concurrent
31//! `fulfill_request` calls never cross responses.
32//!
33//! ## Pacing (bounded step channel)
34//!
35//! The step channel has capacity 1, so a producer cannot publish its next step until
36//! the consumer has pulled the previous one. The consumer therefore *paces* the
37//! algorithm: it advances one step for each `.next().await`. Every producer method is
38//! `async` because publishing a step awaits channel capacity.
39//!
40//! ## Termination
41//!
42//! There is no explicit stop method — the consumer terminates by **dropping the
43//! stream** (and any [`DriverRequest`] it is holding). The producer's next publish
44//! (`fulfill_request`/`info`/`done`/`fail`) then resolves to `Err`, and a producer
45//! awaiting a response sees `Err` once the promise it handed out is dropped. Either
46//! way the algorithm unwinds cooperatively at its next driver interaction. Because the
47//! producer runs on a task the driver does not own, hard cancellation (e.g. mid-compute
48//! that never touches the driver) is the caller's concern — abort the producer task.

61type BoxAny = Box<dyn Any + Send>;
62type StepResult = Result<DriverStep>;

64// TODO make request timeout configurable
65const FULFILL_REQUEST_TIMEOUT: Duration = Duration::from_mins(10);

67/// One item on the stream returned by [`TypeErasedDriver::stream`].
68pub enum DriverStep {
69    /// A request awaiting a response. The consumer downcasts it and fulfills the
70    /// paired promise with [`DriverRequest::respond`].
71    Request(DriverRequest),
72    /// A fire-and-forget payload from a producer; no response is expected.
73    Info(BoxAny),
74    /// A producer's terminal result. The consumer treats it as the last meaningful
75    /// step (the stream itself closes when every [`TypeErasedDriver`] clone drops).
76    Done(BoxAny),
77}
78
79/// The consumer-facing half of one [`TypeErasedDriver::fulfill_request`] call.
80///
81/// Yielded inside [`DriverStep::Request`]. The consumer reads the request via
82/// [`request`](Self::request), does whatever work it names, and fulfills the promise
83/// with [`respond`](Self::respond) — unblocking the producer's `fulfill_request`.
84pub struct DriverRequest {
85    request: BoxAny,
86    // Fulfilled exactly once — `respond` consumes `self` to send, so no `Option`.
87    tx: oneshot::Sender<Result<BoxAny>>,
88}
89
90impl DriverRequest {
91    /// Borrow the request payload as `REQ`. Errors if the producer enqueued a
92    /// different type than the consumer expected.
93    pub fn request<REQ: Any>(&self) -> Result<&REQ> {
94        self.request.downcast_ref::<REQ>().ok_or_else(|| {
95            DriverError::TypeMismatch {
96                expected: std::any::type_name::<REQ>(),
97            }
98            .into()
99        })
100    }
101
102    /// Fulfill the promise with a typed response, or an `Err` to propagate a failure
103    /// back to the producer. Consumes `self`: a promise is fulfilled exactly once.
104    pub fn respond<RES: Any + Send>(self, res: Result<RES>) -> Result<()> {
105        // Erase the response so the single stream item type can carry any RES; the
106        // producer downcasts it back in `fulfill_request`.
107        let boxed: Result<BoxAny> = res.map(|r| Box::new(r) as BoxAny);
108        self.tx
109            .send(boxed)
110            .map_err(|_| DriverError::ResponseDropped.into())
111    }
112}

114/// Internal shared state: the step channel plus the single, take-once receiver.
115struct DriverInner {
116    // Multi-producer: cloned into every `TypeErasedDriver`, so many tasks can enqueue steps.
117    // Capacity 1: a producer blocks publishing its next step until the consumer pulls
118    // the previous one, so the consumer paces the algorithm.
119    step_tx: mpsc::Sender<StepResult>,
120    // Single-consumer: taken out (once) by `stream`. `None` after the first take.
121    step_rx: Mutex<Option<mpsc::Receiver<StepResult>>>,
122}
123
124/// A promise-over-a-stream request pump. See the [module docs](self) for the model.
125///
126/// Cheap to clone (shares one `Arc`); clone it to hand a producer handle to another
127/// task. The consumer calls [`stream`](Self::stream) exactly once to drain steps.
128#[derive(Clone)]
129pub struct TypeErasedDriver {
130    inner: Arc<DriverInner>,
131}
132
133impl TypeErasedDriver {
134    /// Build an empty driver with its step channel ready. Take the consumer stream
135    /// with [`stream`](Self::stream); enqueue work with the other methods.
136    pub fn new() -> Self {
137        let (step_tx, step_rx) = mpsc::channel(1);
138        TypeErasedDriver {
139            inner: Arc::new(DriverInner {
140                step_tx,
141                step_rx: Mutex::new(Some(step_rx)),
142            }),
143        }
144    }
145
146    fn ensure_started(&self) -> Result<()> {
147        let guard = self.inner.step_rx.lock();
148        if guard.is_none() {
149            Ok(())
150        } else {
151            Err(DriverError::NotStarted.into())
152        }
153    }
154
155    /// Enqueue `req` as a [`DriverStep::Request`], await the consumer's response, and
156    /// downcast it to `RES`. Errors if the stream is closed, the promise is dropped
157    /// unfulfilled, the consumer responded with `Err`, or the response was not a `RES`.
158    pub async fn fulfill_request<REQ, RES>(&self, _ctx: Context, req: REQ) -> Result<RES>
159    where
160        REQ: Any + Send + 'static,
161        RES: Any + Send + 'static,
162    {
163        self.ensure_started()?;
164
165        let (tx, rx) = oneshot::channel::<Result<BoxAny>>();
166        let promise = DriverRequest {
167            request: Box::new(req),
168            tx,
169        };
170        self.inner
171            .step_tx
172            .send(Ok(DriverStep::Request(promise)))
173            .await
174            .map_err(|_| DriverError::StreamClosed)?;
175
176        // Outer error: the promise was dropped without a response. Inner error: the
177        // consumer fulfilled it with an explicit `Err` — propagate it as-is.
178        let response = timeout(FULFILL_REQUEST_TIMEOUT, rx)
179            .await
180            .map_err(|_| DriverError::ResponseTimedOut {
181                timeout: FULFILL_REQUEST_TIMEOUT,
182            })?
183            .map_err(|_| DriverError::ResponseDropped)??;
184        response.downcast::<RES>().map(|boxed| *boxed).map_err(|_| {
185            DriverError::TypeMismatch {
186                expected: std::any::type_name::<RES>(),
187            }
188            .into()
189        })
190    }
191
192    /// Push a fire-and-forget [`DriverStep::Info`] payload; there is no promise to
193    /// await for a response. Awaits channel capacity (the consumer pacing the stream)
194    /// and errors only if the stream is closed.
195    pub async fn info<INFO>(&self, _ctx: Context, info: INFO) -> Result<()>
196    where
197        INFO: Any + Send + 'static,
198    {
199        self.ensure_started()?;
200
201        self.inner
202            .step_tx
203            .send(Ok(DriverStep::Info(Box::new(info))))
204            .await
205            .map_err(|_| DriverError::StreamClosed.into())
206    }
207
208    /// Emit the terminal [`DriverStep::Done`] with a final payload. Does not close the
209    /// stream (that happens when every `TypeErasedDriver` clone drops); the consumer treats it
210    /// as the last meaningful step. Awaits channel capacity and errors only if the
211    /// stream is closed.
212    pub async fn done<T>(&self, _ctx: Context, payload: T) -> Result<()>
213    where
214        T: Any + Send + 'static,
215    {
216        self.ensure_started()?;
217
218        self.inner
219            .step_tx
220            .send(Ok(DriverStep::Done(Box::new(payload))))
221            .await
222            .map_err(|_| DriverError::StreamClosed.into())
223    }
224
225    /// Terminate the stream with an error item — the producer-side way to surface a
226    /// failure to the consumer (mirrors how the crate's `run_stream` yields an `Err`
227    /// step). Awaits channel capacity and errors only if the stream is
228    /// already closed.
229    pub async fn fail(&self, _ctx: Context, err: LibsyError) -> Result<()> {
230        self.ensure_started()?;
231
232        self.inner
233            .step_tx
234            .send(Err(err))
235            .await
236            .map_err(|_| DriverError::StreamClosed.into())
237    }
238
239    /// Take the single consumer stream of [`DriverStep`]s. Callable once: a second
240    /// call yields a one-item stream carrying an `Err`, since the receiver is gone.
241    pub fn stream(&self) -> impl Stream<Item = Result<DriverStep>> + use<> {
242        let receiver = self
243            .inner
244            .step_rx
245            .lock()
246            .take()
247            .ok_or_else(|| LibsyError::from(DriverError::StreamAlreadyTaken));
248        match receiver {
249            Ok(rx) => ReceiverStream::new(rx).left_stream(),
250            Err(error) => futures::stream::once(async move { Err(error) }).right_stream(),
251        }
252    }
253}

275    async fn fulfill_request_round_trips_typed_values() -> Result<()> {
276        let driver = TypeErasedDriver::new();
277        let stream = driver.stream();
278
279        // Producer asks for a u32 -> String on its own task.
280        let producer = driver.clone();
281        let handle = tokio::spawn(async move {

471    #[tokio::test(flavor = "multi_thread", worker_threads = 4)]
472    async fn concurrent_producers_do_not_cross_responses() -> Result<()> {
473        const N: usize = 8;
474        let driver = TypeErasedDriver::new();
475        let stream = driver.stream();
476
477        // N producers each fulfill their own request concurrently.
478        let mut handles = Vec::new();
479        for i in 0..N {
480            let producer = driver.clone();
481            handles.push((
482                i,
483                tokio::spawn(async move {
484                    producer
485                        .fulfill_request::<usize, usize>(Context::default(), i)
486                        .await
487                }),
488            ));
489        }
490
491        // Single consumer responds `req * 10` to each request.
492        tokio::pin!(stream);
493        let mut served = 0;
494        while served < N {
495            match stream.next().await.ok_or(DriverError::StreamClosed)?? {
496                DriverStep::Request(promise) => {
497                    let req = *promise.request::<usize>()?;
498                    promise.respond::<usize>(Ok(req * 10))?;
499                    served += 1;
500                }
501                _ => return Err(test_error("expected a Request step")),
502            }
503        }
504
505        // Each producer must see exactly its own response, not another's.
506        for (i, handle) in handles {
507            assert_eq!(handle.await??, i * 10);
508        }
509        Ok(())
510    }

546    #[tokio::test]
547    async fn dropping_the_stream_terminates_the_producer() -> Result<()> {
548        let driver = TypeErasedDriver::new();
549        // Box::pin so the stream is owned here and `drop` actually drops the receiver.
550        // (`tokio::pin!` would rebind to a `Pin<&mut _>`, making `drop` a no-op.)
551        let mut stream = Box::pin(driver.stream());
552
553        // Producer publishes paced steps until the consumer goes away, then reports
554        // how many it managed to send.
555        let producer = driver.clone();
556        let handle = tokio::spawn(async move {
557            let mut sent = 0usize;
558            while producer.info(Context::default(), sent).await.is_ok() {
559                sent += 1;
560            }
561            sent
562        });
563
564        // Pace two steps, then terminate by dropping the stream.
565        for _ in 0..2 {
566            match stream.next().await.ok_or(DriverError::StreamClosed)?? {
567                DriverStep::Info(_) => {}
568                _ => return Err(test_error("expected an Info step")),
569            }
570        }
571        drop(stream);
572
573        // With the consumer gone, the producer's next publish errors and it stops.
574        let sent = handle.await?;
575        assert!(sent >= 2, "producer should have published the paced steps");
576        Ok(())
577    }

581        let driver = TypeErasedDriver::new();

583        // Before the consumer takes the stream, the receiver is still present, so producing
584        // is refused — with a message that points at the fix.
