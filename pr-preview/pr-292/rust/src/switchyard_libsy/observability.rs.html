4//! OpenTelemetry metrics plus `tracing` spans and structured logs for the
5//! algorithm layer.
6//!
7//! The crate's provided run methods call these helpers around the [`Decision`]
8//! hook and the offload boundary, so every algorithm is instrumented from the
9//! outside and carries no telemetry code of its own. Metrics record through the
10//! OpenTelemetry **global** meter provider under the `switchyard` scope — the host
11//! installs an SDK provider and exporters; with none installed, recording is a
12//! no-op. Spans and logs use the `tracing` facade (the async-native surface the
13//! OpenTelemetry ecosystem bridges with `tracing-opentelemetry` /
14//! `opentelemetry-appender-tracing`), so the host's subscriber decides where
15//! they go. Method spans use `#[tracing::instrument]`; the `libsy.run` span is
16//! attached to the spawned run task with [`tracing::Instrument`]. Neither holds
17//! a [`Span::enter`] guard across an `.await` — a suspended task would leave
18//! the span entered on its executor thread, mis-parenting every span other
19//! tasks create there (see the `tracing` docs on spans in asynchronous code).
20//!
21//! Instrument names use the OTel dotted form with the unit baked into the name
22//! (`switchyard.run_duration_ms`), matching the switchyard metric surface; a
23//! Prometheus exporter sanitizes them to `switchyard_run_duration_ms`. Attribute
24//! cardinality is bounded: `algorithm` and `selected_model` are small
25//! configured sets and `outcome` is `ok`/`error`. Nothing per-request becomes a
26//! metric attribute — correlation ids ride on the `libsy.run` span instead.
27//!
28//! Instruments are resolved from the global provider on every record (an
29//! instrument-cache lookup inside the SDK) so recording follows a meter
30//! provider installed at any point in the process lifetime; the cost is
31//! negligible next to a model call.

61/// [`Context::values`] key under which `run_stream` stamps the algorithm's

65/// The algorithm label carried by a request context; empty until stamped.
66pub(crate) fn algorithm_label(ctx: &Context) -> &str {
67    ctx.values
68        .get(ALGORITHM_KEY)

70        .unwrap_or("")

73/// The `libsy`-scoped meter from the globally installed provider.
74fn meter() -> Meter {
75    global::meter(METRICS_SCOPE)
76}

78/// Registers process-wide compatibility gauges with the installed global meter provider.
79pub(crate) fn initialize_metrics() {
80    TOTAL_GAUGES.get_or_init(|| {
81        let meter = meter();
82        let requests = meter
83            .u64_observable_gauge("switchyard.total_requests")
84            .with_callback(|observer| {
85                observer.observe(TOTAL_REQUESTS.load(Ordering::Relaxed), &[]);
86            })
87            .build();
88        let errors = meter
89            .u64_observable_gauge("switchyard.total_errors")
90            .with_callback(|observer| {
91                observer.observe(TOTAL_ERRORS.load(Ordering::Relaxed), &[]);
92            })
93            .build();
94        (requests, errors)
95    });
96}

103/// Span covering one algorithm run (the whole `create_run_task` execution).
104///
105/// Correlation ids from the request [`switchyard_protocol::Metadata`] are recorded as span fields
106/// when present. `tracing` spans cannot grow field names at runtime, so
107/// arbitrary host labels ride in via [`switchyard_protocol::Metadata::extra_metadata`], recorded
108/// whole into the `extra_metadata` field. `outcome` and `error` are filled in
109/// by [`record_run`] when the run ends.
110pub(crate) fn run_span(algorithm: &str, request: &Request) -> Span {
111    let span = tracing::info_span!(
112        target: TRACING_TARGET,
113        "libsy.run",
114        algorithm,
115        switchyard.algorithm = algorithm,
116        openinference.span.kind = "CHAIN",
117        switchyard.route = tracing::field::Empty,
118        session_id = tracing::field::Empty,
119        session.id = tracing::field::Empty,
120        agent_id = tracing::field::Empty,
121        task_id = tracing::field::Empty,
122        correlation_id = tracing::field::Empty,
123        extra_metadata = tracing::field::Empty,
124        outcome = tracing::field::Empty,
125        error = tracing::field::Empty,
126    );
127    if let Some(route) = request.requested_model() {
128        span.record("switchyard.route", route);
129    }
130    if let Some(metadata) = &request.metadata {
131        for (field, value) in [
132            ("session_id", &metadata.session_id),
133            ("agent_id", &metadata.agent_id),
134            ("task_id", &metadata.task_id),
135            ("correlation_id", &metadata.correlation_id),
136        ] {
137            if let Some(value) = value {
138                span.record(field, value.as_str());
139            }
140        }
141        if let Some(session_id) = &metadata.session_id {
142            span.record("session.id", session_id.as_str());
143        }
144        if let Some(extra) = &metadata.extra_metadata {
145            span.record("extra_metadata", tracing::field::debug(extra));
146        }
147    }
148    span
149}

151/// Runs one algorithm task to completion, recording the run counter, duration
152/// histogram, routing overhead, span outcome, and failure log when it resolves.
153/// Executes inside the `libsy.run` span its caller instruments the task with.
154/// `driver` is the run's own, holding the duration of the call that served it.
155pub(crate) async fn observe_run(
156    ctx: Context,
157    driver: Driver,
158    run: impl Future<Output = Result<Response>>,
159) -> Result<Response> {
160    let started = Instant::now();
161    let result = run.await;
162    let duration = started.elapsed();
163    let algorithm = algorithm_label(&ctx);
164    record_run(algorithm, duration, &result, &Span::current());
165    if result.is_ok()
166        && let Some(overhead) =
167            record_routing_overhead(algorithm, duration, driver.routed_call_duration())
168    {
169        driver.observe_routing_overhead(overhead);
170    }
171    result
172}

174/// Records request parameters represented directly by the neutral IR.
175pub(crate) fn record_gen_ai_request(span: &Span, request: &LlmRequest) {
176    if request.stream {
177        span.record("gen_ai.request.stream", true);
178    }
179    if let Some(value) = request.sampling.temperature {
180        span.record("gen_ai.request.temperature", value);
181    }
182    if let Some(value) = request.sampling.top_p {
183        span.record("gen_ai.request.top_p", value);
184    }
185    if let Some(value) = request.sampling.top_k {
186        span.record("gen_ai.request.top_k", value);
187    }
188    if let Some(value) = request.output.max_output_tokens {
189        span.record("gen_ai.request.max_tokens", otel_int(value));
190    }
191    if let Some(value) = request.reasoning.effort.as_deref() {
192        span.record("gen_ai.request.reasoning.level", value);
193    }
194    if let Some(value) = request
195        .output
196        .response_format
197        .as_ref()
198        .and_then(gen_ai_output_type)
199    {
200        span.record("gen_ai.output.type", value);
201    }
202}

204/// Adds terminal response and usage fields to the enclosing `libsy.client_call`
205/// span without consuming or buffering a streaming response.
206pub(crate) fn observe_client_call(result: Result<Response>) -> Result<Response> {
207    let span = Span::current();
208    match result {
209        Ok(mut response) => {
210            match response.llm_response {
211                LlmResponse::Agg(agg) => {
212                    span.record("outcome", "ok");
213                    record_gen_ai_response(&span, &agg);
214                    response.llm_response = LlmResponse::Agg(agg);
215                }
216                LlmResponse::Stream(stream) => {
217                    response.llm_response =
218                        LlmResponse::Stream(observe_client_stream(stream, span));
219                }
220            }
221            Ok(response)
222        }
223        Err(error) => {
224            let error_type = client_call_error_type(&error);
225            record_client_error(&span, &error_type, &error);
226            Err(error)
227        }
228    }
229}

231fn observe_client_stream(stream: LlmResponseStream, span: Span) -> LlmResponseStream {
232    Box::pin(ObservedClientStream {
233        stream,
234        observer: Some(ClientStreamObserver {
235            span,
236            terminal: false,
237        }),
238    })

241fn record_gen_ai_response(span: &Span, response: &AggLlmResponse) {
242    record_optional(span, "gen_ai.response.id", response.id.as_deref());
243    record_optional(span, "gen_ai.response.model", response.model.as_deref());
244    record_finish_reasons(
245        span,
246        response
247            .outputs
248            .iter()
249            .filter_map(|output| output.stop_reason)
250            .map(stop_reason_name),
251    );
252    record_gen_ai_usage(span, &response.usage);
253}

255fn record_gen_ai_usage(span: &Span, usage: &Usage) {
256    let cache_read = usage.cached_input_tokens();
257    let cache_creation = usage.cache_creation_input_tokens();
258    if usage.input_tokens.is_some() || cache_read.is_some() || cache_creation.is_some() {
259        let input_tokens = usage
260            .input_tokens
261            .unwrap_or_default()
262            .saturating_add(cache_read.unwrap_or_default())
263            .saturating_add(cache_creation.unwrap_or_default());
264        span.record("gen_ai.usage.input_tokens", otel_int(input_tokens));
265    }
266    for (field, value) in [
267        ("gen_ai.usage.output_tokens", usage.output_tokens),
268        ("gen_ai.usage.cache_read.input_tokens", cache_read),
269        ("gen_ai.usage.cache_creation.input_tokens", cache_creation),
270        (
271            "gen_ai.usage.reasoning.output_tokens",
272            usage.reasoning_tokens,
273        ),
274    ] {
275        if let Some(value) = value {
276            span.record(field, otel_int(value));
277        }
278    }
279}

292// `tracing` fields cannot preserve a typed string array, so write this attribute directly.
293fn record_finish_reasons(span: &Span, reasons: impl IntoIterator<Item = impl Into<String>>) {
294    let reasons = reasons
295        .into_iter()
296        .map(|reason| StringValue::from(reason.into()))
297        .collect::<Vec<_>>();
298    if !reasons.is_empty() {
299        span.set_attribute(
300            "gen_ai.response.finish_reasons",
301            OtelValue::Array(OtelArray::String(reasons)),
302        );
303    }
304}

331        LibsyError::TargetNotFound { .. } => Cow::Borrowed("target_not_found"),

360// Keep the client span alive until the response is drained, errors, or is abandoned.
361struct ObservedClientStream {
362    stream: LlmResponseStream,
363    observer: Option<ClientStreamObserver>,
364}
365
366impl Stream for ObservedClientStream {
367    type Item = std::result::Result<LlmResponseStreamEvent, LlmClientError>;
368
369    fn poll_next(mut self: Pin<&mut Self>, cx: &mut TaskContext<'_>) -> Poll<Option<Self::Item>> {
370        match self.stream.as_mut().poll_next(cx) {
371            Poll::Ready(Some(item)) => {
372                let failed = self
373                    .observer
374                    .as_mut()
375                    .is_some_and(|observer| observer.observe(&item));
376                if failed {
377                    self.observer.take();
378                }
379                Poll::Ready(Some(item))
380            }
381            Poll::Ready(None) => {
382                if let Some(mut observer) = self.observer.take() {
383                    observer.complete();
384                }
385                Poll::Ready(None)
386            }
387            Poll::Pending => Poll::Pending,
388        }
389    }
390}

392struct ClientStreamObserver {
393    span: Span,
394    terminal: bool,
395}
396
397impl ClientStreamObserver {
398    fn observe(
399        &mut self,
400        item: &std::result::Result<LlmResponseStreamEvent, LlmClientError>,
401    ) -> bool {
402        match item {
403            Ok(event) => {
404                for chunk in event.normalized() {
405                    self.observe_chunk(chunk);
406                    if self.terminal {
407                        break;
408                    }
409                }
410            }
411            Err(error) => {
412                let error_type = llm_client_error_type(error);
413                record_client_error(&self.span, &error_type, error);
414                self.terminal = true;
415            }
416        }
417        self.terminal
418    }
419
420    fn observe_chunk(&mut self, chunk: &LlmResponseChunk) {
421        match chunk {
422            LlmResponseChunk::MessageStart { id, model } => {
423                record_optional(&self.span, "gen_ai.response.id", id.as_deref());
424                record_optional(&self.span, "gen_ai.response.model", model.as_deref());
425            }
426            LlmResponseChunk::Usage(usage) => record_gen_ai_usage(&self.span, usage),
427            LlmResponseChunk::MessageStop { reason } => {
428                record_finish_reasons(&self.span, reason.iter().cloned());
429            }
430            LlmResponseChunk::DecodeError { message } => {
431                record_client_error(&self.span, "response_translation", message);
432                self.terminal = true;
433            }
434            LlmResponseChunk::StreamError { message } => {
435                record_client_error(&self.span, "502", message);
436                self.terminal = true;
437            }
438            _ => {}
439        }
440    }
441
442    fn complete(&mut self) {
443        self.span.record("outcome", "ok");
444        self.terminal = true;
445    }
446}
447
448impl Drop for ClientStreamObserver {
449    fn drop(&mut self) {
450        if !self.terminal {
451            self.span.record("outcome", "cancelled");
452        }
453    }
454}

456fn record_client_error(span: &Span, error_type: &str, error: &dyn std::fmt::Display) {
457    span.record("outcome", "error");
458    span.record("otel.status_code", "ERROR");
459    span.record("error.type", error_type);

463/// Records the end of one algorithm run: the run counter and duration
464/// histogram, the `outcome`/`error` fields on `span`, and a warn log when the
465/// run failed.
466fn record_run(algorithm: &str, duration: Duration, result: &Result<Response>, span: &Span) {
467    let outcome = outcome_value(result);
468    span.record("outcome", outcome);
469    if let Err(error) = result {
470        span.record("error", tracing::field::display(error));
471        tracing::warn!(
472            target: TRACING_TARGET,
473            algorithm,
474            error = %error,
475            "algorithm run failed"
476        );
477    }
478
479    let attributes = [
480        KeyValue::new("algorithm", algorithm.to_string()),
481        KeyValue::new("outcome", outcome),
482    ];
483    let meter = meter();
484    meter
485        .u64_counter("switchyard.runs")
486        .build()
487        .add(1, &attributes);
488    meter
489        .f64_histogram("switchyard.run_duration_ms")
490        .build()
491        .record(duration.as_secs_f64() * 1000.0, &attributes);
492}

494/// Records what routing cost on top of the call that served the run: classifier
495/// calls, target resolution, decision publishing. A run with no routed call has
496/// nothing to subtract, so it records nothing.
497fn record_routing_overhead(
498    algorithm: &str,
499    run: Duration,
500    routed_call: Option<Duration>,
501) -> Option<Duration> {
502    let routed_call = routed_call?;
503    // Saturating: the two clocks start a moment apart, so a run that is all
504    // routed call can come out fractionally negative.
505    let overhead = run.saturating_sub(routed_call);
506    meter()
507        .f64_histogram("switchyard.routing_overhead_ms")
508        .build()
509        .record(
510            overhead.as_secs_f64() * 1000.0,
511            &[KeyValue::new("algorithm", algorithm.to_string())],
512        );
513    Some(overhead)
514}

516/// Records the resolution of one offloaded model call: the call counter and
517/// latency histogram, the `outcome`/`error`/token fields on `span`, and a warn
518/// log when the call failed.
519pub(crate) fn record_llm_call(
520    algorithm: &str,
521    selected_model: &str,
522    tier: Option<&str>,
523    is_routed: bool,
524    duration: Duration,
525    result: &Result<Response>,
526    span: &Span,
527) {
528    let outcome = outcome_value(result);
529    span.record("outcome", outcome);
530
531    let meter = meter();
532    let call_attributes = [
533        KeyValue::new("algorithm", algorithm.to_string()),
534        KeyValue::new("selected_model", selected_model.to_string()),
535        KeyValue::new("outcome", outcome),
536    ];
537    meter
538        .u64_counter("switchyard.llm_calls")
539        .build()
540        .add(1, &call_attributes);
541    meter
542        .f64_histogram("switchyard.llm_call_duration_ms")
543        .build()
544        .record(duration.as_secs_f64() * 1000.0, &call_attributes);
545
546    if is_routed {
547        TOTAL_REQUESTS.fetch_add(1, Ordering::Relaxed);
548        let mut routed_attributes = vec![KeyValue::new("model", selected_model.to_string())];
549        if let Some(tier) = tier {
550            routed_attributes.push(KeyValue::new("tier", tier.to_string()));
551        }
552        if result.is_ok() {
553            meter
554                .u64_counter("switchyard.requests")
555                .build()
556                .add(1, &routed_attributes);
557            meter
558                .f64_histogram("switchyard.model_call_latency_ms")
559                .build()
560                .record(duration.as_secs_f64() * 1000.0, &routed_attributes);
561        } else {
562            TOTAL_ERRORS.fetch_add(1, Ordering::Relaxed);
563            meter
564                .u64_counter("switchyard.errors")
565                .build()
566                .add(1, &routed_attributes);
567        }
568    }
569
570    match result {
571        Ok(response) => {
572            // Token usage exists only once a response is buffered; a streamed
573            // response resolves before its usage is known, so none is recorded.
574            let Some(usage) = response.llm_response.as_agg().map(|agg| &agg.usage) else {
575                return;
576            };
577            for (field, value) in [
578                ("input_tokens", usage.input_tokens),
579                ("output_tokens", usage.output_tokens),
580                ("total_tokens", usage.total_tokens),
581                ("reasoning_tokens", usage.reasoning_tokens),
582            ] {
583                if let Some(value) = value {
584                    span.record(field, value);
585                }
586            }
587        }
588        Err(error) => {
589            span.record("error", tracing::field::display(error));
590            tracing::warn!(
591                target: TRACING_TARGET,
592                algorithm,
593                selected_model,
594                error = %error,
595                "model call failed"
596            );
597        }
598    }
599}

601/// Records one published routing decision: the decision counter plus a
602/// structured debug event carrying the decision's reasoning.
603pub(crate) fn record_decision(ctx: &Context, decision: &dyn Decision) {
604    let algorithm = algorithm_label(ctx);
605    let selected_model = decision.selected_model();
606    tracing::debug!(
607        target: TRACING_TARGET,
608        algorithm,
609        selected_model,
610        reasoning = decision.reasoning().unwrap_or(""),
611        "routing decision"
612    );
613    meter().u64_counter("switchyard.decisions").build().add(
614        1,
615        &[
616            KeyValue::new("algorithm", algorithm.to_string()),
617            KeyValue::new("selected_model", selected_model.to_string()),
618        ],
619    );
620}
