4//! OpenTelemetry metrics plus `tracing` spans and structured logs for the
5//! algorithm layer, including algorithm-defined metrics emitted through the
6//! [`Driver`](crate::Driver).
7//!
8//! The crate's provided run methods instrument the [`Decision`] hook and offload
9//! boundary from the outside. Algorithms emit domain-specific counters and
10//! histograms through the [`Driver`](crate::Driver). Metrics record through the
11//! OpenTelemetry **global** meter provider under the `switchyard` scope — the host
12//! installs an SDK provider and exporters; with none installed, recording is a
13//! no-op. Spans and logs use the `tracing` facade (the async-native surface the
14//! OpenTelemetry ecosystem bridges with `tracing-opentelemetry` /
15//! `opentelemetry-appender-tracing`), so the host's subscriber decides where
16//! they go. Method spans use `#[tracing::instrument]`; the `libsy.run` span is
17//! attached to the spawned run task with [`tracing::Instrument`]. Neither holds
18//! a [`Span::enter`] guard across an `.await` — a suspended task would leave
19//! the span entered on its executor thread, mis-parenting every span other
20//! tasks create there (see the `tracing` docs on spans in asynchronous code).
21//!
22//! Instrument names use the OTel dotted form with the unit baked into the name
23//! (`switchyard.run_duration_ms`), matching the switchyard metric surface; a
24//! Prometheus exporter sanitizes them to `switchyard_run_duration_ms`. Attribute
25//! cardinality is bounded: `algorithm` and `selected_model` are small configured
26//! sets and `outcome` is `ok`/`error`. Algorithm-defined metric attributes follow
27//! the same contract. Nothing per-request becomes a metric attribute — correlation
28//! ids ride on the `libsy.run` span instead.
29//!
30//! Instruments are resolved from the global provider on every record (an
31//! instrument-cache lookup inside the SDK) so recording follows a meter
32//! provider installed at any point in the process lifetime; the cost is
33//! negligible next to a model call.
34

76/// The `libsy`-scoped meter from the globally installed provider.
77fn meter() -> Meter {
78    global::meter(METRICS_SCOPE)
79}

81/// Registers process-wide compatibility gauges with the installed global meter provider.
82pub(crate) fn initialize_metrics() {
83    TOTAL_GAUGES.get_or_init(|| {
84        let meter = meter();
85        let requests = meter
86            .u64_observable_gauge("switchyard.total_requests")
87            .with_callback(|observer| {
88                observer.observe(TOTAL_REQUESTS.load(Ordering::Relaxed), &[]);
89            })
90            .build();
91        let errors = meter
92            .u64_observable_gauge("switchyard.total_errors")
93            .with_callback(|observer| {
94                observer.observe(TOTAL_ERRORS.load(Ordering::Relaxed), &[]);
95            })
96            .build();
97        (requests, errors)
98    });
99}

106/// Span covering one algorithm run (the whole `create_run_task` execution).
107///
108/// Correlation ids from the request [`switchyard_protocol::Metadata`] are recorded as span fields
109/// when present. `tracing` spans cannot grow field names at runtime, so
110/// arbitrary host labels ride in via [`switchyard_protocol::Metadata::extra_metadata`], recorded
111/// whole into the `extra_metadata` field. `outcome` and `error` are filled in
112/// by [`record_run`] when the run ends.
113pub(crate) fn run_span(algorithm: &str, request: &Request) -> Span {
114    let span = tracing::info_span!(
115        target: TRACING_TARGET,
116        "libsy.run",
117        algorithm,
118        switchyard.algorithm = algorithm,
119        openinference.span.kind = "CHAIN",
120        switchyard.route = tracing::field::Empty,
121        session_id = tracing::field::Empty,
122        session.id = tracing::field::Empty,
123        agent_id = tracing::field::Empty,
124        task_id = tracing::field::Empty,
125        correlation_id = tracing::field::Empty,
126        extra_metadata = tracing::field::Empty,
127        outcome = tracing::field::Empty,
128        error = tracing::field::Empty,
129    );
130    if let Some(route) = request.requested_model() {
131        span.record("switchyard.route", route);
132    }
133    if let Some(metadata) = &request.metadata {
134        for (field, value) in [
135            ("session_id", &metadata.session_id),
136            ("agent_id", &metadata.agent_id),
137            ("task_id", &metadata.task_id),
138            ("correlation_id", &metadata.correlation_id),
139        ] {
140            if let Some(value) = value {
141                span.record(field, value.as_str());
142            }
143        }
144        if let Some(session_id) = &metadata.session_id {
145            span.record("session.id", session_id.as_str());
146        }
147        if let Some(extra) = &metadata.extra_metadata {
148            span.record("extra_metadata", tracing::field::debug(extra));
149        }
150    }
151    span
152}

154/// Runs one algorithm task to completion, recording the run counter, duration
155/// histogram, routing overhead, span outcome, and failure log when it resolves.
156/// Executes inside the `libsy.run` span its caller instruments the task with.
157/// `driver` is the run's own, holding the duration of the call that served it.
158pub(crate) async fn observe_run(
159    ctx: Context,
160    driver: Driver,
161    run: impl Future<Output = Result<Response>>,
162) -> Result<Response> {
163    let started = Instant::now();
164    let result = run.await;
165    let duration = started.elapsed();
166    let algorithm = algorithm_label(&ctx);
167    record_run(algorithm, duration, &result, &Span::current());
168    if result.is_ok()
169        && let Some(overhead) =
170            record_routing_overhead(algorithm, duration, driver.routed_call_duration())
171    {
172        driver.observe_routing_overhead(overhead);
173    }
174    result
175}

177/// Records request parameters represented directly by the neutral IR.
178pub(crate) fn record_gen_ai_request(span: &Span, request: &LlmRequest) {
179    if request.stream {
180        span.record("gen_ai.request.stream", true);
181    }
182    if let Some(value) = request.sampling.temperature {
183        span.record("gen_ai.request.temperature", value);
184    }
185    if let Some(value) = request.sampling.top_p {
186        span.record("gen_ai.request.top_p", value);
187    }
188    if let Some(value) = request.sampling.top_k {
189        span.record("gen_ai.request.top_k", value);
190    }
191    if let Some(value) = request.output.max_output_tokens {
192        span.record("gen_ai.request.max_tokens", otel_int(value));
193    }
194    if let Some(value) = request.reasoning.effort.as_deref() {
195        span.record("gen_ai.request.reasoning.level", value);
196    }
197    if let Some(value) = request
198        .output
199        .response_format
200        .as_ref()
201        .and_then(gen_ai_output_type)
202    {
203        span.record("gen_ai.output.type", value);
204    }
205}

207/// Adds terminal response and usage fields to the enclosing `libsy.client_call`
208/// span without consuming or buffering a streaming response.
209pub(crate) fn observe_client_call(result: Result<Response>) -> Result<Response> {
210    let span = Span::current();
211    match result {
212        Ok(mut response) => {
213            match response.llm_response {
214                LlmResponse::Agg(agg) => {
215                    span.record("outcome", "ok");
216                    record_gen_ai_response(&span, &agg);
217                    response.llm_response = LlmResponse::Agg(agg);
218                }
219                LlmResponse::Stream(stream) => {
220                    response.llm_response =
221                        LlmResponse::Stream(observe_client_stream(stream, span));
222                }
223            }
224            Ok(response)
225        }
226        Err(error) => {
227            let error_type = client_call_error_type(&error);
228            record_client_error(&span, &error_type, &error);
229            Err(error)
230        }
231    }
232}

234fn observe_client_stream(stream: LlmResponseStream, span: Span) -> LlmResponseStream {
235    Box::pin(ObservedClientStream {
236        stream,
237        observer: Some(ClientStreamObserver {
238            span,
239            terminal: false,

244fn record_gen_ai_response(span: &Span, response: &AggLlmResponse) {
245    record_optional(span, "gen_ai.response.id", response.id.as_deref());
246    record_optional(span, "gen_ai.response.model", response.model.as_deref());
247    record_finish_reasons(
248        span,
249        response
250            .outputs
251            .iter()
252            .filter_map(|output| output.stop_reason)
253            .map(stop_reason_name),
254    );
255    record_gen_ai_usage(span, &response.usage);
256}

258fn record_gen_ai_usage(span: &Span, usage: &Usage) {
259    let cache_read = usage.cached_input_tokens();
260    let cache_creation = usage.cache_creation_input_tokens();
261    if usage.input_tokens.is_some() || cache_read.is_some() || cache_creation.is_some() {
262        let input_tokens = usage
263            .input_tokens
264            .unwrap_or_default()
265            .saturating_add(cache_read.unwrap_or_default())
266            .saturating_add(cache_creation.unwrap_or_default());
267        span.record("gen_ai.usage.input_tokens", otel_int(input_tokens));
268    }
269    for (field, value) in [
270        ("gen_ai.usage.output_tokens", usage.output_tokens),
271        ("gen_ai.usage.cache_read.input_tokens", cache_read),
272        ("gen_ai.usage.cache_creation.input_tokens", cache_creation),
273        (
274            "gen_ai.usage.reasoning.output_tokens",
275            usage.reasoning_tokens,
276        ),
277    ] {
278        if let Some(value) = value {
279            span.record(field, otel_int(value));
280        }
281    }
282}

295// `tracing` fields cannot preserve a typed string array, so write this attribute directly.
296fn record_finish_reasons(span: &Span, reasons: impl IntoIterator<Item = impl Into<String>>) {
297    let reasons = reasons
298        .into_iter()
299        .map(|reason| StringValue::from(reason.into()))
300        .collect::<Vec<_>>();
301    if !reasons.is_empty() {
302        span.set_attribute(
303            "gen_ai.response.finish_reasons",
304            OtelValue::Array(OtelArray::String(reasons)),
305        );
306    }
307}

363// Keep the client span alive until the response is drained, errors, or is abandoned.
364struct ObservedClientStream {
365    stream: LlmResponseStream,
366    observer: Option<ClientStreamObserver>,
367}
368
369impl Stream for ObservedClientStream {
370    type Item = std::result::Result<LlmResponseStreamEvent, LlmClientError>;
371
372    fn poll_next(mut self: Pin<&mut Self>, cx: &mut TaskContext<'_>) -> Poll<Option<Self::Item>> {
373        match self.stream.as_mut().poll_next(cx) {
374            Poll::Ready(Some(item)) => {
375                let failed = self
376                    .observer
377                    .as_mut()
378                    .is_some_and(|observer| observer.observe(&item));
379                if failed {
380                    self.observer.take();
381                }
382                Poll::Ready(Some(item))
383            }
384            Poll::Ready(None) => {
385                if let Some(mut observer) = self.observer.take() {
386                    observer.complete();
387                }
388                Poll::Ready(None)
389            }
390            Poll::Pending => Poll::Pending,
391        }
392    }
393}
394
395struct ClientStreamObserver {
396    span: Span,
397    terminal: bool,
398}
399
400impl ClientStreamObserver {
401    fn observe(
402        &mut self,
403        item: &std::result::Result<LlmResponseStreamEvent, LlmClientError>,
404    ) -> bool {
405        match item {
406            Ok(event) => {
407                for chunk in event.normalized() {
408                    self.observe_chunk(chunk);
409                    if self.terminal {
410                        break;
411                    }
412                }
413            }
414            Err(error) => {
415                let error_type = llm_client_error_type(error);
416                record_client_error(&self.span, &error_type, error);
417                self.terminal = true;
418            }
419        }
420        self.terminal
421    }
422
423    fn observe_chunk(&mut self, chunk: &LlmResponseChunk) {
424        match chunk {
425            LlmResponseChunk::MessageStart { id, model } => {
426                record_optional(&self.span, "gen_ai.response.id", id.as_deref());
427                record_optional(&self.span, "gen_ai.response.model", model.as_deref());
428            }
429            LlmResponseChunk::Usage(usage) => record_gen_ai_usage(&self.span, usage),
430            LlmResponseChunk::MessageStop { reason } => {
431                record_finish_reasons(&self.span, reason.iter().cloned());
432            }
433            LlmResponseChunk::DecodeError { message } => {
434                record_client_error(&self.span, "response_translation", message);
435                self.terminal = true;
436            }
437            LlmResponseChunk::StreamError { message } => {
438                record_client_error(&self.span, "502", message);
439                self.terminal = true;
440            }
441            _ => {}
442        }
443    }
444
445    fn complete(&mut self) {
446        self.span.record("outcome", "ok");
447        self.terminal = true;
448    }
449}
450
451impl Drop for ClientStreamObserver {
452    fn drop(&mut self) {
453        if !self.terminal {
454            self.span.record("outcome", "cancelled");
455        }
456    }
457}

459fn record_client_error(span: &Span, error_type: &str, error: &dyn std::fmt::Display) {
460    span.record("outcome", "error");
461    span.record("otel.status_code", "ERROR");
462    span.record("error.type", error_type);

466/// Records the end of one algorithm run: the run counter and duration
467/// histogram, the `outcome`/`error` fields on `span`, and a warn log when the
468/// run failed.
469fn record_run(algorithm: &str, duration: Duration, result: &Result<Response>, span: &Span) {
470    let outcome = outcome_value(result);
471    span.record("outcome", outcome);
472    if let Err(error) = result {
473        span.record("error", tracing::field::display(error));
474        tracing::warn!(
475            target: TRACING_TARGET,
476            algorithm,
477            error = %error,
478            "algorithm run failed"
479        );
480    }
481
482    let attributes = [
483        KeyValue::new("algorithm", algorithm.to_string()),
484        KeyValue::new("outcome", outcome),
485    ];
486    let meter = meter();
487    meter
488        .u64_counter("switchyard.runs")
489        .build()
490        .add(1, &attributes);
491    meter
492        .f64_histogram("switchyard.run_duration_ms")
493        .build()
494        .record(duration.as_secs_f64() * 1000.0, &attributes);
495}

497/// Records what routing cost on top of the call that served the run: classifier
498/// calls, target resolution, decision publishing. A run with no routed call has
499/// nothing to subtract, so it records nothing.
500fn record_routing_overhead(
501    algorithm: &str,
502    run: Duration,
503    routed_call: Option<Duration>,
504) -> Option<Duration> {
505    let routed_call = routed_call?;
506    // Saturating: the two clocks start a moment apart, so a run that is all
507    // routed call can come out fractionally negative.
508    let overhead = run.saturating_sub(routed_call);
509    meter()
510        .f64_histogram("switchyard.routing_overhead_ms")
511        .build()
512        .record(
513            overhead.as_secs_f64() * 1000.0,
514            &[KeyValue::new("algorithm", algorithm.to_string())],
515        );
516    Some(overhead)
517}

519/// Records an algorithm metric after adding its algorithm dimension.
520pub(crate) fn record_algorithm_metric(metric: &AlgorithmMetricObservation) {
521    let mut attributes = Vec::with_capacity(metric.attributes.len() + 1);
522    attributes.push(KeyValue::new("algorithm", metric.algorithm.clone()));
523    attributes.extend(
524        metric
525            .attributes
526            .iter()
527            .map(|attribute| KeyValue::new(attribute.key, attribute.value.clone())),
528    );
529
530    let meter = meter();
531    match metric.value {
532        AlgorithmMetricValue::Counter(delta) => meter
533            .u64_counter(metric.name)
534            .build()
535            .add(delta, &attributes),
536        AlgorithmMetricValue::Histogram(sample) => meter
537            .f64_histogram(metric.name)
538            .build()
539            .record(sample, &attributes),
540    }
541}

543/// Records the resolution of one offloaded model call: the call counter and
544/// latency histogram, the `outcome`/`error`/token fields on `span`, and a warn
545/// log when the call failed.
546pub(crate) fn record_llm_call(
547    algorithm: &str,
548    selected_model: &str,
549    tier: Option<&str>,
550    is_routed: bool,
551    duration: Duration,
552    result: &Result<Response>,
553    span: &Span,
554) {
555    let outcome = outcome_value(result);
556    span.record("outcome", outcome);
557
558    let meter = meter();
559    let call_attributes = [
560        KeyValue::new("algorithm", algorithm.to_string()),
561        KeyValue::new("selected_model", selected_model.to_string()),
562        KeyValue::new("outcome", outcome),
563    ];
564    meter
565        .u64_counter("switchyard.llm_calls")
566        .build()
567        .add(1, &call_attributes);
568    meter
569        .f64_histogram("switchyard.llm_call_duration_ms")
570        .build()
571        .record(duration.as_secs_f64() * 1000.0, &call_attributes);
572
573    if is_routed {
574        TOTAL_REQUESTS.fetch_add(1, Ordering::Relaxed);
575        let mut routed_attributes = vec![KeyValue::new("model", selected_model.to_string())];
576        if let Some(tier) = tier {
577            routed_attributes.push(KeyValue::new("tier", tier.to_string()));
578        }
579        if result.is_ok() {
580            meter
581                .u64_counter("switchyard.requests")
582                .build()
583                .add(1, &routed_attributes);
584            meter
585                .f64_histogram("switchyard.model_call_latency_ms")
586                .build()
587                .record(duration.as_secs_f64() * 1000.0, &routed_attributes);
588        } else {
589            TOTAL_ERRORS.fetch_add(1, Ordering::Relaxed);
590            meter
591                .u64_counter("switchyard.errors")
592                .build()
593                .add(1, &routed_attributes);
594        }
595    }
596
597    match result {
598        Ok(response) => {
599            // Token usage exists only once a response is buffered; a streamed
600            // response resolves before its usage is known, so none is recorded.
601            let Some(usage) = response.llm_response.as_agg().map(|agg| &agg.usage) else {
602                return;
603            };
604            for (field, value) in [
605                ("input_tokens", usage.input_tokens),
606                ("output_tokens", usage.output_tokens),
607                ("total_tokens", usage.total_tokens),
608                ("reasoning_tokens", usage.reasoning_tokens),
609            ] {
610                if let Some(value) = value {
611                    span.record(field, value);
612                }
613            }
614        }
615        Err(error) => {
616            span.record("error", tracing::field::display(error));
617            tracing::warn!(
618                target: TRACING_TARGET,
619                algorithm,
620                selected_model,
621                error = %error,
622                "model call failed"
623            );
624        }
625    }
626}

628/// Records one published routing decision: the decision counter plus a
629/// structured debug event carrying the decision's reasoning.
630pub(crate) fn record_decision(ctx: &Context, decision: &dyn Decision) {
631    let algorithm = algorithm_label(ctx);
632    let selected_model = decision.selected_model();
633    tracing::debug!(
634        target: TRACING_TARGET,
635        algorithm,
636        selected_model,
