4//! Buffered HTTP-client translation with diagnostic metrics and logs.
5
6use std::sync::LazyLock;
7
8use serde_json::Value;
9use switchyard_protocol::{AggLlmResponse, LlmRequest, ProviderExtensions};
10use switchyard_translation::{Result, TranslationEngine, TranslationPolicy, WireFormat};
11
12use crate::metrics;
13
14static ENGINE: LazyLock<TranslationEngine> = LazyLock::new(TranslationEngine::default);
15static POLICY: LazyLock<TranslationPolicy> = LazyLock::new(TranslationPolicy::default);
16
17pub(crate) fn decode_request(format: WireFormat, body: &Value) -> Result<LlmRequest> {
18    let decoded = ENGINE.decode_request(format, body, &POLICY)?;
19    metrics::record_translation_diagnostics(&decoded.diagnostics, "request_decode", format);
20    Ok(decoded.request)
21}
22
23pub(crate) fn encode_request(request: &LlmRequest, format: WireFormat) -> Result<Value> {
24    let encoded = ENGINE.encode_request(format, request, &POLICY)?;
25    metrics::record_translation_diagnostics(&encoded.diagnostics, "request_encode", format);
26    Ok(encoded.body)
27}
28
29pub(crate) fn decode_response(body: &Value, format: WireFormat) -> Result<AggLlmResponse> {
30    let decoded = ENGINE.decode_response(format, body, &POLICY)?;
31    metrics::record_translation_diagnostics(&decoded.diagnostics, "response_decode", format);
32    Ok(decoded.response)
33}
34
35pub(crate) fn encode_response(
36    response: &AggLlmResponse,
37    format: WireFormat,
38    served_model: Option<&str>,
39    request_extensions: &ProviderExtensions,
40) -> Result<Value> {
41    let mut encoded =
42        ENGINE.encode_response_with_extensions(format, response, request_extensions, &POLICY)?;
43    metrics::record_translation_diagnostics(&encoded.diagnostics, "response_encode", format);
44    if let (Some(model), Value::Object(body)) = (served_model, &mut encoded.body) {
45        body.insert("model".to_string(), Value::String(model.to_string()));
46    }
47    Ok(encoded.body)
48}
