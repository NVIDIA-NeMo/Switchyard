4//! Sub-agent override as a single SDK component.
5//!
6//! [`SubagentOverride`] scores one fixed worker target for requests carrying delegated
7//! sub-agent work ([`Metadata::is_subagent_work`]) and abstains for everything else, so a
8//! cascade falls through to its later classifiers on ordinary traffic.
9//!
10//! It is stateless and holds only the worker's *name*: the fall-through cascade resolves it
11//! against its
12//! target set. Keeping the policy independent of any memory of past decisions is what lets
13//! it compose with a stateful classifier such as
14//! [`AffinityRouter`](crate::algorithms::AffinityRouter) — the override decides *which*
15//! target delegated work belongs on, affinity decides *how long* a decision lives, and
16//! neither needs to know about the other.
17

29/// Classifies delegated work from its parent-supplied prompt and abstains otherwise.
30///
31/// The inner classifier receives a prompt-only request clone. The original request remains
32/// unchanged for the selected child model.
33pub struct SubagentGate<S> {
34    inner: Arc<dyn Classifier<S>>,
35}
36
37impl<S> SubagentGate<S> {
38    /// Wraps `inner` with delegated-work detection.
39    pub fn new(inner: Arc<dyn Classifier<S>>) -> Self {
40        Self { inner }
41    }
42}
43
44/// Builds the prompt-only request shown to a delegated-work classifier.
45fn delegated_prompt_request(request: &Request) -> Option<Request> {
46    // Coding harnesses append the parent's task after their injected user context and reminders.
47    let prompt = request
48        .llm_request
49        .messages
50        .iter()
51        .rev()
52        .find(|message| message.role == Role::User)?
53        .content
54        .iter()
55        .rev()
56        .find_map(|block| match block {
57            ContentBlock::Text { text } if !text.trim().is_empty() => Some(text.clone()),
58            _ => None,
59        })?;
60
61    Some(Request {
62        llm_request: LlmRequest {
63            model: request.llm_request.model.clone(),
64            messages: vec![Message::text(Role::User, prompt)],
65            ..LlmRequest::default()
66        },
67        raw_request: None,
68        metadata: request.metadata.clone(),
69    })
70}
71
72#[async_trait]
73impl<S> Classifier<S> for SubagentGate<S>
74where
75    S: Send + 'static,
76{
77    async fn score(
78        &self,
79        state: &mut S,
80        request: &mut Request,
81        driver: Option<&Driver>,
82    ) -> Result<(Classification, Option<Response>)> {
83        if !request
84            .metadata
85            .as_ref()
86            .is_some_and(Metadata::is_subagent_work)
87        {
88            return Ok((Classification::Scores(Vec::new()), None));
89        }
90        let Some(mut classifier_request) = delegated_prompt_request(request) else {
91            return Ok((Classification::Scores(Vec::new()), None));
92        };
93        self.inner
94            .score(state, &mut classifier_request, driver)
95            .await
96    }
97}

99/// Scores a fixed worker target for delegated sub-agent work; abstains otherwise.
100pub struct SubagentOverride {
101    /// Name of the worker target, resolved by the cascade against its target set.
102    worker: ModelId,
103}
104
105impl SubagentOverride {
106    /// Creates an override scoring `worker` for delegated sub-agent work.
107    ///
108    /// `worker` must name a target in the cascade's set, or routing a sub-agent request
109    /// fails with [`LibsyError::TargetNotFound`](crate::LibsyError::TargetNotFound).
110    pub fn new(worker: impl Into<ModelId>) -> Self {
111        Self {
112            worker: worker.into(),
113        }
114    }
115}
116
117#[async_trait]
118impl<S> Classifier<S> for SubagentOverride
119where
120    S: Send + 'static,
121{
122    async fn score(
123        &self,
124        _state: &mut S,
125        request: &mut Request,
126        _driver: Option<&Driver>,
127    ) -> Result<(Classification, Option<Response>)> {
128        // Delegated *work* only. A harness maintenance turn (e.g. Codex `compact`) carries
129        // sub-agent lineage but is not delegated work, so it abstains and routes normally.
130        let is_delegated_work = request
131            .metadata
132            .as_ref()
133            .is_some_and(Metadata::is_subagent_work);
134        Ok((
135            Classification::Scores(if is_delegated_work {
136                vec![Score {
137                    confidence: 1.0,
138                    target: self.worker.clone(),
139                }]
140            } else {
141                Vec::new()
142            }),
143            None,
144        ))
145    }
146}

148#[cfg(test)]
149mod tests {
150    use super::*;
151    use parking_lot::Mutex;
152    use switchyard_protocol::{slice_to_header_map, text_request};
153
154    #[derive(Default)]
155    struct CapturingClassifier {
156        requests: Mutex<Vec<Request>>,
157    }
158
159    #[async_trait]
160    impl Classifier<()> for CapturingClassifier {
161        async fn score(
162            &self,
163            _state: &mut (),
164            request: &mut Request,
165            _driver: Option<&Driver>,
166        ) -> Result<(Classification, Option<Response>)> {
167            self.requests.lock().push(request.clone());
168            Ok((
169                Classification::Scores(vec![Score {
170                    confidence: 1.0,
171                    target: ModelId::from("worker"),
172                }]),
173                None,
174            ))
175        }
176    }
177
178    fn request(headers: &[(&str, &str)]) -> Request {
179        let metadata =
180            (!headers.is_empty()).then(|| Metadata::from_headers(&slice_to_header_map(headers)));
181        Request {
182            llm_request: text_request(Some(ModelId::from("auto").to_string()), "hi"),
183            raw_request: None,
184            metadata,
185        }
186    }
187
188    /// Scores `headers` through the override, returning the winning target if it scored.
189    async fn selected(headers: &[(&str, &str)]) -> Result<Option<ModelId>> {
190        let mut state = ();
191        let classification = SubagentOverride::new("worker")
192            .score(&mut state, &mut request(headers), None)
193            .await?;
194        Ok(classification.0.argmax(false)?.map(|score| score.target))
195    }
196
197    #[tokio::test]
198    async fn requests_without_metadata_abstain() -> Result<()> {
199        assert_eq!(selected(&[]).await?, None);
200        Ok(())
201    }

203    #[tokio::test]
204    async fn subagent_work_scores_the_worker() -> Result<()> {
205        // Claude Code child-agent lineage.
206        let claude = &[
207            ("x-claude-code-session-id", "root"),
208            ("x-claude-code-agent-id", "child-1"),
209        ];
210        assert_eq!(selected(claude).await?, Some(ModelId::from("worker")));
211
212        // Codex delegated-work kinds.
213        assert_eq!(
214            selected(&[("x-openai-subagent", "review")]).await?,
215            Some(ModelId::from("worker"))
216        );
217        assert_eq!(
218            selected(&[("x-openai-subagent", "collab_spawn")]).await?,
219            Some(ModelId::from("worker"))
220        );
221        Ok(())
222    }

226        assert_eq!(selected(&[("x-openai-subagent", "compact")]).await?, None);
227        assert_eq!(
228            selected(&[("x-switchyard-is-subagent", "false")]).await?,

234    #[tokio::test]
235    async fn delegated_work_is_scored_definitively() -> Result<()> {
236        // Confidence 1.0 under `Scores` (never `Ambiguous`), so the cascade stops here
237        // rather than consulting later classifiers.
238        let mut state = ();
239        let classification = SubagentOverride::new("worker")
240            .score(
241                &mut state,
242                &mut request(&[("x-openai-subagent", "review")]),
243                None,
244            )
245            .await?;
246        match classification.0 {
247            Classification::Scores(scores) => {
248                assert_eq!(scores.len(), 1);
249                assert_eq!(scores[0].confidence, 1.0);
250            }
251            Classification::Ambiguous(_) => panic!("override must score definitively"),
252        }
253        Ok(())
254    }

256    #[tokio::test]
257    async fn gate_abstains_when_delegated_work_has_no_text_prompt() -> Result<()> {
258        let classifier = Arc::new(CapturingClassifier::default());
259        let gate = SubagentGate::new(classifier.clone());
260        let mut request = request(&[("x-openai-subagent", "collab_spawn")]);
261        request.llm_request.messages = vec![Message::text(Role::Assistant, "no user prompt")];
262
263        let mut state = ();
264        let (classification, response) = gate.score(&mut state, &mut request, None).await?;
265
266        assert!(classification.argmax(false)?.is_none());
267        assert!(response.is_none());
268        assert!(classifier.requests.lock().is_empty());
269        Ok(())
270    }
