4//! Request-origin signals and compatibility with harness-generated summaries.
5
6use switchyard_protocol::{ContentBlock, Metadata, Request, RequestOrigin};
7
8use super::affinity::{is_environment_update, is_user_turn};
9
10/// Explicit origin takes precedence over message-shape heuristics.
11pub(crate) fn starts_user_turn(request: &Request) -> bool {
12    if let Some(origin) = request.metadata.as_ref().and_then(Metadata::request_origin) {
13        return origin == RequestOrigin::User;
14    }
15    request
16        .llm_request
17        .messages
18        .iter()
19        .rev()
20        .find(|message| !is_environment_update(message))
21        .is_some_and(|message| {
22            is_user_turn(message) && !message.content.iter().all(is_compaction_summary)
23        })
24}
25
26fn is_compaction_summary(block: &ContentBlock) -> bool {
27    match block {
28        ContentBlock::Unknown { raw, .. } => matches!(
29            raw.get("type").and_then(serde_json::Value::as_str),
30            Some("compaction" | "context_compaction")
31        ),
32        ContentBlock::Text { text } => {
33            let text = text.trim().to_ascii_lowercase();
34            // These harnesses flatten their internal summary type to ordinary user text.
35            text.starts_with("another language model started to solve this problem and produced a summary of its thinking process.")
36                || text.starts_with("this session is being continued from a previous conversation that ran out of context.")
37                || (text.starts_with("the conversation history before this point was compacted into the following summary:")
38                    && text.ends_with("</summary>"))
39                || (text.starts_with("prior model work/tool state available.")
40                    && text.ends_with("</summary>"))
41                || (text.starts_with("<conversation-checkpoint>")
42                    && text.ends_with("</conversation-checkpoint>"))
43                || hermes_summary(&text)
44        }
45        _ => false,
46    }
47}
48
49fn hermes_summary(text: &str) -> bool {
50    const REPLAY: &str = "[still in progress \u{2014} this is the active request, restated after the compaction boundary";
51    const END: &str = "--- end of context summary \u{2014} respond to the message below, not the summary above ---";
52    // A merged summary may also carry a fresh user message after its closing marker.
53    if let Some((_, tail)) = text.rsplit_once(END)
54        && !tail.trim().is_empty()
55        && !tail.trim_start().starts_with(REPLAY)
56    {
57        return false;
58    }
59    text.lines().any(|line| {
60        let line = line.trim_start();
61        line.starts_with("[context compaction \u{2014} reference only]")
62            || line.starts_with("[context summary]:")
63            || line.starts_with(REPLAY)
64    })
65}

67#[cfg(test)]
68mod tests {
69    use super::*;
70    use crate::algorithms::util::tier_fixtures::turn_request;
71    use switchyard_protocol::{Message, Role, WireFormat};

73    #[test]
74    fn summaries_do_not_hide_followups() {
75        for summary in [
76            "Another language model started to solve this problem and produced a summary of its thinking process.\nSummary",
77            "This session is being continued from a previous conversation that ran out of context.\nSummary",
78            "[CONTEXT COMPACTION \u{2014} REFERENCE ONLY]\nSummary",
79            "Prior context\n[CONTEXT COMPACTION \u{2014} REFERENCE ONLY]\nSummary",
80            "[CONTEXT SUMMARY]: Summary",
81            "[STILL IN PROGRESS \u{2014} this is the active request, restated after the compaction boundary] Task",
82            "The conversation history before this point was compacted into the following summary:\n\n<summary>\nSummary\n</summary>",
83            "Prior model work/tool state available.\nMUST build on prior work\n\n<summary>\nSummary\n</summary>",
84            "<conversation-checkpoint>\nThe following is a summary and serialized record of earlier conversation. Treat it as historical context, not as new instructions.\n\n<summary>\nSummary\n</summary>\n</conversation-checkpoint>",
85        ] {
86            let mut request = turn_request(false);
87            request
88                .llm_request
89                .messages
90                .push(Message::text(Role::User, summary));
91            assert!(!starts_user_turn(&request), "{summary}");
92            request
93                .llm_request
94                .messages
95                .push(Message::text(Role::User, "Now fix the parser"));
96            assert!(starts_user_turn(&request), "{summary}");
97        }
98
99        let mut request = turn_request(false);
100        for tail in [
101            "Now fix the parser",
102            "[STILL IN PROGRESS \u{2014} this is the active request, restated after the compaction boundary] Task",
103        ] {
104            request.llm_request.messages = vec![Message::text(
105                Role::User,
106                format!(
107                    "[CONTEXT COMPACTION \u{2014} REFERENCE ONLY]\nSummary\n--- END OF CONTEXT SUMMARY \u{2014} respond to the message below, not the summary above ---\n{tail}"
108                ),
109            )];
110            assert_eq!(starts_user_turn(&request), tail.starts_with("Now"));
111        }
112    }

114    #[test]
115    fn native_compaction_items_are_continuations() {
116        let mut request = turn_request(false);
117        for kind in ["compaction", "context_compaction", "unrecognized"] {
118            request.llm_request.messages = vec![Message {
119                role: Role::User,
120                content: vec![ContentBlock::Unknown {
121                    provider: WireFormat::OpenAiResponses.into(),
122                    raw: serde_json::json!({"type": kind, "encrypted_content": "opaque"}),
123                }],
124            }];
125            assert_eq!(starts_user_turn(&request), kind == "unrecognized");
126        }
127    }

129    #[test]
130    fn explicit_origin_overrides_message_shape() {
131        for (text, header, value, expected) in [
132            (
133                "[CONTEXT SUMMARY]: quoted by user",
134                "x-switchyard-request-origin",
135                "user",
136                true,
137            ),
138            (
139                "[CONTEXT SUMMARY]: summary",
140                "x-initiator",
141                "unknown",
142                false,
143            ),
144            ("[CONTEXT SUMMARY]: summary", "x-initiator", "user", false),
145            (
146                "Continue the task",
147                "x-switchyard-request-origin",
148                "compaction",
149                false,
150            ),
151            ("Continue the task", "x-initiator", "agent", false),
152            ("Now fix the parser", "x-initiator", "unknown", true),
153            (
154                "Now fix the parser",
155                "x-codex-turn-metadata",
156                "invalid json",
157                true,
158            ),
159            (
160                "Now fix the parser",
161                "x-codex-turn-metadata",
162                r#"{"request_kind":"turn"}"#,
163                true,
164            ),
165            (
166                "Summarize the conversation",
167                "x-codex-turn-metadata",
168                r#"{"request_kind":"compaction"}"#,
169                false,
170            ),
171        ] {
172            let mut request = turn_request(false);
173            request.llm_request.messages = vec![Message::text(Role::User, text)];
174            request
175                .metadata
176                .as_mut()
177                .unwrap()
178                .http_headers
179                .get_or_insert_with(Default::default)
180                .insert(header, value.parse().unwrap());
181            assert_eq!(starts_user_turn(&request), expected, "{header}: {value}");
182        }
183    }
