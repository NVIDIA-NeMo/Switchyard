21const MAX_BUFFER_BYTES: usize = 64 * 1024;
22const MAX_BUFFER_EVENTS: usize = 128;
23
24/// Limits and confidence policy for [`SpeculativeToolCall`].
25#[derive(Clone, Debug)]
26pub struct SpeculativeToolCallConfig {
27    /// Minimum probability of the selected complete call.
28    pub threshold: f64,
29    /// Maximum delay before the normal stream is released.
30    pub max_hold: Duration,
31    /// Maximum complete calls offered to the judge, excluding abstention.
32    pub max_candidates: usize,
33    /// Completion-target instructions also visible to the decision model.
34    pub target_system_prompt: Option<String>,
35}
36
37impl Default for SpeculativeToolCallConfig {
38    fn default() -> Self {
39        Self {
40            threshold: 0.9,
41            max_hold: Duration::from_millis(400),
42            max_candidates: 64,
43            target_system_prompt: None,
44        }
45    }
46}
47
48/// Returns a confident complete tool call, or continues the already-running normal answer.
49pub struct SpeculativeToolCall {
50    config: SpeculativeToolCallConfig,
51}
52
53impl SpeculativeToolCall {
54    /// Rejects invalid confidence, deadline, and candidate limits.
55    pub fn new(config: SpeculativeToolCallConfig) -> Result<Self> {
56        if !(0.0..=1.0).contains(&config.threshold)
57            || config.max_hold.is_zero()
58            || !(1..=256).contains(&config.max_candidates)
59        {
60            return Err(LibsyError::AlgorithmError {
61                message: "speculative_tool_call requires threshold in [0,1], a positive hold, and 1..=256 candidates".into(),
62            });
63        }
64        Ok(Self { config })
65    }
66
67    fn prediction(
68        &self,
69        response: &DecisionResponse,
70        candidates: &[Candidate],
71    ) -> Option<(usize, f64)> {
72        let DecisionValue::Choice {
73            selected,
74            probabilities: Some(probabilities),
75        } = &response.answers.get("next_call")?.value
76        else {
77            return None;
78        };
79        let probability = probabilities.get(selected)?.0;
80        if !(self.config.threshold..=1.0).contains(&probability) {
81            return None;
82        }
83        candidates
84            .iter()
85            .position(|candidate| &candidate.id == selected)
86            .map(|index| (index, probability))
87    }
88
89    fn answer(
90        driver: &Driver,
91        request: Request,
92        normal: ModelId,
93        judge: ModelId,
94        prediction: (usize, f64),
95        mut candidates: Vec<Candidate>,
96        usage: switchyard_protocol::Usage,
97    ) -> RoutingOutcome {
98        let (index, probability) = prediction;
99        driver.set_evidence(json!({
100            "source": "decision", "decision_model": judge,
101            "probability": probability, "candidate_count": candidates.len(),
102        }));
103        let mut call = candidates.swap_remove(index).call;
104        let id = uuid::Uuid::now_v7().simple().to_string();
105        call.id = format!("call_{id}");
106        let aggregate = AggLlmResponse {
107            id: Some(format!("spec_{id}")),
108            model: Some(judge.to_string()),
109            usage,
110            outputs: vec![ResponseOutput {
111                role: Role::Assistant,
112                content: vec![ContentBlock::ToolCall(call)],
113                stop_reason: Some(StopReason::ToolUse),
114                url_citations: Vec::new(),
115            }],
116            ..Default::default()
117        };
118        let mut response = Response {
119            llm_response: if request.llm_request.stream {
120                LlmResponse::Stream(aggregate.into_stream())
121            } else {
122                LlmResponse::Agg(aggregate)
123            },
124            metadata: request.metadata.clone(),
125            upstream_headers: Default::default(),
126        };
127        response.set_served_model(&judge);
128        response.metadata.get_or_insert_default().served_by_decision = true;
129        RoutingOutcome::answered(normal, request, response)
130    }
131}

133#[async_trait::async_trait]
134impl Algorithm for SpeculativeToolCall {
135    fn name(&self) -> &str {
136        "speculative_tool_call"
137    }
138
139    async fn route(self: Arc<Self>, driver: Driver, request: Request) -> Result<RoutingOutcome> {
140        let normal = driver.first_model_for(&Category::Any)?.clone();
141        let candidates = candidates::candidates(&request.llm_request, self.config.max_candidates);
142        if candidates.is_empty() {
143            driver.set_evidence(json!({"source": "model", "reason": "no_candidates"}));
144            return Ok(RoutingOutcome::route_to(normal, Vec::new(), request));
145        }
146        let judge = driver.first_model_for(&Category::Judge)?.clone();
147        let mut options: Vec<_> = candidates
148            .iter()
149            .map(|candidate| ChoiceOption {
150                id: candidate.id.clone(),
151                description: Some(
152                    json!({"name": candidate.call.name, "arguments": candidate.call.arguments}),
153                ),
154            })
155            .collect();
156        options.push(ChoiceOption {
157            id: "abstain".into(),
158            description: Some(json!("Text, clarification, multiple calls, or any action outside the candidates is needed.")),
159        });
160        let question = DecisionRequest {
161            model: None,
162            context: json!({
163                "instructions": request.llm_request.instructions,
164                "target_instructions": self.config.target_system_prompt,
165                "messages": request.llm_request.messages,
166                "tools": request.llm_request.tools,
167                "tool_choice": request.llm_request.tool_choice,
168            }),
169            questions: BTreeMap::from([(
170                "next_call".into(),
171                DecisionQuestion {
172                    instructions: json!(
173                        "Choose the complete immediate next tool call, including every argument. Follow the conversation's instructions. Abstain if text, clarification, multiple tool calls, or any other action is needed. Never guess a missing argument."
174                    ),
175                    kind: DecisionKind::Choice { options },
176                },
177            )]),
178        };
179        let normal_call = driver.call_model_streaming(request.clone(), vec![normal.clone()]);
180        let prediction = tokio::time::timeout(
181            self.config.max_hold,
182            driver.call_decision(question, judge.clone()),
183        );
184        tokio::pin!(normal_call, prediction);
185        let mut response = tokio::select! {
186            result = &mut normal_call => match result {
187                Ok(response) => response,
188                Err(error) => {
189                    if let Ok(Ok(response)) = prediction.await
190                        && let Some(selected) = self.prediction(&response, &candidates)
191                    {
192                        return Ok(Self::answer(
193                            &driver, request, normal, judge, selected, candidates, response.usage,
194                        ));
195                    }
196                    return Err(error);
197                }
198            },
199            result = &mut prediction => {
200                if let Ok(Ok(response)) = result
201                    && let Some(selected) = self.prediction(&response, &candidates)
202                {
203                    return Ok(Self::answer(
204                        &driver, request, normal, judge, selected, candidates, response.usage,
205                    ));
206                }
207                driver.set_evidence(json!({"source": "model", "reason": "prediction_rejected_or_expired"}));
208                return Ok(RoutingOutcome::answered(normal, request, normal_call.await?));
209            }
210        };
211        let LlmResponse::Stream(mut chunks) = response.llm_response else {
212            driver.set_evidence(json!({"source": "model", "reason": "normal_complete"}));
213            return Ok(RoutingOutcome::answered(normal, request, response));
214        };
215        let mut prefix = Vec::new();
216        let mut bytes = 0usize;
217        loop {
218            tokio::select! {
219                result = &mut prediction => {
220                    if let Ok(Ok(predicted)) = result
221                        && let Some(selected) = self.prediction(&predicted, &candidates)
222                    {
223                        return Ok(Self::answer(
224                            &driver, request, normal, judge, selected, candidates, predicted.usage,
225                        ));
226                    }
227                    break;
228                }
229                event = chunks.next() => {
230                    let Some(event) = event else { break };
231                    let terminal = event.as_ref().map_or(true, |event| {
232                        event.normalized().iter().any(|chunk| matches!(chunk,
233                            LlmResponseChunk::MessageStop { .. }
234                            | LlmResponseChunk::DecodeError { .. }
235                            | LlmResponseChunk::StreamError { .. }
236                        ))
237                    });
238                    bytes = bytes.saturating_add(event.as_ref().ok()
239                        .and_then(|event| serde_json::to_vec(event).ok())
240                        .map_or(MAX_BUFFER_BYTES, |value| value.len()));
241                    prefix.push(event);
242                    if terminal || bytes >= MAX_BUFFER_BYTES || prefix.len() >= MAX_BUFFER_EVENTS {
243                        break;
244                    }
245                }
246            }
247        }
248        driver.set_evidence(json!({"source": "model", "reason": "normal_stream_committed"}));
249        response.llm_response = LlmResponse::Stream(stream::iter(prefix).chain(chunks).boxed());
250        Ok(RoutingOutcome::answered(normal, request, response))
251    }
252}
