4//! Routing that stacks a judge over a stage router.
5//!
6//! The judge runs as a [`Preroute`]: it sets configuration the stage router reads,
7//! and picks no target itself. The tier lives in session state, so a deployment
8//! without a session ID gets a fresh tier on every request.

33/// Sets the stage router's fall-open tier from a judge verdict.
34///
35/// Keeps its own tier per routing identity rather than relying on the composition's
36/// session state, which a request carrying no session ID never retains. The retained
37/// tier is replayed into state on every request so the cascade below reads it.
38struct TierSetter {
39    judge: Arc<dyn Classifier<State>>,
40    targets: StageTargets,
41    trigger: ClassifyTrigger,
42    message_hash_fallback: bool,
43    tiers: Mutex<HashMap<RoutingIdentity, Tier>>,
44}
45
46impl TierSetter {
47    /// The key this request's tier is retained under, or `None` when nothing identifies it.
48    fn identity(&self, request: &Request) -> Option<RoutingIdentity> {
49        RoutingIdentity::from_request(request).or_else(|| {
50            self.message_hash_fallback
51                .then(|| first_user_message_hash(request).map(RoutingIdentity::Session))
52                .flatten()
53        })
54    }
55
56    fn is_due(&self, identity: Option<&RoutingIdentity>, request: &Request) -> bool {
57        match self.trigger {
58            ClassifyTrigger::EveryRequest => true,
59            ClassifyTrigger::UserTurn => has_new_user_turn(&request.llm_request.messages),
60            // Unkeyed requests cannot be told apart, so every one is a new session.
61            ClassifyTrigger::NewSession => {
62                identity.is_none_or(|identity| !self.tiers.lock().contains_key(identity))
63            }
64        }
65    }
66
67    fn retain(&self, identity: RoutingIdentity, tier: Tier) {
68        let mut tiers = self.tiers.lock();
69        if tiers.len() >= MAX_ASSIGNMENTS
70            && let Some(evicted) = tiers.keys().next().cloned()
71        {
72            tiers.remove(&evicted);
73        }
74        tiers.insert(identity, tier);
75    }
76}
77
78#[async_trait]
79impl Preroute<State> for TierSetter {
80    async fn run(&self, state: &mut State, request: &mut Request, driver: &Driver) -> Result<()> {
81        let identity = self.identity(request);
82        if self.is_due(identity.as_ref(), request) {
83            let (classification, _) = self.judge.score(state, request, Some(driver)).await?;
84            if let Some(winner) = classification.argmax(false)?
85                && let Some(tier) = self.targets.tier_for(&winner.target)
86            {
87                set_fall_open(state, tier);
88                if let Some(identity) = identity {
89                    self.retain(identity, tier);
90                }
91            }
92            return Ok(());
93        }
94        // Not this request's turn to judge, so replay whatever the last verdict set.
95        if let Some(tier) = identity.and_then(|identity| self.tiers.lock().get(&identity).copied())
96        {
97            set_fall_open(state, tier);
98        }
99        Ok(())
100    }
101}

103/// The judge that picks the tier.
104pub struct TierClassifier {
105    /// Target the judge is called through. Not a routing destination.
106    pub judge_target: ModelId,
107    /// Judge configuration, including how often `classify_trigger` runs it.
108    pub config: TaskClassifierConfig,
109}
110
111/// A judge stacked over a stage router.
112pub struct HierarchicalRouterConfig {
113    /// Sets the tier below it, as often as its `classify_trigger` says.
114    pub classifier: TierClassifier,
115    /// Serves every request.
116    pub stage: StageRouterConfig,
117}
118
119/// Runs a stage router with a tier the judge picks.
120pub struct HierarchicalRouter {
121    route: FallThrough<State>,
122}
123
124impl HierarchicalRouter {
125    /// Stacks the classifier over a stage router across the same tier pair.
126    ///
127    /// Errors on a stage or judge configuration either algorithm rejects, and on a
128    /// stage router carrying its own judge, which would decide the turns this
129    /// classifier set a tier for.
130    pub fn new(
131        capable: ModelId,
132        efficient: ModelId,
133        config: HierarchicalRouterConfig,
134    ) -> Result<Self> {
135        if config.classifier.config.classify_trigger == ClassifyTrigger::EveryRequest {
136            return Err(LibsyError::AlgorithmError {
137                message: "hierarchical: classify_trigger must be user_turn or new_session"
138                    .to_string(),
139            });
140        }
141        if config.stage.llm_fallback.is_some() {
142            return Err(LibsyError::AlgorithmError {
143                message: "hierarchical: the stage router cannot also carry a judge".to_string(),
144            });
145        }
146        let trigger = config.classifier.config.classify_trigger;
147        let message_hash_fallback = config.classifier.config.message_hash_fallback;
148        let judge = LlmTaskClassifier::new(LlmClassifierConfig::Capability {
149            judge_target: config.classifier.judge_target,
150            efficient_target: efficient.clone(),
151            capable_target: capable.clone(),
152            config: config.classifier.config,
153        })?;
154        let setter = TierSetter {
155            judge: Arc::new(judge),
156            targets: StageTargets::new(capable.clone(), efficient.clone()),
157            trigger,
158            message_hash_fallback,
159            tiers: Mutex::new(HashMap::new()),
160        };
161        let route = build_stage_route(capable, efficient, config.stage)?
162            .with_name(HIERARCHICAL)
163            .with_preroute(Arc::new(setter));
164        Ok(Self { route })
165    }
166}

168#[async_trait]
169impl Algorithm for HierarchicalRouter {
170    fn name(&self) -> &str {
171        HIERARCHICAL
172    }
173
174    async fn route(
175        self: Arc<Self>,
176        driver: Driver,
177        request: Request,
178    ) -> Result<crate::RoutingOutcome> {
179        self.route.execute(driver, request).await
180    }
181}

198    const JUDGE: &str = "judge";
199
200    #[derive(Default)]
201    struct Recorder {
202        targets: Mutex<Vec<String>>,
203        judge_p_solve: Mutex<f64>,
204    }
205
206    impl Recorder {
207        fn routed(&self) -> Vec<String> {
208            self.targets
209                .lock()
210                .iter()
211                .filter(|target| *target != JUDGE)
212                .cloned()
213                .collect()
214        }
215
216        fn judge_calls(&self) -> usize {
217            self.targets
218                .lock()
219                .iter()
220                .filter(|target| *target == JUDGE)
221                .count()
222        }
223
224        fn serve(self: &Arc<Self>) -> impl Serve {
225            let recorder = Arc::clone(self);
226            move |target: ModelId, _request: Request| {
227                let recorder = Arc::clone(&recorder);
228                async move {
229                    let target = target.to_string();
230                    recorder.targets.lock().push(target.clone());
231                    let completion = if target == JUDGE {
232                        let p_solve = *recorder.judge_p_solve.lock();
233                        format!(
234                            r#"{{"crux":"bounded task","primary_rule":"SUP-1","capability_boundary":"supported","p_solve":{p_solve}}}"#
235                        )
236                    } else {
237                        target
238                    };
239                    Ok(reply(completion))
240                }
241            }
242        }
243    }
244
245    fn turn_request(failed: bool) -> Request {
246        let content = if failed {
247            "fatal runtime error: out of memory"
248        } else {
249            "ok"
250        };
251        Request {
252            llm_request: LlmRequest {
253                model: Some("auto".to_string()),
254                messages: vec![
255                    Message::text(Role::User, "fix the build"),
256                    Message {
257                        role: Role::Assistant,
258                        content: vec![ContentBlock::ToolCall(ToolCall {
259                            id: "call_1".to_string(),
260                            name: "Bash".to_string(),
261                            arguments: json!({"command": "cargo test"}),
262                        })],
263                    },
264                    Message {
265                        role: Role::Tool,
266                        content: vec![ContentBlock::ToolResult(ToolResult {
267                            tool_call_id: "call_1".to_string(),
268                            content: vec![ContentBlock::Text {
269                                text: content.to_string(),
270                            }],
271                            is_error: Some(failed),
272                        })],
273                    },
274                ],
275                ..LlmRequest::default()
276            },
277            raw_request: None,
278            metadata: Some(Metadata {
279                wire_format: Some(WireFormat::OpenAiChat),

283        }

295    /// The same request shape with no session ID, so only the hash can key it.

360    #[tokio::test]
361    async fn a_session_without_an_id_keys_on_the_message_hash() -> Result<()> {
362        let recorder = Arc::new(Recorder::default());
363        *recorder.judge_p_solve.lock() = 0.1;
364        let router = hash_keyed_router()?;
365
366        test_drive(
367            router.clone(),
368            unkeyed(user_turn_request()),
369            recorder.serve(),
370        )
371        .await?;
372        test_drive(
373            router.clone(),
374            unkeyed(turn_request(false)),
375            recorder.serve(),
376        )
377        .await?;
378
379        assert_eq!(
380            recorder.judge_calls(),
381            1,
382            "new_session judges once without a session id"
383        );
384        assert_eq!(
385            recorder.routed()[1],
386            "strong",
387            "and the tier survives the tool step"
388        );
389        Ok(())
390    }

392    #[tokio::test]
393    async fn the_judge_sets_the_tier_once_a_turn_and_the_signals_run_within_it() -> Result<()> {
394        let recorder = Arc::new(Recorder::default());
395        *recorder.judge_p_solve.lock() = 0.1;
396        let router = router()?;
397
398        test_drive(router.clone(), user_turn_request(), recorder.serve()).await?;
399        test_drive(router.clone(), turn_request(false), recorder.serve()).await?;
400
401        let routed = recorder.routed();
402        assert_eq!(
403            routed[0], "strong",
404            "a quiet turn falls open to the verdict"
405        );
406        assert_eq!(
407            routed[1], "strong",
408            "which holds across the tool steps after it"
409        );
410        assert_eq!(
411            recorder.judge_calls(),
412            1,
413            "a tool step is not a new user turn"
414        );
415        Ok(())
