4//! Signal-driven stage routing for coding agents.
5//!
6//! [`StageRouter`] is the assembled algorithm: a [`FallThrough`] pre-wired with
7//! the tool-signal processor that reads each turn's tool results and the
8//! [`StageClassifier`] that scores them onto the capable/efficient tiers. The cascade
9//! is an internal detail — callers drive the algorithm, not its parts.
10//!
11//! Signals do not decide every turn. An under-threshold turn abstains and falls
12//! through to the optional [`LlmTaskClassifier`] — the capability route's judge,
13//! joined in unchanged — and then to the picker's default tier, or to an
14//! override a decider ahead of stage set in its place.
15//!

38/// Attributes a turn to the classifier it wraps, when that classifier decides it.
39///
40/// The classifiers themselves are composition-agnostic and write no state; only
41/// this router knows where each sits in its cascade.
42struct SourceStamp {
43    inner: Arc<dyn Classifier<State>>,
44    source: DecisionSource,
45}
46
47#[async_trait]
48impl Classifier<State> for SourceStamp {
49    fn routing_tier(&self, selected_model_id: &ModelId) -> Option<&'static str> {
50        self.inner.routing_tier(selected_model_id)
51    }
52
53    async fn score(
54        &self,
55        state: &mut State,
56        request: &mut Request,
57        driver: Option<&Driver>,
58    ) -> Result<(Classification, Option<Response>)> {
59        let (classification, served) = self.inner.score(state, request, driver).await?;
60        // An abstaining classifier passes the turn on, so it is not its to claim.
61        if let Some(winner) = classification.argmax(false)? {
62            record_decision_source(state, self.source);
63            record_routing_decision(self.source, &winner.target);
64        }
65        Ok((classification, served))
66    }
67}

69/// Closes the cascade at zero confidence: a fallback, not a judgement.
70struct FallOpen {
71    targets: StageTargets,
72    default_tier: Tier,
73}
74
75#[async_trait]
76impl Classifier<State> for FallOpen {
77    async fn score(
78        &self,
79        state: &mut State,
80        _request: &mut Request,
81        _driver: Option<&Driver>,
82    ) -> Result<(Classification, Option<Response>)> {
83        let tier = fall_open_tier(state).unwrap_or(self.default_tier);
84        let target = self.targets.name(tier).clone();
85        Ok((
86            Classification::Scores(vec![Score {
87                target,
88                confidence: 0.0,
89            }]),
90            None,
91        ))
92    }
93}

95/// Sets the fall-open tier from a judge verdict at the start of a user turn.
96///
97/// Never decides a turn: the verdict is a floor for the signals to work above,
98/// so this always abstains and the cascade continues. The tier is session state,
99/// so it holds until the next user turn replaces it.
100struct TierPicker {
101    judge: Arc<dyn Classifier<State>>,
102    targets: StageTargets,
103}
104
105#[async_trait]
106impl Classifier<State> for TierPicker {
107    async fn score(
108        &self,
109        state: &mut State,
110        request: &mut Request,
111        driver: Option<&Driver>,
112    ) -> Result<(Classification, Option<Response>)> {
113        if has_new_user_turn(&request.llm_request.messages) {
114            let (classification, _) = self.judge.score(state, request, driver).await?;
115            if let Some(winner) = classification.argmax(false)?
116                && let Some(tier) = self.targets.tier_for(&winner.target)
117            {
118                set_fall_open(state, tier);
119            }
120        }
121        Ok((Classification::Ambiguous(Vec::new()), None))
122    }
123}

125/// The judge that picks a user turn's tier, ahead of the signals.
126pub struct TierClassifier {
127    /// Target the judge model is called through, not a routing destination.
128    pub judge_target: ModelId,
129    /// Judge configuration. `classify_trigger` has no effect: the turn boundary
130    /// is what triggers this judge.
131    pub config: TaskClassifierConfig,
132}
133
134/// The capability judge a stage router falls through to.
135pub struct LlmFallback {
136    /// Target the judge model is called through. It is not a routing
137    /// destination, so it does not belong in the router's target set.
138    pub judge_target: ModelId,
139    /// Judge configuration. `recent_turn_window` is worth setting to this router's
140    /// `recent_window` so the judge reads the same span the signal scorer scored.
141    /// Note: `classify_trigger = new_session` and `message_hash_fallback` have no effect here —
142    /// the judge runs as a cascade classifier, not a standalone algorithm.
143    pub config: TaskClassifierConfig,
144}

146/// How a stage router scores turns, and what it hands the model it picks.
147pub struct StageRouterConfig {
148    /// Tier a turn falls open to when the scorer is not confident.
149    pub mode: PickerMode,
150    /// How much corroboration a decisive pick needs, in `[0.0, 1.0]`.
151    pub confidence_threshold: f64,
152    /// Trailing tool results the signals are computed over. `None` uses
153    /// [`DEFAULT_RECENT_WINDOW`].
154    pub recent_window: Option<usize>,
155    /// Note handed to the model on a signal-driven escalation, and on a
156    /// hand-back to the efficient tier when a de-escalation note is configured.
157    pub handoff_notes: Option<HandoffNoteConfig>,
158    /// System prompts keyed by target, handed over on every turn that target
159    /// serves. Empty by default.
160    pub tier_prompts: TargetPrompts,
161    /// Capability judge consulted on turns the signals leave undecided — the
162    /// judge's own target, plus the same configuration the standalone capability
163    /// route takes.
164    pub llm_fallback: Option<LlmFallback>,
165    /// Judge that sets the fall-open tier at the start of each user turn. The
166    /// signals still decide the turns they can.
167    pub tier_classifier: Option<TierClassifier>,
168}

171    /// The signal-only configuration: no notes, no per-tier prompts, no judge.
172    /// Set the optional fields to add them.

175            mode,
176            confidence_threshold,
177            recent_window: None,
178            handoff_notes: None,
179            tier_prompts: TargetPrompts::default(),
180            llm_fallback: None,
181            tier_classifier: None,
182        }
183    }

186/// Routes coding-agent turns between a capable and an efficient tier: tool signals
187/// decide first, an optional capability judge takes the turns they cannot, and
188/// the picker's default tier closes the cascade so a turn is never left unrouted.
189pub struct StageRouter {
190    route: FallThrough<State>,
191}
192
193impl StageRouter {
194    /// Routes between the `capable` and `efficient` targets. The
195    /// judge, when configured, is called through its own target and is not a
196    /// routing destination.
197    ///
198    /// Errors if either threshold in `config` is outside `[0.0, 1.0]`.
199    pub fn new(capable: ModelId, efficient: ModelId, config: StageRouterConfig) -> Result<Self> {
200        Ok(Self {
201            route: build_route(capable, efficient, config)?,
202        })
203    }
204}

206#[async_trait]
207impl Algorithm for StageRouter {
208    fn name(&self) -> &str {
209        STAGE_ROUTER
210    }
211
212    async fn route(
213        self: Arc<Self>,
214        driver: Driver,
215        request: Request,
216    ) -> Result<crate::RoutingOutcome> {
217        self.route.execute(driver, request).await
218    }
219}
220
221/// Wires the cascade the wrapper drives.
222fn build_route(
223    capable: ModelId,
224    efficient: ModelId,
225    config: StageRouterConfig,
226) -> Result<FallThrough<State>> {
227    if !(0.0..=1.0).contains(&config.confidence_threshold) {
228        return Err(LibsyError::AlgorithmError {
229            message: format!(
230                "confidence_threshold must be between 0 and 1, got {}",
231                config.confidence_threshold
232            ),
233        });
234    }
235    // The tiers are a fixed pair; their targets are whatever the deployment calls
236    // them, and the classifier scores onto those names.
237    let targets = StageTargets::new(capable.clone(), efficient.clone());
238    let default_tier = config.mode.default_tier();
239    let fall_open = FallOpen {
240        targets: targets.clone(),
241        default_tier,
242    };
243
244    let mut classifier =
245        StageClassifier::new(targets.clone(), config.mode, config.confidence_threshold);
246    if let Some(notes) = config.handoff_notes {
247        classifier = classifier.with_handoff_notes(notes);
248    }
249    let signals = ToolSignalProcessor {
250        recent_window: config.recent_window.unwrap_or(DEFAULT_RECENT_WINDOW),
251    };
252    let target_set = vec![capable.clone(), efficient.clone()];
253    let mut router = FallThrough::<State>::new_with_state(target_set)
254        .with_name(STAGE_ROUTER)
255        .with_processor(Arc::new(signals));
256    // Ahead of the signals so a fresh turn's verdict is in place before they score.
257    if let Some(tier_classifier) = config.tier_classifier {
258        let judge = LlmTaskClassifier::new(LlmClassifierConfig::Capability {
259            judge_target: tier_classifier.judge_target,
260            efficient_target: efficient.clone(),
261            capable_target: capable.clone(),
262            config: tier_classifier.config,
263        })?;
264        router = router.with_classifier(Arc::new(TierPicker {
265            judge: Arc::new(judge),
266            targets,
267        }));
268    }
269    router = router.with_classifier(Arc::new(classifier));
270    if let Some(fallback) = config.llm_fallback {
271        // The capability judge takes its tiers in the same order the capability
272        // route passes them: efficient first, capable second.
273        router = router.with_classifier(Arc::new(SourceStamp {
274            inner: Arc::new(LlmTaskClassifier::new(LlmClassifierConfig::Capability {
275                judge_target: fallback.judge_target,
276                efficient_target: efficient,
277                capable_target: capable,
278                config: fallback.config,
279            })?),
280            source: DecisionSource::LlmClassifier,
281        }));
282    }
283    // Nothing behind this, so no turn is left unrouted.
284    router = router.with_classifier(Arc::new(SourceStamp {
285        inner: Arc::new(fall_open),
286        source: DecisionSource::FallOpen,
287    }));
288    // Runs on the post-decision hook, so it applies to the target the cascade
289    // settled on, whichever classifier picked it. With no prompts configured it
290    // is a no-op, so there is nothing to branch on.
291    router = router.with_processor(Arc::new(SystemPromptProcessor::new(config.tier_prompts)));
292    Ok(router)
293}

422    #[derive(Clone, Debug)]
423    struct Call {
424        target: String,
425        messages: Vec<String>,
426    }
427
428    /// Records what each target receives.
429    #[derive(Default)]
430    struct Recorder {
431        calls: Mutex<Vec<Call>>,
432        judge_p_solve: Mutex<f64>,
433    }
434
435    impl Recorder {
436        fn routed(&self) -> Vec<Call> {
437            self.calls
438                .lock()
439                .iter()
440                .filter(|call| call.target != JUDGE)
441                .cloned()
442                .collect()
443        }
444
445        /// Serves every call, recording it. The judge target gets a structured verdict
446        /// back so the fallback classifier has an answer without a real model.
447        fn serve(self: &Arc<Self>) -> impl Serve {
448            let recorder = Arc::clone(self);
449            move |target: ModelId, request: Request| {
450                let recorder = Arc::clone(&recorder);
451                async move {
452                    let target = target.to_string();
453                    recorder.calls.lock().push(Call {
454                        target: target.clone(),
455                        messages: request
456                            .llm_request
457                            .messages
458                            .iter()
459                            .filter_map(|message| message.text_content("|"))
460                            .collect(),
461                    });
462                    let completion = if target == JUDGE {
463                        let p_solve = *recorder.judge_p_solve.lock();
464                        format!(
465                            r#"{{"crux":"bounded task","primary_rule":"SUP-1","capability_boundary":"supported","p_solve":{p_solve}}}"#
466                        )
467                    } else {
468                        target
469                    };
470                    Ok(reply(completion))
471                }
472            }
473        }
474    }

575    /// Stands in for a decider ahead of stage: sets the override once, clears it
576    /// once, and leaves the turns between alone.
577    #[derive(Default)]
578    struct TierDecider {
579        requests: Mutex<u32>,
580    }
581
582    #[async_trait]
583    impl Processor<State> for TierDecider {
584        async fn process(&self, state: &mut State, event: Event<'_>) -> Result<()> {
585            if matches!(event, Event::Request(_)) {
586                let mut requests = self.requests.lock();
587                *requests += 1;
588                match *requests {
589                    1 => set_fall_open(state, Tier::Efficient),
590                    4 => clear_fall_open(state),
591                    _ => {}
592                }
593            }
594            Ok(())
595        }
596    }
