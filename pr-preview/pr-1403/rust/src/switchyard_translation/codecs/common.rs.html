11/// Scales temperature between ranges with a minimum of zero.
12pub(crate) fn scale_temperature(temperature: f64, source_max: f64, target_max: f64) -> f64 {
13    if source_max == target_max {
14        return temperature;
15    }
16    (temperature / source_max) * target_max
17}

24/// Converts an OpenAI allowed-tools policy to a restricted function list and mode.
25pub(crate) fn allowed_function_tools(
26    request: &LlmRequest,
27) -> Result<Option<(Vec<Tool>, ToolChoice)>> {
28    let Some(ToolChoice::Raw(choice)) = &request.tool_choice else {
29        return Ok(None);
30    };
31    if choice.get("type").and_then(Value::as_str) != Some("allowed_tools") {
32        return Ok(None);
33    }
34    let error = || {
35        TranslationError::LossyConversion(
36            "allowed_tools translation requires auto or required mode \
37             and a nonempty subset of known function tools"
38                .to_string(),
39        )
40    };
41    let policy = choice.get("allowed_tools").unwrap_or(choice);
42    let mode = match policy.get("mode").and_then(Value::as_str) {
43        Some("auto") => ToolChoice::Auto,
44        Some("required") => ToolChoice::Required,
45        _ => return Err(error()),
46    };
47    let allowed = policy
48        .get("tools")
49        .and_then(Value::as_array)
50        .ok_or_else(error)?;
51    if allowed.is_empty() {
52        return Err(error());
53    }
54    let mut names = Vec::with_capacity(allowed.len());
55    for tool in allowed {
56        if tool.get("type").and_then(Value::as_str) != Some("function") {
57            return Err(error());
58        }
59        let function = tool.get("function").unwrap_or(tool);
60        let name = function
61            .get("name")
62            .and_then(Value::as_str)
63            .filter(|name| !name.is_empty())
64            .ok_or_else(error)?;
65        let name = match function.get("namespace").and_then(Value::as_str) {
66            Some(namespace) if !namespace.is_empty() => {
67                crate::codex_namespaces::qualified_tool_name(namespace, name)
68            }
69            _ => name.to_string(),
70        };
71        if !request
72            .tools
73            .iter()
74            .filter_map(Tool::function)
75            .any(|tool| tool.name == name)
76        {
77            return Err(error());
78        }
79        names.push(name);
80    }
81    // Filter the catalog, rather than rebuilding it from selectors, to retain definitions
82    // and avoid duplicating tools when the allowed list repeats a name.
83    let tools = request
84        .tools
85        .iter()
86        .filter(|tool| {
87            tool.function()
88                .is_some_and(|tool| names.contains(&tool.name))
89        })
90        .cloned()
91        .collect();
92    Ok(Some((tools, mode)))
93}

103pub(crate) fn decode_text_block(block: &Map<String, Value>) -> ContentBlock {
104    ContentBlock::Text {
105        text: block
106            .get("text")
107            .and_then(Value::as_str)
108            .unwrap_or_default()
109            .to_owned(),
110        cache_control: block
111            .get("cache_control")
112            .filter(|value| value.is_object())
113            .cloned()
114            .map(Box::new),
115    }
116}

118pub(crate) fn text_part(text: &str, cache_control: Option<&Value>) -> Value {
119    let mut part = json!({"type": "text", "text": text});
120    if let Some(marker) = cache_control {
121        part["cache_control"] = marker.clone();
122    }
123    part
124}
125
126fn text_and_cache_control(block: &ContentBlock) -> Option<(&str, Option<&Value>)> {
127    match block {
128        ContentBlock::Text {
129            text,
130            cache_control,
131        } => Some((text.as_str(), cache_control.as_deref())),
132        ContentBlock::Refusal { text } => Some((text.as_str(), None)),
133        ContentBlock::Unknown { raw, .. } => raw.as_str().map(|text| (text, None)),
134        _ => None,
135    }
136}

138// Keep the block array when joining its text would erase a cache boundary.
139pub(crate) fn encode_text_content<'a>(
140    blocks: impl Iterator<Item = &'a ContentBlock> + Clone,
141    separator: &str,
142    supports_cache_control: bool,
143) -> Value {
144    let parts = blocks.filter_map(text_and_cache_control);
145    if supports_cache_control && parts.clone().any(|(_, marker)| marker.is_some()) {
146        Value::Array(
147            parts
148                .map(|(text, marker)| text_part(text, marker))
149                .collect(),
150        )
151    } else {
152        Value::String(
153            parts
154                .map(|(text, _)| text)
155                .collect::<Vec<_>>()
156                .join(separator),
157        )
158    }
159}

161/// Extracts text-like blocks and joins them for text-only provider fields.
162pub(crate) fn text_from_blocks(content: &[ContentBlock], separator: &str) -> String {
163    content
164        .iter()
165        .filter_map(text_and_cache_control)
166        .map(|(text, _)| text)
167        .collect::<Vec<_>>()
168        .join(separator)
169}

171/// Extracts private reasoning blocks without mixing them into visible text.
172pub(crate) fn reasoning_text_from_blocks(content: &[ContentBlock], separator: &str) -> String {
173    content
174        .iter()
175        .filter_map(|block| match block {
176            ContentBlock::Reasoning { text, .. } => Some(text.as_str()),
177            _ => None,
178        })
179        .collect::<Vec<_>>()
180        .join(separator)
181}

183/// Extracts displayable text from structured reasoning details.
184pub(crate) fn reasoning_text_from_details(details: &[Value]) -> Option<String> {
185    let parts = details
186        .iter()
187        .filter_map(Value::as_object)
188        .filter_map(|detail| {
189            detail
190                .get("text")
191                .and_then(Value::as_str)
192                .filter(|text| !text.is_empty())
193                .or_else(|| {
194                    detail
195                        .get("summary")
196                        .and_then(Value::as_str)
197                        .filter(|summary| !summary.is_empty())
198                })
199        })
200        .collect::<Vec<_>>();
201    (!parts.is_empty()).then(|| parts.join("\n"))
202}

204/// Collects reasoning text from a Responses reasoning item's `content` or `summary`
205/// array, or from a bare string, into `out`. Empty strings are skipped.
206pub(crate) fn collect_responses_reasoning_text(value: Option<&Value>, out: &mut Vec<String>) {
207    match value {
208        Some(Value::String(text)) if !text.is_empty() => out.push(text.clone()),
209        Some(Value::Array(items)) => {
210            for item in items {
211                match item {
212                    Value::String(text) if !text.is_empty() => out.push(text.clone()),
213                    Value::Object(object) => {
214                        if matches!(
215                            object.get("type").and_then(Value::as_str),
216                            Some("reasoning_text" | "summary_text" | "text")
217                        ) && let Some(text) = object.get("text").and_then(Value::as_str)
218                            && !text.is_empty()
219                        {
220                            out.push(text.to_string());
221                        }
222                    }
223                    _ => {}
224                }
225            }
226        }
227        _ => {}
228    }
229}

231/// Returns the opaque payload of the first encrypted reasoning detail, if any.
232///
233/// Two detail shapes are accepted: the documented `{"type": "reasoning.encrypted", "data"}`
234/// object, and a verbatim Responses `reasoning` item carrying `encrypted_content` (the shape
235/// the buffered request decoder stores when it keeps the provider item whole).
236pub(crate) fn encrypted_reasoning_data(details: &[Value]) -> Option<String> {
237    details
238        .iter()
239        .filter_map(Value::as_object)
240        .find_map(|detail| match detail.get("type").and_then(Value::as_str) {
241            Some("reasoning.encrypted") => detail.get("data").and_then(Value::as_str),
242            Some("reasoning") => detail.get("encrypted_content").and_then(Value::as_str),
243            _ => None,
244        })
245        .filter(|data| !data.is_empty())
246        .map(ToOwned::to_owned)
247}

249/// Returns the provider item id recorded on the first `reasoning.encrypted` detail, if any.
250/// Encrypted reasoning is bound to the item id it was issued under, so a replay must reuse it.
251pub(crate) fn encrypted_reasoning_item_id(details: &[Value]) -> Option<String> {
252    details
253        .iter()
254        .filter_map(Value::as_object)
255        .find(|detail| {
256            matches!(
257                detail.get("type").and_then(Value::as_str),
258                Some("reasoning.encrypted" | "reasoning")
259            )
260        })
261        .and_then(|detail| detail.get("id").and_then(Value::as_str))
262        .filter(|id| !id.is_empty())
263        .map(ToOwned::to_owned)
264}

266/// Returns the first non-empty string stored under the requested keys.
267pub(crate) fn first_nonempty_string<'a>(
268    object: &'a Map<String, Value>,
269    keys: &[&str],
270) -> Option<&'a str> {
271    keys.iter().find_map(|key| {
272        object
273            .get(*key)
274            .and_then(Value::as_str)
275            .filter(|value| !value.is_empty())
276    })
277}

279/// Copies unknown provider fields into the IR extension map.
280pub(crate) fn provider_extensions(
281    object: &Map<String, Value>,
282    known: &[&str],
283) -> Map<String, Value> {
284    let mut extensions = Map::new();
285    for (key, value) in object {
286        if key != ANTHROPIC_REQUEST_KEY && !known.contains(&key.as_str()) {
287            extensions.insert(key.clone(), value.clone());
288        }
289    }
290    extensions
291}

293/// Invalid charges remain unknown so accounting can use configured token rates.
294pub(crate) fn provider_cost(value: &Map<String, Value>) -> Option<f64> {
295    value
296        .get("cost")
297        .and_then(Value::as_f64)
298        .filter(|cost| cost.is_finite() && *cost >= 0.0)
299}

301/// Cache duration must survive decoding because one-hour writes can have a different rate.
302pub(crate) fn cache_usage_details(
303    value: &Map<String, Value>,
304    cached: Option<u64>,
305    created: Option<u64>,
306) -> Option<Box<crate::llm::InputCacheUsage>> {
307    let one_hour = value
308        .get("cache_creation")
309        .and_then(|details| details.get("ephemeral_1h_input_tokens"))
310        .and_then(Value::as_u64);
311    let mut cache = crate::llm::Usage::cache_details(cached, created);
312    if let Some(tokens) = one_hour {
313        cache
314            .get_or_insert_with(Default::default)
315            .cache_creation_1h_input_tokens = Some(tokens);
316    }
317    cache
318}

320/// These fields preserve provider charges and cache duration across response formats.
321pub(crate) fn encode_usage_metadata(mut value: Value, usage: &crate::llm::Usage) -> Value {
322    if let Some(cost) = usage
323        .provider_cost
324        .filter(|cost| cost.is_finite() && *cost >= 0.0)
325    {
326        value["cost"] = serde_json::json!(cost);
327    }
328    if let Some(tokens) = usage
329        .cache
330        .as_ref()
331        .and_then(|cache| cache.cache_creation_1h_input_tokens)
332    {
333        value["cache_creation"] = serde_json::json!({"ephemeral_1h_input_tokens": tokens});
334    }
335    value
336}

338#[cfg(test)]
339mod tests {
340    use super::*;
341    use serde_json::json;
342
343    #[test]
344    fn encrypted_reasoning_helpers_accept_both_detail_shapes() {
345        let documented = vec![json!({"type": "reasoning.encrypted", "data": "blob", "id": "rs_1"})];
346        assert_eq!(
347            encrypted_reasoning_data(&documented).as_deref(),
348            Some("blob")
349        );
350        assert_eq!(
351            encrypted_reasoning_item_id(&documented).as_deref(),
352            Some("rs_1")
353        );
354        let verbatim_item = vec![json!({
355            "type": "reasoning", "id": "rs_2", "summary": [], "encrypted_content": "blob2"
356        })];
357        assert_eq!(
358            encrypted_reasoning_data(&verbatim_item).as_deref(),
359            Some("blob2")
360        );
361        assert_eq!(
362            encrypted_reasoning_item_id(&verbatim_item).as_deref(),
363            Some("rs_2")
364        );
365        assert_eq!(encrypted_reasoning_data(&[json!({"type": "other"})]), None);
366    }
