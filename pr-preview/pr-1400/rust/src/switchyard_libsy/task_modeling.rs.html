4//! Build a decision request about the current task and interpret its answers.
5//!
6//! ```no_run
7//! # use switchyard_libsy::{Driver, Result, task_modeling::Rubric};
8//! # use switchyard_protocol::{ModelId, Request};
9//! # async fn example(driver: Driver, request: Request, decision_model: ModelId) -> Result<()> {
10//! let rubric = Rubric::default();
11//! let decision_request = rubric.questions(&request);
12//! let decision_response = driver.call_decision(decision_request, decision_model).await?;
13//! let assessment = rubric.interpret(&request, decision_response)?;
14//! let signals = assessment.signals();
15//! # Ok(())
16//! # }
17//! ```
18
19use std::collections::BTreeMap;
20
21use serde::{Deserialize, Serialize};
22use serde_json::json;
23use switchyard_protocol::{
24    DecisionAnswer, DecisionQuestion, DecisionRequest, DecisionResponse, Request,
25};
26
27use crate::algorithms::util::affinity::starts_user_turn;
28use crate::{LibsyError, Result};
29
30/// The bundled rubric, also usable as a template for a custom TOML file.
31pub const DEFAULT_RUBRIC_TOML: &str = include_str!("task_modeling/default.toml");
32
33/// Each TOML key names a signal in the decision response.
34#[derive(Debug, Deserialize)]
35#[serde(deny_unknown_fields)]
36pub struct Rubric {
37    /// Questions about the user's goal and requirements.
38    pub user: BTreeMap<String, DecisionQuestion>,
39    /// Questions about the latest agent action and remaining work.
40    pub agent: BTreeMap<String, DecisionQuestion>,
41}
42
43/// A completed assessment with the provider's typed answers and usage.
44#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
45#[serde(
46    tag = "boundary",
47    content = "decision_response",
48    rename_all = "snake_case"
49)]
50pub enum TaskAssessment {
51    /// Signals about the incoming user goal.
52    User(DecisionResponse),
53    /// Signals about the state of the agent's work.
54    Agent(DecisionResponse),
55}
56
57impl TaskAssessment {
58    /// Answers keyed by the signal names in the originating rubric.
59    pub fn signals(&self) -> &BTreeMap<String, DecisionAnswer> {
60        match self {
61            Self::User(response) | Self::Agent(response) => &response.answers,
62        }
63    }
64}
65
66impl Default for Rubric {
67    fn default() -> Self {
68        toml::from_str(DEFAULT_RUBRIC_TOML).expect("bundled task modeling rubric must parse")
69    }
70}
71
72impl Rubric {
73    /// Builds a decision request from the normalized conversation. The caller selects the model.
74    pub fn questions(&self, request: &Request) -> DecisionRequest {
75        let questions = if starts_user_turn(request) {
76            &self.user
77        } else {
78            &self.agent
79        };
80        DecisionRequest {
81            model: None,
82            input: json!({ "messages": request.llm_request.messages }),
83            questions: questions.clone(),
84        }
85    }
86
87    /// Interprets a decision response against the same request and rubric used to build it.
88    pub fn interpret(
89        &self,
90        request: &Request,
91        decision_response: DecisionResponse,
92    ) -> Result<TaskAssessment> {
93        let (questions, assessment) = if starts_user_turn(request) {
94            (&self.user, TaskAssessment::User(decision_response))
95        } else {
96            (&self.agent, TaskAssessment::Agent(decision_response))
97        };
98        if !questions.keys().eq(assessment.signals().keys()) {
99            return Err(LibsyError::AlgorithmError {
100                message:
101                    "decision response keys do not match the task modeling rubric for this boundary"
102                        .into(),
103            });
104        }
105        Ok(assessment)
106    }
107}

109#[cfg(test)]
110mod tests {
111    use super::*;
112    use switchyard_protocol::{
113        BooleanEstimate, DecisionValue, Message, Probability, Role, text_request,
114    };
115
116    #[test]
117    fn uses_configured_questions_at_each_boundary() {
118        let custom: Rubric = toml::from_str(
119            r#"
120            [user.needs_patch]
121            instructions = "Is a patch requested?"
122            kind = { type = "boolean", data = {} }
123            [agent.tests_passed]
124            instructions = "Did the requested tests pass?"
125            kind = { type = "boolean", data = {} }
126            "#,
127        )
128        .unwrap();
129        let mut request = Request {
130            llm_request: text_request(None, "Fix the parser"),
131            ..Default::default()
132        };
133        assert!(
134            Rubric::default()
135                .questions(&request)
136                .questions
137                .contains_key("deliverable")
138        );
139        for (user_turn, key, value) in [
140            (
141                true,
142                "needs_patch",
143                DecisionValue::Boolean(BooleanEstimate::ProbabilityTrue(Probability(0.91))),
144            ),
145            (false, "tests_passed", DecisionValue::Refusal),
146        ] {
147            let decision_request = custom.questions(&request);
148            assert_eq!(decision_request.questions.keys().collect::<Vec<_>>(), [key]);
149            assert_eq!(
150                decision_request.input["messages"],
151                json!(request.llm_request.messages)
152            );
153
154            let decision_response = DecisionResponse {
155                id: None,
156                model: None,
157                usage: Default::default(),
158                answers: BTreeMap::from([(
159                    key.into(),
160                    DecisionAnswer {
161                        value: value.clone(),
162                        provider_confidence: None,
163                    },
164                )]),
165            };
166            let assessment = custom
167                .interpret(&request, decision_response.clone())
168                .unwrap();
169            assert_eq!(matches!(assessment, TaskAssessment::User(_)), user_turn);
170            assert_eq!(assessment.signals()[key].value, value);
171
172            if user_turn {
173                request
174                    .llm_request
175                    .messages
176                    .push(Message::text(Role::Tool, "parser test failed"));
177                assert!(custom.interpret(&request, decision_response).is_err());
178            }
179        }
180    }
181}
