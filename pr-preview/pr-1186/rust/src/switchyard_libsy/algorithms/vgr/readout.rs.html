4//! The cheap readout: the probability a verifier put on "yes" at its first token.
5//!
6//! Normalized responses carry no logprobs, so this reads the preserved Chat
7//! body. Anything that cannot be scored is `None`, which never commits.
8//! The readout supports OpenAI Chat-compatible verifiers only.
9
10use serde_json::{Value, json};
11use switchyard_protocol::{AggLlmResponse, FormatId, Request, WireFormat};
12
13pub(super) const MAX_OUTPUT_TOKENS: u64 = 4;
14
15/// Minimum combined yes/no probability for the first token to count as a verdict.
16const MIN_VERDICT_MASS: f64 = 0.5;
17
18/// Disables reasoning for one verifier call, including llama.cpp thinking templates.
19pub(super) fn disable_thinking(request: &mut Request) {
20    request.llm_request.reasoning.effort = Some("none".to_string());
21    request.llm_request.extensions.fields.insert(
22        "chat_template_kwargs".to_string(),
23        json!({"enable_thinking": false}),
24    );
25}
26
27/// Asks for a direct verdict with its alternatives; reasoning stays request-local.
28pub(super) fn request_logprobs(request: &mut Request) {
29    disable_thinking(request);
30    let extensions = &mut request.llm_request.extensions.fields;
31    extensions.insert("logprobs".to_string(), json!(true));
32    extensions.insert("top_logprobs".to_string(), json!(8));
33}
34
35pub(super) fn p_yes(response: &AggLlmResponse) -> Option<f64> {
36    let alternatives = response
37        .preservation
38        .responses
39        .get(&FormatId::from(WireFormat::OpenAiChat))?
40        .pointer("/choices/0/logprobs/content/0/top_logprobs")?
41        .as_array()?;
42    let (mut yes, mut no) = (None, None);
43    for entry in alternatives {
44        let Some(probability) = entry.get("logprob").and_then(Value::as_f64) else {
45            continue;
46        };
47        let token = entry.get("token")?.as_str()?;
48        let slot = match token
49            .trim()
50            .trim_matches(['"', '\'', '.', ','])
51            .to_lowercase()
52            .as_str()
53        {
54            "yes" | "y" | "true" => &mut yes,
55            "no" | "n" | "false" => &mut no,
56            _ => continue,
57        };
58        *slot.get_or_insert(0.0) += probability.exp();
59    }
60    match (yes, no) {
61        (Some(yes), Some(no)) if yes + no >= MIN_VERDICT_MASS => Some(yes / (yes + no)),
62        _ => None,
63    }
64}

66#[cfg(test)]
67mod tests {
68    use super::*;
69
70    fn scored(alternatives: Value) -> Option<f64> {
71        let mut response = AggLlmResponse::default();
72        response.preservation.responses.insert(
73            FormatId::from(WireFormat::OpenAiChat),
74            json!({"choices": [{"logprobs": {"content": [{"top_logprobs": alternatives}]}}]}),
75        );
76        p_yes(&response)
77    }
78
79    #[test]
80    fn verdict_mass_is_scored_against_the_pair() {
81        let half = 0.5_f64.ln();
82        let score = scored(json!([
83            {"token": " Yes", "logprob": half},
84            {"token": "no", "logprob": half},
85            {"token": ".", "logprob": -9.0}
86        ]));
87        assert!(score.is_some_and(|score| (score - 0.5).abs() < 1e-9));
88        assert_eq!(scored(json!([{"token": "yes", "logprob": -3.0}])), None);
89        assert_eq!(scored(json!([{"token": "no", "logprob": -3.0}])), None);
90        assert_eq!(scored(json!([{"token": "maybe", "logprob": 0.0}])), None);
91        assert_eq!(p_yes(&AggLlmResponse::default()), None);
92    }
93
94    #[test]
95    fn a_first_token_spent_elsewhere_is_no_verdict() {
96        let stray = scored(json!([
97            {"token": "Okay", "logprob": -0.05},
98            {"token": "yes", "logprob": 0.05_f64.ln()},
99            {"token": "no", "logprob": 0.001_f64.ln()}
100        ]));
101        assert_eq!(stray, None);
102    }
103
104    #[test]
105    fn unavailable_chat_logprobs_produce_no_score() {
106        let mut response = AggLlmResponse::default();
107        for format in [WireFormat::OpenAiResponses, WireFormat::AnthropicMessages] {
108            response.preservation.responses.insert(
109                FormatId::from(format),
110                json!({"choices": [{"logprobs": {"content": [{"top_logprobs": [
111                    {"token": "yes", "logprob": 0.0}
112                ]}]}}]}),
113            );
114            assert_eq!(p_yes(&response), None);
115        }
116        for body in [
117            json!({"choices": []}),
118            json!({"choices": [{"logprobs": null}]}),
119            json!({"choices": [{"logprobs": {"content": []}}]}),
120        ] {
121            response
122                .preservation
123                .responses
124                .insert(FormatId::from(WireFormat::OpenAiChat), body);
125            assert_eq!(p_yes(&response), None);
126        }
127    }
