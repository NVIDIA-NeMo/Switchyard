24/// Session-state key holding the consecutive-escalate streak.
25const STREAK_KEY: &str = "escalation_streak";
26/// Session-state key recording that the up-front capability gate has run for this session.
27const GATE_KEY: &str = "escalation_gate_done";
28
29fn streak(state: &State) -> u32 {
30    match state.extra.get(STREAK_KEY) {
31        Some(StateValue::Count(n)) => *n,
32        _ => 0,
33    }
34}
35
36fn assistant_message(response: &AggLlmResponse) -> Message {
37    Message {
38        role: Role::Assistant,
39        content: response
40            .first_output()
41            .map(|output| output.content.clone())
42            .unwrap_or_default(),
43    }
44}

46/// Calls the efficient model, judges its response, and latches to capable once the streak
47/// confirms. Returns the efficient response directly when not escalating so the caller does
48/// not pay for a second model call.
49struct EscalationClassifier {
50    judge: JudgeClassifier<EscalationJudge, EscalationPolicy>,
51    /// Optional capability forecast run once per session before the first efficient call.
52    gate: Option<Arc<dyn Classifier<State>>>,
53    capable: ModelId,
54    efficient: ModelId,
55    /// Consecutive escalate verdicts required to latch.
56    confirmations: u32,
57}
58
59/// Builds the escalation classifier used by the shared LLM classifier route shell.
60pub(super) fn build_classifier(
61    gate_judge_target: Option<ModelId>,
62    judge_target: ModelId,
63    efficient_target: &ModelId,
64    capable_target: &ModelId,
65    contract_config: ClassifierContractConfig,
66    config: EscalationJudgeConfig,
67    max_output_tokens: u64,
68) -> Result<Arc<dyn Classifier<State>>> {
69    let confirmations = config.confirmations;
70    let gate = match &config.gate {
71        Some(gate) => Some(llm_class::build_capability_gate(
72            gate_judge_target.unwrap_or_else(|| judge_target.clone()),
73            efficient_target,
74            capable_target,
75            gate,
76            contract_config.response_format_type(),
77            max_output_tokens,
78        )?),
79        None => None,
80    };
81    let classifier: Arc<dyn Classifier<State>> = Arc::new(EscalationClassifier {
82        gate,
83        judge: escalation::build_judge(
84            judge_target,
85            capable_target.clone(),
86            efficient_target.clone(),
87            &contract_config,
88            config,
89            max_output_tokens,
90        )?,
91        capable: capable_target.clone(),
92        efficient: efficient_target.clone(),
93        confirmations,
94    });
95    Ok(classifier)
96}

98#[async_trait]
99impl Classifier<State> for EscalationClassifier {
100    async fn score(
101        &self,
102        state: &mut State,
103        request: &mut Request,
104        driver: Option<&Driver>,
105    ) -> Result<(Classification, Option<Response>)> {
106        let Some(driver) = driver else {
107            return Err(LibsyError::AlgorithmError {
108                message: "escalation classifier requires a driver".into(),
109            });
110        };
111
112        // A confirmed session stays capable without a judge call.
113        if streak(state) >= self.confirmations {
114            driver.set_evidence(serde_json::json!({
115                "source": "escalation",
116                "verdict": "latched",
117            }));
118            return Ok((decisive(&self.capable), None));
119        }
120
121        // The gate forecasts once per session, from the task framing alone, before the efficient
122        // tier spends anything. A verdict below the threshold latches immediately; anything else
123        // (including an unusable verdict or a failed call) falls open to the efficient tier and
124        // leaves the trajectory judge in charge.
125        if let Some(gate) = &self.gate
126            && !matches!(state.extra.get(GATE_KEY), Some(StateValue::Count(_)))
127        {
128            state
129                .extra
130                .insert(GATE_KEY.to_string(), StateValue::Count(1));
131            let mut gate_request = request.clone();
132            match gate.score(state, &mut gate_request, Some(driver)).await {
133                Ok((classification, _)) => {
134                    let to_capable = classification
135                        .argmax(false)?
136                        .is_some_and(|score| score.target == self.capable);
137                    // The forecaster's evidence carries `score` (p_solve) and `threshold`; log it
138                    // so an operator can read the forecast distribution and tune the threshold.
139                    let forecast = driver.evidence().unwrap_or(Value::Null);
140                    tracing::info!(
141                        latched = to_capable,
142                        forecast = %forecast,
143                        "escalation gate verdict"
144                    );
145                    if to_capable {
146                        state.extra.insert(
147                            STREAK_KEY.to_string(),
148                            StateValue::Count(self.confirmations),
149                        );
150                        tracing::info!(
151                            target = %self.capable,
152                            "escalation gate latched the session to the capable tier"
153                        );
154                        let mut evidence = serde_json::json!({
155                            "source": "escalation",
156                            "verdict": "gate",
157                        });
158                        if let (Some(evidence), Some(forecast)) =
159                            (evidence.as_object_mut(), forecast.as_object())
160                        {
161                            for key in ["score", "threshold"] {
162                                if let Some(value) = forecast.get(key) {
163                                    evidence.insert(key.to_string(), value.clone());
164                                }
165                            }
166                        }
167                        driver.set_evidence(evidence);
168                        return Ok((decisive(&self.capable), None));
169                    }
170                }
171                Err(error) => {
172                    tracing::warn!(%error, "escalation gate failed; serving the efficient tier");
173                }
174            }
175        }
176
177        // Call efficient model and buffer the response so the judge can read it.
178        //
179        // If the efficient model exceeds its context window, fall through to capable. This call
180        // deliberately has one candidate so the classifier sees the efficient model's error.
181        tracing::info!(
182            target = %self.efficient,
183            "escalation classifier selected efficient tier"
184        );
185        let efficient_response = match driver
186            .call_model(request.clone(), vec![self.efficient.clone()])
187            .await
188        {
189            Ok(r) => r,
190            Err(LibsyError::ClientCall {
191                source: LlmClientError::ContextWindowExceeded { .. },
192                ..
193            }) => {
194                driver.set_evidence(serde_json::json!({
195                    "source": "fallback",
196                    "reason_code": "context_window",
197                }));
198                return Ok((decisive(&self.capable), None));
199            }
200            Err(e) => return Err(e),
201        };
202        // The call resolves when its stream handle arrives; transport can still fail while
203        // buffering. Fall back only for that availability failure and keep other errors typed.
204        let agg = match efficient_response.llm_response.into_agg().await {
205            Ok(agg) => agg,
206            Err(LlmClientError::Transport { .. }) => {
207                driver.set_evidence(serde_json::json!({
208                    "source": "fallback",
209                    "reason_code": "transport",
210                }));
211                return Ok((decisive(&self.capable), None));
212            }
213            Err(source) => {
214                return Err(LibsyError::client_call(self.efficient.clone(), source));
215            }
216        };
217        // Append the efficient reply so the judge reads this turn's completed trajectory.
218        let mut judge_request = request.clone();
219        judge_request
220            .llm_request
221            .messages
222            .push(assistant_message(&agg));
223        let efficient_response = Response {
224            llm_response: if request.llm_request.stream {
225                LlmResponse::Stream(agg.into_stream())
226            } else {
227                LlmResponse::Agg(agg)
228            },
229            metadata: efficient_response.metadata,
230        };
231
232        let (classification, _) = self
233            .judge
234            .score(state, &mut judge_request, Some(driver))
235            .await?;
236
237        let held = streak(state);
238        let best = classification.argmax(false)?;
239        let (escalate, pending) = match &best {
240            Some(score) if score.target == self.capable => (true, held + 1),
241            Some(_) => (false, 0),
242            None => (false, held),
243        };
244        state
245            .extra
246            .insert(STREAK_KEY.to_string(), StateValue::Count(pending));
247
248        if escalate && pending >= self.confirmations {
249            // Streak confirmed: drop the efficient response, caller will serve capable.
250            driver.set_evidence(serde_json::json!({
251                "source": "escalation",
252                "verdict": "escalate",
253            }));
254            return Ok((decisive(&self.capable), None));
255        }
256
257        if escalate {
258            driver.set_evidence(serde_json::json!({
259                "source": "escalation",
260                "verdict": "pending",
261            }));
262        }
263
264        Ok((decisive(&self.efficient), Some(efficient_response)))
265    }
266}

285    /// A queue of replies, drained in order.
286    struct Queue(Mutex<VecDeque<String>>);
287
288    impl Queue {
289        fn new(replies: impl IntoIterator<Item = &'static str>) -> Arc<Self> {
290            Arc::new(Self(Mutex::new(
291                replies.into_iter().map(String::from).collect(),
292            )))
293        }
294
295        fn take(&self) -> String {
296            self.0
297                .lock()
298                .pop_front()
299                .unwrap_or_else(|| "unexpected call".to_string())
300        }
301    }

303    /// Serves the judge and model targets from separate reply queues.
304    fn queued(model: Arc<Queue>, judge: Arc<Queue>) -> impl Serve {
305        move |target: ModelId, request: Request| {
306            let queue = if target == "judge" {
307                Arc::clone(&judge)
308            } else {
309                Arc::clone(&model)
310            };
311            async move {
312                Ok(Response {
313                    llm_response: LlmResponse::Agg(text_response(None, queue.take())),
314                    metadata: request.metadata,
315                })
316            }
317        }
318    }

676            match target.as_str() {
677                "efficient" => Ok(streamed_then_error(LlmClientError::InvalidResponse {
678                    source: Box::new(std::io::Error::other("invalid stream event")),
679                })),
680                other => panic!("unexpected call to {other}"),
681            }
682        };
683        let mut request = classify_request();
684        request.llm_request.stream = true;
685
686        match test_drive(escalation_router()?, request, serve).await {
687            Err(LibsyError::ClientCall {
688                target,
689                source: LlmClientError::InvalidResponse { .. },
690            }) => {
691                assert_eq!(target, "efficient");
692                Ok(())
693            }
694            Err(other) => panic!("expected InvalidResponse client error, got {other:?}"),
695            Ok(_) => panic!("expected stream aggregation to fail"),
696        }
697    }
