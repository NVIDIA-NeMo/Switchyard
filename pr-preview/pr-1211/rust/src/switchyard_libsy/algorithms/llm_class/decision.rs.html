22/// Trailing turns sent to a decision service when the route sets no window.
23const DEFAULT_TURN_WINDOW: usize = 10;
24
25/// Instructions, evidence, and cutoff policy for a decision judge.
26///
27/// Candidate labels keep model names out of the generated context. Evidence must use
28/// the same labels, with unknown outcomes left unknown. Only the first runtime
29/// capable and efficient targets are compared; extra candidates do not add routes.
30#[derive(Clone, Debug)]
31pub struct DecisionJudgeConfig {
32    /// Route capable only when its choice score is strictly above this cutoff.
33    /// This score is not a calibrated solve probability. Choose a cutoff from evaluations.
34    pub cutoff: f64,
35    /// Replaces the packaged instructions. Without custom criteria, preserve
36    /// the meaning of `advantage` and `no_advantage`.
37    pub instructions: Option<Value>,
38    /// Two choice descriptions keyed by candidate label. Requires custom instructions.
39    pub criteria: Option<BTreeMap<String, Value>>,
40    /// Maps anonymous labels used in evidence (e.g. `"a"`) to runtime model IDs.
41    /// Every compared target needs a unique label; extra candidates provide context.
42    pub candidates: BTreeMap<String, ModelId>,
43    /// JSON passed unchanged to the judge, such as candidate descriptions, reference
44    /// cases, outcome/cost summaries, and selection notes. The judge interprets it
45    /// using the instructions; the router neither reads its fields nor derives statistics.
46    pub evidence: Value,
47}
48
49impl DecisionJudgeConfig {
50    pub(super) fn validate(&self) -> Result<()> {
51        if !(0.0..=1.0).contains(&self.cutoff) {
52            return Err(LibsyError::AlgorithmError {
53                message: "decision cutoff must be between 0 and 1".into(),
54            });
55        }
56        if self.candidates.values().collect::<HashSet<_>>().len() != self.candidates.len() {
57            return Err(LibsyError::AlgorithmError {
58                message: "decision candidates must map to distinct targets".into(),
59            });
60        }
61        if let Some(criteria) = &self.criteria
62            && (self.instructions.is_none()
63                || criteria.len() != 2
64                || !criteria.keys().eq(self.candidates.keys()))
65        {
66            return Err(LibsyError::AlgorithmError {
67                message: "custom decision criteria require instructions and exactly two matching candidates".into(),
68            });
69        }
70        Ok(())
71    }
72
73    fn candidate(&self, model: &ModelId) -> Result<&str> {
74        self.candidates
75            .iter()
76            .find(|(_, target)| *target == model)
77            .map(|(id, _)| id.as_str())
78            .ok_or_else(|| LibsyError::AlgorithmError {
79                message: format!("decision candidate is missing for target {model}"),
80            })
81    }
82}

91impl DecisionClassifier {
92    pub(super) fn new(
93        mut config: DecisionJudgeConfig,
94        mut input: TaskInput,
95        fail_open: bool,
96    ) -> Result<Self> {
97        input.recent_turn_window.get_or_insert(DEFAULT_TURN_WINDOW);
98        let instructions = match config.instructions.take() {
99            Some(instructions) => instructions,
100            None => serde_json::from_str(include_str!(
101                "../../prompts/capability-classifier/relative_advantage.json"
102            ))
103            .map_err(|error| LibsyError::external("loading decision judge instructions", error))?,
104        };
105        let options = match &config.criteria {
106            Some(criteria) => criteria
107                .iter()
108                .map(|(id, description)| ChoiceOption {
109                    id: id.clone(),
110                    description: Some(description.clone()),
111                })
112                .collect(),
113            None => [
114                ("advantage", "The capable candidate succeeds AND the efficient candidate fails."),
115                ("no_advantage", "The efficient candidate succeeds OR the capable candidate fails, including shared failure."),
116            ]
117            .into_iter()
118            .map(|(id, description)| ChoiceOption {
119                id: id.into(),
120                description: Some(json!(description)),
121            })
122            .collect(),
123        };
124        let question = DecisionQuestion {
125            instructions,
126            kind: DecisionKind::Choice { options },
127        };
128        Ok(Self {
129            config,
130            input,
131            fail_open,
132            question,
133        })
134    }
135
136    async fn try_score(
137        &self,
138        decision: DecisionRequest,
139        driver: &Driver,
140        judge: &ModelId,
141        score_option: &str,
142    ) -> Result<f64> {
143        let response = driver.call_decision(decision, judge.clone()).await?;
144        let answer = response
145            .answers
146            .get("route")
147            .ok_or_else(|| LibsyError::invalid_decision(judge.clone(), "missing route answer"))?;
148        let DecisionValue::Choice {
149            probabilities: Some(probabilities),
150            ..
151        } = &answer.value
152        else {
153            return Err(LibsyError::invalid_decision(
154                judge.clone(),
155                "expected a choice with probabilities",
156            ));
157        };
158        // The selected option can differ from the cutoff event, so only the
159        // requested option's probability is used.
160        let score = probabilities
161            .get(score_option)
162            .map(|p| p.0)
163            .ok_or_else(|| {
164                LibsyError::invalid_decision(
165                    judge.clone(),
166                    format!("missing probability for {score_option}"),
167                )
168            })?;
169        if !(0.0..=1.0).contains(&score) {
170            return Err(LibsyError::invalid_decision(
171                judge.clone(),
172                "decision probability must be between 0 and 1",
173            ));
174        }
175        Ok(score)
176    }
177}

179#[async_trait]
180impl Classifier for DecisionClassifier {
181    async fn score(
182        &self,
183        _state: &mut State,
184        request: &mut Request,
185        driver: &Driver,
186    ) -> Result<(Classification, Option<Response>)> {
187        let judge = driver.first_model_for(&Category::Judge)?;
188        let capable = driver.first_model_for(&Category::Capable)?;
189        let efficient = driver.first_model_for(&Category::Efficient)?;
190        let score_option = if self.config.criteria.is_some() {
191            self.config.candidate(capable)?
192        } else {
193            "advantage"
194        };
195        let decision = DecisionRequest {
196            model: None,
197            context: json!({
198                "task": self.input.messages(request),
199                "session": request.metadata.as_ref().and_then(|m| m.session_id.as_deref()),
200                "candidates": self.config.candidates.keys().collect::<Vec<_>>(),
201                "comparison": {
202                    "capable": self.config.candidate(capable)?,
203                    "efficient": self.config.candidate(efficient)?,
204                },
205                "evidence": self.config.evidence,
206            }),
207            questions: BTreeMap::from([("route".into(), self.question.clone())]),
208        };
209        let score = match self.try_score(decision, driver, judge, score_option).await {
210            Ok(score) => score,
211            Err(error) if self.fail_open => {
212                return Ok(unavailable(
213                    driver,
214                    judge,
215                    safe_error_summary(&error),
216                    libsy_error_reason(&error),
217                ));
218            }
219            Err(error) => return Err(error),
220        };
221        let (target, category) = if score > self.config.cutoff {
222            (capable, Category::Capable)
223        } else {
224            (efficient, Category::Efficient)
225        };
226        let evidence = json!({
227            "source": "decision_classifier",
228            "verdict": if self.config.criteria.is_some() { "candidate_choice" } else { "relative_advantage" },
229            "selected_candidate": self.config.candidate(target)?,
230            "score_option": score_option,
231            "score": score,
232            "threshold": self.config.cutoff,
233        });
234        let selected = Score {
235            target: target.clone(),
236            confidence: 1.0,
237            category: Some(category),
238        };
239        driver.record_judgment(Some(selected.clone()), Some(evidence.clone()));
240        driver.set_evidence(evidence);
241        Ok((Classification::Scores(vec![selected]), None))
242    }
243}

245/// Settings for a decision service that selects one model category.
246#[derive(Clone, Debug)]
247pub struct DirectClassifierConfig {
248    /// Categories the service may select. Their names are the option IDs.
249    pub options: Vec<Category>,
250    /// Routes to the default target when the decision call fails or returns an
251    /// invalid choice.
252    pub fail_open: bool,
253    /// How often the decision service re-decides this session's target.
254    pub classify_trigger: ClassifyTrigger,
255    /// Uses the first user message when session metadata is unavailable.
256    pub message_hash_fallback: bool,
257    /// Trailing conversation turns sent to the decision service. Defaults to 10.
258    pub recent_turn_window: Option<usize>,
259}
260
261pub(super) struct DirectClassifier {
262    config: DirectClassifierConfig,
263    input: TaskInput,
264    question: DecisionQuestion,
265}
266
267impl DirectClassifier {
268    pub(super) fn new(config: DirectClassifierConfig) -> Result<Self> {
269        if config.options.len() < 2
270            || config
271                .options
272                .iter()
273                .any(|option| matches!(option, Category::Judge | Category::Any))
274        {
275            return Err(LibsyError::AlgorithmError {
276                message: "direct mode needs at least two options other than judge and any".into(),
277            });
278        }
279        let question = DecisionQuestion {
280            instructions: json!("Select the option that should serve this request."),
281            kind: DecisionKind::Choice {
282                options: config
283                    .options
284                    .iter()
285                    .map(|option| ChoiceOption {
286                        id: option.as_str().into(),
287                        description: None,
288                    })
289                    .collect(),
290            },
291        };
292        Ok(Self {
293            input: TaskInput {
294                recent_turn_window: Some(config.recent_turn_window.unwrap_or(DEFAULT_TURN_WINDOW)),
295            },
296            config,
297            question,
298        })
299    }
300
301    async fn try_select(
302        &self,
303        decision: DecisionRequest,
304        driver: &Driver,
305        judge: &ModelId,
306    ) -> Result<&Category> {
307        let response = driver.call_decision(decision, judge.clone()).await?;
308        let Some(DecisionValue::Choice { selected, .. }) =
309            response.answers.get("route").map(|answer| &answer.value)
310        else {
311            return Err(LibsyError::invalid_decision(
312                judge.clone(),
313                "expected a route choice",
314            ));
315        };
316        self.config
317            .options
318            .iter()
319            .find(|option| option.as_str() == selected)
320            .ok_or_else(|| {
321                LibsyError::invalid_decision(judge.clone(), "route choice is not an option")
322            })
323    }
324}

326#[async_trait]
327impl Classifier for DirectClassifier {
328    async fn score(
329        &self,
330        _state: &mut State,
331        request: &mut Request,
332        driver: &Driver,
333    ) -> Result<(Classification, Option<Response>)> {
334        let judge = driver.first_model_for(&Category::Judge)?;
335        let decision = DecisionRequest {
336            model: None,
337            context: json!({
338                "task": self.input.messages(request),
339                "session": request.metadata.as_ref().and_then(|m| m.session_id.as_deref()),
340            }),
341            questions: BTreeMap::from([("route".into(), self.question.clone())]),
342        };
343        let category = match self.try_select(decision, driver, judge).await {
344            Ok(category) => category,
345            Err(error) if self.config.fail_open => {
346                return Ok(unavailable(
347                    driver,
348                    judge,
349                    safe_error_summary(&error),
350                    libsy_error_reason(&error),
351                ));
352            }
353            Err(error) => return Err(error),
354        };
355        let evidence = json!({
356            "source": "decision_classifier",
357            "verdict": "direct",
358            "selected": category.as_str(),
359        });
360        let selected = Score {
361            target: driver.first_model_for(category)?.clone(),
362            confidence: 1.0,
363            category: Some(category.clone()),
364        };
365        driver.record_judgment(Some(selected.clone()), Some(evidence.clone()));
366        driver.set_evidence(evidence);
367        Ok((Classification::Scores(vec![selected]), None))
368    }
369}
370
371fn unavailable(
372    driver: &Driver,
373    judge: &ModelId,
374    error: String,
375    reason: &'static str,
376) -> (Classification, Option<Response>) {
377    report_fail_open(judge.as_str(), error, reason);
378    let evidence = json!({"source": "fail_open", "reason_code": reason});
379    driver.record_judgment(None, Some(evidence.clone()));
380    driver.set_evidence_if_empty(evidence);
381    (Classification::Ambiguous(vec![]), None)
382}
