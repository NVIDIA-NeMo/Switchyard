4//! The trigger classifier: decides, from the gate's signals, whether the
5//! buffered turn warrants a review, and which trigger claims it.
6//!
7//! Runs after the executor call, on response-side signals: a request-side
8//! trigger would review one turn late for mid-task turns and never for the
9//! terminal turn — the premature-"done" moment the gate exists to catch.
10//! Pure over its inputs; the once-per-conversation stall latch and the
11//! review budget live in the ledger.
12
13use crate::Result;
14
15use super::signals::GateSignals;
16use super::{AdvisorGateConfig, GateTrigger, algorithm_error};
17
18/// The trigger with its pattern compiled once at construction.
19enum CompiledTrigger {
20    NoToolCall,
21    Pattern(regex::Regex),
22}
23
24/// Classifies buffered turns against the configured trigger and the stall
25/// checkpoint threshold.
26pub(super) struct TriggerClassifier {
27    trigger: CompiledTrigger,
28    /// For the `no_tool_call` trigger: tool results the conversation must
29    /// carry before a terminal turn is reviewable.
30    min_tool_results: u32,
31    /// Assistant turns at which the stall checkpoint is reached; 0 disables.
32    stall_turns: u32,
33}
34
35/// What the classifier concluded about one buffered turn.
36pub(super) struct TriggerDecision {
37    /// The terminal-turn trigger that fired, as the telemetry `trigger`
38    /// label ("no_tool_call" | "pattern"); `None` when the turn is not
39    /// terminal.
40    pub(super) fired: Option<&'static str>,
41    /// The conversation reached the stall checkpoint. Threshold only — the
42    /// once-per-conversation latch belongs to the budget ledger.
43    pub(super) stalled: bool,
44}
45
46impl TriggerClassifier {
47    /// Validates and compiles the configured trigger.
48    pub(super) fn new(config: &AdvisorGateConfig) -> Result<Self> {
49        let trigger = match &config.gate_trigger {
50            GateTrigger::NoToolCall => CompiledTrigger::NoToolCall,
51            GateTrigger::Pattern(pattern) => {
52                if pattern.is_empty() {
53                    return Err(algorithm_error(
54                        "gate_trigger 'pattern' requires a non-empty gate_trigger_pattern",
55                    ));
56                }
57                CompiledTrigger::Pattern(regex::Regex::new(pattern).map_err(|error| {
58                    algorithm_error(format!(
59                        "gate_trigger_pattern is not a valid regex: {error}"
60                    ))
61                })?)
62            }
63        };
64        Ok(Self {
65            trigger,
66            min_tool_results: config.gate_min_tool_results,
67            stall_turns: config.gate_stall_turns,
68        })
69    }
70
71    /// Classifies the turn: does it warrant a review, and on which trigger?
72    pub(super) fn classify(&self, signals: &GateSignals) -> TriggerDecision {
73        let fired = match &self.trigger {
74            CompiledTrigger::Pattern(pattern) => pattern
75                .is_match(signals.turn.visible_text.as_deref().unwrap_or(""))
76                .then_some("pattern"),
77            CompiledTrigger::NoToolCall => (!signals.turn.has_tool_use
78                && signals.conversation.tool_result_count >= self.min_tool_results)
79                .then_some("no_tool_call"),
80        };
81        let stalled =
82            self.stall_turns > 0 && signals.conversation.assistant_turn_count >= self.stall_turns;
83        TriggerDecision { fired, stalled }
84    }
85}

87#[cfg(test)]
88mod tests {
89    use super::super::signals::TurnSignals;
90    use super::*;
91    use crate::algorithms::util::tool_signals::ToolSignals;
92
93    fn classifier(config: AdvisorGateConfig) -> TriggerClassifier {
94        TriggerClassifier::new(&config).expect("test config is valid")
95    }
96
97    fn signals(
98        has_tool_use: bool,
99        visible_text: Option<&str>,
100        tool_results: u32,
101        assistant_turns: u32,
102    ) -> GateSignals {
103        GateSignals {
104            conversation: ToolSignals {
105                tool_result_count: tool_results,
106                assistant_turn_count: assistant_turns,
107                ..ToolSignals::default()
108            },
109            turn: TurnSignals {
110                has_tool_use,
111                visible_text: visible_text.map(str::to_string),
112            },
113        }
114    }
115
116    #[test]
117    fn no_tool_call_fires_on_tool_less_turns_past_the_guard() {
118        let classifier = classifier(AdvisorGateConfig {
119            gate_min_tool_results: 2,
120            ..AdvisorGateConfig::default()
121        });
122        // A tool-less turn under the guard stays quiet; past it, it fires.
123        assert!(
124            classifier
125                .classify(&signals(false, None, 1, 0))
126                .fired
127                .is_none()
128        );
129        assert_eq!(
130            classifier.classify(&signals(false, None, 2, 0)).fired,
131            Some("no_tool_call")
132        );
133        // Tool use exempts the turn regardless of the guard.
134        assert!(
135            classifier
136                .classify(&signals(true, None, 5, 0))
137                .fired
138                .is_none()
139        );
140    }
141
142    #[test]
143    fn pattern_reads_text_only_and_ignores_tool_use() {
144        let classifier = classifier(AdvisorGateConfig {
145            gate_trigger: GateTrigger::Pattern("task_complete".to_string()),
146            ..AdvisorGateConfig::default()
147        });
148        assert_eq!(
149            classifier
150                .classify(&signals(true, Some("task_complete: true"), 0, 0))
151                .fired,
152            Some("pattern")
153        );
154        assert!(
155            classifier
156                .classify(&signals(false, Some("still working"), 0, 0))
157                .fired
158                .is_none()
159        );
160        // No visible text matches against the empty string, not a panic.
161        assert!(
162            classifier
163                .classify(&signals(false, None, 0, 0))
164                .fired
165                .is_none()
166        );
167    }
168
169    #[test]
170    fn the_stall_threshold_is_reached_at_the_configured_turn_count() {
171        let classifier = classifier(AdvisorGateConfig {
172            gate_stall_turns: 3,
173            ..AdvisorGateConfig::default()
174        });
175        assert!(!classifier.classify(&signals(true, None, 0, 2)).stalled);
176        assert!(classifier.classify(&signals(true, None, 0, 3)).stalled);
177    }
178
179    #[test]
180    fn a_zero_stall_threshold_disables_the_checkpoint() {
181        let classifier = classifier(AdvisorGateConfig::default());
182        assert!(!classifier.classify(&signals(true, None, 0, 100)).stalled);
183    }
184}
