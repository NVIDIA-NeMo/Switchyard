4//! Executor gated by a once-per-session advisor review.
5//!
6//! The executor answers every client-visible turn. Turns with tool calls pass
7//! through unreviewed; the first *terminal* turn — no tool calls (or a text
8//! match under the `pattern` trigger) — is buffered and shown to a stronger
9//! advisor model together with the full transcript. `APPROVE` releases the
10//! buffered turn unchanged; `REDO` appends the discarded turn's text and the
11//! advisor's plan as feedback, then re-invokes the executor so it keeps
12//! working. Each budget scope (one benchmark evaluation, one session, or the
13//! whole instance — see [`budget::budget_scope`]) is reviewed at most
14//! `max_reviews` times; afterwards every call is a pure passthrough.
15//!
16//! This design is a near-superset of solo executor behavior: identical until
17//! the executor first claims to be done, plus one quality gate that catches
18//! premature convergence. Front-loading advice was measured to suppress the
19//! executor's own test-and-iterate loop, so no advice is injected up front.
20//!
21//! Failure posture: executor errors always propagate (including
22//! `ContextWindowExceeded`, which hosts map to a client-visible 400 so agent
23//! harnesses can compact). Advisor errors honor `fail_open` — the buffered
24//! turn passes through as an implicit APPROVE — refund the consumed review,
25//! and count toward a per-scope failure cap that stops consulting a down
26//! advisor entirely.
27//!
28//! Structure: [`AdvisorGate`] is a thin orchestrator over the gate's
29//! components. The [`signals`] processor folds each event's facts into the
30//! per-turn state — conversation counts on [`Event::Request`], the generated
31//! turn's shape on [`Event::ModelResponse`] — the [`trigger`] classifier
32//! reads them after the executor call, and the [`budget`] ledger holds the
33//! only mutable state: the reserve/refund review budget and the stall latch.

79/// Labels the executor's internal reasoning when a turn has no visible text,
80/// so the advisor still has evidence to review (reasoning models on vLLM/NIM
81/// can emit turns whose only output is reasoning).
82const REASONING_TAIL_LABEL: &str =
83    "(the executor produced no visible text this turn; its internal reasoning follows)\n";
84/// REDO echo when the discarded turn had neither text nor reasoning; strict
85/// endpoints (Anthropic) reject empty text blocks, so never echo "".
86const EMPTY_ECHO_PLACEHOLDER: &str = "(the executor produced no output this turn)";
87/// Benchmark harnesses stamp every request of one evaluation — sub-agents
88/// included — with this header, so it is the review budget's first-choice
89/// scope: "reviews for *this* task" survives gateways shared by many tasks.
90const BENCH_SESSION_HEADER: &str = "proxy_x_session_id";

92/// How the gate decides a buffered executor turn is terminal.
93#[derive(Clone, Debug, PartialEq)]
94pub enum GateTrigger {
95    /// First turn without tool calls (subject to `gate_min_tool_results`).
96    NoToolCall,
97    /// First turn whose visible text matches this regex (searched, not anchored) —
98    /// for text-protocol harnesses where every turn lacks tool calls and
99    /// completion is declared with a textual marker instead.
100    Pattern(String),
101}

103/// Gate knobs; defaults mirror the benchmarked Python advisor configuration.
104#[derive(Clone, Debug)]
105pub struct AdvisorGateConfig {
106    /// System prompt for the advisor's review call; states the APPROVE/REDO contract.
107    pub reviewer_system_prompt: String,
108    /// Prepended to the advisor's REDO plan when fed back to the executor.
109    pub redo_feedback_prefix: String,
110    /// What fires the review.
111    pub gate_trigger: GateTrigger,
112    /// Reviews allowed per budget scope. 1 keeps the original once-per-task
113    /// gate; higher values re-review later terminal turns, making the gate a
114    /// sequential best-of-(N+1) with the advisor as judge.
115    pub max_reviews: u32,
116    /// When > 0, additionally review (once per conversation, consuming budget)
117    /// the first request already carrying at least this many assistant turns —
118    /// a mid-task checkpoint for executors that grind without declaring
119    /// completion. 0 disables.
120    pub gate_stall_turns: u32,
121    /// For the `no_tool_call` trigger: only review once the conversation
122    /// carries at least this many tool results, skipping early commentary
123    /// turns on chatty harnesses. 0 reviews from the first terminal turn.
124    pub gate_min_tool_results: u32,
125    /// Cap on the advisor's output per consult.
126    pub advisor_max_tokens: u64,
127    /// Sampling temperature for the consult; `None` omits the field on the wire.
128    pub advisor_temperature: Option<f64>,
129    /// Cap on the serialized transcript handed to the advisor; the middle of
130    /// an over-cap conversation is dropped (task head + recent tail survive).
131    pub transcript_max_chars: usize,
132    /// When true (default), an advisor failure degrades to APPROVE; when
133    /// false, it propagates as the turn's error.
134    pub fail_open: bool,
135}

154/// Advisor review gate: executor turns pass through until the first terminal
155/// turn, which a stronger advisor reviews once per scope budget (APPROVE
156/// releases it, REDO feeds the plan back and re-invokes the executor).
157pub struct AdvisorGate {
158    executor: ModelId,
159    advisor: ModelId,
160    config: AdvisorGateConfig,
161    /// Folds request- and response-side facts into the per-turn [`GateSignals`].
162    signals: GateSignalProcessor,
163    /// Decides, from the signals, whether the buffered turn warrants review.
164    trigger: TriggerClassifier,
165    /// Reserve/refund review ledger and stall latch — the gate's only mutable state.
166    budget: ReviewBudget,
167    verdict_re: regex::Regex,
168}
169
170impl AdvisorGate {
171    /// Validates ranges and compiles the trigger and verdict patterns.
172    pub fn new(executor: ModelId, advisor: ModelId, config: AdvisorGateConfig) -> Result<Self> {
173        if config.max_reviews < 1 {
174            return Err(algorithm_error("max_reviews must be at least 1"));
175        }
176        if config.advisor_max_tokens < 1 {
177            return Err(algorithm_error("advisor_max_tokens must be at least 1"));
178        }
179        if config.transcript_max_chars < 256 {
180            return Err(algorithm_error("transcript_max_chars must be at least 256"));
181        }
182        let trigger = TriggerClassifier::new(&config)?;
183        let verdict_re = regex::Regex::new(VERDICT_PATTERN).map_err(|error| {
184            algorithm_error(format!("verdict pattern failed to compile: {error}"))
185        })?;
186        let budget = ReviewBudget::new(config.max_reviews);
187        Ok(Self {
188            executor,
189            advisor,
190            config,
191            signals: GateSignalProcessor,
192            trigger,
193            budget,
194            verdict_re,
195        })
196    }

200    async fn route_inner(
201        &self,
202        driver: &Driver,
203        request: Request,
204        scope: &ScopeKey,
205    ) -> Result<RoutingOutcome> {
206        // Spent budget (or failure cap): pure passthrough — live stream,
207        // verbatim preserved-body replay, zero buffering. Executor errors
208        // (including ContextWindowExceeded) propagate for the host's
209        // client-visible mapping.
210        if self.budget.check_exhausted(scope) {
211            return Ok(RoutingOutcome::route_to(
212                self.executor.clone(),
213                Vec::new(),
214                request,
215            ));
216        }
217
218        // Request-side signals fold in before the executor runs: conversation
219        // counts from the shared ToolSignals extraction — the same definition
220        // of tool-result/turn counting the stage router uses.
221        let mut request = request;
222        let mut signals = GateSignals::default();
223        self.signals
224            .process(
225                &mut signals,
226                Event::Request {
227                    request: &mut request,
228                    driver: Some(driver),
229                },
230            )
231            .await?;
232
233        // Gated phase: generate the turn once, fully buffered, so the gate
234        // can inspect it before the client sees anything.
235        let response = driver
236            .call_model(request.clone(), vec![self.executor.clone()])
237            .await?;
238        let turn = buffer_turn(self.executor.as_str(), response).await?;
239
240        // Response-side signals fold in after it: a terminal turn never
241        // appears on any subsequent request — the harness stops once the
242        // executor answers without tools — so the trigger classifier can only
243        // run on the model-response event.
244        self.signals
245            .process(&mut signals, Event::ModelResponse(&turn.agg))
246            .await?;
247
248        let decision = self.trigger.classify(&signals);
249        // The stall checkpoint fires once per conversation regardless of the
250        // turn's shape — even a tool-call turn — for executors that grind
251        // without ever declaring completion.
252        let stall_key = stall_key(&request);
253        let stall = decision.stalled && !self.budget.stall_already_fired(stall_key);
254        if decision.fired.is_none() && !stall {
255            return Ok(RoutingOutcome::answered(
256                self.executor.clone(),
257                request,
258                turn.into_response(),
259            ));
260        }
261        // A stall consumed by a simultaneous trigger does not latch, so the
262        // checkpoint can still fire later if this review is refunded.
263        if stall && decision.fired.is_none() {
264            self.budget.mark_stall_fired(stall_key);
265        }
266        if !self.budget.try_reserve(scope) {
267            return Ok(RoutingOutcome::answered(
268                self.executor.clone(),
269                request,
270                turn.into_response(),
271            ));
272        }
273
274        let trigger_label = decision.fired.unwrap_or("stall");
275        let review_tail = visible_text(&turn.agg).or_else(|| {
276            reasoning_text(&turn.agg).map(|reasoning| format!("{REASONING_TAIL_LABEL}{reasoning}"))
277        });
278        match self
279            .consult(driver, &request, review_tail.as_deref(), trigger_label)
280            .await
281        {
282            Ok(ConsultOutcome::Approve) => Ok(RoutingOutcome::answered(
283                self.executor.clone(),
284                request,
285                turn.into_response(),
286            )),
287            Ok(ConsultOutcome::Redo { plan }) => Ok(self.redo(request, turn, &plan)),
288            Ok(ConsultOutcome::Failed) => {
289                self.budget.refund_failure(scope);
290                Ok(RoutingOutcome::answered(
291                    self.executor.clone(),
292                    request,
293                    turn.into_response(),
294                ))
295            }
296            Err(error) => {
297                self.budget.refund_failure(scope);
298                Err(error)
299            }
300        }
301    }

303    /// REDO: the client never sees the gated turn. Its text (or reasoning) is
304    /// echoed as an assistant message, the advisor's plan follows as user
305    /// feedback, and the executor continues as a pure passthrough call.
306    fn redo(&self, request: Request, turn: GatedTurn, plan: &str) -> RoutingOutcome {
307        record_discarded(&turn.agg.usage);
308        emit_discarded_audit(self.executor.as_str(), &turn.agg.usage);
309        let echo = visible_text(&turn.agg)
310            .or_else(|| reasoning_text(&turn.agg))
311            .unwrap_or_else(|| EMPTY_ECHO_PLACEHOLDER.to_string());
312        let mut redo = request;
313        redo.llm_request
314            .messages
315            .push(Message::text(Role::Assistant, echo));
316        redo.llm_request.messages.push(Message::text(
317            Role::User,
318            format!("{}{}", self.config.redo_feedback_prefix, plan),
319        ));
320        // Mandatory after any message mutation: codecs otherwise replay the
321        // preserved pre-surgery body verbatim and the feedback never reaches
322        // the executor.
323        crate::algorithms::util::prompts::drop_exact_replay(&mut redo);
324        RoutingOutcome::route_to(self.executor.clone(), Vec::new(), redo)
325    }

327    /// Consults the advisor over the buffered transcript and parses the
328    /// verdict. `Ok(Failed)` covers fail-open errors and unparseable replies
329    /// (the caller refunds); fail-closed errors return `Err`.
330    async fn consult(
331        &self,
332        driver: &Driver,
333        base: &Request,
334        review_tail: Option<&str>,
335        trigger: &'static str,
336    ) -> Result<ConsultOutcome> {
337        // The advisor reviews the FULL transcript: system/developer content is
338        // normalized out of `messages` into `instructions`, so prepend it back
339        // as leading messages (identical {role, content} shape) — the task
340        // constraints the verdict must check against usually live there.
341        let transcript_messages: Vec<Message> = base
342            .llm_request
343            .instructions
344            .iter()
345            .map(|block| Message {
346                role: block.role,
347                content: block.content.clone(),
348            })
349            .chain(base.llm_request.messages.iter().cloned())
350            .collect();
351        let transcript = review_transcript(
352            &transcript_messages,
353            review_tail,
354            self.config.transcript_max_chars,
355        );
356        let consult_request = self.build_consult_request(base, transcript);
357        let started = Instant::now();
358        let reply = match driver
359            .call_model(consult_request, vec![self.advisor.clone()])
360            .await
361        {
362            Ok(response) => response
363                .llm_response
364                .into_agg()
365                .await
366                .map_err(|source| LibsyError::client_call(self.advisor.clone(), source)),
367            Err(error) => Err(error),
368        };
369        let latency_ms = started.elapsed().as_secs_f64() * 1000.0;
370        let agg = match reply {
371            Ok(agg) => agg,
372            Err(error) => {
373                record_consult_failure(crate::algorithms::util::llm_judge::libsy_error_reason(
374                    &error,
375                ));
376                if !self.config.fail_open {
377                    // Surface as an algorithm failure (5xx), never as the
378                    // advisor's own client error: a typed ContextWindowExceeded
379                    // from the consult would otherwise reach the client as 400
380                    // context_length_exceeded and trigger compaction of a
381                    // healthy conversation.
382                    return Err(algorithm_error(format!(
383                        "advisor consult failed (fail_open = false): {error}"
384                    )));
385                }
386                tracing::warn!(
387                    target: "libsy",
388                    error = %error,
389                    "advisor gate: consult failed; passing the turn through (fail open)"
390                );
391                emit_review_audit(ReviewAudit {
392                    verdict: "APPROVE",
393                    error: Some(error.to_string()),
394                    latency_ms,
395                    reply_head: None,
396                    usage: None,
397                });
398                return Ok(ConsultOutcome::Failed);
399            }
400        };
401        let reply_text = advisor_reply_text(&agg);
402        let reply_head: String = reply_text.chars().take(160).collect();
403        match parse_verdict(&self.verdict_re, &reply_text) {
404            Some(Verdict::Approve) => {
405                record_review("approve", trigger);
406                emit_review_audit(ReviewAudit {
407                    verdict: "APPROVE",
408                    error: None,
409                    latency_ms,
410                    reply_head: Some(reply_head),
411                    usage: Some(&agg.usage),
412                });
413                Ok(ConsultOutcome::Approve)
414            }
415            Some(Verdict::Redo { plan }) => {
416                record_review("redo", trigger);
417                emit_review_audit(ReviewAudit {
418                    verdict: "REDO",
419                    error: None,
420                    latency_ms,
421                    reply_head: Some(reply_head),
422                    usage: Some(&agg.usage),
423                });
424                Ok(ConsultOutcome::Redo { plan })
425            }
426            None => {
427                // The advisor spent real tokens on a reply the gate cannot
428                // act on; the observer already recorded them. Refunded by
429                // the caller so a flaky advisor cannot burn the budget.
430                record_review("unparseable", trigger);
431                emit_review_audit(ReviewAudit {
432                    verdict: "UNPARSEABLE",
433                    error: None,
434                    latency_ms,
435                    reply_head: Some(reply_head),
436                    usage: Some(&agg.usage),
437                });
438                Ok(ConsultOutcome::Failed)
439            }
440        }
441    }

443    /// A fresh, buffered, tool-free request carrying the reviewer contract and
444    /// the serialized transcript; metadata is kept for session correlation.
445    fn build_consult_request(&self, base: &Request, transcript: String) -> Request {
446        Request {
447            llm_request: LlmRequest {
448                model: base.llm_request.model.clone(),
449                instructions: vec![InstructionBlock {
450                    role: Role::System,
451                    content: vec![ContentBlock::Text {
452                        text: self.config.reviewer_system_prompt.clone(),
453                    }],
454                }],
455                messages: vec![Message::text(Role::User, transcript)],
456                sampling: SamplingParams {
457                    temperature: self.config.advisor_temperature,
458                    ..SamplingParams::default()
459                },
460                output: OutputParams {
461                    max_output_tokens: Some(self.config.advisor_max_tokens),
462                    response_format: None,
463                },
464                ..LlmRequest::default()
465            },
466            raw_request: None,
467            metadata: base.metadata.clone(),
468        }
469    }

472#[async_trait::async_trait]
473impl Algorithm for AdvisorGate {
474    fn name(&self) -> &str {
475        "advisor_gate"
476    }
477
478    async fn route(self: Arc<Self>, driver: Driver, request: Request) -> Result<RoutingOutcome> {
479        let scope = budget_scope(&request);
480        let session_final = request
481            .metadata
482            .as_ref()
483            .and_then(|metadata| metadata.session_final)
484            == Some(true);
485        let result = self.route_inner(&driver, request, &scope).await;
486        if session_final {
487            self.budget.evict_scope(&scope);
488        }
489        result
490    }
491}

493/// Outcome of one consult; `Failed` = fail-open error or unparseable reply.
494enum ConsultOutcome {
495    Approve,
496    Redo { plan: String },
497    Failed,
498}

501    LibsyError::AlgorithmError {
502        message: message.into(),
