4//! The judged views a verifier reads.
5//!
6//! Each section is redacted before its budget applies, and user requirements
7//! render in full while trajectory and attempt evidence clip.
8
9use switchyard_protocol::{ContentBlock, Request, Role};
10
11use super::text::{Turn, clip_mid, content_text, redact};
12
13const ASSISTANT_TURN_BUDGET: usize = 1200;
14pub(super) const ATTEMPT_BUDGET: usize = 2200;
15const AGENTIC_TRAJECTORY_BUDGET: usize = 12_000;
16const AGENTIC_EVENT_BUDGET: usize = 1400;
17
18/// Redacted non-empty turns, and the index of the latest user turn among them.
19fn rendered(turns: &[Turn]) -> (Vec<(Role, String)>, Option<usize>) {
20    let rendered: Vec<(Role, String)> = turns
21        .iter()
22        .filter(|turn| !turn.text.trim().is_empty())
23        .map(|turn| (turn.role, redact(&turn.text)))
24        .collect();
25    let latest_user = rendered.iter().rposition(|(role, _)| *role == Role::User);
26    (rendered, latest_user)
27}
28
29fn role_label(role: Role) -> &'static str {
30    match role {
31        Role::System => "system",
32        Role::Developer => "developer",
33        Role::User => "user",
34        Role::Assistant => "assistant",
35        Role::Tool => "tool",
36    }
37}
38
39fn attempt_section(attempt: &str) -> String {
40    clip_mid(&redact(attempt), ATTEMPT_BUDGET, 1.0 / 3.0)
41}
42
43/// Earlier turns first, then the latest user instruction and the attempt.
44pub(super) fn render_session(turns: &[Turn], attempt: &str) -> String {
45    let (turns, latest_user) = rendered(turns);
46    let mut lines: Vec<String> = turns
47        .iter()
48        .enumerate()
49        .filter(|(index, _)| Some(*index) != latest_user)
50        .map(|(_, (role, text))| {
51            let text = if matches!(role, Role::User | Role::System | Role::Developer) {
52                text.clone()
53            } else {
54                clip_mid(text, ASSISTANT_TURN_BUDGET, 0.5)
55            };
56            format!("[{}] {text}", role_label(*role))
57        })
58        .collect();
59    if let Some(index) = latest_user {
60        lines.push(format!("[user (latest)] {}", turns[index].1));
61    }
62    lines.push(format!("[assistant attempt] {}", attempt_section(attempt)));
63    lines.join("\n")
64}

66/// The user task and updates, the tool trajectory, and the attempt.
67///
68/// Framework instructions are omitted.
69pub(super) fn render_agentic_view(request: &Request, turns: &[Turn], attempt: &str) -> String {
70    let (turns, _) = rendered(turns);
71    let requirements = turns
72        .iter()
73        .filter(|(role, _)| *role == Role::User)
74        .enumerate()
75        .map(|(index, (_, text))| match index {
76            0 => format!("[user task] {text}"),
77            _ => format!("[user update {index}] {text}"),
78        })
79        .collect::<Vec<_>>()
80        .join("\n");
81    let trajectory = agentic_trajectory(request);
82    let mut parts = vec![format!("USER TASK AND UPDATES:\n{requirements}")];
83    if !trajectory.is_empty() {
84        parts.push(format!(
85            "TOOL TRAJECTORY:\n{}",
86            clip_mid(&trajectory.join("\n"), AGENTIC_TRAJECTORY_BUDGET, 1.0 / 3.0)
87        ));
88    }
89    parts.push(format!("CURRENT ATTEMPT:\n{}", attempt_section(attempt)));
90    parts.join("\n\n")
91}
92
93/// Bounded assistant and tool events after the first user task.
94fn agentic_trajectory(request: &Request) -> Vec<String> {
95    let mut messages = request.llm_request.messages.iter();
96    for message in messages.by_ref() {
97        if message.role == Role::User && !content_text(&message.content).0.trim().is_empty() {
98            break;
99        }
100    }
101    let mut events = Vec::new();
102    for message in messages {
103        for block in &message.content {
104            let event = match block {
105                ContentBlock::Text { text } | ContentBlock::Refusal { text } => {
106                    match message.role {
107                        Role::Assistant => Some(("assistant", text.clone())),
108                        Role::Tool => Some(("tool result", text.clone())),
109                        _ => None,
110                    }
111                }
112                ContentBlock::ToolCall(call) => {
113                    Some(("tool call", format!("{}({})", call.name, call.arguments)))
114                }
115                ContentBlock::ToolResult(result) => Some((
116                    if result.is_error == Some(true) {
117                        "tool error"
118                    } else {
119                        "tool result"
120                    },
121                    content_text(&result.content).0,
122                )),
123                _ => None,
124            };
125            if let Some((label, text)) = event {
126                events.push(format!(
127                    "[{label}] {}",
128                    clip_mid(&redact(&text), AGENTIC_EVENT_BUDGET, 0.5)
129                ));
130            }
131        }
132    }
133    events
134}
