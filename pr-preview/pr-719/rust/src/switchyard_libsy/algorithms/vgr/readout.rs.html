4//! The cheap readout: the probability a verifier put on "yes" at its first token.
5//!
6//! Normalized responses carry no logprobs, so this reads the preserved Chat
7//! body. Anything that cannot be scored is `None`, which never commits.
8
9use serde_json::{Value, json};
10use switchyard_protocol::{AggLlmResponse, FormatId, Request, WireFormat};
11
12pub(super) const MAX_OUTPUT_TOKENS: u64 = 4;
13
14/// Asks for a direct verdict with its alternatives; reasoning stays request-local.
15pub(super) fn request_logprobs(request: &mut Request) {
16    request.llm_request.reasoning.effort = Some("none".to_string());
17    let extensions = &mut request.llm_request.extensions.fields;
18    extensions.insert("logprobs".to_string(), json!(true));
19    extensions.insert("top_logprobs".to_string(), json!(8));
20}
21
22pub(super) fn p_yes(response: &AggLlmResponse) -> Option<f64> {
23    let alternatives = response
24        .preservation
25        .responses
26        .get(&FormatId::from(WireFormat::OpenAiChat))?
27        .pointer("/choices/0/logprobs/content/0/top_logprobs")?
28        .as_array()?;
29    let (mut yes, mut no) = (None, None);
30    for entry in alternatives {
31        let Some(probability) = entry.get("logprob").and_then(Value::as_f64) else {
32            continue;
33        };
34        let token = entry.get("token")?.as_str()?;
35        let slot = match token
36            .trim()
37            .trim_matches(['"', '\'', '.', ','])
38            .to_lowercase()
39            .as_str()
40        {
41            "yes" | "y" | "true" => &mut yes,
42            "no" | "n" | "false" => &mut no,
43            _ => continue,
44        };
45        *slot.get_or_insert(0.0) += probability.exp();
46    }
47    match (yes, no) {
48        (Some(yes), Some(no)) if yes + no > 0.0 => Some(yes / (yes + no)),
49        (Some(_), None) => Some(1.0),
50        (None, Some(_)) => Some(0.0),
51        _ => None,
52    }
53}

55#[cfg(test)]
56mod tests {
57    use super::*;
58
59    fn scored(alternatives: Value) -> Option<f64> {
60        let mut response = AggLlmResponse::default();
61        response.preservation.responses.insert(
62            FormatId::from(WireFormat::OpenAiChat),
63            json!({"choices": [{"logprobs": {"content": [{"top_logprobs": alternatives}]}}]}),
64        );
65        p_yes(&response)
66    }
67
68    #[test]
69    fn verdict_mass_is_scored_against_the_pair() {
70        let half = 0.5_f64.ln();
71        let score = scored(json!([
72            {"token": " Yes", "logprob": half},
73            {"token": "no", "logprob": half},
74            {"token": ".", "logprob": -9.0}
75        ]));
76        assert!(score.is_some_and(|score| (score - 0.5).abs() < 1e-9));
77        assert_eq!(
78            scored(json!([{"token": "yes", "logprob": -3.0}])),
79            Some(1.0)
80        );
81        assert_eq!(scored(json!([{"token": "maybe", "logprob": 0.0}])), None);
82        assert_eq!(p_yes(&AggLlmResponse::default()), None);
83    }
84}
