15/// This check reports metadata loss or rejects conversion before grouping can discard the fields.
16pub(crate) fn check_responses_metadata_loss(
17    request: &LlmRequest,
18    target: WireFormat,
19    diagnostics: &mut Vec<TranslationDiagnostic>,
20    policy: &TranslationPolicy,
21) -> Result<()> {
22    let mut has_conversation = false;
23    for message in &request.messages {
24        let Some(metadata) = &message.responses else {
25            has_conversation = true;
26            continue;
27        };
28        if metadata.fields.get("type").and_then(Value::as_str) == Some("additional_tools") {
29            continue;
30        }
31        if metadata.fields.contains_key("phase") {
32            push_lossy(
33                diagnostics,
34                policy,
35                format!("Responses assistant phase is not supported by {target}"),
36            )?;
37        }
38        // Reasoning item IDs live in the block's details and can survive Chat conversion.
39        let loses_reasoning_id = target != WireFormat::OpenAiChat
40            && message.content.iter().any(|block| {
41                matches!(block, ContentBlock::Reasoning { details, .. }
42                    if details.iter().any(|item| item.get("id").is_some()))
43            });
44        if metadata.fields.contains_key("id") || loses_reasoning_id {
45            push_lossy(
46                diagnostics,
47                policy,
48                format!(
49                    "Responses item id is not supported by {target}; tool call_id remains separate"
50                ),
51            )?;
52        }
53        if matches!(message.role, Role::System | Role::Developer) {
54            if has_conversation {
55                push_lossy(
56                    diagnostics,
57                    policy,
58                    format!("Responses inline instruction position changes when encoding {target}"),
59                )?;
60            }
61            if message.role == Role::Developer && target == WireFormat::AnthropicMessages {
62                push_lossy(
63                    diagnostics,
64                    policy,
65                    "Responses developer instructions become Anthropic system instructions",
66                )?;
67            }
68        } else {
69            has_conversation = true;
70        }
71    }
72    Ok(())
73}

83/// Converts an OpenAI allowed-tools policy to a restricted function list and mode.
84pub(crate) fn allowed_function_tools(
85    request: &LlmRequest,
86) -> Result<Option<(Vec<Tool>, ToolChoice)>> {
87    let Some(ToolChoice::Raw(choice)) = &request.tool_choice else {
88        return Ok(None);
89    };
90    if choice.get("type").and_then(Value::as_str) != Some("allowed_tools") {
91        return Ok(None);
92    }
93    let error = || {
94        TranslationError::LossyConversion(
95            "allowed_tools translation requires auto or required mode \
96             and a nonempty subset of known function tools"
97                .to_string(),
98        )
99    };
100    let policy = choice.get("allowed_tools").unwrap_or(choice);
101    let mode = match policy.get("mode").and_then(Value::as_str) {
102        Some("auto") => ToolChoice::Auto,
103        Some("required") => ToolChoice::Required,
104        _ => return Err(error()),
105    };
106    let allowed = policy
107        .get("tools")
108        .and_then(Value::as_array)
109        .ok_or_else(error)?;
110    if allowed.is_empty() {
111        return Err(error());
112    }
113    let mut names = Vec::with_capacity(allowed.len());
114    for tool in allowed {
115        if tool.get("type").and_then(Value::as_str) != Some("function") {
116            return Err(error());
117        }
118        let function = tool.get("function").unwrap_or(tool);
119        let name = function
120            .get("name")
121            .and_then(Value::as_str)
122            .filter(|name| !name.is_empty())
123            .ok_or_else(error)?;
124        let name = match function.get("namespace").and_then(Value::as_str) {
125            Some(namespace) if !namespace.is_empty() => {
126                crate::codex_namespaces::qualified_tool_name(namespace, name)
127            }
128            _ => name.to_string(),
129        };
130        if !request
131            .tools
132            .iter()
133            .filter_map(Tool::function)
134            .any(|tool| tool.name == name)
135        {
136            return Err(error());
137        }
138        names.push(name);
139    }
140    // Filter the catalog, rather than rebuilding it from selectors, to retain definitions
141    // and avoid duplicating tools when the allowed list repeats a name.
142    let tools = request
143        .tools
144        .iter()
145        .filter(|tool| {
146            tool.function()
147                .is_some_and(|tool| names.contains(&tool.name))
148        })
149        .cloned()
150        .collect();
151    Ok(Some((tools, mode)))
152}

162/// Extracts text-like blocks and joins them for text-only provider fields.
163pub(crate) fn text_from_blocks(content: &[ContentBlock], separator: &str) -> String {
164    content
165        .iter()
166        .filter_map(|block| match block {
167            ContentBlock::Text { text, .. } => Some(text.as_str()),
168            ContentBlock::Refusal { text } => Some(text.as_str()),
169            ContentBlock::Unknown { raw, .. } => raw.as_str(),
170            _ => None,
171        })
172        .collect::<Vec<_>>()
173        .join(separator)
174}

176/// Extracts private reasoning blocks without mixing them into visible text.
177pub(crate) fn reasoning_text_from_blocks(content: &[ContentBlock], separator: &str) -> String {
178    content
179        .iter()
180        .filter_map(|block| match block {
181            ContentBlock::Reasoning { text, .. } => Some(text.as_str()),
182            _ => None,
183        })
184        .collect::<Vec<_>>()
185        .join(separator)
186}

188/// Extracts displayable text from structured reasoning details.
189pub(crate) fn reasoning_text_from_details(details: &[Value]) -> Option<String> {
190    let parts = details
191        .iter()
192        .filter_map(Value::as_object)
193        .filter_map(|detail| {
194            detail
195                .get("text")
196                .and_then(Value::as_str)
197                .filter(|text| !text.is_empty())
198                .or_else(|| {
199                    detail
200                        .get("summary")
201                        .and_then(Value::as_str)
202                        .filter(|summary| !summary.is_empty())
203                })
204        })
205        .collect::<Vec<_>>();
206    (!parts.is_empty()).then(|| parts.join("\n"))
207}

209/// Collects reasoning text from a Responses reasoning item's `content` or `summary`
210/// array, or from a bare string, into `out`. Empty strings are skipped.
211pub(crate) fn collect_responses_reasoning_text(value: Option<&Value>, out: &mut Vec<String>) {
212    match value {
213        Some(Value::String(text)) if !text.is_empty() => out.push(text.clone()),
214        Some(Value::Array(items)) => {
215            for item in items {
216                match item {
217                    Value::String(text) if !text.is_empty() => out.push(text.clone()),
218                    Value::Object(object) => {
219                        if matches!(
220                            object.get("type").and_then(Value::as_str),
221                            Some("reasoning_text" | "summary_text" | "text")
222                        ) && let Some(text) = object.get("text").and_then(Value::as_str)
223                            && !text.is_empty()
224                        {
225                            out.push(text.to_string());
226                        }
227                    }
228                    _ => {}
229                }
230            }
231        }
232        _ => {}
233    }
234}

236/// Returns the opaque payload of the first encrypted reasoning detail, if any.
237///
238/// Two detail shapes are accepted: the documented `{"type": "reasoning.encrypted", "data"}`
239/// object, and a verbatim Responses `reasoning` item carrying `encrypted_content` (the shape
240/// the buffered request decoder stores when it keeps the provider item whole).
241pub(crate) fn encrypted_reasoning_data(details: &[Value]) -> Option<String> {
242    details
243        .iter()
244        .filter_map(Value::as_object)
245        .find_map(|detail| match detail.get("type").and_then(Value::as_str) {
246            Some("reasoning.encrypted") => detail.get("data").and_then(Value::as_str),
247            Some("reasoning") => detail.get("encrypted_content").and_then(Value::as_str),
248            _ => None,
249        })
250        .filter(|data| !data.is_empty())
251        .map(ToOwned::to_owned)
252}

254/// Returns the provider item id recorded on the first `reasoning.encrypted` detail, if any.
255/// Encrypted reasoning is bound to the item id it was issued under, so a replay must reuse it.
256pub(crate) fn encrypted_reasoning_item_id(details: &[Value]) -> Option<String> {
257    details
258        .iter()
259        .filter_map(Value::as_object)
260        .find(|detail| {
261            matches!(
262                detail.get("type").and_then(Value::as_str),
263                Some("reasoning.encrypted" | "reasoning")
264            )
265        })
266        .and_then(|detail| detail.get("id").and_then(Value::as_str))
267        .filter(|id| !id.is_empty())
268        .map(ToOwned::to_owned)
269}

271/// Returns the first non-empty string stored under the requested keys.
272pub(crate) fn first_nonempty_string<'a>(
273    object: &'a Map<String, Value>,
274    keys: &[&str],
275) -> Option<&'a str> {
276    keys.iter().find_map(|key| {
277        object
278            .get(*key)
279            .and_then(Value::as_str)
280            .filter(|value| !value.is_empty())
281    })
282}

284/// Copies unknown provider fields into the IR extension map.
285pub(crate) fn provider_extensions(
286    object: &Map<String, Value>,
287    known: &[&str],
288) -> Map<String, Value> {
289    let mut extensions = Map::new();
290    for (key, value) in object {
291        if key != ANTHROPIC_REQUEST_KEY && !known.contains(&key.as_str()) {
292            extensions.insert(key.clone(), value.clone());
293        }
294    }
295    extensions
296}

298/// Invalid charges remain unknown so accounting can use configured token rates.
299pub(crate) fn provider_cost(value: &Map<String, Value>) -> Option<f64> {
300    value
301        .get("cost")
302        .and_then(Value::as_f64)
303        .filter(|cost| cost.is_finite() && *cost >= 0.0)
304}

306/// Cache duration must survive decoding because one-hour writes can have a different rate.
307pub(crate) fn cache_usage_details(
308    value: &Map<String, Value>,
309    cached: Option<u64>,
310    created: Option<u64>,
311) -> Option<Box<crate::llm::InputCacheUsage>> {
312    let one_hour = value
313        .get("cache_creation")
314        .and_then(|details| details.get("ephemeral_1h_input_tokens"))
315        .and_then(Value::as_u64);
316    let mut cache = crate::llm::Usage::cache_details(cached, created);
317    if let Some(tokens) = one_hour {
318        cache
319            .get_or_insert_with(Default::default)
320            .cache_creation_1h_input_tokens = Some(tokens);
321    }
322    cache
323}

325/// These fields preserve provider charges and cache duration across response formats.
326pub(crate) fn encode_usage_metadata(mut value: Value, usage: &crate::llm::Usage) -> Value {
327    if let Some(cost) = usage
328        .provider_cost
329        .filter(|cost| cost.is_finite() && *cost >= 0.0)
330    {
331        value["cost"] = serde_json::json!(cost);
332    }
333    if let Some(tokens) = usage
334        .cache
335        .as_ref()
336        .and_then(|cache| cache.cache_creation_1h_input_tokens)
337    {
338        value["cache_creation"] = serde_json::json!({"ephemeral_1h_input_tokens": tokens});
339    }
340    value
341}

343#[cfg(test)]
344mod tests {
345    use super::*;
346    use serde_json::json;
347
348    #[test]
349    fn encrypted_reasoning_helpers_accept_both_detail_shapes() {
350        let documented = vec![json!({"type": "reasoning.encrypted", "data": "blob", "id": "rs_1"})];
351        assert_eq!(
352            encrypted_reasoning_data(&documented).as_deref(),
353            Some("blob")
354        );
355        assert_eq!(
356            encrypted_reasoning_item_id(&documented).as_deref(),
357            Some("rs_1")
358        );
359        let verbatim_item = vec![json!({
360            "type": "reasoning", "id": "rs_2", "summary": [], "encrypted_content": "blob2"
361        })];
362        assert_eq!(
363            encrypted_reasoning_data(&verbatim_item).as_deref(),
364            Some("blob2")
365        );
366        assert_eq!(
367            encrypted_reasoning_item_id(&verbatim_item).as_deref(),
368            Some("rs_2")
369        );
370        assert_eq!(encrypted_reasoning_data(&[json!({"type": "other"})]), None);
371    }
