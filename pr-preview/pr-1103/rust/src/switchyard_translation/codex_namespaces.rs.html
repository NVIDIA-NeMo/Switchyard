4//! Codex tool namespace preservation across a Responses/Chat translation.
5//!
6//! Codex groups tools into non-standard ``namespace`` containers — MCP servers,
7//! usually behind an `mcp__` prefix, plus builtin groups such as
8//! `multi_agent_v1` — and dispatches on the pair of tool name and namespace. It
9//! therefore expects the namespace back on the call it receives:
10//! `{"type": "function_call", "name": "search", "namespace": "mcp__docs"}`.
11//!
12//! OpenAI-compatible upstreams accept only flat `function` tools, so the request
13//! codec flattens the containers and names each child `<namespace>__<tool>`.
14//! Two tools differing only by namespace therefore stay distinct upstream, and
15//! the conversation history uses the same spelling so the transcript never
16//! teaches the model a name the upstream was not offered.
17//!
18//! The qualified-name to namespace mapping rides in the request's
19//! [`ProviderExtensions`], so no provider-neutral type grows a Codex-specific
20//! field. No codec copies unknown extension keys into an outbound body, so the
21//! mapping reaches neither the upstream nor the client.
22//!
23//! No container is filtered by its name, because Codex resolves a call with no
24//! namespace against its default group: dropping a builtin group's namespace
25//! would look the call up in the wrong place.
26
27use std::collections::{HashMap, HashSet};

29use serde_json::{Map, Value};
30use switchyard_protocol::ProviderExtensions;

32pub use switchyard_protocol::codex_namespaces::{
33    NAMESPACE_SEPARATOR, TOOL_NAMESPACES_KEY, split_qualified_name, tool_namespaces,
34};

36/// Joins a namespace and a tool name into the name used on the wire.
37pub fn qualified_tool_name(namespace: &str, tool: &str) -> String {
38    format!("{namespace}{NAMESPACE_SEPARATOR}{tool}")
39}
40
41/// Records that `qualified` was produced by flattening a tool out of `namespace`.
42pub fn record_tool_namespace(
43    namespaces: &mut Map<String, Value>,
44    qualified: &str,
45    namespace: &str,
46    description: Option<&str>,
47) {
48    let value = description.map_or_else(
49        || Value::String(namespace.to_string()),
50        |description| {
51            serde_json::json!({
52                "namespace": namespace,
53                "description": description,
54            })
55        },
56    );
57    namespaces.insert(qualified.to_string(), value);
58}
59
60/// Stores a collected mapping on a request's extensions, when it has entries.
61pub fn attach_tool_namespaces(extensions: &mut ProviderExtensions, namespaces: Map<String, Value>) {
62    if !namespaces.is_empty() {
63        extensions
64            .fields
65            .insert(TOOL_NAMESPACES_KEY.to_string(), Value::Object(namespaces));
66    }
67}

69/// Returns a retained description for `namespace`.
70pub fn namespace_description<'a>(
71    namespaces: &'a Map<String, Value>,
72    namespace: &str,
73) -> Option<&'a str> {
74    namespaces.values().find_map(|value| {
75        let object = value.as_object()?;
76        (object.get("namespace").and_then(Value::as_str) == Some(namespace))
77            .then(|| object.get("description").and_then(Value::as_str))
78            .flatten()
79    })
80}

82/// Reverse map from an upstream tool name to its Codex tool name and namespace.
83///
84/// The exact qualified name is always registered. A model often returns a near
85/// miss instead, so two fallback spellings are registered when neither can be
86/// confused with another tool:
87///
88/// * the qualified name without the `mcp__` prefix
89/// * the bare tool name, only when exactly one namespace claims it
90pub fn qualified_tool_origins(
91    extensions: &ProviderExtensions,
92) -> HashMap<String, (String, String)> {
93    let Some(namespaces) = tool_namespaces(extensions) else {
94        return HashMap::new();
95    };
96    let split: Vec<(String, String, String)> = namespaces
97        .keys()
98        .filter_map(|qualified| {
99            let (tool, namespace) = split_qualified_name(namespaces, qualified)?;
100            Some((qualified.clone(), tool.to_string(), namespace.to_string()))
101        })
102        .collect();
103
104    // A bare name claimed by more than one namespace cannot be guessed, and a
105    // fallback must never shadow a name that is itself a real qualified tool.
106    let mut bare_claims: HashMap<&str, usize> = HashMap::new();
107    for (_, tool, _) in &split {
108        *bare_claims.entry(tool.as_str()).or_default() += 1;
109    }
110    let qualified_names: HashSet<&str> = split
111        .iter()
112        .map(|(qualified, _, _)| qualified.as_str())
113        .collect();
114
115    let mut origins = HashMap::new();
116    for (qualified, tool, namespace) in &split {
117        let origin = (tool.clone(), namespace.clone());
118        if let Some(stripped) = namespace.strip_prefix("mcp__") {
119            let spelling = qualified_tool_name(stripped, tool);
120            if !qualified_names.contains(spelling.as_str()) {
121                origins.entry(spelling).or_insert_with(|| origin.clone());
122            }
123        }
124        if bare_claims.get(tool.as_str()) == Some(&1) && !qualified_names.contains(tool.as_str()) {
125            origins
126                .entry(tool.clone())
127                .or_insert_with(|| origin.clone());
128        }
129        // The exact spelling always wins over a fallback.
130        origins.insert(qualified.clone(), origin);
131    }
132    origins
133}

135/// Restore the Codex tool name and namespace on function calls and argument completions.
136///
137/// Walks the whole value, covering `function_call` items in buffered responses
138/// and streaming events, plus the top-level name in
139/// `response.function_call_arguments.done`. Preserves an existing `namespace`.
140pub fn restore_qualified_tool_names(body: &mut Value, origins: &HashMap<String, (String, String)>) {
141    if origins.is_empty() {
142        return;
143    }
144    match body {
145        Value::Array(values) => {
146            for value in values {
147                restore_qualified_tool_names(value, origins);
148            }
149        }
150        Value::Object(object) => {
151            if matches!(
152                object.get("type").and_then(Value::as_str),
153                Some("function_call" | "response.function_call_arguments.done")
154            ) && let Some(name) = object.get("name").and_then(Value::as_str)
155                && let Some((tool, namespace)) = origins.get(name)
156            {
157                object.insert("name".to_string(), Value::String(tool.clone()));
158                object
159                    .entry("namespace".to_string())
160                    .or_insert_with(|| Value::String(namespace.clone()));
161            }
162            for value in object.values_mut() {
163                restore_qualified_tool_names(value, origins);
164            }
165        }
166        _ => {}
167    }
168}

170#[cfg(test)]
171mod tests {
172    use serde_json::{Map, json};
173    use switchyard_protocol::ProviderExtensions;
174
175    use super::{
176        attach_tool_namespaces, qualified_tool_name, qualified_tool_origins, record_tool_namespace,
177        restore_qualified_tool_names, split_qualified_name, tool_namespaces,
178    };
179
180    fn extensions(pairs: &[(&str, &str)]) -> ProviderExtensions {
181        let mut namespaces = Map::new();
182        for (namespace, tool) in pairs {
183            record_tool_namespace(
184                &mut namespaces,
185                &qualified_tool_name(namespace, tool),
186                namespace,
187                None,
188            );
189        }
190        let mut extensions = ProviderExtensions::default();
191        attach_tool_namespaces(&mut extensions, namespaces);
192        extensions
193    }
194
195    // A tool name may itself contain the separator, so the namespace is matched
196    // as a prefix rather than by splitting on the first `__`.
197    #[test]
198    fn splits_a_qualified_name_back_apart() {
199        let simple = extensions(&[("mcp__docs", "search")]);
200        let simple = tool_namespaces(&simple).expect("mapping present");
201        assert_eq!(
202            split_qualified_name(simple, "mcp__docs__search"),
203            Some(("search", "mcp__docs"))
204        );
205        assert_eq!(split_qualified_name(simple, "unknown_tool"), None);
206
207        let nested = extensions(&[("mcp__docs", "fetch__raw")]);
208        let nested = tool_namespaces(&nested).expect("mapping present");
209        assert_eq!(
210            split_qualified_name(nested, "mcp__docs__fetch__raw"),
211            Some(("fetch__raw", "mcp__docs"))
212        );
213    }
214
215    // A bare name claimed by two namespaces must not be guessed: a wrong guess
216    // dispatches the call to the wrong server.
217    #[test]
218    fn leaves_an_ambiguous_bare_name_alone() {
219        let origins =
220            qualified_tool_origins(&extensions(&[("mcp__a", "search"), ("mcp__b", "search")]));
221        let mut response = json!({"output": [{"type": "function_call", "name": "search"}]});
222        let before = response.clone();
223
224        restore_qualified_tool_names(&mut response, &origins);
225
226        assert_eq!(response, before);
227    }
228
229    // Codex namespaces builtin groups too, so nothing may key on `mcp__`.
230    #[test]
231    fn resolves_namespaces_that_are_not_mcp_servers() {
232        let origins = qualified_tool_origins(&extensions(&[("multi_agent_v1", "spawn_agent")]));
233        let mut response = json!({
234            "output": [{"type": "function_call", "name": "multi_agent_v1__spawn_agent"}]
235        });
236
237        restore_qualified_tool_names(&mut response, &origins);
238
239        assert_eq!(response["output"][0]["name"], "spawn_agent");
240        assert_eq!(response["output"][0]["namespace"], "multi_agent_v1");
241    }
