140    pub(crate) started: bool,
141    pub(crate) content_index: Option<usize>,
142    /// Position in the OpenAI Chat `tool_calls` array, assigned when ENCODING. Chat numbers
143    /// tool calls on their own, while Anthropic and Responses index the whole content array,
144    /// so the source index cannot be reused.
145    pub(crate) chat_tool_index: Option<usize>,
146    pub(crate) response_output_index: Option<usize>,
147    pub(crate) response_item_id: Option<String>,

167/// Codec contract for one provider streaming event format.
168pub trait StreamCodec: Send + Sync {
169    /// Returns the stream format handled by this codec.
170    fn format(&self) -> FormatId;
171
172    /// Decodes one provider event into zero or more neutral events.
173    fn decode_event(
174        &self,
175        state: &mut StreamTranslationState,
176        event: &Value,
177    ) -> Vec<LlmResponseChunk>;
178
179    /// Encodes one neutral event into zero or more provider events.
180    fn encode_event(
181        &self,
182        state: &mut StreamTranslationState,
183        event: LlmResponseChunk,
184    ) -> Vec<Value>;
185
186    /// Advances encoder state after an exact same-format event replay.
187    ///
188    /// Exact replay returns the preserved provider JSON instead of the JSON emitted by
189    /// [`Self::encode_event`]. The encoder must nevertheless observe the normalized chunks so
190    /// [`Self::finish`] can close an incomplete stream without duplicating an already replayed
191    /// terminal event. Codecs whose terminal state cannot be inferred from `MessageStop` alone
192    /// may override this hook and inspect `raw`.
193    fn observe_replayed_event(
194        &self,
195        state: &mut StreamTranslationState,
196        _raw: &Value,
197        normalized: Vec<LlmResponseChunk>,
198    ) {
199        let replayed_terminal = normalized
200            .iter()
201            .any(|chunk| matches!(chunk, LlmResponseChunk::MessageStop { .. }));
202        for chunk in normalized {
203            drop(self.encode_event(state, chunk));
204        }
205        if replayed_terminal {
206            state.finished = true;
207        }
208    }
209
210    /// Emits any terminal provider events needed after the source stream ends.
211    ///
212    /// This is intentionally required on every codec. Some target formats
213    /// need explicit terminal events after the source closes (for example,
214    /// Anthropic ``message_delta``/``message_stop`` or Responses
215    /// ``response.completed``). Formats that have no source-close work should
216    /// return an empty vector explicitly so the no-op behavior is a conscious
217    /// codec-level choice.
218    fn finish(&self, state: &mut StreamTranslationState) -> Vec<Value>;
219}

221/// Registry mapping stream wire formats to stream codecs.
222#[derive(Default)]
223pub struct StreamCodecRegistry {
224    codecs: BTreeMap<FormatId, Arc<dyn StreamCodec>>,
225}
226
227impl StreamCodecRegistry {
228    /// Creates an empty stream codec registry.
229    pub fn new() -> Self {
230        Self::default()
231    }
232
233    /// Creates a registry populated with built-in stream codecs.
234    pub fn with_builtins() -> Self {
235        let mut registry = Self::new();
236        registry.register(OpenAiChatStreamCodec);
237        registry.register(AnthropicMessagesStreamCodec);
238        registry.register(OpenAiResponsesStreamCodec);
239        registry
240    }
241
242    /// Registers or replaces a stream codec for its declared format.
243    pub fn register(&mut self, codec: impl StreamCodec + 'static) {
244        self.codecs.insert(codec.format(), Arc::new(codec));
245    }
246
247    /// Looks up a stream codec by format identifier.
248    pub fn codec(&self, format: impl Into<FormatId>) -> Result<Arc<dyn StreamCodec>> {
249        let format = format.into();
250        self.codecs.get(&format).cloned().ok_or_else(|| {
251            TranslationError::Other(format!("no stream codec registered for {format}"))
252        })
253    }
254}
255
256impl StreamTranslationEngine {
257    /// Creates a streaming engine from an explicit codec registry.
258    pub fn new(registry: StreamCodecRegistry) -> Self {
259        Self {
260            engine: TranslationEngine::with_registries(FormatRegistry::with_builtins(), registry),
261        }
262    }
263
264    /// Translates one source provider event into target provider events.
265    pub fn translate_event(
266        &self,
267        state: &mut StreamTranslationState,
268        source: impl Into<FormatId>,
269        target: impl Into<FormatId>,
270        event: &Value,
271    ) -> Result<Vec<Value>> {
272        self.engine.translate_event(state, source, target, event)
273    }
274
275    /// Finishes target-provider stream emission after the source stream closes.
276    pub fn finish(
277        &self,
278        state: &mut StreamTranslationState,
279        target: impl Into<FormatId>,
280    ) -> Result<Vec<Value>> {
281        self.engine.finish_stream(state, target)
282    }
283
284    /// Convenience helper using built-in codecs and error events instead of `Result`.
285    pub fn translate_event_with_builtins(
286        state: &mut StreamTranslationState,
287        source: impl Into<FormatId>,
288        target: impl Into<FormatId>,
289        event: &Value,
290    ) -> Vec<Value> {
291        Self::default()
292            .translate_event(state, source, target, event)
293            .unwrap_or_else(|error| vec![json!({"error": {"message": error.to_string()}})])
294    }
295}

297/// Decodes one provider stream event with the built-in codec registry.
298pub fn decode_stream_event(
299    state: &mut StreamTranslationState,
300    source: impl Into<FormatId>,
301    event: &Value,
302) -> Vec<LlmResponseChunk> {
303    let source = source.into();
304    StreamCodecRegistry::with_builtins()
305        .codec(source)
306        .map(|codec| codec.decode_event(state, event))
307        .unwrap_or_else(|error| {
308            // A missing codec is a translation-side failure, not something the
309            // upstream sent, so it decodes to `DecodeError` rather than `StreamError`.
310            vec![LlmResponseChunk::DecodeError {
311                message: error.to_string(),
312            }]
313        })
314}

316pub(crate) fn encode_response_stream_event(
317    state: &mut StreamTranslationState,
318    target_codec: &dyn StreamCodec,
319    target: &FormatId,
320    event: crate::LlmResponseStreamEvent,
321) -> Vec<Value> {
322    if state.errored {
323        return Vec::new();
324    }
325    let (preservation, normalized) = event.into_parts();
326    if let Some(preservation) = preservation {
327        let (source, raw) = preservation.into_parts();
328        if let Err(error) = super::responses::validate_stream_output(&source, target, &raw) {
329            return target_codec.encode_event(
330                state,
331                LlmResponseChunk::DecodeError {
332                    message: error.to_string(),
333                },
334            );
335        }
336        // Invalid protocol data must become an error frame, not be replayed as ordinary data.
337        let has_decode_error = normalized
338            .iter()
339            .any(|chunk| matches!(chunk, LlmResponseChunk::DecodeError { .. }));
340        if &source == target && !has_decode_error {
341            // Exact replay bypasses the target encoder's emitted JSON, but the encoder must
342            // still observe every normalized chunk. Otherwise `finish` starts from empty state:
343            // a clean EOF after a nonterminal provider event can omit or synthesize malformed
344            // terminal events, while a replayed terminal can be emitted twice.
345            target_codec.observe_replayed_event(state, &raw, normalized);
346            return vec![raw];
347        }
348    }
349
350    normalized
351        .into_iter()
352        .flat_map(|chunk| target_codec.encode_event(state, chunk))
353        .collect()
354}

356/// Encodes one neutral stream event with the built-in codec registry.
357pub fn encode_stream_event(
358    state: &mut StreamTranslationState,
359    target: impl Into<FormatId>,
360    event: LlmResponseChunk,
361) -> Vec<Value> {
362    StreamCodecRegistry::with_builtins()
363        .codec(target)
364        .map(|codec| codec.encode_event(state, event))
365        .unwrap_or_else(|error| vec![json!({"error": {"message": error.to_string()}})])
366}

368// Records source-provider identity carried by decoded stream events.
369pub(crate) fn record_source_identity(
370    state: &mut StreamTranslationState,
371    id: Option<String>,
372    model: Option<String>,
373) {
374    if id.is_some() {
375        state.message_id = id;
376    }
377    if model.is_some() {
378        state.model = model;
379    }
380}

382// Returns the served model when supplied, otherwise the upstream model.
383pub(crate) fn target_model_or_source_model(state: &StreamTranslationState) -> String {
384    state
385        .target_model
386        .clone()
387        .or_else(|| state.model.clone())
388        .unwrap_or_else(|| "unknown".to_string())
389}

391// Returns the target/client ID when supplied, otherwise the upstream ID.
392pub(crate) fn target_message_id_or_source_message_id(
393    state: &StreamTranslationState,
394) -> Option<&str> {
395    state
396        .target_message_id
397        .as_deref()
398        .or(state.message_id.as_deref())
399}

401// Checks whether the current source format matches a built-in format.
402pub(crate) fn state_source_is(state: &StreamTranslationState, format: WireFormat) -> bool {
403    let format_id: FormatId = format.into();
404    match &state.source {
405        Some(source) => source == &format_id,
406        None => false,
407    }
408}

410// Reads a non-empty string field from an event object.
411pub(crate) fn string_field(object: &Map<String, Value>, key: &str) -> Option<String> {
412    object
413        .get(key)
414        .and_then(Value::as_str)
415        .filter(|value| !value.is_empty())
416        .map(ToOwned::to_owned)
