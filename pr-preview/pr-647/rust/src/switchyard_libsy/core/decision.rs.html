4//! Optional metadata explaining a routing decision.
5
6/// Metadata attached by libsy to a successful routing outcome.
7#[derive(Clone, Debug, PartialEq, Eq)]
8pub struct DecisionMetadata {
9    /// Unique identifier for this invocation of
10    /// [`Algorithm::run_stream`](crate::Algorithm::run_stream).
11    pub decision_id: String,
12    /// Stable name of the algorithm that made the decision.
13    pub algorithm: String,
14    /// Optional bounded, machine-readable evidence produced by the algorithm.
15    pub evidence: Option<String>,
16}
