20/// Status reported for an upstream error delivered inside a streaming body. The
21/// upstream already sent a success status line before failing, so there is no real
22/// code to propagate; 502 matches how a failed upstream call surfaces elsewhere.
23const MID_STREAM_UPSTREAM_STATUS: u16 = 502;
24
25/// A boxed, `Send` stream of [`LlmResponseChunk`]s — the token-by-token output of a
26/// streaming backend. Each item may fail independently mid-stream.
27pub type LlmResponseStream =
28    Pin<Box<dyn Stream<Item = Result<LlmResponseChunk, LlmClientError>> + Send>>;
29
30/// A model response: either a live [`Stream`](LlmResponse::Stream) of chunks or the
31/// terminal buffered [`Agg`](LlmResponse::Agg)regate.
32///
33/// Not `Clone` — the `Stream` variant owns a single-consumption stream. A buffered
34/// backend returns `Agg` directly; a streaming one returns `Stream` and the consumer
35/// drives it, folding to an [`AggLlmResponse`] when it needs the whole response.
36pub enum LlmResponse {
37    /// Live, single-consumption response stream.
38    Stream(LlmResponseStream),
39    /// Fully buffered terminal response.
40    Agg(AggLlmResponse),
41}
42
43impl LlmResponse {
44    /// Borrow the aggregate; `None` while this is still a stream.
45    pub fn as_agg(&self) -> Option<&AggLlmResponse> {
46        match self {
47            LlmResponse::Agg(agg) => Some(agg),
48            LlmResponse::Stream(_) => None,
49        }
50    }
51
52    /// Reduce to the buffered aggregate: return an `Agg` unchanged, or drive a `Stream`
53    /// to completion, folding its chunks into an [`AggLlmResponse`] via
54    /// [`ResponseAccumulator`]. A stream item error aborts with `Err`, as does an
55    /// in-band [`LlmResponseChunk::DecodeError`] (as `ResponseTranslation`) or
56    /// [`LlmResponseChunk::StreamError`] (as `UpstreamHttp`).
57    pub async fn into_agg(self) -> Result<AggLlmResponse, LlmClientError> {
58        match self {
59            LlmResponse::Agg(agg) => Ok(agg),
60            LlmResponse::Stream(mut stream) => {
61                let mut accumulator = ResponseAccumulator::new();
62                while let Some(item) = stream.next().await {
63                    match item? {
64                        LlmResponseChunk::DecodeError { message } => {
65                            return Err(LlmClientError::ResponseTranslation(message));
66                        }
67                        LlmResponseChunk::StreamError { message } => {
68                            // The upstream reported the failure inside the response body, so
69                            // there is no real status line to carry; 502 stands in for "the
70                            // upstream failed" the same way a failed non-streaming call would.
71                            return Err(LlmClientError::UpstreamHttp {
72                                status: MID_STREAM_UPSTREAM_STATUS,
73                                body: message,
74                            });
75                        }
76                        chunk => accumulator.push(chunk),
77                    }
78                }
79                Ok(accumulator.finish())
80            }
81        }
82    }
83
84    /// Returns the model recorded by a buffered response.
85    ///
86    /// Returns `None` for a live stream because reading its model-bearing
87    /// [`LlmResponseChunk::MessageStart`] event would consume the stream.
88    pub fn selected_model(&self) -> Option<&str> {
89        match self {
90            LlmResponse::Agg(agg) => agg.model.as_deref(),
91            // TODO: How do we get the model name on a stream?
92            LlmResponse::Stream(_) => None,
93        }
94    }
95}

97impl AggLlmResponse {
98    /// Converts a fully-buffered response into a synthetic chunk stream.
99    ///
100    /// Useful when a caller had to buffer the response (e.g. to judge it) but the
101    /// downstream expects `LlmResponse::Stream` — for instance when `stream: true` was
102    /// requested and the algorithm had to aggregate before it could return.
103    ///
104    /// This conversion is lossy: only text, reasoning, and tool-call content has
105    /// a synthetic chunk representation. Refusals, tool results, media, files,
106    /// unknown blocks, response extensions, and preservation metadata are omitted.
107    pub fn into_stream(self) -> LlmResponseStream {
108        let mut chunks: Vec<LlmResponseChunk> = Vec::new();
109        chunks.push(LlmResponseChunk::MessageStart {
110            id: self.id,
111            model: self.model,
112        });
113        let mut tool_call_index = 0usize;
114        for (output_index, output) in self.outputs.into_iter().enumerate() {
115            for block in output.content {
116                match block {
117                    ContentBlock::Text { text } => {
118                        chunks.push(LlmResponseChunk::TextDelta {
119                            index: output_index,
120                            text,
121                        });
122                    }
123                    ContentBlock::Reasoning { text, .. } => {
124                        chunks.push(LlmResponseChunk::ReasoningDelta {
125                            index: output_index,
126                            text,
127                        });
128                    }
129                    ContentBlock::ToolCall(tool) => {
130                        let args = serde_json::to_string(&tool.arguments).unwrap_or_default();
131                        chunks.push(LlmResponseChunk::ToolCallDelta {
132                            index: tool_call_index,
133                            id: Some(tool.id),
134                            name: Some(tool.name),
135                            arguments_delta: Some(args),
136                        });
137                        tool_call_index += 1;
138                    }
139                    // Other variants (Image, ToolResult, Refusal, etc.) don't have
140                    // a streaming chunk representation and don't appear in assistant outputs.
141                    _ => {}
142                }
143            }
144            chunks.push(LlmResponseChunk::MessageStop {
145                reason: output.stop_reason.and_then(|r| {
146                    serde_json::to_value(r)
147                        .ok()
148                        .and_then(|v| v.as_str().map(String::from))
149                }),
150            });
151        }
152        chunks.push(LlmResponseChunk::Usage(self.usage));
153        Box::pin(futures::stream::iter(chunks.into_iter().map(Ok)))
154    }
155}

157/// One provider-neutral streaming event — the normalized counterpart to
158/// [`AggLlmResponse`], sitting between stream decoders and encoders.
159#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
160pub enum LlmResponseChunk {
161    /// Starts a response message.
162    MessageStart {
163        /// Provider response identifier.
164        id: Option<String>,
165        /// Model reported by the provider.
166        model: Option<String>,
167    },
168    /// Adds text to one output index.
169    TextDelta {
170        /// Provider output index.
171        index: usize,
172        /// Text fragment.
173        text: String,
174    },
175    /// Adds reasoning text to one output index.
176    ReasoningDelta {
177        /// Provider output index.
178        index: usize,
179        /// Reasoning fragment.
180        text: String,
181    },
182    /// Adds or updates a tool call at one index.
183    ToolCallDelta {
184        /// Tool-call index within the response.
185        index: usize,
186        /// Tool-call identifier, normally supplied by the first delta.
187        id: Option<String>,
188        /// Tool name, normally supplied by the first delta.
189        name: Option<String>,
190        /// Fragment of the serialized tool arguments.
191        arguments_delta: Option<String>,
192    },
193    /// Reports token usage, normally near the end of the stream.
194    Usage(Usage),
195    /// Ends a response message.
196    MessageStop {
197        /// Provider stop-reason string before normalization.
198        reason: Option<String>,
199    },
200    /// Reports that an inbound stream event could not be decoded.
201    DecodeError {
202        /// Human-readable decoding failure.
203        message: String,
204    },
205    /// Reports an upstream failure delivered inside an otherwise successful stream.
206    StreamError {
207        /// Human-readable upstream failure.
208        message: String,
209    },
210}

212/// Folds a sequence of [`LlmResponseChunk`]s into the terminal [`AggLlmResponse`].
213///
214/// Text and reasoning deltas concatenate; tool-call deltas assemble by index (name,
215/// id, and a growing arguments string parsed as JSON at the end); `MessageStart`,
216/// `Usage`, and `MessageStop` set the corresponding fields. `Error` chunks are
217/// ignored here — a driver consuming the stream is expected to surface them.
218///
219/// Folding is lossy for multiple outputs: text and reasoning indices are ignored,
220/// and [`finish`](Self::finish) produces one assistant output. Prefer
221/// [`LlmResponse::into_agg`] when stream errors must be surfaced.
222///
223/// Drive it by `push`-ing each chunk in order, then call [`finish`](Self::finish).
224#[derive(Default)]
225pub struct ResponseAccumulator {
226    id: Option<String>,
227    model: Option<String>,
228    text: String,
229    reasoning: Option<String>,
230    tool_calls: BTreeMap<usize, PartialToolCall>,
231    usage: Usage,
232    stop_reason: Option<StopReason>,
233}
234
235/// A tool call being assembled from streamed [`LlmResponseChunk::ToolCallDelta`]s.
236#[derive(Default)]
237struct PartialToolCall {
238    id: Option<String>,
239    name: Option<String>,
240    arguments: String,
241}
242
243impl ResponseAccumulator {
244    /// A fresh accumulator with no chunks applied.
245    pub fn new() -> Self {
246        Self::default()
247    }
248
249    /// Apply one chunk. Later `MessageStart`/`Usage`/`MessageStop` fields overwrite
250    /// earlier ones; text, reasoning, and tool-call arguments append.
251    pub fn push(&mut self, chunk: LlmResponseChunk) {
252        match chunk {
253            LlmResponseChunk::MessageStart { id, model } => {
254                if id.is_some() {
255                    self.id = id;
256                }
257                if model.is_some() {
258                    self.model = model;
259                }
260            }
261            LlmResponseChunk::TextDelta { text, .. } => self.text.push_str(&text),
262            LlmResponseChunk::ReasoningDelta { text, .. } => {
263                self.reasoning
264                    .get_or_insert_with(String::new)
265                    .push_str(&text);
266            }
267            LlmResponseChunk::ToolCallDelta {
268                index,
269                id,
270                name,
271                arguments_delta,
272            } => {
273                let call = self.tool_calls.entry(index).or_default();
274                if id.is_some() {
275                    call.id = id;
276                }
277                if name.is_some() {
278                    call.name = name;
279                }
280                if let Some(delta) = arguments_delta {
281                    call.arguments.push_str(&delta);
282                }
283            }
284            LlmResponseChunk::Usage(usage) => self.usage = usage,
285            LlmResponseChunk::MessageStop { reason } => {
286                self.stop_reason = Some(stop_reason_from_str(reason.as_deref()));
287            }
288            LlmResponseChunk::DecodeError { .. } | LlmResponseChunk::StreamError { .. } => {}
289        }
290    }
291
292    /// Build the buffered response. Content is ordered reasoning, then text, then
293    /// tool calls (by ascending delta index) — a single assistant output.
294    pub fn finish(self) -> AggLlmResponse {
295        let mut content = Vec::new();
296        if let Some(reasoning) = self.reasoning {
297            content.push(ContentBlock::Reasoning {
298                text: reasoning,
299                signature: None,
300            });
301        }
302        if !self.text.is_empty() {
303            content.push(ContentBlock::Text { text: self.text });
304        }
305        for call in self.tool_calls.into_values() {
306            content.push(ContentBlock::ToolCall(ToolCall {
307                id: call.id.unwrap_or_default(),
308                name: call.name.unwrap_or_default(),
309                arguments: parse_tool_arguments(&call.arguments),
310            }));
311        }
312        AggLlmResponse {
313            id: self.id,
314            model: self.model,
315            outputs: vec![ResponseOutput {
316                role: Role::Assistant,
317                content,
318                stop_reason: self.stop_reason,
319            }],
320            usage: self.usage,
321            ..AggLlmResponse::default()
322        }
323    }
324}

326/// Parse an assembled tool-call arguments string as JSON, falling back to a JSON
327/// string when it is not valid JSON and to an empty object when it is empty.
328fn parse_tool_arguments(arguments: &str) -> Value {
329    if arguments.is_empty() {
330        return Value::Object(serde_json::Map::new());
331    }
332    serde_json::from_str(arguments).unwrap_or_else(|_| Value::String(arguments.to_string()))
333}
334
335/// Map a provider stop-reason string (as carried by [`LlmResponseChunk::MessageStop`])
336/// to a normalized [`StopReason`], covering the common OpenAI and Anthropic spellings.
337fn stop_reason_from_str(reason: Option<&str>) -> StopReason {
338    match reason {
339        Some("length" | "max_tokens") => StopReason::MaxTokens,
340        Some("tool_calls" | "function_call" | "tool_use") => StopReason::ToolUse,
341        Some("content_filter") => StopReason::ContentFilter,
342        Some("stop" | "end_turn" | "stop_sequence") | None => StopReason::EndTurn,
343        Some(_) => StopReason::Unknown,
344    }
345}

355    fn fold(chunks: Vec<LlmResponseChunk>) -> AggLlmResponse {
356        let mut accumulator = ResponseAccumulator::new();
357        for chunk in chunks {
358            accumulator.push(chunk);
359        }
360        accumulator.finish()
361    }

363    #[test]
364    fn folds_text_usage_and_stop_reason() {
365        let agg = fold(vec![
366            LlmResponseChunk::MessageStart {
367                id: Some("id1".to_string()),
368                model: Some("m".to_string()),
369            },
370            LlmResponseChunk::TextDelta {
371                index: 0,
372                text: "Hel".to_string(),
373            },
374            LlmResponseChunk::TextDelta {
375                index: 0,
376                text: "lo".to_string(),
377            },
378            LlmResponseChunk::Usage(Usage {
379                output_tokens: Some(2),
380                ..Usage::default()
381            }),
382            LlmResponseChunk::MessageStop {
383                reason: Some("length".to_string()),
384            },
385        ]);
386        assert_eq!(agg.id.as_deref(), Some("id1"));
387        assert_eq!(agg.model.as_deref(), Some("m"));
388        assert_eq!(agg.usage.output_tokens, Some(2));
389        assert_eq!(agg.outputs[0].stop_reason, Some(StopReason::MaxTokens));
390        assert_eq!(
391            agg.outputs[0].content,
392            vec![ContentBlock::Text {
393                text: "Hello".to_string()
394            }]
395        );
396    }

455    #[test]
456    fn into_stream_round_trips_through_into_agg() {
457        let original = AggLlmResponse {
458            id: Some("id1".to_string()),
459            model: Some("m".to_string()),
460            outputs: vec![ResponseOutput {
461                role: Role::Assistant,
462                content: vec![ContentBlock::Text {
463                    text: "hello".to_string(),
464                }],
465                stop_reason: Some(StopReason::EndTurn),
466            }],
467            usage: Usage {
468                output_tokens: Some(3),
469                ..Usage::default()
470            },
471            ..AggLlmResponse::default()
472        };
473        let stream = LlmResponse::Stream(original.clone().into_stream());
474        let recovered = block_on(stream.into_agg()).expect("into_agg failed");
475        assert_eq!(recovered.id, original.id);
476        assert_eq!(recovered.model, original.model);
477        assert_eq!(recovered.usage.output_tokens, original.usage.output_tokens);
478        assert_eq!(
479            recovered.outputs[0].stop_reason,
480            original.outputs[0].stop_reason
481        );
482        assert_eq!(recovered.outputs[0].content, original.outputs[0].content);
483    }
