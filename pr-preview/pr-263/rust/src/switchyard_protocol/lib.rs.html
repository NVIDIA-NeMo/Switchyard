6//! Provider-neutral request, response, streaming, routing, and metadata types shared by
7//! Switchyard algorithms, clients, and translation codecs. This crate defines contracts;
8//! it does not route requests, translate wire payloads, or perform network calls.
9//!
10//! ## Setup
11//!
12//! ```toml
13//! [dependencies]
14//! switchyard-protocol = { git = "https://github.com/NVIDIA-NeMo/Switchyard.git" }
15//! serde_json = "1"
16//! ```
17//!
18//! ## Simple request
19//!
20//! ```
21//! use switchyard_protocol::{prompt_text, text_request};
22//!
23//! let request = text_request(Some("provider/model".into()), "Explain tail latency");
24//!
25//! assert_eq!(request.model.as_deref(), Some("provider/model"));
26//! assert_eq!(prompt_text(&request), "Explain tail latency");
27//! ```
28//!
29//! ## Detailed request
30//!
31//! Use [`Request`] when routing also needs correlation metadata, and construct
32//! [`LlmRequest`] directly for instructions, tools, and generation controls:
33//!
34//! ```
35//! use serde_json::json;
36//! use switchyard_protocol::{
37//!     ContentBlock, InstructionBlock, LlmRequest, Message, Metadata, OutputParams,
38//!     Request, Role, SamplingParams, ToolChoice, ToolDefinition,
39//! };
40//!
41//! let request = Request {
42//!     llm_request: LlmRequest {
43//!         model: Some("provider/model".into()),
44//!         instructions: vec![InstructionBlock {
45//!             role: Role::System,
46//!             content: vec![ContentBlock::Text {
47//!                 text: "Answer with concise operational guidance.".into(),
48//!             }],
49//!         }],
50//!         messages: vec![Message::text(Role::User, "Why is p99 latency rising?")],
51//!         tools: vec![ToolDefinition {
52//!             name: "lookup_metric".into(),
53//!             description: Some("Read one service metric".into()),
54//!             parameters: json!({
55//!                 "type": "object",
56//!                 "properties": { "name": { "type": "string" } },
57//!                 "required": ["name"]
58//!             }),
59//!             strict: Some(true),
60//!         }],
61//!         tool_choice: Some(ToolChoice::Auto),
62//!         sampling: SamplingParams {
63//!             temperature: Some(0.2),
64//!             ..SamplingParams::default()
65//!         },
66//!         output: OutputParams {
67//!             max_output_tokens: Some(512),
68//!             ..OutputParams::default()
69//!         },
70//!         stream: true,
71//!         ..LlmRequest::default()
72//!     },
73//!     metadata: Some(Metadata {
74//!         session_id: Some("session-42".into()),
75//!         correlation_id: Some("request-7".into()),
76//!         ..Metadata::default()
77//!     }),
78//!     ..Request::default()
79//! };
80//!
81//! assert_eq!(request.llm_request.tools[0].name, "lookup_metric");
82//! assert_eq!(
83//!     request
84//!         .metadata
85//!         .as_ref()
86//!         .and_then(|metadata| metadata.session_id.as_deref()),
87//!     Some("session-42")
88//! );
89//! ```
90//!
91//! [`LlmResponse`] represents either a buffered [`AggLlmResponse`] or a
92//! single-consumption stream of [`LlmResponseChunk`]s. See each type's documentation for
93//! aggregation and preservation behavior.

109/// Builds a single-turn request: one user message carrying `prompt`, for `model`.
110///
111/// This deliberately creates only the common text shape. Construct [`LlmRequest`]
112/// directly for instructions, tools, multimodal content, or sampling controls.
113pub fn text_request(model: Option<String>, prompt: impl Into<String>) -> LlmRequest {
114    LlmRequest {
115        model,
116        messages: vec![Message::text(Role::User, prompt)],
117        ..LlmRequest::default()
118    }
119}
120
121/// Returns a lossy text view of all user messages, joined by newlines.
122///
123/// Only text and refusal blocks are included. Instructions, tool content,
124/// reasoning, and media are omitted. Returns an empty string when no user text exists.
125pub fn prompt_text(request: &LlmRequest) -> String {
126    request
127        .messages
128        .iter()
129        .filter(|message| message.role == Role::User)
130        .filter_map(|message| message.text_content("\n"))
131        .collect::<Vec<_>>()
132        .join("\n")
133}

135/// Builds a single-turn response: one assistant message carrying `completion`, for `model`.
136///
137/// Construct [`AggLlmResponse`] directly when usage, tools, reasoning, or multiple
138/// output items must be represented.
139pub fn text_response(model: Option<String>, completion: impl Into<String>) -> AggLlmResponse {
140    AggLlmResponse {
141        model,
142        outputs: vec![ResponseOutput {
143            role: Role::Assistant,
144            content: vec![ContentBlock::Text {
145                text: completion.into(),
146            }],
147            stop_reason: None,
148        }],
149        ..AggLlmResponse::default()
150    }
151}
152
153/// Returns a lossy text view of the first assistant output.
154///
155/// Only text blocks from the first output are concatenated. Refusals, reasoning,
156/// tools, media, and additional outputs are omitted. Returns an empty string when
157/// no such text exists.
158pub fn completion_text(response: &AggLlmResponse) -> String {
159    response
160        .outputs
161        .first()
162        .map(|output| {
163            output
164                .content
165                .iter()
166                .filter_map(|block| match block {
167                    ContentBlock::Text { text } => Some(text.as_str()),
168                    _ => None,
169                })
170                .collect::<String>()
171        })
172        .unwrap_or_default()
173}

175#[cfg(test)]
176mod tests {
177    use super::*;
178
179    #[test]
180    fn request_round_trips_prompt_text() {
181        let req = text_request(Some("m".to_string()), "hello world");
182        assert_eq!(req.model.as_deref(), Some("m"));
183        assert_eq!(prompt_text(&req), "hello world");
184    }
185
186    #[test]
187    fn response_round_trips_completion_text() {
188        let resp = text_response(None, "the answer");
189        assert_eq!(completion_text(&resp), "the answer");
190    }
191
192    #[test]
193    fn empty_text_helpers_are_empty_strings() {
194        assert_eq!(prompt_text(&LlmRequest::default()), "");
195        assert_eq!(completion_text(&AggLlmResponse::default()), "");
196    }
