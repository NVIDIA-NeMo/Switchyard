13/// Inclusive prompt tokens: non-cached input plus both cache buckets, the
14/// same fold the routing log uses, so advisor and executor rows reconcile.
15pub(super) fn inclusive_prompt_tokens(usage: &Usage) -> u64 {
16    usage
17        .input_tokens
18        .unwrap_or(0)
19        .saturating_add(usage.cached_input_tokens().unwrap_or(0))
20        .saturating_add(usage.cache_creation_input_tokens().unwrap_or(0))
21}

23pub(super) fn record_review(verdict: &'static str, trigger: &'static str) {
24    observability::meter()
25        .u64_counter("switchyard.advisor_gate.reviews")
26        .build()
27        .add(
28            1,
29            &[
30                KeyValue::new("verdict", verdict),
31                KeyValue::new("trigger", trigger),
32            ],
33        );
34}
35
36pub(super) fn record_consult_failure(reason: &'static str) {
37    observability::meter()
38        .u64_counter("switchyard.advisor_gate.consult_failures")
39        .build()
40        .add(1, &[KeyValue::new("reason", reason)]);
41}

43/// Counts a REDO-discarded executor turn and its tokens; the client never
44/// sees the turn, so the host's terminal usage accounting never prices it.
45pub(super) fn record_discarded(usage: &Usage) {
46    let meter = observability::meter();
47    meter
48        .u64_counter("switchyard.advisor_gate.discarded_turns")
49        .build()
50        .add(1, &[]);
51    let tokens = meter
52        .u64_counter("switchyard.advisor_gate.discarded_tokens")
53        .build();
54    for (kind, value) in [
55        ("input", usage.input_tokens.unwrap_or(0)),
56        ("cached", usage.cached_input_tokens().unwrap_or(0)),
57        (
58            "cache_creation",
59            usage.cache_creation_input_tokens().unwrap_or(0),
60        ),
61        ("output", usage.output_tokens.unwrap_or(0)),
62    ] {
63        if value > 0 {
64            tokens.add(value, &[KeyValue::new("kind", kind)]);
65        }
66    }
67}

69/// One review consult's audit payload.
70pub(super) struct ReviewAudit<'a> {
71    pub(super) verdict: &'static str,
72    pub(super) error: Option<String>,
73    pub(super) latency_ms: f64,
74    pub(super) reply_head: Option<String>,
75    pub(super) usage: Option<&'a Usage>,
76}
77
78/// Emits the one-line sorted-key JSON audit record benchmark tooling greps
79/// for (`advisor_review=`).
80pub(super) fn emit_review_audit(audit: ReviewAudit<'_>) {
81    let mut payload = serde_json::Map::new();
82    payload.insert("advisor_review".to_string(), true.into());
83    payload.insert(
84        "latency_ms".to_string(),
85        ((audit.latency_ms * 10.0).round() / 10.0).into(),
86    );
87    payload.insert("verdict".to_string(), audit.verdict.into());
88    if let Some(error) = audit.error {
89        payload.insert("error".to_string(), error.into());
90    }
91    if let Some(head) = audit.reply_head
92        && !head.is_empty()
93    {
94        payload.insert("reply_head".to_string(), head.into());
95    }
96    if let Some(usage) = audit.usage {
97        payload.insert(
98            "prompt_tokens".to_string(),
99            inclusive_prompt_tokens(usage).into(),
100        );
101        payload.insert(
102            "completion_tokens".to_string(),
103            usage.output_tokens.unwrap_or(0).into(),
104        );
105        let cached = usage.cached_input_tokens().unwrap_or(0);
106        if cached > 0 {
107            payload.insert("cached_tokens".to_string(), cached.into());
108        }
109        let creation = usage.cache_creation_input_tokens().unwrap_or(0);
110        if creation > 0 {
111            payload.insert("cache_creation_tokens".to_string(), creation.into());
112        }
113    }
114    tracing::info!(
115        target: "libsy",
116        "advisor_review={}",
117        serde_json::Value::Object(payload)
118    );
119}

121/// Emits the discarded-turn audit record (`advisor_discarded=`), the gate's
122/// own accounting for a turn no host-side observer can price.
123pub(super) fn emit_discarded_audit(model: &str, usage: &Usage) {
124    let mut payload = serde_json::Map::new();
125    payload.insert("advisor_discarded".to_string(), true.into());
126    payload.insert("model".to_string(), model.into());
127    payload.insert(
128        "prompt_tokens".to_string(),
129        inclusive_prompt_tokens(usage).into(),
130    );
131    payload.insert(
132        "cached_tokens".to_string(),
133        usage.cached_input_tokens().unwrap_or(0).into(),
134    );
135    payload.insert(
136        "cache_creation_tokens".to_string(),
137        usage.cache_creation_input_tokens().unwrap_or(0).into(),
138    );
139    payload.insert(
140        "completion_tokens".to_string(),
141        usage.output_tokens.unwrap_or(0).into(),
142    );
143    tracing::info!(
144        target: "libsy",
145        "advisor_discarded={}",
146        serde_json::Value::Object(payload)
147    );
148}
