4//! Hardening helpers shared by classifier-style inner-LLM callers.
5//!
6//! Consulting a model to make a routing decision fails in ways the routed call does
7//! not, and those failures are logged on a path that runs for every turn. The
8//! natural `Display` of an error is not safe there: an upstream error body is echoed
9//! verbatim, and both bodies and decode failures can embed the conversation or the
10//! model's reply. [`safe_error_summary`] is the redaction those log sites use.
11
12use switchyard_protocol::LlmClientError;
13
14use crate::LibsyError;
15
16/// A loggable description of `error` that carries no conversation or response content.
17///
18/// Each case contributes only vetted operational detail: the failure's class, the
19/// target or model it concerns, and a status code where one exists. Anything sourced
20/// from outside the process is reduced to its class.
21///
22/// The match is deliberately exhaustive rather than defaulting to `to_string()`, so a
23/// new [`LibsyError`] variant cannot start leaking content by omission.
24pub(crate) fn safe_error_summary(error: &LibsyError) -> String {
25    match error {
26        LibsyError::ClientCall { target, source } => {
27            format!(
28                "client call to target {target:?} failed: {}",
29                safe_client_error(source)
30            )
31        }
32        // `message` is a `String` with no constraint on its contents. Callers build it
33        // from static text today, but nothing stops one embedding a response or an
34        // upstream error, so only the class is reported.
35        LibsyError::AlgorithmError { .. } => "algorithm error".to_string(),
36        LibsyError::Driver(_) => "algorithm driver failed".to_string(),
37        // The operation is a static label, but the boxed source comes from a user
38        // extension and is unvetted.
39        LibsyError::External { operation, .. } => format!("{operation} failed"),
40        // The remaining variants render only structural detail: configured target
41        // names and fixed strings, nothing request-derived.
42        LibsyError::TargetNotFound { .. }
43        | LibsyError::NoTargets
44        | LibsyError::MissingFinalResponse => error.to_string(),
45    }
46}
47
48/// The client half of [`safe_error_summary`].
49///
50/// `UpstreamHttp` is the sharp edge: its `Display` interpolates the raw upstream
51/// body, which routinely quotes the request back. Boxed transport, decode, and FFI
52/// sources are reduced for the same reason.
53pub(crate) fn safe_client_error(error: &LlmClientError) -> String {
54    match error {
55        LlmClientError::UpstreamHttp { status, .. } => format!("upstream HTTP {status}"),
56        LlmClientError::ContextWindowExceeded { model, .. } => {
57            format!("context window exceeded for model {model}")
58        }
59        LlmClientError::Timeout { .. } => "upstream request timed out".to_string(),
60        LlmClientError::Transport { .. } => "upstream transport error".to_string(),
61        LlmClientError::InvalidResponse { .. } => "invalid upstream response".to_string(),
62        LlmClientError::Ffi { .. } => "foreign function interface error".to_string(),
63        LlmClientError::InvalidRequest { .. } => "invalid request".to_string(),
64        LlmClientError::RequestTranslation(_) => "request translation failed".to_string(),
65        LlmClientError::RequestEncoding(_) => "outbound request encoding failed".to_string(),
66        LlmClientError::ResponseTranslation(_) => "response translation failed".to_string(),
67        // `message` is unconstrained here too, so the class is reported without it.
68        LlmClientError::Configuration { .. } => "client configuration error".to_string(),
69        // Free-form by definition.
70        LlmClientError::General(_) => "client call failed".to_string(),
71        // `LlmClientError` is `#[non_exhaustive]`, so a wildcard is required. A future
72        // variant is unvetted until it is given an arm above.
73        _ => "client call failed".to_string(),
74    }
75}

77#[cfg(test)]
78mod tests {
79    use switchyard_protocol::ModelId;
80
81    use super::*;
82
83    /// The conversation text a leaking log line would expose.
84    const SECRET: &str = "patient name is Jane Doe";
85
86    #[test]
87    fn an_upstream_body_never_reaches_the_summary() {
88        let error = LibsyError::ClientCall {
89            target: ModelId::new("weak"),
90            source: LlmClientError::UpstreamHttp {
91                status: http::StatusCode::BAD_REQUEST,
92                body: format!(r#"{{"error":{{"message":"bad request: {SECRET}"}}}}"#),
93            },
94        };
95
96        let summary = safe_error_summary(&error);
97
98        // The status and target survive; the body does not.
99        assert!(summary.contains("upstream HTTP 400"), "{summary}");
100        assert!(summary.contains("weak"), "{summary}");
101        assert!(!summary.contains(SECRET), "{summary}");
102        // Guard against the redaction silently regressing to `Display`.
103        assert!(error.to_string().contains(SECRET));
104    }
105
106    #[test]
107    fn boxed_sources_are_reduced_to_their_class() {
108        for source in [
109            LlmClientError::Transport {
110                source: std::io::Error::other(SECRET).into(),
111            },
112            LlmClientError::InvalidResponse {
113                source: std::io::Error::other(SECRET).into(),
114            },
115            LlmClientError::Timeout {
116                source: std::io::Error::other(SECRET).into(),
117            },
118        ] {
119            let summary = safe_client_error(&source);
120            assert!(!summary.contains(SECRET), "{summary}");
121            assert!(!summary.is_empty());
122        }
123    }
124
125    #[test]
126    fn context_overflow_keeps_the_model_but_not_the_message() {
127        let summary = safe_client_error(&LlmClientError::ContextWindowExceeded {
128            model: ModelId::new("weak"),
129            message: SECRET.to_string(),
130        });
131        assert!(summary.contains("weak"), "{summary}");
132        assert!(!summary.contains(SECRET), "{summary}");
133    }
134
135    #[test]
136    fn an_algorithm_error_message_is_reduced_to_its_class() {
137        // `AlgorithmError::message` is an unconstrained `String`. Callers build it from
138        // static text today, but the summary must not depend on that holding.
139        let summary = safe_error_summary(&LibsyError::AlgorithmError {
140            message: format!("judge reply did not parse: {SECRET}"),
141        });
142        assert_eq!(summary, "algorithm error");
143        assert!(!summary.contains(SECRET), "{summary}");
144    }
145
146    #[test]
147    fn a_client_configuration_message_is_reduced_to_its_class() {
148        let summary = safe_client_error(&LlmClientError::Configuration {
149            message: SECRET.to_string(),
150        });
151        assert_eq!(summary, "client configuration error");
152    }
153
154    #[test]
155    fn a_general_error_is_reduced_to_its_class() {
156        let summary = safe_client_error(&LlmClientError::General(SECRET.to_string()));
157        assert!(!summary.contains(SECRET), "{summary}");
158    }
159
160    #[test]
161    fn an_extension_failure_keeps_its_label_but_not_its_source() {
162        let error = LibsyError::external("loading extension", std::io::Error::other(SECRET));
163
164        let summary = safe_error_summary(&error);
165
166        assert_eq!(summary, "loading extension failed");
167        assert!(error.to_string().contains(SECRET));
168    }
169
170    #[test]
171    fn structural_variants_keep_their_detail() {
172        let summary = safe_error_summary(&LibsyError::TargetNotFound {
173            target: ModelId::new("strong"),
174        });
175        assert!(summary.contains("strong"), "{summary}");
176    }
177}
