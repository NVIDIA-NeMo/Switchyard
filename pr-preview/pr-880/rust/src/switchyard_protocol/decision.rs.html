4//! Provider-neutral decision questions and answers, separate from LLM messages.
5//!
6//! Enums use snake-case `type` tags and a `data` payload in serialized form.
7//! Numeric bounds are checked during deserialization. Public fields allow direct
8//! construction; callers are responsible for valid values in that case.
9
10use std::collections::BTreeMap;
11
12use serde::{Deserialize, Deserializer, Serialize, de};
13use serde_json::Value;
14
15use crate::{ModelId, Usage};
16
17/// Answer probability on a `[0, 1]` scale.
18#[derive(Clone, Copy, Debug, PartialEq, Serialize, Deserialize)]
19#[serde(transparent)]
20pub struct Probability(#[serde(deserialize_with = "deserialize_probability")] pub f64);
21
22/// Position in the request's rubric, including fractional positions.
23#[derive(Clone, Copy, Debug, PartialEq, Serialize, Deserialize)]
24#[serde(transparent)]
25pub struct ScoreValue(#[serde(deserialize_with = "deserialize_score")] pub f64);
26
27/// Provider confidence; its scale and meaning are provider-specific.
28#[derive(Clone, Copy, Debug, PartialEq, Serialize, Deserialize)]
29#[serde(transparent)]
30pub struct ProviderConfidence(#[serde(deserialize_with = "deserialize_confidence")] pub f64);
31
32fn deserialize_probability<'de, D: Deserializer<'de>>(deserializer: D) -> Result<f64, D::Error> {
33    let value = f64::deserialize(deserializer)?;
34    if !(0.0..=1.0).contains(&value) {
35        return Err(de::Error::custom(
36            "probability must be finite and in [0, 1]",
37        ));
38    }
39    Ok(value)
40}
41
42fn deserialize_score<'de, D: Deserializer<'de>>(deserializer: D) -> Result<f64, D::Error> {
43    let value = f64::deserialize(deserializer)?;
44    if !value.is_finite() || value < 0.0 {
45        return Err(de::Error::custom("score must be finite and nonnegative"));
46    }
47    Ok(value)
48}
49
50fn deserialize_confidence<'de, D: Deserializer<'de>>(deserializer: D) -> Result<f64, D::Error> {
51    let value = f64::deserialize(deserializer)?;
52    if !value.is_finite() {
53        return Err(de::Error::custom("provider confidence must be finite"));
54    }
55    Ok(value)
56}

58/// Shared context evaluated against independent, named questions.
59#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
60pub struct DecisionRequest {
61    /// Optional until a target is selected.
62    pub model: Option<ModelId>,
63    /// Conversation, application state, or other material to evaluate.
64    pub context: Value,
65    /// Independent questions keyed by their IDs.
66    pub questions: BTreeMap<String, DecisionQuestion>,
67}
68
69/// Instructions and the expected answer shape for one question.
70#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
71pub struct DecisionQuestion {
72    /// Structured or textual instructions shared with the provider.
73    pub instructions: Value,
74    /// Expected answer shape.
75    pub kind: DecisionKind,
76}
77
78/// The answer shape and any options or ordered rubric levels.
79#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
80#[serde(tag = "type", content = "data", rename_all = "snake_case")]
81pub enum DecisionKind {
82    /// A Boolean judgment or probability of true.
83    Boolean {
84        /// Meaning of a true answer, when needed.
85        true_description: Option<Value>,
86        /// Meaning of a false answer, when needed.
87        false_description: Option<Value>,
88    },
89    /// Select one of the declared options.
90    Choice {
91        /// Nonempty options with unique IDs, preserving caller order.
92        options: Vec<ChoiceOption>,
93    },
94    /// A position on an ordered rubric, not an arbitrary numeric measurement.
95    Score {
96        /// At least two levels, ordered low to high and indexed from zero.
97        levels: Vec<Value>,
98    },
99}
100
101/// An identified choice with an optional structured description.
102#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
103pub struct ChoiceOption {
104    /// Stable identifier used by choice answers and distributions.
105    pub id: String,
106    /// Meaning of this option, when its ID alone is insufficient.
107    pub description: Option<Value>,
108}
109
110/// Answers keyed by the matching request's question IDs.
111#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
112pub struct DecisionResponse {
113    /// Provider-reported response identifier.
114    pub id: Option<String>,
115    /// Provider-reported model identifier.
116    pub model: Option<ModelId>,
117    /// Typed answers corresponding to the request's questions.
118    pub answers: BTreeMap<String, DecisionAnswer>,
119    /// Available token counts; absent counts remain unknown.
120    #[serde(default)]
121    pub usage: Usage,
122}
123
124/// An answer and separate, optional provider confidence.
125#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
126pub struct DecisionAnswer {
127    /// The estimate for the matching question.
128    pub value: DecisionValue,
129    /// Provider-specific confidence, distinct from answer probabilities.
130    pub provider_confidence: Option<ProviderConfidence>,
131}
132
133/// A typed estimate; missing distributions remain unknown.
134#[derive(Clone, Debug, PartialEq, Serialize, Deserialize)]
135#[serde(tag = "type", content = "data", rename_all = "snake_case")]
136pub enum DecisionValue {
137    /// A Boolean judgment or probability, without an implicit threshold.
138    Boolean(BooleanEstimate),
139    /// One selected option with an optional complete distribution.
140    Choice {
141        /// Must name an option in the matching question.
142        selected: String,
143        /// Maps every declared option ID to its probability when available.
144        probabilities: Option<BTreeMap<String, Probability>>,
145    },
146    /// A fractional position in the matching request's rubric.
147    Score {
148        /// Must lie in `0..=N-1` for the request's N levels.
149        value: ScoreValue,
150        /// Follows the request's level order. Retain that request to interpret it.
151        probabilities: Option<Vec<Probability>>,
152    },
153}
154
155/// Preserves Boolean-only answers without inventing probability or certainty.
156#[derive(Clone, Copy, Debug, PartialEq, Serialize, Deserialize)]
157#[serde(tag = "type", content = "data", rename_all = "snake_case")]
158pub enum BooleanEstimate {
159    /// A Boolean judgment with no probability supplied.
160    Value(bool),
161    /// Probability of true; algorithms choose their own thresholds.
162    ProbabilityTrue(Probability),
163}
