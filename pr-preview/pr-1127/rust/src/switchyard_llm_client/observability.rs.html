18/// Records request parameters represented directly by the neutral IR.
19pub(crate) fn record_gen_ai_request(span: &Span, request: &LlmRequest) {
20    if request.stream {
21        span.record("gen_ai.request.stream", true);
22    }
23    if let Some(value) = request.sampling.temperature {
24        span.record("gen_ai.request.temperature", value);
25    }
26    if let Some(value) = request.sampling.top_p {
27        span.record("gen_ai.request.top_p", value);
28    }
29    if let Some(value) = request.sampling.top_k {
30        span.record("gen_ai.request.top_k", value);
31    }
32    if let Some(value) = request.output.max_output_tokens {
33        span.record("gen_ai.request.max_tokens", otel_int(value));
34    }
35    if let Some(value) = request.reasoning.effort.as_deref() {
36        span.record("gen_ai.request.reasoning.level", value);
37    }
38    if let Some(value) = request
39        .output
40        .response_format
41        .as_ref()
42        .and_then(gen_ai_output_type)
43    {
44        span.record("gen_ai.output.type", value);
45    }
46}

48/// Adds terminal response and usage fields to the enclosing `libsy.client_call`
49/// span without consuming or buffering a streaming response.
50pub(crate) fn observe_client_call(result: Result<Response>) -> Result<Response> {
51    let span = Span::current();
52    match result {
53        Ok(mut response) => {
54            match response.llm_response {
55                LlmResponse::Agg(agg) => {
56                    span.record("outcome", "ok");
57                    record_gen_ai_response(&span, &agg);
58                    response.llm_response = LlmResponse::Agg(agg);
59                }
60                LlmResponse::Stream(stream) => {
61                    response.llm_response =
62                        LlmResponse::Stream(observe_client_stream(stream, span));
63                }
64            }
65            Ok(response)
66        }
67        Err(error) => {
68            let error_type = client_call_error_type(&error);
69            match &error {
70                LibsyError::ClientCall {
71                    target,
72                    source: LlmClientError::UpstreamHttp { status, .. },
73                } => record_client_error(
74                    &span,
75                    &error_type,
76                    &format_args!(
77                        "client call to target {target:?} failed: upstream HTTP {status}"
78                    ),
79                ),
80                LibsyError::ClientCall {
81                    source: LlmClientError::Host { .. },
82                    ..
83                } => record_host_error(&span),
84                _ => record_client_error(&span, &error_type, &error),
85            }
86            Err(error)
87        }
88    }
89}

91fn observe_client_stream(stream: LlmResponseStream, span: Span) -> LlmResponseStream {
92    Box::pin(ObservedClientStream {
93        stream,
94        observer: Some(ClientStreamObserver {
95            span,
96            outcome: Outcome::Open,
97        }),
98    })
99}

101fn record_gen_ai_response(span: &Span, response: &AggLlmResponse) {
102    record_optional(span, "gen_ai.response.id", response.id.as_deref());
103    record_optional(span, "gen_ai.response.model", response.model.as_deref());
104    record_finish_reasons(
105        span,
106        response
107            .outputs
108            .iter()
109            .filter_map(|output| output.stop_reason)
110            .map(stop_reason_name),
111    );
112    record_gen_ai_usage(span, &response.usage);
113}

115fn record_gen_ai_usage(span: &Span, usage: &Usage) {
116    let cache_read = usage.cached_input_tokens();
117    let cache_creation = usage.cache_creation_input_tokens();
118    if usage.input_tokens.is_some() || cache_read.is_some() || cache_creation.is_some() {
119        let input_tokens = usage
120            .input_tokens
121            .unwrap_or_default()
122            .saturating_add(cache_read.unwrap_or_default())
123            .saturating_add(cache_creation.unwrap_or_default());
124        span.record("gen_ai.usage.input_tokens", otel_int(input_tokens));
125    }
126    for (field, value) in [
127        ("gen_ai.usage.output_tokens", usage.output_tokens),
128        ("gen_ai.usage.cache_read.input_tokens", cache_read),
129        ("gen_ai.usage.cache_creation.input_tokens", cache_creation),
130        (
131            "gen_ai.usage.reasoning.output_tokens",
132            usage.reasoning_tokens,
133        ),
134    ] {
135        if let Some(value) = value {
136            span.record(field, otel_int(value));
137        }
138    }
139}

141// OpenTelemetry integer attributes are signed; token counts are unsigned in the IR.
142fn otel_int(value: u64) -> i64 {
143    value.min(i64::MAX as u64) as i64
144}
145
146fn record_optional(span: &Span, field: &str, value: Option<&str>) {
147    if let Some(value) = value {
148        span.record(field, value);
149    }
150}

152// `tracing` fields cannot preserve a typed string array, so write this attribute directly.
153fn record_finish_reasons(span: &Span, reasons: impl IntoIterator<Item = impl Into<String>>) {
154    let reasons = reasons
155        .into_iter()
156        .map(|reason| StringValue::from(reason.into()))
157        .collect::<Vec<_>>();
158    if !reasons.is_empty() {
159        span.set_attribute(
160            "gen_ai.response.finish_reasons",
161            OtelValue::Array(OtelArray::String(reasons)),
162        );
163    }
164}

166fn stop_reason_name(reason: StopReason) -> &'static str {
167    match reason {
168        StopReason::EndTurn => "end_turn",
169        StopReason::MaxTokens => "max_tokens",
170        StopReason::ToolUse => "tool_use",
171        StopReason::ContentFilter => "content_filter",
172        StopReason::Error => "error",
173        StopReason::Unknown => "unknown",
174    }
175}

177fn gen_ai_output_type(response_format: &serde_json::Value) -> Option<&'static str> {
178    match response_format
179        .get("type")
180        .and_then(serde_json::Value::as_str)
181    {
182        Some("json" | "json_object" | "json_schema") => Some("json"),
183        Some("text") => Some("text"),
184        _ => None,
185    }
186}

188fn client_call_error_type(error: &LibsyError) -> Cow<'static, str> {
189    match error {
190        LibsyError::ClientCall { source, .. } => llm_client_error_type(source),
191        LibsyError::TargetNotFound { .. } => Cow::Borrowed("target_not_found"),
192        LibsyError::NoTargets => Cow::Borrowed("no_targets"),
193        LibsyError::AlgorithmError { .. } => Cow::Borrowed("algorithm_error"),
194        LibsyError::Driver(_) => Cow::Borrowed("driver_error"),
195        LibsyError::MissingFinalResponse => Cow::Borrowed("missing_final_response"),
196        LibsyError::External { .. } => Cow::Borrowed("_OTHER"),
197    }
198}

200fn llm_client_error_type(error: &LlmClientError) -> Cow<'static, str> {
201    match error {
202        LlmClientError::InvalidRequest { .. } => Cow::Borrowed("invalid_request"),
203        LlmClientError::RequestTranslation(_) => Cow::Borrowed("request_translation"),
204        LlmClientError::RequestEncoding(_) => Cow::Borrowed("request_encoding"),
205        LlmClientError::ResponseTranslation(_) => Cow::Borrowed("response_translation"),
206        LlmClientError::Configuration { .. } => Cow::Borrowed("configuration"),
207        LlmClientError::Transport { .. } => Cow::Borrowed("transport"),
208        LlmClientError::TemporarilyUnavailable => Cow::Borrowed("temporarily_unavailable"),
209        LlmClientError::Timeout { .. } => Cow::Borrowed("timeout"),
210        LlmClientError::ContextWindowExceeded { .. } => Cow::Borrowed("context_window_exceeded"),
211        LlmClientError::UpstreamHttp { status, .. } => Cow::Owned(status.as_str().to_owned()),
212        LlmClientError::InvalidResponse { .. } => Cow::Borrowed("invalid_response"),
213        LlmClientError::Host { .. } => Cow::Borrowed("host"),
214        LlmClientError::Ffi { .. } => Cow::Borrowed("ffi"),
215        _ => Cow::Borrowed("_OTHER"),
216    }
217}

219// Keep the client span alive until the response is drained, errors, or is abandoned.
220struct ObservedClientStream {
221    stream: LlmResponseStream,
222    observer: Option<ClientStreamObserver>,
223}
224
225impl Stream for ObservedClientStream {
226    type Item = std::result::Result<LlmResponseStreamEvent, LlmClientError>;
227
228    fn poll_next(mut self: Pin<&mut Self>, cx: &mut TaskContext<'_>) -> Poll<Option<Self::Item>> {
229        match self.stream.as_mut().poll_next(cx) {
230            Poll::Ready(Some(item)) => {
231                let failed = self
232                    .observer
233                    .as_mut()
234                    .is_some_and(|observer| observer.observe(&item));
235                if failed {
236                    self.observer.take();
237                }
238                Poll::Ready(Some(item))
239            }
240            Poll::Ready(None) => {
241                if let Some(mut observer) = self.observer.take() {
242                    observer.complete();
243                }
244                Poll::Ready(None)
245            }
246            Poll::Pending => Poll::Pending,
247        }
248    }
249}

251/// What the span has recorded for this stream so far.
252#[derive(PartialEq)]
253enum Outcome {
254    /// Still streaming. Dropping the response now is a cancellation.
255    Open,
256    /// The model finished its message, so `outcome` is already `ok` and dropping the
257    /// response is no longer a cancellation. Observation continues: OpenAI Chat sends usage
258    /// in a chunk *after* the one carrying `finish_reason`, so stopping here would lose it.
259    Completed,
260    /// An error is recorded and terminal; nothing later is worth observing.
261    Failed,
262}
263
264struct ClientStreamObserver {
265    span: Span,
266    outcome: Outcome,
267}
268
269impl ClientStreamObserver {
270    /// Records what `item` says about the response. Returns whether the stream has failed —
271    /// the only state in which the caller should stop observing.
272    fn observe(
273        &mut self,
274        item: &std::result::Result<LlmResponseStreamEvent, LlmClientError>,
275    ) -> bool {
276        match item {
277            Ok(event) => {
278                for chunk in event.normalized() {
279                    self.observe_chunk(chunk);
280                    if self.outcome == Outcome::Failed {
281                        break;
282                    }
283                }
284            }
285            Err(error) => {
286                let error_type = llm_client_error_type(error);
287                match error {
288                    LlmClientError::Host { .. } => record_host_error(&self.span),
289                    _ => record_client_error(&self.span, &error_type, error),
290                }
291                self.outcome = Outcome::Failed;
292            }
293        }
294        self.outcome == Outcome::Failed
295    }
296
297    fn observe_chunk(&mut self, chunk: &LlmResponseChunk) {
298        match chunk {
299            LlmResponseChunk::MessageStart { id, model } => {
300                record_optional(&self.span, "gen_ai.response.id", id.as_deref());
301                record_optional(&self.span, "gen_ai.response.model", model.as_deref());
302            }
303            LlmResponseChunk::Usage(usage) => record_gen_ai_usage(&self.span, usage),
304            LlmResponseChunk::MessageStop { reason } => {
305                record_finish_reasons(&self.span, reason.iter().cloned());
306                self.complete();
307            }
308            LlmResponseChunk::DecodeError { message } => {
309                record_client_error(&self.span, "response_translation", message);
310                self.outcome = Outcome::Failed;
311            }
312            LlmResponseChunk::StreamError { message } => {
313                record_client_error(&self.span, "502", message);
314                self.outcome = Outcome::Failed;
315            }
316            _ => {}
317        }
318    }
319
320    /// Records the normal terminal outcome, unless an error already claimed it.
321    fn complete(&mut self) {
322        if self.outcome == Outcome::Open {
323            self.span.record("outcome", "ok");
324            self.outcome = Outcome::Completed;
325        }
326    }
327}
328
329impl Drop for ClientStreamObserver {
330    fn drop(&mut self) {
331        if self.outcome == Outcome::Open {
332            self.span.record("outcome", "cancelled");
333        }
334    }
335}

337// Host error messages may contain private request data.
338fn record_host_error(span: &Span) {
339    record_client_error(span, "host", &format_args!("host client error"));
340}
341
342fn record_client_error(span: &Span, error_type: &str, error: &dyn std::fmt::Display) {
343    span.record("outcome", "error");
344    span.record("otel.status_code", "ERROR");
345    span.record("error.type", error_type);
346    span.record("error", tracing::field::display(error));
347}
