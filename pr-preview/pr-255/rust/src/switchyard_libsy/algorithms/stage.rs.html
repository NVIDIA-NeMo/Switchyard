4//! Signal-driven stage routing for coding agents.
5//!
6//! [`StageRouter`] is the assembled algorithm: a [`FallThrough`] pre-wired with
7//! the tool-signal processor that reads each turn's tool results and the
8//! [`StageClassifier`] that scores them onto the capable/efficient tiers. The cascade
9//! is an internal detail — callers drive the algorithm, not its parts.
10//!
11//! Signals do not decide every turn. An under-threshold turn abstains and falls
12//! through to the optional [`LlmTaskClassifier`] — the capability route's judge,
13//! joined in unchanged — and then to the picker's default tier. The judge is
14//! asked per turn and its verdict is never pinned to the session.
15//!

34/// Telemetry name for a router this module assembles.
35const STAGE_ROUTER: &str = "stage_router";

37/// Attributes a turn to the classifier it wraps, when that classifier decides it.
38///
39/// The classifiers themselves are composition-agnostic and write no state; only
40/// this router knows where each sits in its cascade.
41struct SourceStamp {
42    inner: Arc<dyn Classifier<State>>,
43    source: DecisionSource,
44}
45
46#[async_trait]
47impl Classifier<State> for SourceStamp {
48    fn routing_tier(&self, selected_model: &str) -> Option<&'static str> {
49        self.inner.routing_tier(selected_model)
50    }
51
52    async fn score(
53        &self,
54        state: &mut State,
55        request: &mut Request,
56        driver: Option<&Driver>,
57    ) -> Result<(Classification, Option<Response>)> {
58        let (classification, served) = self.inner.score(state, request, driver).await?;
59        // An abstaining classifier passes the turn on, so it is not its to claim.
60        if matches!(&classification, Classification::Scores(scores) if !scores.is_empty()) {
61            record_decision_source(state, self.source);
62        }
63        Ok((classification, served))
64    }
65}

67/// The capability judge a stage router falls through to.
68pub struct LlmFallback {
69    /// Target the judge model is called through. It is not a routing
70    /// destination, so it does not belong in the router's target set.
71    pub judge_target: LlmTarget,
72    /// Judge configuration. `recent_turn_window` is worth setting to this router's
73    /// `recent_window` so the judge reads the same span the signal scorer scored.
74    /// Note: `session_affinity` and `message_hash_fallback` have no effect here —
75    /// the judge runs as a cascade classifier, not a standalone algorithm.
76    pub config: TaskClassifierConfig,
77}

79/// How a stage router scores turns, and what it hands the model it picks.
80pub struct StageRouterConfig {
81    /// Tier a turn falls open to when the scorer is not confident.
82    pub mode: PickerMode,
83    /// How much corroboration a decisive pick needs, in `[0.0, 1.0]`.
84    pub confidence_threshold: f64,
85    /// Trailing tool results the signals are computed over. `None` uses
86    /// [`DEFAULT_RECENT_WINDOW`].
87    pub recent_window: Option<usize>,
88    /// Note handed to the model on a signal-driven escalation, and on a
89    /// hand-back to the efficient tier when a de-escalation note is configured.
90    pub handoff_notes: Option<HandoffNoteConfig>,
91    /// System prompts keyed by target, handed over on every turn that target
92    /// serves. Empty by default.
93    pub tier_prompts: TargetPrompts,
94    /// Capability judge consulted on turns the signals leave undecided — the
95    /// judge's own target, plus the same configuration the standalone capability
96    /// route takes.
97    pub llm_fallback: Option<LlmFallback>,
98}

100impl StageRouterConfig {
101    /// The signal-only configuration: no notes, no per-tier prompts, no judge.
102    /// Set the optional fields to add them.

105            mode,
106            confidence_threshold,
107            recent_window: None,
108            handoff_notes: None,
109            tier_prompts: TargetPrompts::default(),
110            llm_fallback: None,
111        }
112    }

115/// Routes coding-agent turns between a capable and an efficient tier: tool signals
116/// decide first, an optional capability judge takes the turns they cannot, and
117/// the picker's default tier closes the cascade so a turn is never left unrouted.
118pub struct StageRouter {
119    route: FallThrough<State>,
120}
121
122impl StageRouter {
123    /// Routes between the `capable` and `efficient` targets. The
124    /// judge, when configured, is called through its own target and is not a
125    /// routing destination.
126    ///
127    /// Errors if either threshold in `config` is outside `[0.0, 1.0]`.
128    pub fn new(
129        capable: LlmTarget,
130        efficient: LlmTarget,
131        config: StageRouterConfig,
132    ) -> Result<Self> {
133        Ok(Self {
134            route: build_route(capable, efficient, config)?,
135        })
136    }
137}

139#[async_trait]
140impl Algorithm for StageRouter {
141    fn name(&self) -> &str {
142        STAGE_ROUTER
143    }
144
145    fn count_tokens_client(&self) -> Option<Arc<dyn RoutedLlmClient>> {
146        self.route.count_tokens_client()
147    }
148
149    async fn create_run_task(
150        self: Arc<Self>,
151        ctx: Context,
152        driver: Driver,
153        request: Request,
154    ) -> Result<Response> {
155        self.route.execute(ctx, driver, request).await
156    }
157}

159/// Wires the cascade the wrapper drives.
160fn build_route(
161    capable: LlmTarget,
162    efficient: LlmTarget,
163    config: StageRouterConfig,
164) -> Result<FallThrough<State>> {
165    if !(0.0..=1.0).contains(&config.confidence_threshold) {
166        return Err(LibsyError::AlgorithmError {
167            message: format!(
168                "confidence_threshold must be between 0 and 1, got {}",
169                config.confidence_threshold
170            ),
171        });
172    }
173    // The tiers are a fixed pair; their targets are whatever the deployment calls
174    // them, and the classifier scores onto those names.
175    let targets = StageTargets::new(
176        capable.semantic_name.clone(),
177        efficient.semantic_name.clone(),
178    );
179    // The picker's mode fixes the fallback tier up front, so the terminal
180    // classifier is a constant rather than a per-turn lookup.
181    let fall_open = targets.name(config.mode.default_tier()).to_string();
182
183    let mut classifier = StageClassifier::new(targets, config.mode, config.confidence_threshold);
184    if let Some(notes) = config.handoff_notes {
185        classifier = classifier.with_handoff_notes(notes);
186    }
187    let signals = ToolSignalProcessor {
188        recent_window: config.recent_window.unwrap_or(DEFAULT_RECENT_WINDOW),
189    };
190
191    let target_set = LlmTargetSet::new(vec![capable.clone(), efficient.clone()]);
192    let mut router = FallThrough::<State>::new_with_state(target_set)
193        .with_name(STAGE_ROUTER)
194        .with_processor(Arc::new(signals))
195        .with_classifier(Arc::new(classifier));
196    if let Some(fallback) = config.llm_fallback {
197        // The capability judge takes its tiers in the same order the capability
198        // route passes them: efficient first, capable second.
199        router = router.with_classifier(Arc::new(SourceStamp {
200            inner: Arc::new(LlmTaskClassifier::new(
201                fallback.judge_target,
202                efficient,
203                capable,
204                fallback.config,
205            )?),
206            source: DecisionSource::LlmClassifier,
207        }));
208    }
209    // Nothing behind this, so the turn lands on the picker's default tier —
210    // including when the judge could not tell.
211    router = router.with_classifier(Arc::new(SourceStamp {
212        inner: Arc::new(DefaultTarget::new(fall_open)),
213        source: DecisionSource::FallOpen,
214    }));
215    // Runs on the post-decision hook, so it applies to the target the cascade
216    // settled on, whichever classifier picked it. With no prompts configured it
217    // is a no-op, so there is nothing to branch on.
218    router = router.with_processor(Arc::new(SystemPromptProcessor::new(config.tier_prompts)));
219    Ok(router)
220}

358    const ESCALATION: &str = "the previous model was stalling; pick up the diagnosis";
359    const JUDGE: &str = "judge";

361    #[derive(Clone, Debug)]
362    struct Call {
363        target: String,
364        messages: Vec<String>,
365    }
366
367    /// Records what each target receives. When called as the judge target it
368    /// replies with a structured verdict so the fallback classifier gets an answer
369    /// without a real model.
370    #[derive(Default)]
371    struct RecordingClient {
372        calls: Mutex<Vec<Call>>,
373        judge_p_solve: Mutex<f64>,
374    }
375
376    impl RecordingClient {
377        fn routed(&self) -> Vec<Call> {
378            self.calls
379                .lock()
380                .iter()
381                .filter(|call| call.target != JUDGE)
382                .cloned()
383                .collect()
384        }
385    }
386
387    #[async_trait]
388    impl RoutedLlmClient for RecordingClient {
389        async fn call(
390            &self,
391            _ctx: Context,
392            request: Request,
393            decision: Arc<dyn Decision>,
394        ) -> std::result::Result<Response, switchyard_protocol::LlmClientError> {
395            let target = decision.selected_model().to_string();
396            self.calls.lock().push(Call {
397                target: target.clone(),
398                messages: request
399                    .llm_request
400                    .messages
401                    .iter()
402                    .filter_map(|message| message.text_content("|"))
403                    .collect(),
404            });
405            let completion = if target == JUDGE {
406                let p_solve = *self.judge_p_solve.lock();
407                format!(
408                    r#"{{"recommended_route":"efficient","p_solve":{p_solve},"confidence":0.9,"abstain":false,"capability_boundary":"supported","primary_rule":"SUP-1","crux":"bounded task"}}"#
409                )
410            } else {
411                target
412            };
413            Ok(Response {
414                llm_response: LlmResponse::Agg(text_response(None, completion)),
415                metadata: None,
416            })
417        }
418    }

458    fn turn_request(failed: bool) -> Request {
459        let content = if failed {
460            "fatal runtime error: out of memory"
461        } else {
462            "ok"
463        };
464        Request {
465            llm_request: LlmRequest {
466                model: Some("auto".to_string()),
467                messages: vec![
468                    Message::text(Role::User, "fix the build"),
469                    Message {
470                        role: Role::Assistant,
471                        content: vec![ContentBlock::ToolCall(ToolCall {
472                            id: "call_1".to_string(),
473                            name: "Bash".to_string(),
474                            arguments: json!({"command": "cargo test"}),
475                        })],
476                    },
477                    Message {
478                        role: Role::Tool,
479                        content: vec![ContentBlock::ToolResult(ToolResult {
480                            tool_call_id: "call_1".to_string(),
481                            content: vec![ContentBlock::Text {
482                                text: content.to_string(),
483                            }],
484                            is_error: Some(failed),
485                        })],
486                    },
487                ],
488                ..LlmRequest::default()
489            },
490            raw_request: Some(json!({
491                "model": "auto",
492                "messages": [
493                    {"role": "user", "content": "fix the build"},
494                    {"role": "assistant", "tool_calls": [{"id": "call_1", "type": "function",
495                        "function": {"name": "Bash", "arguments": "{\"command\": \"cargo test\"}"}}]},
496                    {"role": "tool", "tool_call_id": "call_1", "content": content},
497                ],
498            })),
499            metadata: Some(Metadata {
500                wire_format: Some(WireFormat::OpenAiChat),
501                session_id: Some("session-1".to_string()),
502                ..Default::default()
503            }),
504        }
505    }

508    async fn a_signal_driven_escalation_hands_the_note_to_the_model() -> Result<()> {
509        let client = Arc::new(RecordingClient::default());
510        let router = recording_router(client.clone(), config_with_notes())?;
511        let ctx = Context::default();
512
513        router.clone().run(ctx.clone(), turn_request(false)).await?;
514        router.run(ctx, turn_request(true)).await?;
515
516        let calls = client.routed();
517        assert_eq!(calls[0].target, "weak");
518        assert_eq!(calls[1].target, "strong");
519        assert!(
520            !calls[0].messages.iter().any(|t| t.contains(ESCALATION)),
521            "steady-state turn should carry no note: {:?}",
522            calls[0].messages
523        );
524        assert!(
525            calls[1]
526                .messages
527                .last()
528                .is_some_and(|t| t.ends_with(ESCALATION)),
529            "escalating turn should carry the note last: {:?}",
530            calls[1].messages
531        );
532        Ok(())
533    }
