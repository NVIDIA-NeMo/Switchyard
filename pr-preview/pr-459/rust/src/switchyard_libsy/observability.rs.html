4//! OpenTelemetry metrics plus `tracing` spans and structured logs for the
5//! algorithm layer.
6//!
7//! [`Algorithm::run_stream`](crate::Algorithm::run_stream) and [`Driver`] call these
8//! helpers around the routing outcome and the offload boundary, so every algorithm is
9//! instrumented from the outside and carries no telemetry code of its own. The provider
10//! call on the other side of the offload belongs to the host, and is instrumented by
11//! whoever makes it. Metrics record through the
12//! OpenTelemetry **global** meter provider under the `switchyard` scope — the host
13//! installs an SDK provider and exporters; with none installed, recording is a
14//! no-op. Spans and logs use the `tracing` facade (the async-native surface the
15//! OpenTelemetry ecosystem bridges with `tracing-opentelemetry` /
16//! `opentelemetry-appender-tracing`), so the host's subscriber decides where
17//! they go. Method spans use `#[tracing::instrument]`; the `libsy.run` span is
18//! attached to the spawned run task with [`tracing::Instrument`]. Neither holds
19//! a [`Span::enter`] guard across an `.await` — a suspended task would leave
20//! the span entered on its executor thread, mis-parenting every span other
21//! tasks create there (see the `tracing` docs on spans in asynchronous code).
22//!
23//! Instrument names use the OTel dotted form with the unit baked into the name
24//! (`switchyard.run_duration_ms`), matching the switchyard metric surface; a
25//! Prometheus exporter sanitizes them to `switchyard_run_duration_ms`. Attribute
26//! cardinality is bounded: `algorithm` and `selected_model` are small
27//! configured sets and `outcome` is `ok`/`error`. Nothing per-request becomes a
28//! metric attribute — correlation ids ride on the `libsy.run` span instead.
29//!
30//! Instruments are resolved from the global provider on every record (an
31//! instrument-cache lookup inside the SDK) so recording follows a meter
32//! provider installed at any point in the process lifetime; the cost is
33//! negligible next to a model call.
34

45const METRICS_SCOPE: &str = "switchyard";
46const TRACING_TARGET: &str = "libsy";
47
48/// The `libsy`-scoped meter from the globally installed provider.
49pub(crate) fn meter() -> Meter {
50    global::meter(METRICS_SCOPE)
51}
52
53/// `outcome` attribute value for a result: `ok` or `error`.
54pub(crate) fn outcome_value<T>(result: &Result<T>) -> &'static str {
55    match result {
56        Ok(_) => "ok",
57        Err(_) => "error",
58    }
59}

61/// Span covering one algorithm run (the whole `route` execution).
62///
63/// Correlation ids from the request [`switchyard_protocol::Metadata`] are recorded as span fields
64/// when present. `tracing` spans cannot grow field names at runtime, so
65/// arbitrary host labels ride in via [`switchyard_protocol::Metadata::extra_metadata`], recorded
66/// whole into the `extra_metadata` field. `outcome` and `error` are filled in
67/// by [`record_run`] when the run ends.
68pub(crate) fn run_span(algorithm: &str, request: &Request) -> Span {
69    let span = tracing::info_span!(
70        target: TRACING_TARGET,
71        "libsy.run",
72        algorithm,
73        switchyard.algorithm = algorithm,
74        openinference.span.kind = "CHAIN",
75        switchyard.route = tracing::field::Empty,
76        session_id = tracing::field::Empty,
77        session.id = tracing::field::Empty,
78        agent_id = tracing::field::Empty,
79        task_id = tracing::field::Empty,
80        correlation_id = tracing::field::Empty,
81        extra_metadata = tracing::field::Empty,
82        outcome = tracing::field::Empty,
83        error = tracing::field::Empty,
84    );
85    if let Some(route) = request.model_id() {
86        span.record("switchyard.route", route.as_ref());
87    }
88    if let Some(metadata) = &request.metadata {
89        for (field, value) in [
90            ("session_id", &metadata.session_id),
91            ("agent_id", &metadata.agent_id),
92            ("task_id", &metadata.task_id),
93            ("correlation_id", &metadata.correlation_id),
94        ] {
95            if let Some(value) = value {
96                span.record(field, value.as_str());
97            }
98        }
99        if let Some(session_id) = &metadata.session_id {
100            span.record("session.id", session_id.as_str());
101        }
102        if let Some(extra) = &metadata.extra_metadata {
103            span.record("extra_metadata", tracing::field::debug(extra));
104        }
105    }
106    span
107}

109/// Runs one algorithm task to completion, recording the run counter, duration
110/// histogram, span outcome, and failure log when it resolves.
111/// Executes inside the `libsy.run` span its caller instruments the task with.
112pub(crate) async fn observe_run<T>(
113    algorithm: &str,
114    run: impl Future<Output = Result<T>>,
115) -> Result<T> {
116    let started = Instant::now();
117    let result = run.await;
118    let duration = started.elapsed();
119    record_run(algorithm, duration, &result, &Span::current());
120    result
121}

123/// Records the end of one algorithm run: the run counter and duration
124/// histogram, the `outcome`/`error` fields on `span`, and a warn log when the
125/// run failed.
126fn record_run<T>(algorithm: &str, duration: Duration, result: &Result<T>, span: &Span) {
127    let outcome = outcome_value(result);
128    span.record("outcome", outcome);
129    if let Err(error) = result {
130        span.record("error", tracing::field::display(error));
131        tracing::warn!(
132            target: TRACING_TARGET,
133            algorithm,
134            error = %error,
135            "algorithm run failed"
136        );
137    }
138
139    let attributes = [
140        KeyValue::new("algorithm", algorithm.to_string()),
141        KeyValue::new("outcome", outcome),
142    ];
143    let meter = meter();
144    meter
145        .u64_counter("switchyard.runs")
146        .build()
147        .add(1, &attributes);
148    meter
149        .f64_histogram("switchyard.run_duration_ms")
150        .build()
151        .record(duration.as_secs_f64() * 1000.0, &attributes);
152}

154/// Records a judge failure that made the classifier route without a verdict.
155pub(crate) fn record_classifier_fail_open(judge_model: &str, reason: &'static str) {
156    meter()
157        .u64_counter("switchyard.classifier_fail_open")
158        .build()
159        .add(

161            &[
162                KeyValue::new("judge_model", judge_model.to_string()),

168/// Records the resolution of one offloaded model call: the call counter and
169/// latency histogram, the `outcome`/`error`/token fields on `span`, and a warn
170/// log when the call failed.
171pub(crate) fn record_llm_call(
172    algorithm: &str,
173    selected_model: &str,
174    duration: Duration,
175    result: &Result<Response>,
176    span: &Span,
177) {
178    let outcome = outcome_value(result);
179    span.record("outcome", outcome);
180
181    let meter = meter();
182    let call_attributes = [
183        KeyValue::new("algorithm", algorithm.to_string()),
184        KeyValue::new("selected_model", selected_model.to_string()),
185        KeyValue::new("outcome", outcome),
186    ];
187    meter
188        .u64_counter("switchyard.llm_calls")
189        .build()
190        .add(1, &call_attributes);
191    meter
192        .f64_histogram("switchyard.llm_call_duration_ms")
193        .build()
194        .record(duration.as_secs_f64() * 1000.0, &call_attributes);
195
196    match result {
197        Ok(response) => {
198            // Token usage exists only once a response is buffered; a streamed
199            // response resolves before its usage is known, so none is recorded.
200            let Some(usage) = response.llm_response.as_agg().map(|agg| &agg.usage) else {
201                return;
202            };
203            for (field, value) in [
204                ("input_tokens", usage.input_tokens),
205                ("output_tokens", usage.output_tokens),
206                ("total_tokens", usage.total_tokens),
207                ("reasoning_tokens", usage.reasoning_tokens),
208            ] {
209                if let Some(value) = value {
210                    span.record(field, value);
211                }
212            }
213        }
214        Err(error) => {
215            span.record("error", tracing::field::display(error));
216            tracing::warn!(
217                target: TRACING_TARGET,
218                algorithm,
219                selected_model,
220                error = %error,
221                "model call failed"
222            );
223        }
224    }
225}

227/// Records one published routing decision: the decision counter plus a structured debug event.
228pub(crate) fn record_decision(algorithm: &str, selected_model: &ModelId) {
229    tracing::debug!(
230        target: TRACING_TARGET,
231        algorithm,
232        selected_model = %selected_model,
233        "routing decision"
234    );
235    meter().u64_counter("switchyard.decisions").build().add(
236        1,
237        &[
238            KeyValue::new("algorithm", algorithm.to_string()),
239            KeyValue::new("selected_model", selected_model.to_string()),
240        ],
241    );
