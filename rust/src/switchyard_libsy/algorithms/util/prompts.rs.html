11/// Prepends a system prompt and updates supported preserved provider bodies.
12/// It removes bodies it cannot edit so encoding rebuilds them from the updated request.
13/// Call it once per candidate on a request that has not already received a target prompt.
14pub fn prepend_system_prompt(request: &mut LlmRequest, prompt: &str) {
15    request.instructions.insert(
16        0,
17        InstructionBlock {
18            role: Role::System,
19            content: vec![ContentBlock::Text {
20                text: prompt.to_string(),
21            }],
22        },
23    );
24    request.preservation.requests.retain(|format, body| {
25        let Some(body) = body.as_object_mut() else {
26            return false;
27        };
28        let format = format.as_str();
29        if format == WireFormat::OpenAiChat.as_str() {
30            prepend_openai_chat_system(body, prompt)
31        } else if format == WireFormat::OpenAiResponses.as_str() {
32            prepend_text_field(body, "instructions", prompt)
33        } else if format == WireFormat::AnthropicMessages.as_str() {
34            body.remove(ANTHROPIC_REQUEST_KEY);
35            prepend_anthropic_system(body, prompt)
36        } else {
37            false
38        }
39    });
40}

42fn prepend_openai_chat_system(body: &mut Map<String, Value>, prompt: &str) -> bool {
43    let message = json!({"role": "system", "content": prompt});
44    match body.get_mut("messages") {
45        None | Some(Value::Null) => {
46            body.insert("messages".to_string(), Value::Array(vec![message]));
47        }
48        Some(Value::Array(messages)) => messages.insert(0, message),
49        Some(_) => return false,
50    }
51    true
52}

54// Preserved Responses instructions and Anthropic system prompts can be strings, missing, or null.
55fn prepend_text_field(body: &mut Map<String, Value>, key: &str, prompt: &str) -> bool {
56    match body.get_mut(key) {
57        None | Some(Value::Null) => {
58            body.insert(key.to_string(), Value::String(prompt.to_string()));
59        }
60        Some(Value::String(text)) if text.is_empty() => {
61            *text = prompt.to_string();
62        }
63        Some(Value::String(text)) => text.insert_str(0, &format!("{prompt}\n\n")),
64        Some(_) => return false,
65    }
66    true
67}
68
69fn prepend_anthropic_system(body: &mut Map<String, Value>, prompt: &str) -> bool {
70    if let Some(Value::Array(blocks)) = body.get_mut("system") {
71        blocks.insert(0, json!({"type":"text","text":prompt}));
72        true
73    } else {
74        prepend_text_field(body, "system", prompt)
75    }
76}

78/// Appends temporary guidance and updates preserved provider bodies.
79/// It removes bodies it cannot edit so encoding rebuilds them from the updated request.
80pub(crate) fn append_note(request: &mut LlmRequest, note: &str) {
81    // A separate message preserves the caller's final message boundary for OpenAI caching.
82    request.messages.push(Message::text(Role::User, note));
83    request.temporary_note_count += 1;
84    request.preservation.requests.retain(|format, body| {
85        let Some(body) = body.as_object_mut() else {
86            return false;
87        };
88        let format = format.as_str();
89        let (key, content_type) = if format == WireFormat::OpenAiChat.as_str()
90            || format == WireFormat::AnthropicMessages.as_str()
91        {
92            ("messages", "text")
93        } else if format == WireFormat::OpenAiResponses.as_str() {
94            ("input", "input_text")
95        } else {
96            return false;
97        };
98        // Responses also accepts a string as a single user input.
99        if key == "input"
100            && let Some(Value::String(text)) = body.get_mut(key)
101        {
102            let original = std::mem::take(text);
103            body.insert(key.into(), json!([{"role":"user", "content":original}]));
104        }
105        if key == "input" && body.get(key).is_none_or(Value::is_null) {
106            body.insert(key.into(), Value::Array(Vec::new()));
107        }
108        let Some(messages) = body.get_mut(key).and_then(Value::as_array_mut) else {
109            return false;
110        };
111        if format == WireFormat::AnthropicMessages.as_str()
112            && let Some(last) = messages.last_mut()
113            && last.get("role").and_then(Value::as_str) == Some("user")
114        {
115            // Anthropic guidance follows existing content, including any tool result.
116            match last.get_mut("content") {
117                Some(Value::String(text)) => {
118                    let original = std::mem::take(text);
119                    last["content"] = json!([
120                        {"type":"text", "text":original},
121                        {"type":"text", "text":note}
122                    ]);
123                }
124                Some(Value::Array(blocks)) => blocks.push(json!({"type":"text", "text":note})),
125                _ => return false,
126            }
127        } else {
128            messages.push(json!({"role":"user", "content":[{"type":content_type,"text":note}]}));
129        }
130        true
131    });
132}

134/// Clears preserved bodies and the note count after other conversation edits.
135/// Subsequent encoding uses the edited request instead of replaying stale provider bodies.
136pub(crate) fn drop_exact_replay(request: &mut LlmRequest) {
137    request.preservation.requests.clear();
138    request.temporary_note_count = 0;
139}

148    // Native bodies retain provider fields and tool argument text that decoding may not preserve.
149    #[test]
150    fn edits_preserve_native_prefixes_and_provider_fields() -> Result<(), Box<dyn std::error::Error>>
151    {
152        let engine = TranslationEngine::default();
153        let policy = TranslationPolicy::default();
154        let fixtures = [
155            (
156                WireFormat::OpenAiChat,
157                json!({
158                    "model":"auto", "messages":[
159                        {"role":"developer","content":[{"type":"text","text":"Keep this instruction.","vendor":"keep"}]},
160                        {"role":"assistant","tool_calls":[{"id":"call_1","type":"function","function":{"name":"run","arguments":"{ \"cmd\": \"test\" }"}}]},
161                        {"role":"tool","tool_call_id":"call_1","content":"exit 1","vendor":"keep"}
162                    ], "metadata":{"caller":"keep"}, "temperature":0.2
163                }),
164            ),
165            (
166                WireFormat::OpenAiResponses,
167                json!({
168                    "model":"auto", "instructions":"Keep this instruction.", "input":[
169                        {"type":"message","role":"developer","content":[{"type":"input_text","text":"Keep this too."}]},
170                        {"type":"function_call","call_id":"call_1","name":"run","arguments":"{ \"cmd\": \"test\" }"},
171                        {"type":"function_call_output","call_id":"call_1","output":"exit 1"}
172                    ], "metadata":{"caller":"keep"}, "store":false
173                }),
174            ),
175            (
176                WireFormat::AnthropicMessages,
177                json!({
178                    "model":"auto", "max_tokens":32,
179                    "system":[{"type":"text","text":"Keep this instruction.","cache_control":{"type":"ephemeral","ttl":"1h"}}],
180                    "messages":[
181                        {"role":"assistant","content":[{"type":"tool_use","id":"call_1","name":"run","input":{"cmd":"test"}}]},
182                        {"role":"user","content":[{"type":"tool_result","tool_use_id":"call_1","content":"exit 1","is_error":true,"cache_control":{"type":"ephemeral"}}]}
183                    ], "metadata":{"user_id":"keep"}
184                }),
185            ),
186        ];
187        for (format, original) in fixtures {
188            let mut request = engine.decode_request(format, &original, &policy)?.request;
189            append_note(&mut request, "Reassess the failure.");
190            let encoded = engine.encode_request(format, &request, &policy)?.body;
191            let mut expected = original.clone();
192            match format {
193                WireFormat::OpenAiChat => expected["messages"].as_array_mut().ok_or("messages")?.push(json!({"role":"user","content":[{"type":"text","text":"Reassess the failure."}]})),
194                WireFormat::OpenAiResponses => expected["input"].as_array_mut().ok_or("input")?.push(json!({"role":"user","content":[{"type":"input_text","text":"Reassess the failure."}]})),
195                WireFormat::AnthropicMessages => expected["messages"][1]["content"].as_array_mut().ok_or("content")?.push(json!({"type":"text","text":"Reassess the failure."})),
196            }
197            assert_eq!(encoded, expected, "{format}: only the note may change");
198            prepend_system_prompt(&mut request, "Use the capable tier.");
199            let edited = engine.encode_request(format, &request, &policy)?.body;
200            // The system prompt edit must retain the earlier note and all conversation and tool items.
201            let key = if format == WireFormat::OpenAiResponses {
202                "input"
203            } else {
204                "messages"
205            };
206            if format == WireFormat::OpenAiChat {
207                assert_eq!(
208                    edited[key].as_array().ok_or("messages")?[1..],
209                    encoded[key].as_array().ok_or("messages")?[..]
210                );
211            } else {
212                assert_eq!(edited[key], encoded[key]);
213            }
214            assert_eq!(edited["metadata"], original["metadata"]);
215        }
216        Ok(())
217    }

219    #[test]
220    fn string_inputs_and_assistant_endings_keep_the_original_message()
221    -> Result<(), Box<dyn std::error::Error>> {
222        let engine = TranslationEngine::default();
223        let policy = TranslationPolicy::default();
224        for (format, body, key) in [
225            (
226                WireFormat::OpenAiChat,
227                json!({"model":"m","messages":[{"role":"assistant","content":"done"}]}),
228                "messages",
229            ),
230            (
231                WireFormat::AnthropicMessages,
232                json!({"model":"m","max_tokens":8,"messages":[{"role":"user","content":"question"},{"role":"assistant","content":"done"}]}),
233                "messages",
234            ),
235            (
236                WireFormat::OpenAiResponses,
237                json!({"model":"m","input":"question"}),
238                "input",
239            ),
240        ] {
241            let mut request = engine.decode_request(format, &body, &policy)?.request;
242            append_note(&mut request, "note one");
243            append_note(&mut request, "note two");
244            let encoded = engine.encode_request(format, &request, &policy)?.body;
245            let items = encoded[key].as_array().ok_or("items")?;
246            if body[key].is_array() {
247                let original = body[key].as_array().ok_or("original")?;
248                assert_eq!(&items[..original.len()], original);
249            } else {
250                assert_eq!(items[0], json!({"role":"user","content":"question"}));
251            }
252        }
253        Ok(())
254    }
255    // Unsupported preserved bodies must not cause encoding to omit the note.
256    #[test]
257    fn invalid_preservation_falls_back_without_losing_the_note()
258    -> Result<(), Box<dyn std::error::Error>> {
259        let engine = TranslationEngine::default();
260        let policy = TranslationPolicy::default();
261        let source = json!({"model":"m","messages":[{"role":"user","content":"original task"}]});
262        let mut request = engine
263            .decode_request(WireFormat::OpenAiChat, &source, &policy)?
264            .request;
265        request
266            .preservation
267            .requests
268            .insert("custom".into(), json!({"messages":"unsupported"}));
269        request.preservation.requests.insert(
270            WireFormat::OpenAiChat.into(),
271            json!({"model":"m","messages":"unsupported"}),
272        );
273        append_note(&mut request, "reassess the failure");
274        assert!(request.preservation.requests.is_empty());
275        for target in [
276            WireFormat::OpenAiChat,
277            WireFormat::OpenAiResponses,
278            WireFormat::AnthropicMessages,
279        ] {
280            let body = engine.encode_request(target, &request, &policy)?.body;
281            let decoded = engine.decode_request(target, &body, &policy)?.request;
282            let text: Vec<_> = decoded
283                .messages
284                .iter()
285                .filter_map(|message| message.text_content(""))
286                .collect();
287            assert_eq!(
288                text,
289                vec!["original task", "reassess the failure"],
290                "{target}"
291            );
292        }
293        Ok(())
294    }
