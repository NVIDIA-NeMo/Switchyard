4//! Codex tool namespace preservation across a Responses/Chat translation.
5//!
6//! Codex groups tools into non-standard ``namespace`` containers — MCP servers,
7//! usually behind an `mcp__` prefix, plus builtin groups such as
8//! `multi_agent_v1` — and dispatches on the pair of tool name and namespace. It
9//! therefore expects the namespace back on the call it receives:
10//! `{"type": "function_call", "name": "search", "namespace": "mcp__docs"}`.
11//!
12//! OpenAI-compatible upstreams accept only flat `function` tools, so the request
13//! codec flattens the containers and names each child `<namespace>__<tool>`.
14//! Two tools differing only by namespace therefore stay distinct upstream, and
15//! the conversation history uses the same spelling so the transcript never
16//! teaches the model a name the upstream was not offered.
17//!
18//! The qualified-name to namespace mapping rides in the request's
19//! [`ProviderExtensions`], so no provider-neutral type grows a Codex-specific
20//! field. No codec copies unknown extension keys into an outbound body, so the
21//! mapping reaches neither the upstream nor the client.
22//!
23//! No container is filtered by its name, because Codex resolves a call with no
24//! namespace against its default group: dropping a builtin group's namespace
25//! would look the call up in the wrong place.
26
27use std::collections::HashMap;
28
29use serde_json::{Map, Value};
30use switchyard_protocol::ProviderExtensions;
31
32pub use switchyard_protocol::codex_namespaces::{
33    NAMESPACE_SEPARATOR, TOOL_NAMESPACES_KEY, split_qualified_name, tool_namespaces,
34};
35
36/// Joins a namespace and a tool name into the name used on the wire.
37pub fn qualified_tool_name(namespace: &str, tool: &str) -> String {
38    format!("{namespace}{NAMESPACE_SEPARATOR}{tool}")
39}
40
41/// Records that `qualified` was produced by flattening a tool out of `namespace`.
42pub fn record_tool_namespace(
43    namespaces: &mut Map<String, Value>,
44    qualified: &str,
45    namespace: &str,
46    description: Option<&str>,
47) {
48    let value = description.map_or_else(
49        || Value::String(namespace.to_string()),
50        |description| {
51            serde_json::json!({
52                "namespace": namespace,
53                "description": description,
54            })
55        },
56    );
57    namespaces.insert(qualified.to_string(), value);
58}
59
60/// Stores a collected mapping on a request's extensions, when it has entries.
61pub fn attach_tool_namespaces(extensions: &mut ProviderExtensions, namespaces: Map<String, Value>) {
62    if !namespaces.is_empty() {
63        extensions
64            .fields
65            .insert(TOOL_NAMESPACES_KEY.to_string(), Value::Object(namespaces));
66    }
67}
68
69/// Returns a retained description for `namespace`.
70pub fn namespace_description<'a>(
71    namespaces: &'a Map<String, Value>,
72    namespace: &str,
73) -> Option<&'a str> {
74    namespaces.values().find_map(|value| {
75        let object = value.as_object()?;
76        (object.get("namespace").and_then(Value::as_str) == Some(namespace))
77            .then(|| object.get("description").and_then(Value::as_str))
78            .flatten()
79    })
80}
81
82/// Maps exact qualified tool names to their Codex tool names and namespaces.
83pub fn qualified_tool_origins(
84    extensions: &ProviderExtensions,
85) -> HashMap<String, (String, String)> {
86    let Some(namespaces) = tool_namespaces(extensions) else {
87        return HashMap::new();
88    };
89    namespaces
90        .keys()
91        .filter_map(|qualified| {
92            let (tool, namespace) = split_qualified_name(namespaces, qualified)?;
93            Some((qualified.clone(), (tool.to_string(), namespace.to_string())))
94        })
95        .collect()
96}
97
98/// Restore the Codex tool name and namespace on function calls and argument completions.
99///
100/// Walks the whole value, covering `function_call` items in buffered responses
101/// and streaming events, plus the top-level name in
102/// `response.function_call_arguments.done`. Preserves an existing `namespace`.
103pub fn restore_qualified_tool_names(body: &mut Value, origins: &HashMap<String, (String, String)>) {
104    if origins.is_empty() {
105        return;
106    }
107    match body {
108        Value::Array(values) => {
109            for value in values {
110                restore_qualified_tool_names(value, origins);
111            }
112        }
113        Value::Object(object) => {
114            if matches!(
115                object.get("type").and_then(Value::as_str),
116                Some("function_call" | "response.function_call_arguments.done")
117            ) && let Some(name) = object.get("name").and_then(Value::as_str)
118                && let Some((tool, namespace)) = origins.get(name)
119            {
120                object.insert("name".to_string(), Value::String(tool.clone()));
121                object
122                    .entry("namespace".to_string())
123                    .or_insert_with(|| Value::String(namespace.clone()));
124            }
125            for value in object.values_mut() {
126                restore_qualified_tool_names(value, origins);
127            }
128        }
129        _ => {}
130    }
131}

133#[cfg(test)]
134mod tests {
135    use serde_json::{Map, json};
136    use switchyard_protocol::ProviderExtensions;
137
138    use super::{
139        attach_tool_namespaces, qualified_tool_name, qualified_tool_origins, record_tool_namespace,
140        restore_qualified_tool_names, split_qualified_name, tool_namespaces,
141    };
142
143    fn extensions(pairs: &[(&str, &str)]) -> ProviderExtensions {
144        let mut namespaces = Map::new();
145        for (namespace, tool) in pairs {
146            record_tool_namespace(
147                &mut namespaces,
148                &qualified_tool_name(namespace, tool),
149                namespace,
150                None,
151            );
152        }
153        let mut extensions = ProviderExtensions::default();
154        attach_tool_namespaces(&mut extensions, namespaces);
155        extensions
156    }
157
158    // A tool name may itself contain the separator, so the namespace is matched
159    // as a prefix rather than by splitting on the first `__`.
160    #[test]
161    fn splits_a_qualified_name_back_apart() {
162        let simple = extensions(&[("mcp__docs", "search")]);
163        let simple = tool_namespaces(&simple).expect("mapping present");
164        assert_eq!(
165            split_qualified_name(simple, "mcp__docs__search"),
166            Some(("search", "mcp__docs"))
167        );
168        assert_eq!(split_qualified_name(simple, "unknown_tool"), None);
169
170        let nested = extensions(&[("mcp__docs", "fetch__raw")]);
171        let nested = tool_namespaces(&nested).expect("mapping present");
172        assert_eq!(
173            split_qualified_name(nested, "mcp__docs__fetch__raw"),
174            Some(("fetch__raw", "mcp__docs"))
175        );
176    }
177
178    #[test]
179    fn leaves_unqualified_names_alone() {
180        let origins = qualified_tool_origins(&extensions(&[("mcp__docs", "search")]));
181        let mut response = json!([
182            {"output": [{"type": "function_call", "name": "search"}]},
183            {"type": "response.output_item.done", "item": {
184                "type": "function_call", "name": "docs__search"
185            }},
186            {"type": "response.function_call_arguments.done", "name": "docs__search"}
187        ]);
188        let before = response.clone();
189
190        restore_qualified_tool_names(&mut response, &origins);
191
192        assert_eq!(response, before);
193    }
194
195    // Codex namespaces builtin groups too, so nothing may key on `mcp__`.
196    #[test]
197    fn resolves_namespaces_that_are_not_mcp_servers() {
198        let origins = qualified_tool_origins(&extensions(&[("multi_agent_v1", "spawn_agent")]));
199        let mut response = json!({
200            "output": [{"type": "function_call", "name": "multi_agent_v1__spawn_agent"}]
201        });
202
203        restore_qualified_tool_names(&mut response, &origins);
204
205        assert_eq!(response["output"][0]["name"], "spawn_agent");
206        assert_eq!(response["output"][0]["namespace"], "multi_agent_v1");
207    }
208}
