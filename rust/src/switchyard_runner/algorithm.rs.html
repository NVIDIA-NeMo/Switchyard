27/// Error returned when an algorithm description cannot be constructed.
28#[derive(Debug)]
29pub struct AlgorithmConfigError {
30    message: String,
31    source: Option<Box<dyn Error + Send + Sync>>,
32}
33
34impl AlgorithmConfigError {
35    fn new(message: impl Into<String>) -> Self {
36        Self {
37            message: message.into(),
38            source: None,
39        }
40    }
41
42    fn with_source(message: impl Into<String>, source: impl Error + Send + Sync + 'static) -> Self {
43        Self {
44            message: message.into(),
45            source: Some(Box::new(source)),
46        }
47    }
48}
49
50impl Display for AlgorithmConfigError {
51    fn fmt(&self, formatter: &mut Formatter<'_>) -> std::fmt::Result {
52        formatter.write_str(&self.message)
53    }
54}
55
56impl Error for AlgorithmConfigError {
57    fn source(&self) -> Option<&(dyn Error + 'static)> {
58        self.source.as_deref().map(|source| source as _)
59    }
60}

115impl CapabilityClassifierRouteConfig {
116    fn into_task_config(
117        self,
118        route_name: &str,
119        targets: &BTreeMap<String, ModelId>,
120    ) -> AlgorithmResult<TaskClassifierConfig> {
121        let config = self;
122        let judge = match config.judge {
123            CapabilityJudgeRouteConfig::Llm(judge) => CapabilityJudgeConfig::Llm(judge),
124            CapabilityJudgeRouteConfig::Decision(judge) => {
125                for target in [&config.strong_target, &config.weak_target] {
126                    if !judge.candidates.values().any(|name| name == target) {
127                        return Err(AlgorithmConfigError::new(format!(
128                            "route {route_name}: decision.candidates must include target {target}"
129                        )));
130                    }
131                }
132                let candidates = judge
133                    .candidates
134                    .into_iter()
135                    .map(|(label, name)| {
136                        resolve_target_model_id(route_name, &name, targets)
137                            .map(|model| (label, model))
138                    })
139                    .collect::<AlgorithmResult<_>>()?;
140                CapabilityJudgeConfig::Decision(DecisionJudgeConfig {
141                    cutoff: judge.cutoff,
142                    instructions: judge.instructions,
143                    candidates,
144                    evidence: judge.evidence,
145                })
146            }
147        };
148        Ok(TaskClassifierConfig {
149            judge,
150            fail_open: config.fail_open,
151            classify_trigger: config.classify_trigger,
152            message_hash_fallback: config.message_hash_fallback,
153            recent_turn_window: config.recent_turn_window,
154        })
155    }

188/// Runtime model groups for a custom classifier, keyed by group name.
189///
190/// `any` and `judge` are required; `capable` and `efficient` carry tier meaning
191/// when present. Any other key is a deployment-defined group the policy may
192/// select by name, which is what lets one route choose between more than two
193/// models. Ordered so error messages and derived target lists do not depend on
194/// hash iteration.
195#[derive(Clone, Debug, Default, Deserialize)]
196#[serde(transparent)]
197pub struct CategoryModelConfig(BTreeMap<String, Vec<String>>);
198
199impl CategoryModelConfig {
200    fn validate(&self, route_name: &str, default_target: &Category) -> AlgorithmResult<()> {
201        for category in [Category::Any, Category::Judge] {
202            if self.get(&category).is_empty() {
203                return Err(AlgorithmConfigError::new(format!(
204                    "llm_classifier route {route_name} models.{} must contain at least one target",
205                    category.as_str()
206                )));
207            }
208        }
209        if self.get(default_target).is_empty() {
210            return Err(AlgorithmConfigError::new(format!(
211                "llm_classifier route {route_name} models.{} must contain at least one target because it is the default_target",
212                default_target.as_str()
213            )));
214        }
215        // Every group name parses, so a typo would otherwise build fine and then
216        // abstain on every request. `any` is the fallback pool the router checks
217        // the selected target against, so a group outside it can never be served.
218        let any = self.get(&Category::Any);
219        for (name, models) in &self.0 {
220            if name == Category::Judge.as_str() || name == Category::Any.as_str() {
221                continue;
222            }
223            if let Some(missing) = models.iter().find(|model| !any.contains(model)) {
224                return Err(AlgorithmConfigError::new(format!(
225                    "llm_classifier route {route_name} models.{name} lists target {missing}, which must also appear in models.any"
226                )));
227            }
228        }
229        Ok(())
230    }
231
232    fn get(&self, category: &Category) -> &[String] {
233        self.0.get(category.as_str()).map_or(&[], Vec::as_slice)
234    }
235
236    /// Every configured target name, judge included.
237    fn all_names(&self) -> Vec<&str> {
238        Self::deduped(self.0.values().flatten())
239    }
240
241    /// The completion targets: every group except the judge's own candidates.
242    ///
243    /// `any` leads because it is the deployment's stated fallback order, and
244    /// callers read this order to pick a route's representative target.
245    fn routing_names(&self) -> Vec<&str> {
246        let others = self
247            .0
248            .iter()
249            .filter(|(name, _)| {
250                *name != Category::Judge.as_str() && *name != Category::Any.as_str()
251            })
252            .flat_map(|(_, models)| models);
253        Self::deduped(self.get(&Category::Any).iter().chain(others))
254    }
255
256    fn deduped<'a>(names: impl Iterator<Item = &'a String>) -> Vec<&'a str> {
257        let mut seen = BTreeSet::new();
258        names
259            .map(String::as_str)
260            .filter(|name| seen.insert(*name))
261            .collect()
262    }
263
264    fn groups(&self) -> impl Iterator<Item = (Category, Vec<String>)> + '_ {
265        self.0
266            .iter()
267            .filter_map(|(name, models)| Some((name.parse::<Category>().ok()?, models.clone())))
268    }
269}

370                .map(|models| models.get(&Category::Judge))
371                .unwrap_or_default()
372                .iter()
373                .map(String::as_str)
374                .collect(),
375        }
376    }

892            }

899            }
900            Self::Advisor { advisor_target, .. } => names.push(advisor_target),

911        }
912        | Self::StageRouter {
913            subagents: Some(subagents),
914            ..
915        }
916        | Self::Composite {
917            subagents: Some(subagents),
918            ..
919        } = self

926    /// Target names grouped as the runtime [`Driver`](libsy::Driver) expects them.
927    fn runtime_model_names(&self, route_name: &str) -> AlgorithmResult<RuntimeModelNames> {
928        let parent = match self {
929            Self::Noop { .. } => HashMap::new(),
930            Self::Random { targets, .. } | Self::PrefillRouter { targets, .. } => {
931                category_models([(Category::Any, targets.clone())])
932            }
933            Self::Passthrough { target, .. } => {
934                category_models([(Category::Any, vec![target.clone()])])
935            }
936            Self::PlanExecute {
937                capable_target,
938                efficient_target,
939                ..
940            } => category_models([
941                (Category::Capable, vec![capable_target.clone()]),
942                (Category::Efficient, vec![efficient_target.clone()]),
943                (
944                    Category::Any,
945                    vec![capable_target.clone(), efficient_target.clone()],
946                ),
947            ]),
948            Self::LlmClassifier { config } => {
949                classifier_runtime_model_names(config.validated_classifier_mode(route_name)?)
950            }
951            Self::StageRouter {
952                tiers, classifier, ..
953            } => {
954                let mut models = category_models([
955                    (Category::Capable, vec![tiers.capable_target.clone()]),
956                    (Category::Efficient, vec![tiers.efficient_target.clone()]),
957                    (
958                        Category::Any,
959                        vec![tiers.capable_target.clone(), tiers.efficient_target.clone()],
960                    ),
961                ]);
962                if let Some(classifier) = classifier {
963                    models.insert(Category::Judge, vec![classifier.target.clone()]);
964                }
965                models
966            }
967            Self::Auto {
968                capable_target,
969                efficient_target,
970            } => category_models([
971                (Category::Capable, vec![capable_target.clone()]),
972                (Category::Efficient, vec![efficient_target.clone()]),
973                (
974                    Category::Any,
975                    vec![capable_target.clone(), efficient_target.clone()],
976                ),
977            ]),
978            Self::Composite {
979                classifier, stage, ..
980            } => category_models([
981                (Category::Judge, vec![classifier.target.clone()]),
982                (Category::Capable, vec![stage.capable_target.clone()]),
983                (Category::Efficient, vec![stage.efficient_target.clone()]),
984                (
985                    Category::Any,
986                    vec![stage.capable_target.clone(), stage.efficient_target.clone()],
987                ),
988            ]),
989            Self::Advisor {
990                executor_target,
991                advisor_target,
992                ..
993            } => category_models([
994                (Category::Efficient, vec![executor_target.clone()]),
995                (Category::Any, vec![executor_target.clone()]),
996                (Category::Judge, vec![advisor_target.clone()]),
997            ]),
998            Self::Vgr { config } => {
999                let judge = config
1000                    .judge_target
1001                    .as_ref()
1002                    .unwrap_or(&config.efficient_target);
1003                let judges = std::iter::once(judge)
1004                    .chain(&config.capable_judge_target)
1005                    .cloned()
1006                    .collect();
1007                category_models([
1008                    (Category::Efficient, vec![config.efficient_target.clone()]),
1009                    (Category::Capable, vec![config.capable_target.clone()]),
1010                    (
1011                        Category::Any,
1012                        vec![
1013                            config.efficient_target.clone(),
1014                            config.capable_target.clone(),
1015                        ],
1016                    ),
1017                    (Category::Judge, judges),
1018                ])
1019            }
1020        };
1021
1022        let subagents = match self {
1023            Self::Passthrough { subagents, .. }
1024            | Self::StageRouter { subagents, .. }
1025            | Self::Composite { subagents, .. } => subagents.as_ref(),
1026            _ => None,
1027        };
1028        // Sub-agent groups stay separate from the parent's. Merging them would let a
1029        // sub-agent's `capable` resolve to the parent's, and would put models only the
1030        // sub-agents were given into the parent's `Any` fallback list.
1031        let subagent = subagents
1032            .map(|subagents| subagent_runtime_model_names(subagents, route_name))
1033            .transpose()?;
1034        Ok(RuntimeModelNames { parent, subagent })
1035    }

1163impl LlmClassifierRouteConfig {
1164    fn classifier_mode(&self) -> ClassifierMode {
1165        let default = if self.escalation.is_some() {
1166            ClassifierMode::Escalation
1167        } else {
1168            ClassifierMode::Capability
1169        };
1170        self.mode.unwrap_or(default)
1171    }
1172
1173    fn validated_classifier_mode(
1174        &self,
1175        route_name: &str,
1176    ) -> AlgorithmResult<LlmClassifierModeConfig> {
1177        let Self {
1178            classifier_target,
1179            mode,
1180            strong_target,
1181            weak_target,
1182            base_threshold,
1183            threshold_step,
1184            fail_open,
1185            classify_trigger,
1186            message_hash_fallback,
1187            recent_turn_window,
1188            prompt,
1189            response_format_type,
1190            max_output_tokens,
1191            escalation,
1192            decision,
1193            models,
1194            default_target,
1195            response_schema,
1196            policy,
1197        } = self;
1198
1199        let selected_mode = match (mode, escalation.is_some()) {
1200            (Some(mode), _) => *mode,
1201            (None, true) => ClassifierMode::Escalation,
1202            (None, false) => ClassifierMode::Capability,
1203        };
1204
1205        if decision.is_some() && !matches!(selected_mode, ClassifierMode::Capability) {
1206            return Err(AlgorithmConfigError::new(format!(
1207                "llm_classifier route {route_name}: decision requires capability mode"
1208            )));
1209        }
1210
1211        if !matches!(selected_mode, ClassifierMode::Capability) && *fail_open == Some(true) {
1212            return Err(AlgorithmConfigError::new(format!(
1213                "llm_classifier route {route_name}: fail_open = true requires capability mode"
1214            )));
1215        }
1216
1217        match selected_mode {
1218            ClassifierMode::Capability => {
1219                if escalation.is_some() {
1220                    return Err(classifier_field_error(
1221                        route_name,
1222                        "escalation",
1223                        "capability",
1224                    ));
1225                }
1226                reject_custom_fields(
1227                    route_name,
1228                    "capability",
1229                    models,
1230                    default_target,
1231                    response_schema,
1232                    policy,
1233                )?;
1234                let judge = if let Some(decision) = decision {
1235                    if base_threshold.is_some()
1236                        || threshold_step.is_some()
1237                        || prompt.is_some()
1238                        || *response_format_type != ClassifierResponseFormat::JsonSchema
1239                        || *max_output_tokens != default_classifier_max_output_tokens()
1240                    {
1241                        return Err(AlgorithmConfigError::new(format!(
1242                            "llm_classifier route {route_name}: decision cannot use LLM judge settings; use decision.cutoff and decision.instructions"
1243                        )));
1244                    }
1245                    CapabilityJudgeRouteConfig::Decision(decision.clone())
1246                } else {
1247                    CapabilityJudgeRouteConfig::Llm(LlmCapabilityConfig {
1248                        base_threshold: required_classifier_field(
1249                            route_name,
1250                            "base_threshold",
1251                            base_threshold,
1252                        )?,
1253                        threshold_step: threshold_step.unwrap_or_default(),
1254                        contract: classifier_contract(prompt.as_deref())
1255                            .with_response_format_type(*response_format_type),
1256                        max_output_tokens: *max_output_tokens,
1257                    })
1258                };
1259                Ok(LlmClassifierModeConfig::Capability(
1260                    CapabilityClassifierRouteConfig {
1261                        classifier_target: classifier_target.clone(),
1262                        fail_open: fail_open.unwrap_or(true),
1263                        strong_target: required_classifier_field(
1264                            route_name,
1265                            "strong_target",
1266                            strong_target,
1267                        )?,
1268                        weak_target: required_classifier_field(
1269                            route_name,
1270                            "weak_target",
1271                            weak_target,
1272                        )?,
1273                        judge,
1274                        classify_trigger: *classify_trigger,
1275                        message_hash_fallback: *message_hash_fallback,
1276                        recent_turn_window: *recent_turn_window,
1277                    },
1278                ))
1279            }
1280            ClassifierMode::Escalation => {
1281                reject_custom_fields(
1282                    route_name,
1283                    "escalation",
1284                    models,
1285                    default_target,
1286                    response_schema,
1287                    policy,
1288                )?;
1289                if *classify_trigger != ClassifyTrigger::EveryRequest {
1290                    return Err(AlgorithmConfigError::new(format!(
1291                        "llm_classifier route {route_name} mode escalation cannot use classify_trigger"
1292                    )));
1293                }
1294                let has_capability_settings = base_threshold.is_some()
1295                    || threshold_step.is_some()
1296                    || fail_open.is_some()
1297                    || *message_hash_fallback
1298                    || recent_turn_window.is_some();
1299                if mode.is_some() && has_capability_settings {
1300                    return Err(AlgorithmConfigError::new(format!(
1301                        "llm_classifier route {route_name} mode escalation cannot use capability routing settings"
1302                    )));
1303                }
1304                if has_capability_settings {
1305                    // Existing files that select escalation through the `escalation` table alone
1306                    // keep loading, but their capability keys do nothing in this mode.
1307                    tracing::warn!(
1308                        "llm_classifier route {route_name} runs in escalation mode because it has an escalation table; base_threshold, threshold_step, fail_open, recent_turn_window and message_hash_fallback are ignored in this mode. Set mode = \"escalation\" and remove them."
1309                    );
1310                }
1311                Ok(LlmClassifierModeConfig::Escalation(
1312                    EscalationClassifierRouteConfig {
1313                        classifier_target: classifier_target.clone(),
1314                        strong_target: required_classifier_field(
1315                            route_name,
1316                            "strong_target",
1317                            strong_target,
1318                        )?,
1319                        weak_target: required_classifier_field(
1320                            route_name,
1321                            "weak_target",
1322                            weak_target,
1323                        )?,
1324                        prompt: prompt.clone(),
1325                        response_format_type: *response_format_type,
1326                        max_output_tokens: *max_output_tokens,
1327                        judge: required_classifier_field(route_name, "escalation", escalation)?,
1328                    },
1329                ))
1330            }
1331            ClassifierMode::Custom => {
1332                if !classifier_target.is_empty()
1333                    || strong_target.is_some()
1334                    || weak_target.is_some()
1335                    || base_threshold.is_some()
1336                    || threshold_step.is_some()
1337                    || escalation.is_some()
1338                    || *response_format_type != ClassifierResponseFormat::JsonSchema
1339                {
1340                    return Err(AlgorithmConfigError::new(format!(
1341                        "llm_classifier route {route_name} mode custom cannot use capability or escalation fields and response_format_type must be 'json_schema'"
1342                    )));
1343                }
1344                let models = required_classifier_field(route_name, "models", models)?;
1345                let default_target: Category = required_classifier_field(
1346                    route_name,
1347                    "default_target",
1348                    default_target,
1349                )?
1350                .parse()
1351                .map_err(|error| {
1352                    AlgorithmConfigError::new(format!(
1353                        "llm_classifier route {route_name} has invalid default_target: {error}"
1354                    ))
1355                })?;
1356                if default_target == Category::Judge {
1357                    return Err(AlgorithmConfigError::new(format!(
1358                        "llm_classifier route {route_name} default_target cannot be judge"
1359                    )));
1360                }
1361                models.validate(route_name, &default_target)?;
1362                Ok(LlmClassifierModeConfig::Custom(
1363                    CustomClassifierRouteConfig {
1364                        models,
1365                        default_target,
1366                        prompt: required_classifier_field(route_name, "prompt", prompt)?,
1367                        response_schema: required_classifier_field(
1368                            route_name,
1369                            "response_schema",
1370                            response_schema,
1371                        )?,
1372                        policy: required_classifier_field(route_name, "policy", policy)?,
1373                        classify_trigger: *classify_trigger,
1374                        message_hash_fallback: *message_hash_fallback,
1375                        recent_turn_window: *recent_turn_window,
1376                        max_output_tokens: *max_output_tokens,
1377                    },
1378                ))
1379            }
1380        }
1381    }
1382}

1419fn build_subagent_router_config(
1420    route_name: &str,
1421    config: &SubagentRouteConfig,
1422    targets: &BTreeMap<String, ModelId>,
1423) -> AlgorithmResult<SubagentRouterConfig> {
1424    match config {
1425        SubagentRouteConfig::Passthrough { target } => {
1426            resolve_target_model_id(route_name, target, targets)?;
1427            Ok(SubagentRouterConfig::fixed_target())
1428        }
1429        SubagentRouteConfig::LlmClassifier(config) => {
1430            let LlmClassifierModeConfig::Custom(config) =
1431                config.validated_classifier_mode(route_name)?
1432            else {
1433                return Err(AlgorithmConfigError::new(format!(
1434                    "route {route_name}: subagents llm_classifier only supports mode custom"
1435                )));
1436            };
1437            for name in config.models.all_names() {
1438                resolve_target_model_id(route_name, name, targets)?;
1439            }
1440            if config.models.get(&config.default_target).is_empty() {
1441                return Err(AlgorithmConfigError::new(format!(
1442                    "route {route_name}: subagents llm_classifier has no model for default category {}",
1443                    config.default_target.as_str()
1444                )));
1445            }
1446            let response_schema =
1447                serde_json::from_str(&config.response_schema).map_err(|error| {
1448                    AlgorithmConfigError::with_source(
1449                        format!(
1450                            "route {route_name}: subagents llm_classifier response_schema is invalid JSON: {error}"
1451                        ),
1452                        error,
1453                    )
1454                })?;
1455            warn_single_target_classifier(route_name, &config.models);
1456            let mut classifier_config = CustomClassifierConfig::new(
1457                config.prompt,
1458                response_schema,
1459                config.policy.into_libsy(),
1460            );
1461            classifier_config.recent_turn_window = config.recent_turn_window;
1462            classifier_config.max_output_tokens = config.max_output_tokens;
1463            let classifier = Arc::new(
1464                LlmTaskClassifier::new(LlmClassifierConfig::Custom {
1465                    default_target: config.default_target.clone(),
1466                    config: classifier_config,
1467                })
1468                .map_err(|error| {
1469                    AlgorithmConfigError::with_source(
1470                        format!("route {route_name}: subagents llm_classifier: {error}"),
1471                        error,
1472                    )
1473                })?,
1474            );
1475            Ok(SubagentRouterConfig {
1476                classifier,
1477                default_target: config.default_target,
1478                classify_trigger: config.classify_trigger,
1479                message_hash_fallback: config.message_hash_fallback,
1480            })
1481        }
1482    }
1483}

1504fn build_algorithm(
1505    route_name: &str,
1506    config: &AlgorithmSpec,
1507    targets: &BTreeMap<String, ModelId>,
1508) -> AlgorithmResult<Arc<dyn Algorithm>> {
1509    match config {
1510        AlgorithmSpec::Noop { .. } => Ok(Arc::new(Noop {})),
1511        AlgorithmSpec::Random {
1512            targets: names,
1513            weights,
1514            seed,
1515        } => {
1516            // The algorithm only sees its targets at request time, so a bad pairing
1517            // would otherwise fail every request instead of failing to start.
1518            if let Some(weights) = weights
1519                && weights.len() != names.len()
1520            {
1521                return Err(AlgorithmConfigError::new(format!(
1522                    "random route {route_name}: expected {} weights, got {}",
1523                    names.len(),
1524                    weights.len()
1525                )));
1526            }
1527            let mut seen = BTreeSet::new();
1528            if let Some(duplicate) = names.iter().find(|name| !seen.insert(*name)) {
1529                return Err(AlgorithmConfigError::new(format!(
1530                    "random route {route_name}: targets must be unique, {duplicate} is repeated"
1531                )));
1532            }
1533            let algorithm = Random::new(weights.clone(), *seed).map_err(|error| {
1534                AlgorithmConfigError::with_source(
1535                    format!("random route {route_name}: {error}"),
1536                    error,
1537                )
1538            })?;
1539            Ok(Arc::new(algorithm))
1540        }
1541        AlgorithmSpec::Passthrough { subagents, .. } => {
1542            let algorithm = Passthrough;
1543            let parent: Arc<dyn Algorithm> = Arc::new(algorithm);
1544            attach_subagent_router(route_name, parent, subagents.as_ref(), targets)
1545        }
1546        AlgorithmSpec::PlanExecute {
1547            tool_semantics,
1548            planning_prompt,
1549            handoff_prompt,
1550            planner_reasoning_as_text,
1551            ..
1552        } => {
1553            let mut config = PlanExecuteConfig::default();
1554            config.tool_semantics.clone_from(tool_semantics);
1555            if let Some(prompt) = planning_prompt {
1556                config.planning_prompt = prompt.clone();
1557            }
1558            config.handoff_prompt.clone_from(handoff_prompt);
1559            config.planner_reasoning_as_text = *planner_reasoning_as_text;
1560            let algorithm = PlanExecute::new(config).map_err(|error| {
1561                AlgorithmConfigError::with_source(
1562                    format!("plan_execute route {route_name}: {error}"),
1563                    error,
1564                )
1565            })?;
1566            Ok(Arc::new(algorithm))
1567        }
1568        AlgorithmSpec::LlmClassifier {
1569            config: classifier_config,
1570            ..
1571        } => {
1572            let mode = classifier_config.validated_classifier_mode(route_name)?;
1573            let algorithm = match mode {
1574                LlmClassifierModeConfig::Capability(config) => {
1575                    let classifier_config = config.into_task_config(route_name, targets)?;
1576                    LlmTaskClassifier::new(LlmClassifierConfig::Capability {
1577                        config: classifier_config,
1578                    })
1579                }
1580                LlmClassifierModeConfig::Escalation(config) => {
1581                    LlmTaskClassifier::new(LlmClassifierConfig::Escalation {
1582                        contract: classifier_contract(config.prompt.as_deref())
1583                            .with_response_format_type(config.response_format_type),
1584                        config: config.judge,
1585                        max_output_tokens: config.max_output_tokens,
1586                    })
1587                }
1588                LlmClassifierModeConfig::Custom(config) => {
1589                    let response_schema = serde_json::from_str(&config.response_schema).map_err(
1590                        |error| {
1591                            AlgorithmConfigError::with_source(
1592                                format!(
1593                                    "llm_classifier route {route_name}: response_schema is invalid JSON: {error}"
1594                                ),
1595                                error,
1596                            )
1597                        },
1598                    )?;
1599                    warn_single_target_classifier(route_name, &config.models);
1600                    let mut classifier_config = CustomClassifierConfig::new(
1601                        config.prompt,
1602                        response_schema,
1603                        config.policy.into_libsy(),
1604                    );
1605                    classifier_config.classify_trigger = config.classify_trigger;
1606                    classifier_config.message_hash_fallback = config.message_hash_fallback;
1607                    classifier_config.recent_turn_window = config.recent_turn_window;
1608                    classifier_config.max_output_tokens = config.max_output_tokens;
1609                    LlmTaskClassifier::new(LlmClassifierConfig::Custom {
1610                        default_target: config.default_target,
1611                        config: classifier_config,
1612                    })
1613                }
1614            }
1615            .map_err(|error| {
1616                AlgorithmConfigError::with_source(
1617                    format!("llm_classifier route {route_name}: {error}"),
1618                    error,
1619                )
1620            })?;
1621            Ok(Arc::new(algorithm))
1622        }
1623        AlgorithmSpec::StageRouter {
1624            tiers,
1625            picker,
1626            classifier,
1627            subagents,
1628            ..
1629        } => {
1630            let StageTierConfig {
1631                confidence_threshold,
1632                recent_turn_window,
1633                tool_semantics,
1634                capable_hold_turns,
1635                efficient_hold_turns,
1636                handoff_notes,
1637                ..
1638            } = tiers;
1639            if matches!(picker, PickerMode::CapableFirst) {
1640                tracing::warn!(
1641                    "stage_router route {route_name} uses picker \"capable_first\", which is experimental: published thresholds and routing results all come from \"efficient_first\", so there is no calibrated confidence_threshold for it and no measured accuracy or cost. Use \"efficient_first\" unless you are running your own calibration."
1642                );
1643            }
1644            tiers.warn_identical_tiers(route_name);
1645            let threshold = confidence_threshold.ok_or_else(|| {
1646                AlgorithmConfigError::new(format!(
1647                    "stage_router route {route_name}: confidence_threshold is required"
1648                ))
1649            })?;
1650            let mut config = StageRouterConfig::new(*picker, threshold);
1651            config.capable_first_threshold = tiers.capable_first_threshold;
1652            config.recent_window = *recent_turn_window;
1653            if let Some(turns) = capable_hold_turns {
1654                config.capable_hold_turns = *turns;
1655            }
1656            config.efficient_hold_turns = *efficient_hold_turns;
1657            config.tool_semantics = tool_semantics.clone();
1658            config.handoff_notes = handoff_notes.clone();
1659            // The judge is called through its own target, so it is not a routing
1660            // destination and stays out of the tier pair.
1661            config.llm_fallback = classifier
1662                .as_ref()
1663                .map(|classifier| {
1664                    classifier
1665                        .task_classifier_config(route_name, tiers, targets)
1666                        .map(|config| LlmFallback { config })
1667                })
1668                .transpose()?;
1669            let algorithm = StageRouter::new(config).map_err(|error| {
1670                AlgorithmConfigError::with_source(
1671                    format!("stage_router route {route_name}: {error}"),
1672                    error,
1673                )
1674            })?;
1675            let parent: Arc<dyn Algorithm> = Arc::new(algorithm);
1676            attach_subagent_router(route_name, parent, subagents.as_ref(), targets)
1677        }
1678        AlgorithmSpec::Auto {
1679            capable_target,
1680            efficient_target,
1681        } => {
1682            warn_identical_tiers(route_name, capable_target, efficient_target);
1683            let config = StageRouterConfig::new(PickerMode::EfficientFirst, 0.5);
1684            let algorithm = StageRouter::new(config).map_err(|error| {
1685                AlgorithmConfigError::with_source(
1686                    format!("auto route {route_name}: {error}"),
1687                    error,
1688                )
1689            })?;
1690            Ok(Arc::new(algorithm))
1691        }
1692        AlgorithmSpec::Composite {
1693            classifier,
1694            stage,
1695            subagents,
1696        } => {
1697            stage.warn_identical_tiers(route_name);
1698            let mut stage_config = StageRouterConfig::new(
1699                PickerMode::EfficientFirst,
1700                stage.confidence_threshold.unwrap_or(0.5),
1701            );
1702            stage_config.capable_first_threshold = stage.capable_first_threshold;
1703            stage_config.recent_window = stage.recent_turn_window;
1704            if let Some(turns) = stage.capable_hold_turns {
1705                stage_config.capable_hold_turns = turns;
1706            }
1707            stage_config.efficient_hold_turns = stage.efficient_hold_turns;
1708            stage_config.tool_semantics = stage.tool_semantics.clone();
1709            stage_config.handoff_notes = stage.handoff_notes.clone();
1710            let judge = if matches!(classifier.mode, Some(ClassifierMode::Custom)) {
1711                CompositeJudgeConfig::Custom(classifier.custom_classifier_config(route_name)?)
1712            } else {
1713                CompositeJudgeConfig::Capability(
1714                    classifier.task_classifier_config(route_name, stage, targets)?,
1715                )
1716            };
1717            let algorithm = CompositeRouter::new(CompositeRouterConfig {
1718                judge,
1719                stage: stage_config,
1720            })
1721            .map_err(|error| {
1722                AlgorithmConfigError::with_source(
1723                    format!("composite route {route_name}: {error}"),
1724                    error,
1725                )
1726            })?;
1727            let parent: Arc<dyn Algorithm> = Arc::new(algorithm);
1728            attach_subagent_router(route_name, parent, subagents.as_ref(), targets)
1729        }
1730        AlgorithmSpec::Advisor {
1731            reviewer_system_prompt,
1732            redo_feedback_prefix,
1733            gate_trigger,
1734            gate_trigger_pattern,
1735            max_reviews,
1736            gate_stall_turns,
1737            gate_min_tool_results,
1738            advisor_max_tokens,
1739            advisor_temperature,
1740            transcript_max_chars,
1741            fail_open,
1742            ..
1743        } => {
1744            // A pattern set under the default trigger would be silently
1745            // ignored; reject the misconfiguration instead.
1746            if *gate_trigger == AdvisorTriggerConfig::NoToolCall && gate_trigger_pattern.is_some() {
1747                return Err(AlgorithmConfigError::new(format!(
1748                    "advisor route {route_name}: gate_trigger_pattern requires \
1749                     gate_trigger = \"pattern\""
1750                )));
1751            }
1752            let mut config = AdvisorGateConfig::default();
1753            if let Some(prompt) = reviewer_system_prompt {
1754                config.reviewer_system_prompt = prompt.clone();
1755            }
1756            if let Some(prefix) = redo_feedback_prefix {
1757                config.redo_feedback_prefix = prefix.clone();
1758            }
1759            config.gate_trigger = match gate_trigger {
1760                AdvisorTriggerConfig::NoToolCall => GateTrigger::NoToolCall,
1761                AdvisorTriggerConfig::Pattern => {
1762                    GateTrigger::Pattern(gate_trigger_pattern.clone().unwrap_or_default())
1763                }
1764            };
1765            config.max_reviews = *max_reviews;
1766            config.gate_stall_turns = *gate_stall_turns;
1767            config.gate_min_tool_results = *gate_min_tool_results;
1768            config.advisor_max_tokens = *advisor_max_tokens;
1769            config.advisor_temperature = *advisor_temperature;
1770            config.transcript_max_chars = *transcript_max_chars;
1771            config.fail_open = *fail_open;
1772            let algorithm = AdvisorGate::new(config).map_err(|error| {
1773                AlgorithmConfigError::with_source(
1774                    format!("advisor route {route_name}: {error}"),
1775                    error,
1776                )
1777            })?;
1778            Ok(Arc::new(algorithm))
1779        }
1780        AlgorithmSpec::Vgr { config } => build_vgr(route_name, config, targets),
1781        AlgorithmSpec::PrefillRouter {
1782            targets: names,
1783            checkpoint,
1784            device,
1785            cache_dir,
1786            max_length,
1787            batch_size,
1788        } => {
1789            #[cfg(feature = "prefill-router")]
1790            {
1791                let targets = names
1792                    .iter()
1793                    .map(|name| resolve_target_model_id(route_name, name, targets))
1794                    .collect::<AlgorithmResult<Vec<_>>>()?;
1795                let mut config = prefill_router::PrefillRouterConfig::new(targets, checkpoint);
1796                config.device.clone_from(device);
1797                config.cache_dir.clone_from(cache_dir);
1798                if let Some(max_length) = max_length {
1799                    config.max_length = *max_length;
1800                }
1801                if let Some(batch_size) = batch_size {
1802                    config.batch_size = *batch_size;
1803                }
1804                let algorithm = config.build().map_err(|error| {
1805                    AlgorithmConfigError::with_source(
1806                        format!("prefill_router route {route_name}: {error}"),
1807                        error,
1808                    )
1809                })?;
1810                Ok(Arc::new(algorithm))
1811            }
1812
1813            #[cfg(not(feature = "prefill-router"))]
1814            {
1815                let _ = (
1816                    names, checkpoint, device, cache_dir, max_length, batch_size, targets,
1817                );
1818                Err(AlgorithmConfigError::new(format!(
1819                    "prefill_router route {route_name} requires the `prefill-router` Cargo feature"
1820                )))
1821            }
1822        }
1823    }
1824}

1826fn build_vgr(
1827    route_name: &str,
1828    config: &VgrRouteConfig,
1829    targets: &BTreeMap<String, ModelId>,
1830) -> AlgorithmResult<Arc<dyn Algorithm>> {
1831    let efficient = resolve_target_model_id(route_name, &config.efficient_target, targets)?;
1832    let capable = resolve_target_model_id(route_name, &config.capable_target, targets)?;
1833    let optional = |name: &Option<String>| {
1834        name.as_deref()
1835            .map(|name| resolve_target_model_id(route_name, name, targets))
1836            .transpose()
1837    };
1838    let mut runtime = VgrConfig::new(efficient, capable);
1839    runtime.targets.judge = optional(&config.judge_target)?;
1840    runtime.targets.capable_judge = optional(&config.capable_judge_target)?;
1841    runtime.mode = match config.mode {
1842        VgrModeConfig::Off => libsy::ServingMode::Off,
1843        VgrModeConfig::Evaluate => libsy::ServingMode::Evaluate,
1844        VgrModeConfig::Shadow => libsy::ServingMode::Shadow,
1845        VgrModeConfig::Active => libsy::ServingMode::Active {
1846            approval: config.active_approval.clone().unwrap_or_default(),
1847        },
1848    };
1849    runtime.deadline = vgr_duration(route_name, "deadline_seconds", config.deadline_seconds)?;
1850    runtime.bounded_efficient = config.bounded_efficient;
1851    runtime.task_typing = config.task_typing;
1852    runtime.confirmed_recovery_min_clean_tail = config.confirmed_recovery_min_clean_tail;
1853    runtime.agentic_handoff = config.agentic_handoff;
1854    runtime.compact_handoff = config.compact_handoff;
1855    runtime.latch_escalation = config.latch_escalation;
1856    runtime.efficient_turn_budget = config
1857        .efficient_turn_budget_seconds
1858        .map(|seconds| vgr_duration(route_name, "efficient_turn_budget_seconds", seconds))
1859        .transpose()?;
1860    runtime.breaker = BreakerConfig {
1861        threshold: config.breaker_threshold,
1862        cooldown: vgr_duration(
1863            route_name,
1864            "breaker_cooldown_seconds",
1865            config.breaker_cooldown_seconds,
1866        )?,
1867    };
1868    Vgr::new(runtime)
1869        .map(|algorithm| Arc::new(algorithm) as Arc<dyn Algorithm>)
1870        .map_err(|error| {
1871            AlgorithmConfigError::with_source(format!("vgr route {route_name}: {error}"), error)
1872        })
1873}
1874
1875fn vgr_duration(route: &str, field: &str, seconds: f64) -> AlgorithmResult<std::time::Duration> {
1876    let message = || format!("vgr route {route}: {field} must be a positive number");
1877    let duration = std::time::Duration::try_from_secs_f64(seconds)
1878        .map_err(|error| AlgorithmConfigError::with_source(message(), error))?;
1879    if duration.is_zero() {
1880        return Err(AlgorithmConfigError::new(message()));
1881    }
1882    Ok(duration)
1883}

1961fn resolve_runtime_models(
1962    route_name: &str,
1963    names: HashMap<Category, Vec<String>>,
1964    targets: &BTreeMap<String, ModelId>,
1965) -> AlgorithmResult<HashMap<Category, Vec<ModelId>>> {
1966    names
1967        .into_iter()
1968        .map(|(category, names)| {
1969            let models = names
1970                .into_iter()
1971                .map(|name| resolve_target_model_id(route_name, &name, targets))
1972                .collect::<AlgorithmResult<Vec<_>>>()?;
1973            Ok((category, models))
1974        })
1975        .collect()
